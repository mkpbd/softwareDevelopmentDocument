(এখানে শুধু পরিবর্তিত/গুরুত্বপূর্ণ টেবিলগুলো দেখানো হলো)

**`salary_components`** — (আগের মতোই, কোনো পরিবর্তন নেই)

**`payroll_runs`** — (আগের মতোই)

**`payroll_details`** — (আগের মতোই)

**`payroll_detail_components`** — (আগের মতোই)

**`journal_vouchers`** — (আগের মতোই)

**`journal_voucher_lines`** — (আগের মতোই)

**`payment_transactions`** _(নতুন যোগ করা হয়েছে)_

SQL

```
id                      BIGINT PRIMARY KEY AUTO_INCREMENT,
payroll_detail_id       BIGINT NOT NULL,
transaction_date        DATE NOT NULL,
amount                  DECIMAL(15,2) NOT NULL,
payment_method          ENUM('Bank','Cash','Mobile_Banking','Cheque') NOT NULL,
transaction_id          VARCHAR(100) NULL,           -- Bank ref or bkash/nagad trx id
status                  ENUM('Pending','Success','Failed') DEFAULT 'Pending',
remarks                 TEXT NULL,
created_at              TIMESTAMP DEFAULT CURRENT_TIMESTAMP,
FOREIGN KEY (payroll_detail_id) REFERENCES payroll_details(id)
```

```sql
	
-- =============================================
-- Integrated HRM + Accounting System
-- PostgreSQL DDL Script (Version 14+)
-- =============================================

-- Disable foreign key checks for initial setup
SET session_replication_role = 'replica';

-- =============================================
-- 1. Company Structure
-- =============================================

CREATE TABLE organizations (
    id              BIGSERIAL PRIMARY KEY,
    name            VARCHAR(150) NOT NULL,
    legal_name      VARCHAR(200),
    bin             VARCHAR(20),
    tin             VARCHAR(20),
    created_at      TIMESTAMP DEFAULT CURRENT_TIMESTAMP,
    updated_at      TIMESTAMP DEFAULT CURRENT_TIMESTAMP
);

CREATE TABLE branches (
    id                BIGSERIAL PRIMARY KEY,
    organization_id   BIGINT NOT NULL,
    name              VARCHAR(100) NOT NULL,
    branch_code       VARCHAR(20) NOT NULL UNIQUE,
    address           TEXT,
    is_head_office   BOOLEAN DEFAULT FALSE,
    created_at        TIMESTAMP DEFAULT CURRENT_TIMESTAMP,
    FOREIGN KEY (organization_id) REFERENCES organizations(id) ON DELETE RESTRICT
);

CREATE TABLE departments (
    id                BIGSERIAL PRIMARY KEY,
    branch_id         BIGINT NOT NULL,
    name              VARCHAR(80) NOT NULL,
    department_code   VARCHAR(20),
    cost_center_code  VARCHAR(20),
    created_at        TIMESTAMP DEFAULT CURRENT_TIMESTAMP,
    FOREIGN KEY (branch_id) REFERENCES branches(id) ON DELETE RESTRICT
);

CREATE TABLE designations (
    id          BIGSERIAL PRIMARY KEY,
    name        VARCHAR(100) NOT NULL UNIQUE,
    rank_level  INTEGER,
    created_at  TIMESTAMP DEFAULT CURRENT_TIMESTAMP
);

-- =============================================
-- 2. Employee Management
-- =============================================

CREATE TABLE employees (
    id                  BIGSERIAL PRIMARY KEY,
    employee_id         VARCHAR(30) NOT NULL UNIQUE,
    full_name           VARCHAR(150) NOT NULL,
    email               VARCHAR(100) UNIQUE,
    mobile              VARCHAR(15) UNIQUE,
    nid                 VARCHAR(20),
    department_id       BIGINT NOT NULL,
    designation_id      BIGINT NOT NULL,
    joining_date        DATE NOT NULL,
    resignation_date    DATE,
    status              VARCHAR(20) DEFAULT 'Active',
    created_at          TIMESTAMP DEFAULT CURRENT_TIMESTAMP,
    updated_at          TIMESTAMP DEFAULT CURRENT_TIMESTAMP,
    FOREIGN KEY (department_id) REFERENCES departments(id),
    FOREIGN KEY (designation_id) REFERENCES designations(id)
);

CREATE TABLE employee_documents (
    id              BIGSERIAL PRIMARY KEY,
    employee_id     BIGINT NOT NULL,
    document_type   VARCHAR(50) NOT NULL,
    file_path       VARCHAR(255) NOT NULL,
    uploaded_at     TIMESTAMP DEFAULT CURRENT_TIMESTAMP,
    FOREIGN KEY (employee_id) REFERENCES employees(id) ON DELETE CASCADE
);

-- =============================================
-- 3. Attendance & Leave
-- =============================================

CREATE TABLE attendance (
    id                  BIGSERIAL PRIMARY KEY,
    employee_id         BIGINT NOT NULL,
    attendance_date     DATE NOT NULL,
    check_in            TIME,
    check_out           TIME,
    status              VARCHAR(20) NOT NULL,
    overtime_hours      DECIMAL(5,2) DEFAULT 0.00,
    created_at          TIMESTAMP DEFAULT CURRENT_TIMESTAMP,
    FOREIGN KEY (employee_id) REFERENCES employees(id),
    UNIQUE (employee_id, attendance_date)
);

CREATE TABLE leave_types (
    id                  BIGSERIAL PRIMARY KEY,
    leave_name          VARCHAR(50) NOT NULL UNIQUE,
    total_days_per_year INTEGER NOT NULL,
    is_paid             BOOLEAN DEFAULT TRUE,
    created_at          TIMESTAMP DEFAULT CURRENT_TIMESTAMP
);

CREATE TABLE leave_applications (
    id              BIGSERIAL PRIMARY KEY,
    employee_id     BIGINT NOT NULL,
    leave_type_id   BIGINT NOT NULL,
    from_date       DATE NOT NULL,
    to_date         DATE NOT NULL,
    total_days      DECIMAL(5,2) NOT NULL,
    reason          TEXT,
    status          VARCHAR(20) DEFAULT 'Pending',
    approved_by     BIGINT,
    approved_at     TIMESTAMP,
    created_at      TIMESTAMP DEFAULT CURRENT_TIMESTAMP,
    FOREIGN KEY (employee_id) REFERENCES employees(id),
    FOREIGN KEY (leave_type_id) REFERENCES leave_types(id)
);

-- =============================================
-- 4. Payroll & Accounting
-- =============================================

CREATE TABLE salary_components (
    id                      BIGSERIAL PRIMARY KEY,
    component_name          VARCHAR(80) NOT NULL UNIQUE,
    component_type          VARCHAR(20) NOT NULL,
    is_taxable              BOOLEAN DEFAULT FALSE,
    is_pf_applicable        BOOLEAN DEFAULT FALSE,
    is_loan_deduction       BOOLEAN DEFAULT FALSE,
    calculation_type        VARCHAR(20) NOT NULL,
    formula                 TEXT,
    sort_order              INTEGER DEFAULT 0,
    is_active               BOOLEAN DEFAULT TRUE,
    created_at              TIMESTAMP DEFAULT CURRENT_TIMESTAMP
);

CREATE TABLE chart_of_accounts (
    id                  BIGSERIAL PRIMARY KEY,
    account_code        VARCHAR(20) NOT NULL UNIQUE,
    account_name        VARCHAR(150) NOT NULL,
    account_type        VARCHAR(20) NOT NULL,
    parent_id           BIGINT,
    is_control_account  BOOLEAN DEFAULT FALSE,
    is_active           BOOLEAN DEFAULT TRUE,
    created_at          TIMESTAMP DEFAULT CURRENT_TIMESTAMP,
    FOREIGN KEY (parent_id) REFERENCES chart_of_accounts(id)
);

CREATE TABLE account_mapping (
    id                      BIGSERIAL PRIMARY KEY,
    salary_component_id     BIGINT NOT NULL,
    debit_account_id        BIGINT NOT NULL,
    credit_account_id       BIGINT NOT NULL,
    is_active               BOOLEAN DEFAULT TRUE,
    created_at              TIMESTAMP DEFAULT CURRENT_TIMESTAMP,
    FOREIGN KEY (salary_component_id) REFERENCES salary_components(id) ON DELETE CASCADE,
    FOREIGN KEY (debit_account_id) REFERENCES chart_of_accounts(id),
    FOREIGN KEY (credit_account_id) REFERENCES chart_of_accounts(id),
    UNIQUE (salary_component_id)
);

CREATE TABLE payroll_runs (
    id              BIGSERIAL PRIMARY KEY,
    month           INTEGER NOT NULL CHECK (month BETWEEN 1 AND 12),
    year            INTEGER NOT NULL,
    run_date        DATE NOT NULL,
    status          VARCHAR(20) DEFAULT 'Draft',
    posted_at       TIMESTAMP,
    posted_by       BIGINT,
    created_at      TIMESTAMP DEFAULT CURRENT_TIMESTAMP,
    UNIQUE (month, year)
);

CREATE TABLE payroll_details (
    id                      BIGSERIAL PRIMARY KEY,
    payroll_run_id          BIGINT NOT NULL,
    employee_id             BIGINT NOT NULL,
    gross_salary            DECIMAL(15,2) NOT NULL,
    total_deduction         DECIMAL(15,2) DEFAULT 0.00,
    net_salary              DECIMAL(15,2) NOT NULL,
    payment_status          VARCHAR(20) DEFAULT 'Unpaid',
    journal_voucher_id      BIGINT,
    created_at              TIMESTAMP DEFAULT CURRENT_TIMESTAMP,
    FOREIGN KEY (payroll_run_id) REFERENCES payroll_runs(id) ON DELETE CASCADE,
    FOREIGN KEY (employee_id) REFERENCES employees(id),
    FOREIGN KEY (journal_voucher_id) REFERENCES journal_vouchers(id)
);

CREATE TABLE payroll_detail_components (
    id                        BIGSERIAL PRIMARY KEY,
    payroll_detail_id         BIGINT NOT NULL,
    salary_component_id       BIGINT NOT NULL,
    amount                    DECIMAL(15,2) NOT NULL,
    calculated_value          DECIMAL(15,2),
    FOREIGN KEY (payroll_detail_id) REFERENCES payroll_details(id) ON DELETE CASCADE,
    FOREIGN KEY (salary_component_id) REFERENCES salary_components(id)
);

-- =============================================
-- 5. Accounting & Transaction Tables
-- =============================================

CREATE TABLE journal_vouchers (
    id                  BIGSERIAL PRIMARY KEY,
    voucher_no          VARCHAR(50) NOT NULL UNIQUE,
    voucher_date        DATE NOT NULL,
    reference_type      VARCHAR(30) DEFAULT 'Payroll',
    reference_id        BIGINT,
    narration           TEXT,
    status              VARCHAR(20) DEFAULT 'Draft',
    created_by          BIGINT NOT NULL,
    created_at          TIMESTAMP DEFAULT CURRENT_TIMESTAMP,
    posted_at           TIMESTAMP
);

CREATE TABLE journal_voucher_lines (
    id                      BIGSERIAL PRIMARY KEY,
    journal_voucher_id      BIGINT NOT NULL,
    account_id              BIGINT NOT NULL,
    debit_amount            DECIMAL(15,2) DEFAULT 0.00,
    credit_amount           DECIMAL(15,2) DEFAULT 0.00,
    department_id           BIGINT,
    employee_id             BIGINT,
    narration               TEXT,
    FOREIGN KEY (journal_voucher_id) REFERENCES journal_vouchers(id) ON DELETE CASCADE,
    FOREIGN KEY (account_id) REFERENCES chart_of_accounts(id),
    FOREIGN KEY (department_id) REFERENCES departments(id),
    FOREIGN KEY (employee_id) REFERENCES employees(id)
);

CREATE TABLE employee_loans (
    id                    BIGSERIAL PRIMARY KEY,
    employee_id           BIGINT NOT NULL,
    loan_no               VARCHAR(30) NOT NULL UNIQUE,
    loan_amount           DECIMAL(15,2) NOT NULL,
    installment_amount    DECIMAL(15,2) NOT NULL,
    total_installments    INTEGER NOT NULL,
    paid_installments     INTEGER DEFAULT 0,
    remaining_amount      DECIMAL(15,2) NOT NULL,
    status                VARCHAR(20) DEFAULT 'Active',
    created_at            TIMESTAMP DEFAULT CURRENT_TIMESTAMP,
    FOREIGN KEY (employee_id) REFERENCES employees(id)
);

CREATE TABLE provident_fund (
    id                          BIGSERIAL PRIMARY KEY,
    employee_id                 BIGINT NOT NULL,
    employee_contribution_rate  DECIMAL(5,2) DEFAULT 10.00,
    company_contribution_rate   DECIMAL(5,2) DEFAULT 10.00,
    total_balance               DECIMAL(15,2) DEFAULT 0.00,
    last_deposit_date           DATE,
    FOREIGN KEY (employee_id) REFERENCES employees(id),
    UNIQUE (employee_id)
);

CREATE TABLE tax_calculations (
    id                    BIGSERIAL PRIMARY KEY,
    payroll_detail_id     BIGINT NOT NULL,
    taxable_income        DECIMAL(15,2) NOT NULL,
    tax_amount            DECIMAL(15,2) NOT NULL,
    tax_rule_version      VARCHAR(20) NOT NULL,
    calculated_at         TIMESTAMP DEFAULT CURRENT_TIMESTAMP,
    FOREIGN KEY (payroll_detail_id) REFERENCES payroll_details(id) ON DELETE CASCADE
);

CREATE TABLE payment_transactions (
    id                      BIGSERIAL PRIMARY KEY,
    payroll_detail_id       BIGINT NOT NULL,
    transaction_date        DATE NOT NULL,
    amount                  DECIMAL(15,2) NOT NULL,
    payment_method          VARCHAR(20) NOT NULL,
    transaction_id          VARCHAR(100),
    status                  VARCHAR(20) DEFAULT 'Pending',
    remarks                 TEXT,
    created_at              TIMESTAMP DEFAULT CURRENT_TIMESTAMP,
    FOREIGN KEY (payroll_detail_id) REFERENCES payroll_details(id)
);

-- =============================================
-- Re-enable foreign key checks
-- =============================================
SET session_replication_role = 'origin';

```


### **যদি আপনি Enum Types চান:**

আপনি যদি সঠিক PostgreSQL Enum টাইপ ব্যবহার করতে চান, তাহলে প্রথমে এই টাইপগুলো তৈরি করুন:

```sql
-- Create Enum Types
CREATE TYPE employee_status AS ENUM ('Active', 'Resigned', 'Terminated', 'OnLeave');
CREATE TYPE component_type AS ENUM ('Earning', 'Deduction', 'Statutory');
CREATE TYPE calculation_type AS ENUM ('Fixed', 'Percentage', 'Formula');
CREATE TYPE account_type AS ENUM ('Asset', 'Liability', 'Equity', 'Income', 'Expense');
CREATE TYPE payroll_status AS ENUM ('Draft', 'Processed', 'Posted', 'Failed');
CREATE TYPE payment_status AS ENUM ('Unpaid', 'Paid');
CREATE TYPE voucher_status AS ENUM ('Draft', 'Posted', 'Cancelled');
CREATE TYPE attendance_status AS ENUM ('Present', 'Absent', 'Late', 'HalfDay', 'Holiday', 'Weekend');
CREATE TYPE leave_status AS ENUM ('Pending', 'Approved', 'Rejected', 'Cancelled');
CREATE TYPE payment_method AS ENUM ('Bank', 'Cash', 'Mobile_Banking', 'Cheque');
CREATE TYPE transaction_status AS ENUM ('Pending', 'Success', 'Failed');

-- Then modify the tables to use these types
ALTER TABLE employees 
    ALTER COLUMN status TYPE employee_status 
    USING status::employee_status;

-- Similar modifications for other tables...
```
