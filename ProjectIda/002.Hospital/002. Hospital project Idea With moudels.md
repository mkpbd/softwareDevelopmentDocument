-- ============================================
-- SCHEMA SETUP

CREATE SCHEMA IF NOT EXISTS clinical;
CREATE SCHEMA IF NOT EXISTS finance;

CREATE SCHEMA IF NOT EXISTS hr;

-- Extensions
CREATE EXTENSION IF NOT EXISTS "uuid-ossp";
CREATE EXTENSION IF NOT EXISTS "pgcrypto";
CREATE EXTENSION IF NOT EXISTS "pg_trgm";

-- ============================================
-- TABLE: clinical.patient_categories
-- PURPOSE: VIP/Regular/Corporate/Foreign
-- ============================================
CREATE TABLE clinical.patient_categories (
    category_id         SERIAL PRIMARY KEY,
    tenant_id           UUID NOT NULL,
    category_name       VARCHAR(100) NOT NULL,
    category_code       VARCHAR(20) NOT NULL,
    category_color      VARCHAR(10),
    discount_percent    DECIMAL(5,2) DEFAULT 0,
    priority_level      INT DEFAULT 1,
    special_services    JSONB DEFAULT '{}',
    is_active           BOOLEAN DEFAULT TRUE,
    created_at          TIMESTAMPTZ DEFAULT NOW(),
    UNIQUE(tenant_id, category_code)
);

-- ============================================
-- TABLE: clinical.patients
-- PURPOSE: Core patient master record
-- ============================================
CREATE TABLE clinical.patients (
    patient_id          SERIAL PRIMARY KEY,
    tenant_id           UUID NOT NULL,
    branch_id           INT NOT NULL,
    -- Identifiers
    mrn                 VARCHAR(30) NOT NULL,
    mrn_barcode         VARCHAR(200),
    mrn_qr_code         TEXT,
    -- Personal Info
    first_name          VARCHAR(100) NOT NULL,
    last_name           VARCHAR(100) NOT NULL,
    first_name_bn       VARCHAR(100),
    last_name_bn        VARCHAR(100),
    full_name           VARCHAR(200)
                        GENERATED ALWAYS AS (
                            first_name || ' ' || last_name
                        ) STORED,
    date_of_birth       DATE NOT NULL,
    age_years           INT
                        GENERATED ALWAYS AS (
                            EXTRACT(YEAR FROM AGE(date_of_birth))::INT
                        ) STORED,
    gender              VARCHAR(10) NOT NULL
                        CHECK (gender IN ('male','female','other')),
    blood_group         VARCHAR(5)
                        CHECK (blood_group IN (
                            'A+','A-','B+','B-',
                            'AB+','AB-','O+','O-','unknown'
                        )),
    -- Identity Documents
    national_id         VARCHAR(30),
    birth_certificate   VARCHAR(30),
    passport_number     VARCHAR(30),
    passport_expiry     DATE,
    visa_number         VARCHAR(30),
    visa_expiry         DATE,
    driving_license     VARCHAR(30),
    -- Contact
    primary_phone       VARCHAR(20) NOT NULL,
    secondary_phone     VARCHAR(20),
    whatsapp_number     VARCHAR(20),
    email               VARCHAR(150),
    -- Address
    present_address     TEXT,
    present_city        VARCHAR(100),
    present_district    VARCHAR(100),
    present_upazila     VARCHAR(100),
    permanent_address   TEXT,
    permanent_city      VARCHAR(100),
    permanent_district  VARCHAR(100),
    -- Demographics
    marital_status      VARCHAR(20)
                        CHECK (marital_status IN (
                            'single','married',
                            'divorced','widowed','separated'
                        )),
    religion            VARCHAR(50),
    nationality         VARCHAR(50) DEFAULT 'Bangladeshi',
    occupation          VARCHAR(100),
    education_level     VARCHAR(50),
    monthly_income      DECIMAL(12,2),
    -- Patient Classification
    category_id         INT REFERENCES
                            clinical.patient_categories(category_id),
    patient_type        VARCHAR(20) DEFAULT 'regular'
                        CHECK (patient_type IN (
                            'regular','vip','foreign',
                            'corporate','government',
                            'staff','complimentary'
                        )),
    corporate_id        INT,
    -- Family
    family_group_id     INT,
    relationship_head   VARCHAR(50),
    -- Photo & Media
    photo_url           VARCHAR(500),
    signature_url       VARCHAR(500),
    -- Portal Access
    portal_enabled      BOOLEAN DEFAULT FALSE,
    portal_username     VARCHAR(100),
    portal_password_hash VARCHAR(255),
    portal_pin          VARCHAR(6),
    -- Status
    registration_type   VARCHAR(20) DEFAULT 'walk_in'
                        CHECK (registration_type IN (
                            'walk_in','online','phone',
                            'referred','emergency'
                        )),
    is_deceased         BOOLEAN DEFAULT FALSE,
    deceased_at         TIMESTAMPTZ,
    death_certificate_no VARCHAR(50),
    is_active           BOOLEAN DEFAULT TRUE,
    is_blacklisted      BOOLEAN DEFAULT FALSE,
    blacklist_reason    TEXT,
    -- MLC
    is_mlc_patient      BOOLEAN DEFAULT FALSE,
    -- Tracking
    registered_by       INT,
    registered_at       TIMESTAMPTZ DEFAULT NOW(),
    last_visit_date     DATE,
    total_visits        INT DEFAULT 0,
    created_at          TIMESTAMPTZ DEFAULT NOW(),
    updated_at          TIMESTAMPTZ DEFAULT NOW(),
    UNIQUE(tenant_id, mrn)
);

-- Performance Indexes
CREATE INDEX idx_pat_tenant
    ON clinical.patients(tenant_id);
CREATE INDEX idx_pat_mrn
    ON clinical.patients(mrn);
CREATE INDEX idx_pat_phone
    ON clinical.patients(primary_phone);
CREATE INDEX idx_pat_name
    ON clinical.patients
    USING gin(full_name gin_trgm_ops);
CREATE INDEX idx_pat_nid
    ON clinical.patients(national_id)
    WHERE national_id IS NOT NULL;
CREATE INDEX idx_pat_dob
    ON clinical.patients(date_of_birth);
CREATE INDEX idx_pat_active
    ON clinical.patients(tenant_id, is_active);

-- ============================================
-- TABLE: clinical.patient_family_groups
-- PURPOSE: Link family members together
-- ============================================
CREATE TABLE clinical.patient_family_groups (
    group_id            SERIAL PRIMARY KEY,
    tenant_id           UUID NOT NULL,
    group_name          VARCHAR(200) NOT NULL,
    head_patient_id     INT REFERENCES
                            clinical.patients(patient_id),
    total_members       INT DEFAULT 1,
    created_at          TIMESTAMPTZ DEFAULT NOW()
);

ALTER TABLE clinical.patients
    ADD CONSTRAINT fk_patient_family
    FOREIGN KEY (family_group_id)
    REFERENCES clinical.patient_family_groups(group_id);

-- ============================================
-- TABLE: clinical.patient_emergency_contacts
-- PURPOSE: Emergency contact persons
-- ============================================
CREATE TABLE clinical.patient_emergency_contacts (
    contact_id          SERIAL PRIMARY KEY,
    patient_id          INT NOT NULL REFERENCES
                            clinical.patients(patient_id)
                            ON DELETE CASCADE,
    contact_name        VARCHAR(200) NOT NULL,
    relationship        VARCHAR(50) NOT NULL,
    primary_phone       VARCHAR(20) NOT NULL,
    secondary_phone     VARCHAR(20),
    whatsapp            VARCHAR(20),
    email               VARCHAR(150),
    address             TEXT,
    occupation          VARCHAR(100),
    is_primary          BOOLEAN DEFAULT FALSE,
    is_legal_guardian   BOOLEAN DEFAULT FALSE,
    created_at          TIMESTAMPTZ DEFAULT NOW()
);

-- ============================================
-- TABLE: clinical.patient_allergies
-- PURPOSE: Known allergies with severity
-- ============================================
CREATE TABLE clinical.patient_allergies (
    allergy_id          SERIAL PRIMARY KEY,
    patient_id          INT NOT NULL REFERENCES
                            clinical.patients(patient_id)
                            ON DELETE CASCADE,
    allergen_type       VARCHAR(30) NOT NULL
                        CHECK (allergen_type IN (

-- ============================================
CREATE OR REPLACE FUNCTION clinical.generate_mrn()
RETURNS TRIGGER AS $$
DECLARE
    v_prefix    VARCHAR(10);
    v_year      VARCHAR(4);
    v_sequence  VARCHAR(8);
    v_mrn       VARCHAR(30);
BEGIN
    v_prefix := 'MRN';
    v_year := TO_CHAR(NOW(), 'YYYY');
    v_sequence := LPAD(NEW.patient_id::TEXT, 6, '0');
    v_mrn := v_prefix || '-' || v_year || '-' || v_sequence;
    NEW.mrn := v_mrn;
    RETURN NEW;
END;
$$ LANGUAGE plpgsql;

CREATE TRIGGER trg_generate_mrn
    BEFORE INSERT ON clinical.patients
    FOR EACH ROW
    WHEN (NEW.mrn IS NULL OR NEW.mrn = '')
    EXECUTE FUNCTION clinical.generate_mrn();

-- ============================================
-- TRIGGER: Update last visit date
-- ============================================
CREATE OR REPLACE FUNCTION clinical.update_patient_visit_stats()
RETURNS TRIGGER AS $$
BEGIN
    UPDATE clinical.patients
    SET
        last_visit_date = NEW.visit_date,
        total_visits = total_visits + 1,
        updated_at = NOW()
    WHERE patient_id = NEW.patient_id;
    RETURN NEW;
END;
$$ LANGUAGE plpgsql;

-- VIEWS: Bed Status Dashboard
-- ============================================
CREATE VIEW clinical.bed_status_dashboard AS
SELECT
    b.tenant_id,
    b.branch_id,
    w.ward_name,
    w.ward_type,
    COUNT(b.bed_id)                              AS total_beds,
    COUNT(b.bed_id) FILTER (
        WHERE b.status = 'available'
    )                                            AS available_beds,
    COUNT(b.bed_id) FILTER (
        WHERE b.status = 'occupied'
    )                                            AS occupied_beds,
    COUNT(b.bed_id) FILTER (
        WHERE b.status = 'reserved'
    )                                            AS reserved_beds,
    COUNT(b.bed_id) FILTER (
        WHERE b.status = 'cleaning'
    )                                            AS cleaning_beds,
    COUNT(b.bed_id) FILTER (
        WHERE b.status = 'maintenance'
    )                                            AS maintenance_beds,
    ROUND(
        COUNT(b.bed_id) FILTER (
            WHERE b.status = 'occupied'
        )::DECIMAL /
        NULLIF(COUNT(b.bed_id), 0) * 100, 2
    )                                            AS occupancy_rate
FROM clinical.beds b
JOIN clinical.wards w ON w.ward_id = b.ward_id
WHERE b.is_active = TRUE
GROUP BY
    b.tenant_id, b.branch_id,
    w.ward_id, w.ward_name, w.ward_type
ORDER BY w.ward_name;

-- ============================================
-- TRIGGER: Auto update bed status on transfer
-- ============================================
CREATE OR REPLACE FUNCTION clinical.execute_bed_transfer()
RETURNS TRIGGER AS $$
BEGIN
    IF NEW.status = 'completed'
       AND OLD.status != 'completed'
    THEN
        -- Free old bed
        UPDATE clinical.beds
        SET
            status = 'cleaning',
            current_admission_id = NULL,
            occupied_since = NULL
        WHERE bed_id = NEW.from_bed_id;

        -- Occupy new bed
        UPDATE clinical.beds
        SET
            status = 'occupied',
            current_admission_id = NEW.admission_id,
            occupied_since = NEW.executed_at
        WHERE bed_id = NEW.to_bed_id;
    END IF;
    RETURN NEW;
END;
$$ LANGUAGE plpgsql;

CREATE TRIGGER trg_bed_transfer_execute
    AFTER UPDATE ON clinical.bed_transfers
    FOR EACH ROW
    EXECUTE FUNCTION clinical.execute_bed_transfer();

    ins.company_name,
    COUNT(ic.claim_id)          AS total_claims,
    SUM(ic.claimed_amount)      AS total_claimed,
    SUM(ic.settled_amount)      AS total_settled,
    SUM(ic.claimed_amount) -
    COALESCE(SUM(ic.settled_amount), 0)
                                AS outstanding,
    COUNT(ic.claim_id) FILTER (
        WHERE ic.status = 'submitted'
    )                           AS pending_claims,
    COUNT(ic.claim_id) FILTER (
        WHERE ic.status = 'query_raised'
    )                           AS query_raised,
    COUNT(ic.claim_id) FILTER (
        WHERE ic.status = 'rejected'
    )                           AS rejected_claims,
    AVG(
        EXTRACT(EPOCH FROM (
            ic.updated_at - ic.submitted_at
        )) / 86400
    )                           AS avg_settlement_days
FROM finance.insurance_claims ic
JOIN finance.patient_insurance_policies pip
    ON pip.policy_id = ic.policy_id
JOIN finance.insurance_companies ins
    ON ins.company_id = pip.company_id

CREATE VIEW pharmacy.expiry_alerts AS
SELECT
    ms.tenant_id,
    ms.branch_id,
    pl.location_name,
    m.medicine_code,
    m.medicine_name,
    m.generic_name,
    ms.batch_number,
    ms.expiry_date,
    ms.current_quantity,
    ms.available_quantity,
    ms.selling_price,
    (ms.expiry_date - CURRENT_DATE)     AS days_to_expiry,
    CASE
        WHEN ms.expiry_date < CURRENT_DATE
            THEN 'EXPIRED'
        WHEN ms.expiry_date <=
             CURRENT_DATE + INTERVAL '30 days'
            THEN 'CRITICAL'
        WHEN ms.expiry_date <=
             CURRENT_DATE + INTERVAL '60 days'
            THEN 'WARNING'
        ELSE 'NOTICE'
    END                                 AS alert_level
FROM pharmacy.medicine_stock ms
JOIN pharmacy.medicines m
    ON m.medicine_id = ms.medicine_id
JOIN pharmacy.pharmacy_locations pl
    ON pl.location_id = ms.location_id
WHERE ms.status = 'active'
  AND ms.current_quantity > 0
  AND ms.expiry_date <= CURRENT_DATE + INTERVAL '90 days'
ORDER BY ms.expiry_date;

-- ============================================
-- TRIGGER: Auto deduct stock on dispensing
-- ============================================
CREATE OR REPLACE FUNCTION pharmacy.deduct_stock_on_dispensing()
RETURNS TRIGGER AS $$
BEGIN
    UPDATE pharmacy.medicine_stock
    SET
        current_quantity = current_quantity - NEW.quantity_dispensed,
        updated_at = NOW()
    WHERE stock_id = NEW.stock_id;

    INSERT INTO pharmacy.stock_transactions (
        tenant_id, branch_id, location_id,
        medicine_id, stock_id,
        txn_type, reference_type,
        reference_id, quantity_out,
        balance_quantity, unit_price,
        total_value, batch_number,
        expiry_date, patient_id,
        created_by
    )
    SELECT
        ms.tenant_id, ms.branch_id,
        ms.location_id, NEW.medicine_id,
        NEW.stock_id, 'dispensing',
        'dispensing_session',
        NEW.session_id,
        NEW.quantity_dispensed,
        ms.current_quantity,
        NEW.unit_price,
        NEW.total_price,
        ms.batch_number,
        ms.expiry_date,
        ds.patient_id,
        ds.dispensed_by
    FROM pharmacy.medicine_stock ms
    JOIN pharmacy.dispensing_sessions ds
        ON ds.session_id = NEW.session_id
    WHERE ms.stock_id = NEW.stock_id;

    RETURN NEW;
END;
$$ LANGUAGE plpgsql;

CREATE TRIGGER trg_deduct_stock
    AFTER INSERT ON pharmacy.dispensing_items
    FOR EACH ROW
    EXECUTE FUNCTION pharmacy.deduct_stock_on_dispensing();

        WHEN SUM(ist.quantity_available) <= 0
            THEN 'OUT_OF_STOCK'

            THEN 'CRITICAL'

            THEN 'REORDER_NOW'

    ON i.item_id = ist.item_id

    ON w.warehouse_id = ist.warehouse_id

    w.warehouse_id, w.warehouse_name,
    i.item_id, i.item_code, i.item_name,
    i.item_type, i.reorder_point,
    i.reorder_quantity, i.min_stock
HAVING SUM(ist.quantity_available) <= i.reorder_point
ORDER BY alert_status, i.item_name;

-- ============================================
-- TRIGGER: Calculate payroll totals
-- ============================================
CREATE OR REPLACE FUNCTION hr.calc_payroll_totals()
RETURNS TRIGGER AS $$
BEGIN
    NEW.gross_salary :=
        COALESCE(NEW.basic_salary, 0) +
        COALESCE(NEW.house_rent, 0) +
        COALESCE(NEW.medical_allowance, 0) +
        COALESCE(NEW.transport_allowance, 0) +
        COALESCE(NEW.food_allowance, 0) +
        COALESCE(NEW.mobile_allowance, 0) +
        COALESCE(NEW.overtime_pay, 0) +
        COALESCE(NEW.night_allowance, 0) +
        COALESCE(NEW.festival_bonus, 0) +
        COALESCE(NEW.performance_bonus, 0) +
        COALESCE(NEW.arrear_amount, 0) +
        COALESCE(NEW.other_earnings, 0);

    NEW.total_deductions :=
        COALESCE(NEW.income_tax, 0) +
        COALESCE(NEW.provident_fund, 0) +
        COALESCE(NEW.absent_deduction, 0) +
        COALESCE(NEW.loan_deduction, 0) +
        COALESCE(NEW.advance_deduction, 0) +
        COALESCE(NEW.other_deductions, 0);

    NEW.net_salary :=
        NEW.gross_salary - NEW.total_deductions;

    RETURN NEW;
END;
$$ LANGUAGE plpgsql;

CREATE TRIGGER trg_payroll_totals
    BEFORE INSERT OR UPDATE
    ON hr.payroll_details
    FOR EACH ROW
    EXECUTE FUNCTION hr.calc_payroll_totals();

-- ============================================
-- MATERIALIZED VIEW: Hospital KPI Master
-- ============================================
CREATE MATERIALIZED VIEW compliance.hospital_kpi_master AS
SELECT
    NOW()                               AS refreshed_at,
    -- Patient
    (
        SELECT COUNT(*)
        FROM clinical.patients
        WHERE DATE(registered_at) = CURRENT_DATE
    )                                   AS new_registrations_today,
    -- Beds
    (
        SELECT COUNT(*)
        FROM clinical.beds
        WHERE status = 'occupied'
          AND is_active = TRUE
    )                                   AS occupied_beds,
    (
        SELECT COUNT(*)
        FROM clinical.beds
        WHERE is_active = TRUE
    )                                   AS total_beds,
    -- Lab
    (
        SELECT COUNT(*)
        FROM clinical.lab_orders
        WHERE DATE(order_datetime) = CURRENT_DATE
    )                                   AS lab_orders_today,
    -- Critical Alerts
    (
        SELECT COUNT(*)
        FROM clinical.test_results
        WHERE is_critical = TRUE
          AND critical_notified = FALSE
    )                                   AS unnotified_critical_results,
    -- Maintenance
    (
        SELECT COUNT(*)
        FROM facility.maintenance_tickets
        WHERE status IN ('open','assigned')
          AND priority IN ('critical','high')
    )                                   AS critical_maintenance_tickets,
    -- Complaints
    (
        SELECT COUNT(*)
        FROM compliance.complaints
        WHERE status NOT IN ('closed','withdrawn')
          AND is_sla_breached = TRUE
    )                                   AS sla_breached_complaints;

-- Refresh every 5 minutes
COMMENT ON MATERIALIZED VIEW compliance.hospital_kpi_master IS
    'Refresh with: REFRESH MATERIALIZED VIEW CONCURRENTLY compliance.hospital_kpi_master';

-- ============================================
-- TRIGGER: Generic Audit Log Function
-- ============================================
CREATE OR REPLACE FUNCTION compliance.log_changes()
RETURNS TRIGGER AS $$
BEGIN
    INSERT INTO compliance.audit_logs (
        tenant_id, user_id, action,
        schema_name, table_name,
        record_id, old_data, new_data
    )
    VALUES (
        current_setting('app.current_tenant_id',TRUE)::UUID,
        current_setting('app.current_user_id', TRUE)::INT,
        TG_OP,
        TG_TABLE_SCHEMA,
        TG_TABLE_NAME,
        COALESCE(
            (row_to_json(NEW) ->> 'id'),
            (row_to_json(OLD) ->> 'id')
        ),
        CASE WHEN TG_OP != 'INSERT'
            THEN row_to_json(OLD)
            ELSE NULL
        END,
        CASE WHEN TG_OP != 'DELETE'
            THEN row_to_json(NEW)
            ELSE NULL
        END
    );
    RETURN COALESCE(NEW, OLD);
END;
$$ LANGUAGE plpgsql SECURITY DEFINER;
