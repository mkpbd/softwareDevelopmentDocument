

```sql
-- =====================================================
-- ENTERPRISE HOSPITAL ERP SYSTEM - POSTGRESQL SCHEMA
-- =====================================================

-- Enable UUID extension
CREATE EXTENSION IF NOT EXISTS "uuid-ossp";

-- =====================================================
-- ENUM TYPES (Statuses and Categories)
-- =====================================================

-- Patient related
CREATE TYPE patient_status AS ENUM ('active', 'inactive', 'deceased', 'merged');
CREATE TYPE blood_group AS ENUM ('A+', 'A-', 'B+', 'B-', 'AB+', 'AB-', 'O+', 'O-', 'unknown');
CREATE TYPE payer_category AS ENUM ('self_pay', 'insurance', 'corporate', 'government', 'subscription', 'charity');

-- Encounter types
CREATE TYPE encounter_type AS ENUM ('opd', 'ipd', 'er', 'telemedicine');
CREATE TYPE encounter_status AS ENUM ('registered', 'checked_in', 'active', 'pending_orders', 'completed', 'closed', 'cancelled');

-- Appointment related
CREATE TYPE appointment_status AS ENUM ('requested', 'confirmed', 'arrived', 'waiting', 'in_consultation', 'completed', 'cancelled', 'no_show', 'rescheduled');

-- Bed related
CREATE TYPE bed_status AS ENUM ('available', 'reserved', 'occupied', 'vacated_cleaning', 'cleaning', 'maintenance', 'blocked');
CREATE TYPE bed_class AS ENUM ('general', 'semi_private', 'private', 'icu', 'hdu', 'isolation');

-- Clinical order types
CREATE TYPE order_type AS ENUM ('lab', 'radiology', 'procedure', 'blood_bank', 'diet', 'nursing');
CREATE TYPE order_status AS ENUM ('ordered', 'scheduled', 'in_progress', 'completed', 'cancelled', 'verified', 'released');

-- Prescription related
CREATE TYPE prescription_status AS ENUM ('draft', 'submitted', 'pharmacist_review', 'approved', 'partially_dispensed', 'fully_dispensed', 'administered', 'completed', 'cancelled', 'rejected');

-- Lab specific
CREATE TYPE lab_sample_status AS ENUM ('pending', 'collected', 'received', 'processing', 'rejected', 'completed');
CREATE TYPE critical_value_status AS ENUM ('normal', 'critical', 'alert_sent', 'acknowledged');

-- Blood bank
CREATE TYPE blood_component AS ENUM ('whole_blood', 'prbc', 'ffp', 'platelets', 'cryo');
CREATE TYPE crossmatch_status AS ENUM ('pending', 'compatible', 'incompatible', 'not_performed');

-- Admission/Discharge
CREATE TYPE admission_status AS ENUM ('requested', 'financial_review', 'bed_reserved', 'admitted', 'under_treatment', 'discharge_planned', 'medical_clearance', 'financial_clearance', 'discharged', 'closed', 'cancelled', 'transferred', 'absconded', 'expired');

-- Financial
CREATE TYPE bill_status AS ENUM ('draft', 'provisional', 'finalized', 'partially_paid', 'paid', 'disputed', 'written_off');
CREATE TYPE payment_method AS ENUM ('cash', 'card', 'bkash', 'nagad', 'bank_transfer', 'insurance', 'corporate_credit', 'deposit_adjustment');
CREATE TYPE claim_status AS ENUM ('prepared', 'submitted', 'under_review', 'query_raised', 'resubmitted', 'approved', 'partially_approved', 'rejected', 'payment_received', 'closed');

-- Procurement
CREATE TYPE requisition_status AS ENUM ('draft', 'submitted', 'under_review', 'approved', 'rejected', 'rfq_issued', 'vendor_selected', 'po_issued', 'partially_received', 'fully_received', 'closed');
CREATE TYPE po_status AS ENUM ('created', 'approved', 'sent_to_vendor', 'partially_delivered', 'delivered', 'invoice_matched', 'closed', 'cancelled');

-- Inventory
CREATE TYPE stock_movement_type AS ENUM ('receipt', 'issue', 'transfer_in', 'transfer_out', 'return', 'adjustment', 'expiry_writeoff', 'disposal');
CREATE TYPE item_category AS ENUM ('medicine', 'consumable', 'surgical', 'lab_reagent', 'equipment', 'linen', 'stationery', 'other');

-- HR
CREATE TYPE employment_status AS ENUM ('candidate', 'hired', 'onboarding', 'active', 'on_leave', 'suspended', 'transferred', 'resigned', 'terminated', 'archived');
CREATE TYPE leave_type AS ENUM ('annual', 'sick', 'casual', 'maternity', 'paternity', 'bereavement', 'unpaid');

-- Asset/Maintenance
CREATE TYPE asset_status AS ENUM ('procured', 'received', 'installed', 'active', 'under_maintenance', 'retired', 'disposed');
CREATE TYPE maintenance_type AS ENUM ('preventive', 'corrective', 'emergency');

-- Helpdesk
CREATE TYPE ticket_status AS ENUM ('logged', 'assigned', 'in_progress', 'resolved', 'confirmed', 'closed', 'reopened', 'escalated');
CREATE TYPE ticket_priority AS ENUM ('low', 'medium', 'high', 'critical');

-- =====================================================
-- CORE MASTER TABLES
-- =====================================================

-- Branches / Facilities (for multi-branch operations)
CREATE TABLE branches (
    id UUID PRIMARY KEY DEFAULT uuid_generate_v4(),
    code VARCHAR(50) UNIQUE NOT NULL,
    name VARCHAR(255) NOT NULL,
    address TEXT,
    city VARCHAR(100),
    state VARCHAR(100),
    postal_code VARCHAR(20),
    country VARCHAR(100),
    phone VARCHAR(50),
    email VARCHAR(255),
    is_headquarters BOOLEAN DEFAULT FALSE,
    is_active BOOLEAN DEFAULT TRUE,
    created_at TIMESTAMPTZ DEFAULT NOW(),
    updated_at TIMESTAMPTZ DEFAULT NOW()
);

-- Departments
CREATE TABLE departments (
    id UUID PRIMARY KEY DEFAULT uuid_generate_v4(),
    branch_id UUID REFERENCES branches(id),
    code VARCHAR(50) UNIQUE NOT NULL,
    name VARCHAR(255) NOT NULL,
    parent_department_id UUID REFERENCES departments(id),
    cost_center_code VARCHAR(50),
    head_user_id UUID, -- References users table (defined later)
    is_active BOOLEAN DEFAULT TRUE,
    created_at TIMESTAMPTZ DEFAULT NOW(),
    updated_at TIMESTAMPTZ DEFAULT NOW()
);

-- Doctors / Providers
CREATE TABLE doctors (
    id UUID PRIMARY KEY DEFAULT uuid_generate_v4(),
    employee_id UUID, -- References employees table
    registration_number VARCHAR(100) UNIQUE NOT NULL,
    registration_body VARCHAR(100), -- e.g., BMDC
    registration_expiry DATE,
    specialty VARCHAR(255),
    sub_specialty VARCHAR(255),
    qualification TEXT,
    consultation_fee DECIMAL(10,2),
    follow_up_fee DECIMAL(10,2),
    is_consultant BOOLEAN DEFAULT FALSE,
    max_opd_per_day INT DEFAULT 30,
    telemedicine_enabled BOOLEAN DEFAULT FALSE,
    is_active BOOLEAN DEFAULT TRUE,
    created_at TIMESTAMPTZ DEFAULT NOW(),
    updated_at TIMESTAMPTZ DEFAULT NOW()
);

-- Doctor schedules
CREATE TABLE doctor_schedules (
    id UUID PRIMARY KEY DEFAULT uuid_generate_v4(),
    doctor_id UUID REFERENCES doctors(id),
    branch_id UUID REFERENCES branches(id),
    day_of_week INT CHECK (day_of_week BETWEEN 0 AND 6), -- 0=Sunday, 6=Saturday
    start_time TIME NOT NULL,
    end_time TIME NOT NULL,
    slot_duration_minutes INT DEFAULT 15,
    is_telemedicine_slot BOOLEAN DEFAULT FALSE,
    max_appointments INT DEFAULT 20,
    is_active BOOLEAN DEFAULT TRUE,
    valid_from DATE,
    valid_to DATE,
    created_at TIMESTAMPTZ DEFAULT NOW()
);

-- =====================================================
-- PATIENT MANAGEMENT
-- =====================================================

-- Patients table (master patient index)
CREATE TABLE patients (
    id UUID PRIMARY KEY DEFAULT uuid_generate_v4(),
    mrn VARCHAR(50) UNIQUE NOT NULL, -- Medical Record Number
    temporary_id BOOLEAN DEFAULT FALSE,
    first_name VARCHAR(100) NOT NULL,
    last_name VARCHAR(100),
    date_of_birth DATE,
    age_years INT,
    age_months INT,
    gender VARCHAR(20),
    blood_group blood_group DEFAULT 'unknown',
    mobile VARCHAR(20) NOT NULL,
    alternate_mobile VARCHAR(20),
    email VARCHAR(255),
    national_id VARCHAR(50),
    passport_number VARCHAR(50),
    address_line1 TEXT,
    address_line2 TEXT,
    city VARCHAR(100),
    state VARCHAR(100),
    postal_code VARCHAR(20),
    country VARCHAR(100),
    occupation VARCHAR(100),
    marital_status VARCHAR(20),
    religion VARCHAR(50),
    emergency_contact_name VARCHAR(255),
    emergency_contact_relation VARCHAR(100),
    emergency_contact_phone VARCHAR(20),
    primary_payer payer_category DEFAULT 'self_pay',
    insurance_id UUID, -- References insurance_policies
    corporate_id UUID, -- References corporate_clients
    subscription_id UUID, -- References subscriptions
    is_vip BOOLEAN DEFAULT FALSE,
    is_confidential BOOLEAN DEFAULT FALSE,
    is_deceased BOOLEAN DEFAULT FALSE,
    deceased_date DATE,
    status patient_status DEFAULT 'active',
    merged_into_patient_id UUID REFERENCES patients(id),
    created_by UUID,
    created_at TIMESTAMPTZ DEFAULT NOW(),
    updated_at TIMESTAMPTZ DEFAULT NOW()
);

-- Patient alerts (allergies, risks, etc.)
CREATE TABLE patient_alerts (
    id UUID PRIMARY KEY DEFAULT uuid_generate_v4(),
    patient_id UUID REFERENCES patients(id),
    alert_type VARCHAR(50), -- allergy, infection_risk, fall_risk, violence_risk, medico_legal
    alert_value TEXT,
    severity VARCHAR(20), -- mild, moderate, severe, critical
    clinical_notes TEXT,
    created_by UUID,
    created_at TIMESTAMPTZ DEFAULT NOW(),
    updated_at TIMESTAMPTZ DEFAULT NOW()
);

-- Patient allergies
CREATE TABLE patient_allergies (
    id UUID PRIMARY KEY DEFAULT uuid_generate_v4(),
    patient_id UUID REFERENCES patients(id),
    allergen_type VARCHAR(50), -- drug, food, latex, contrast, other
    allergen_name VARCHAR(255) NOT NULL,
    reaction TEXT,
    severity VARCHAR(20),
    recorded_date DATE,
    recorded_by UUID,
    is_active BOOLEAN DEFAULT TRUE,
    created_at TIMESTAMPTZ DEFAULT NOW()
);

-- =====================================================
-- ENCOUNTERS (OPD, IPD, ER)
-- =====================================================

-- Encounters table (central to all patient interactions)
CREATE TABLE encounters (
    id UUID PRIMARY KEY DEFAULT uuid_generate_v4(),
    patient_id UUID REFERENCES patients(id),
    branch_id UUID REFERENCES branches(id),
    encounter_number VARCHAR(50) UNIQUE NOT NULL,
    encounter_type encounter_type NOT NULL,
    status encounter_status DEFAULT 'registered',
    visit_date DATE NOT NULL,
    check_in_time TIMESTAMPTZ,
    check_out_time TIMESTAMPTZ,
    appointment_id UUID, -- References appointments
    referring_doctor_id UUID REFERENCES doctors(id),
    admitting_doctor_id UUID REFERENCES doctors(id),
    discharge_summary TEXT,
    created_by UUID,
    created_at TIMESTAMPTZ DEFAULT NOW(),
    updated_at TIMESTAMPTZ DEFAULT NOW()
);

-- =====================================================
-- APPOINTMENT & QUEUE MANAGEMENT
-- =====================================================

CREATE TABLE appointments (
    id UUID PRIMARY KEY DEFAULT uuid_generate_v4(),
    patient_id UUID REFERENCES patients(id),
    doctor_id UUID REFERENCES doctors(id),
    branch_id UUID REFERENCES branches(id),
    encounter_id UUID REFERENCES encounters(id),
    appointment_number VARCHAR(50) UNIQUE NOT NULL,
    appointment_date DATE NOT NULL,
    start_time TIME NOT NULL,
    end_time TIME NOT NULL,
    visit_type VARCHAR(50), -- new, follow_up, telemedicine
    status appointment_status DEFAULT 'requested',
    queue_token INT,
    cancellation_reason TEXT,
    no_show_reason TEXT,
    rescheduled_from_id UUID REFERENCES appointments(id),
    created_by UUID,
    created_at TIMESTAMPTZ DEFAULT NOW(),
    updated_at TIMESTAMPTZ DEFAULT NOW()
);

-- Queue management
CREATE TABLE queue_tokens (
    id UUID PRIMARY KEY DEFAULT uuid_generate_v4(),
    appointment_id UUID REFERENCES appointments(id),
    encounter_id UUID REFERENCES encounters(id),
    doctor_id UUID REFERENCES doctors(id),
    token_number INT NOT NULL,
    priority_level INT DEFAULT 5, -- 1=highest (emergency), 5=normal
    status VARCHAR(20), -- waiting, called, in_consultation, completed, skipped
    called_at TIMESTAMPTZ,
    consultation_started_at TIMESTAMPTZ,
    consultation_ended_at TIMESTAMPTZ,
    created_at TIMESTAMPTZ DEFAULT NOW()
);

-- =====================================================
-- BED MANAGEMENT
-- =====================================================

CREATE TABLE beds (
    id UUID PRIMARY KEY DEFAULT uuid_generate_v4(),
    branch_id UUID REFERENCES branches(id),
    ward_name VARCHAR(100),
    room_number VARCHAR(50),
    bed_number VARCHAR(50) NOT NULL,
    bed_class bed_class NOT NULL,
    status bed_status DEFAULT 'available',
    is_isolated BOOLEAN DEFAULT FALSE,
    isolation_type VARCHAR(50),
    gender_restriction VARCHAR(10), -- male, female, none
    has_ventilator BOOLEAN DEFAULT FALSE,
    has_oxygen BOOLEAN DEFAULT FALSE,
    has_monitor BOOLEAN DEFAULT FALSE,
    daily_rate DECIMAL(10,2),
    created_at TIMESTAMPTZ DEFAULT NOW(),
    updated_at TIMESTAMPTZ DEFAULT NOW(),
    UNIQUE(branch_id, ward_name, room_number, bed_number)
);

-- Bed assignments (admissions)
CREATE TABLE bed_assignments (
    id UUID PRIMARY KEY DEFAULT uuid_generate_v4(),
    admission_id UUID, -- References admissions
    bed_id UUID REFERENCES beds(id),
    patient_id UUID REFERENCES patients(id),
    assigned_at TIMESTAMPTZ NOT NULL,
    released_at TIMESTAMPTZ,
    is_current BOOLEAN DEFAULT TRUE,
    transfer_reason TEXT,
    created_at TIMESTAMPTZ DEFAULT NOW()
);

-- =====================================================
-- IPD ADMISSIONS
-- =====================================================

CREATE TABLE admissions (
    id UUID PRIMARY KEY DEFAULT uuid_generate_v4(),
    patient_id UUID REFERENCES patients(id),
    encounter_id UUID REFERENCES encounters(id),
    admission_number VARCHAR(50) UNIQUE NOT NULL,
    status admission_status DEFAULT 'requested',
    admission_date DATE NOT NULL,
    admission_time TIMESTAMPTZ NOT NULL,
    discharge_date DATE,
    discharge_time TIMESTAMPTZ,
    admitting_doctor_id UUID REFERENCES doctors(id),
    attending_doctor_id UUID REFERENCES doctors(id),
    provisional_diagnosis TEXT,
    final_diagnosis TEXT,
    bed_class_requested bed_class,
    advance_deposit DECIMAL(10,2) DEFAULT 0,
    is_emergency_admission BOOLEAN DEFAULT FALSE,
    discharge_summary_id UUID, -- References documents
    discharge_type VARCHAR(50), -- planned, lama, expired, transferred
    created_by UUID,
    created_at TIMESTAMPTZ DEFAULT NOW(),
    updated_at TIMESTAMPTZ DEFAULT NOW()
);

-- Patient transfers (between beds/wards)
CREATE TABLE patient_transfers (
    id UUID PRIMARY KEY DEFAULT uuid_generate_v4(),
    admission_id UUID REFERENCES admissions(id),
    from_bed_id UUID REFERENCES beds(id),
    to_bed_id UUID REFERENCES beds(id),
    transfer_reason TEXT,
    transferred_by UUID,
    transferred_at TIMESTAMPTZ NOT NULL,
    created_at TIMESTAMPTZ DEFAULT NOW()
);

-- =====================================================
-- CLINICAL: EMR, PRESCRIPTIONS, ORDERS
-- =====================================================

-- Clinical notes (EMR)
CREATE TABLE clinical_notes (
    id UUID PRIMARY KEY DEFAULT uuid_generate_v4(),
    encounter_id UUID REFERENCES encounters(id),
    patient_id UUID REFERENCES patients(id),
    doctor_id UUID REFERENCES doctors(id),
    note_type VARCHAR(50), -- admission_note, progress_note, consultation_note, discharge_note
    chief_complaint TEXT,
    history_of_present_illness TEXT,
    past_medical_history TEXT,
    examination_findings TEXT,
    diagnosis TEXT,
    treatment_plan TEXT,
    follow_up_instructions TEXT,
    is_finalized BOOLEAN DEFAULT FALSE,
    finalized_at TIMESTAMPTZ,
    created_at TIMESTAMPTZ DEFAULT NOW(),
    updated_at TIMESTAMPTZ DEFAULT NOW()
);

-- Diagnoses (ICD-10/11 coded)
CREATE TABLE diagnoses (
    id UUID PRIMARY KEY DEFAULT uuid_generate_v4(),
    encounter_id UUID REFERENCES encounters(id),
    patient_id UUID REFERENCES patients(id),
    icd_code VARCHAR(20) NOT NULL,
    diagnosis_description TEXT,
    diagnosis_type VARCHAR(30), -- primary, secondary, provisional, final
    diagnosed_by UUID REFERENCES doctors(id),
    diagnosed_at TIMESTAMPTZ DEFAULT NOW(),
    is_active BOOLEAN DEFAULT TRUE,
    created_at TIMESTAMPTZ DEFAULT NOW()
);

-- Prescriptions
CREATE TABLE prescriptions (
    id UUID PRIMARY KEY DEFAULT uuid_generate_v4(),
    encounter_id UUID REFERENCES encounters(id),
    patient_id UUID REFERENCES patients(id),
    doctor_id UUID REFERENCES doctors(id),
    prescription_number VARCHAR(50) UNIQUE NOT NULL,
    status prescription_status DEFAULT 'draft',
    notes TEXT,
    is_controlled_substance BOOLEAN DEFAULT FALSE,
    requires_second_approval BOOLEAN DEFAULT FALSE,
    approved_by_doctor_id UUID REFERENCES doctors(id),
    approved_at TIMESTAMPTZ,
    created_at TIMESTAMPTZ DEFAULT NOW(),
    updated_at TIMESTAMPTZ DEFAULT NOW()
);

-- Prescription items (medicines)
CREATE TABLE prescription_items (
    id UUID PRIMARY KEY DEFAULT uuid_generate_v4(),
    prescription_id UUID REFERENCES prescriptions(id),
    medicine_id UUID, -- References inventory_items
    medicine_name VARCHAR(255) NOT NULL,
    strength VARCHAR(100),
    dosage_form VARCHAR(50),
    route VARCHAR(50),
    frequency VARCHAR(100),
    duration_days INT,
    quantity INT NOT NULL,
    quantity_dispensed INT DEFAULT 0,
    instructions TEXT,
    is_substituted BOOLEAN DEFAULT FALSE,
    substituted_from_medicine VARCHAR(255),
    created_at TIMESTAMPTZ DEFAULT NOW()
);

-- Medication Administration Record (MAR)
CREATE TABLE medication_administration (
    id UUID PRIMARY KEY DEFAULT uuid_generate_v4(),
    prescription_item_id UUID REFERENCES prescription_items(id),
    patient_id UUID REFERENCES patients(id),
    encounter_id UUID REFERENCES encounters(id),
    nurse_id UUID,
    scheduled_time TIMESTAMPTZ NOT NULL,
    administered_time TIMESTAMPTZ,
    administered_quantity INT,
    administration_route VARCHAR(50),
    status VARCHAR(30), -- administered, missed, refused, held, delayed
    refusal_reason TEXT,
    notes TEXT,
    witnessed_by UUID,
    created_at TIMESTAMPTZ DEFAULT NOW()
);

-- BLOOD BANK MODULE

    id UUID PRIMARY KEY DEFAULT uuid_generate_v4(),

    unit_number VARCHAR(50) UNIQUE NOT NULL,

    collection_date DATE,

    requesting_doctor_id UUID REFERENCES doctors(id),

    crossmatch_status crossmatch_status DEFAULT 'pending',

    transfusion_reaction BOOLEAN DEFAULT FALSE,
    reaction_notes TEXT,
    created_at TIMESTAMPTZ DEFAULT NOW()
);

-- =====================================================
-- PHARMACY & INVENTORY
-- =====================================================

-- Inventory items (medicines, consumables, etc.)
CREATE TABLE inventory_items (
    id UUID PRIMARY KEY DEFAULT uuid_generate_v4(),
    branch_id UUID REFERENCES branches(id),
    item_code VARCHAR(50) UNIQUE NOT NULL,
    item_name VARCHAR(255) NOT NULL,
    generic_name VARCHAR(255),
    category item_category NOT NULL,
    manufacturer VARCHAR(255),
    unit_of_measure VARCHAR(20),
    reorder_level INT DEFAULT 0,
    reorder_quantity INT DEFAULT 0,
    safety_stock INT DEFAULT 0,
    storage_condition VARCHAR(100),
    requires_batch_tracking BOOLEAN DEFAULT TRUE,
    requires_expiry_tracking BOOLEAN DEFAULT TRUE,
    is_controlled_substance BOOLEAN DEFAULT FALSE,
    is_active BOOLEAN DEFAULT TRUE,
    created_at TIMESTAMPTZ DEFAULT NOW(),
    updated_at TIMESTAMPTZ DEFAULT NOW()
);

-- Stock batches (with expiry tracking)
CREATE TABLE stock_batches (
    id UUID PRIMARY KEY DEFAULT uuid_generate_v4(),
    item_id UUID REFERENCES inventory_items(id),
    branch_id UUID REFERENCES branches(id),
    batch_number VARCHAR(100) NOT NULL,
    manufacturing_date DATE,
    expiry_date DATE NOT NULL,
    quantity_received INT NOT NULL,
    quantity_available INT NOT NULL,
    unit_cost DECIMAL(10,2),
    selling_price DECIMAL(10,2),
    is_quarantined BOOLEAN DEFAULT FALSE,
    quarantine_reason TEXT,
    created_at TIMESTAMPTZ DEFAULT NOW(),
    updated_at TIMESTAMPTZ DEFAULT NOW()
);

-- Stock movements (transactions)
CREATE TABLE stock_movements (
    id UUID PRIMARY KEY DEFAULT uuid_generate_v4(),
    batch_id UUID REFERENCES stock_batches(id),
    item_id UUID REFERENCES inventory_items(id),
    branch_id UUID REFERENCES branches(id),
    movement_type stock_movement_type NOT NULL,
    quantity INT NOT NULL,
    reference_document_type VARCHAR(50), -- prescription, purchase_order, transfer, adjustment
    reference_document_id UUID,
    unit_price DECIMAL(10,2),
    total_amount DECIMAL(10,2),
    performed_by UUID,
    notes TEXT,
    created_at TIMESTAMPTZ DEFAULT NOW()
);

-- =====================================================
-- BILLING & FINANCIAL MODULES

-- Charge master (service pricing)
CREATE TABLE charge_master (
    id UUID PRIMARY KEY DEFAULT uuid_generate_v4(),
    branch_id UUID REFERENCES branches(id),
    service_code VARCHAR(50) UNIQUE NOT NULL,

-- Daily OPD statistics materialized view
CREATE MATERIALIZED VIEW mv_daily_opd_stats AS
SELECT 
    DATE(e.visit_date) as visit_date,
    e.branch_id,
    COUNT(DISTINCT e.id) as total_encounters,
    COUNT(DISTINCT e.patient_id) as unique_patients,
    COUNT(DISTINCT CASE WHEN e.status = 'completed' THEN e.id END) as completed_visits,
    AVG(EXTRACT(EPOCH FROM (e.check_out_time - e.check_in_time))/60) as avg_consultation_minutes
FROM encounters e
WHERE e.encounter_type = 'opd'
GROUP BY DATE(e.visit_date), e.branch_id;

-- Refresh function (run daily)
CREATE OR REPLACE FUNCTION refresh_daily_stats()
RETURNS TRIGGER AS $$
BEGIN
    REFRESH MATERIALIZED VIEW CONCURRENTLY mv_daily_opd_stats;
    RETURN NULL;
END;
$$ LANGUAGE plpgsql;

RETURNS TRIGGER AS $$
BEGIN

-- Trigger to prevent dispensing expired medicines
CREATE OR REPLACE FUNCTION check_medicine_expiry_before_dispense()
RETURNS TRIGGER AS $$
BEGIN
    IF EXISTS (
        SELECT 1 FROM stock_batches sb
        JOIN prescription_items pi ON pi.medicine_id = sb.item_id
        WHERE sb.id = NEW.batch_id 
        AND sb.expiry_date < CURRENT_DATE
    ) THEN
        RAISE EXCEPTION 'Cannot dispense expired medicine';
    END IF;
    RETURN NEW;
END;
$$ LANGUAGE plpgsql;

CREATE TRIGGER prevent_expired_dispense
BEFORE INSERT ON stock_movements
FOR EACH ROW
WHEN (NEW.movement_type = 'issue')
EXECUTE FUNCTION check_medicine_expiry_before_dispense();

## Key Design Decisions

### 1. **Patient-Centric Design**

- Central `patients` table with unique MRN
    
- `encounters` table links all patient interactions (OPD/IPD/ER)
    
- Supports temporary IDs for emergency cases
    

### 2. **State Machines Implemented via Enums**

- All critical lifecycles use PostgreSQL ENUMs matching the business logic
    
- Examples: `appointment_status`, `bed_status`, `admission_status`, `claim_status`
    

### 3. **Audit Compliance**

- `audit_logs` table with JSONB for before/after data
    
- All critical tables have `created_at` and `updated_at` timestamps
    
- User tracking (`created_by`, `updated_by` references)
    

### 4. **Multi-Branch/Multi-Tenant Support**

- `branches` table for physical locations
    
- Most tables have `branch_id` for data isolation
    
- User roles can be scoped to specific branches
    

### 5. **Financial Integrity**

- 3-way match enforced via `purchase_orders`, `goods_receipt_notes`, and vendor invoices
    
- Payment reconciliation via `payments` and `refunds` tables
    
- Insurance claims with full tracking (`pre_authorizations`, `insurance_claims`)
    

### 6. **Clinical Safety**

- `patient_allergies` table linked to prescription checks
    
- `critical_value_alerts` for lab results
    
- Medication administration with `medication_administration` table (MAR)
    

### 7. **Inventory Management**

- Batch and expiry tracking via `stock_batches`
    
- Stock movements with complete traceability
    
- Reorder level alerts (can be implemented as scheduled jobs)
    

### 8. **Performance Optimizations**

- Strategic indexes on frequently queried columns
    
- Materialized view for daily statistics
    
- Partitioning can be added later for large tables like `audit_logs`
    

This schema supports the complete hospital ERP functionality described in both documents, including OPD/IPD workflows, emergency handling, insurance claims, procurement cycles, inventory management, HR/payroll, and multi-branch operations.

-- Daily patient census
CREATE VIEW vw_daily_census AS
SELECT

SELECT

    d.name as department_name,
    SUM(bi.total_price) as revenue,
    COUNT(DISTINCT b.id) as bill_count
FROM bills b
JOIN bill_items bi ON bi.bill_id = b.id
LEFT JOIN charge_master cm ON cm.id = bi.charge_id
LEFT JOIN departments d ON d.id = cm.branch_id -- simplified
WHERE b.status = 'paid'

SELECT 
    ic.claim_number,
    p.first_name || ' ' || p.last_name as patient_name,
    ic.claim_amount,
    ic.status,
    ic.submission_date,
    NOW() - ic.submission_date as days_pending,
    ic.rejection_reason
FROM insurance_claims ic
JOIN patients p ON p.id = ic.patient_id
WHERE ic.status NOT IN ('payment_received', 'closed')

CREATE VIEW vw_bed_occupancy AS
SELECT 
    b.branch_id,
    b.ward_name,
    b.bed_class,
    COUNT(*) as total_beds,
    COUNT(CASE WHEN b.status = 'occupied' THEN 1 END) as occupied_beds,
    ROUND(100.0 * COUNT(CASE WHEN b.status = 'occupied' THEN 1 END) / COUNT(*), 2) as occupancy_percentage
FROM beds b
GROUP BY b.branch_id, b.ward_name, b.bed_class;

-- Doctor productivity
CREATE VIEW vw_doctor_productivity AS
SELECT 
    d.id as doctor_id,
    d.registration_number,
    COUNT(DISTINCT e.id) as total_encounters,
    COUNT(DISTINCT p.id) as total_prescriptions,
    COUNT(DISTINCT lo.id) as lab_orders,
    COUNT(DISTINCT ro.id) as radiology_orders,
    SUM(bi.total_price) as revenue_generated

-- Function to automatically generate next MRN
CREATE OR REPLACE FUNCTION generate_mrn()
RETURNS TRIGGER AS $$
DECLARE
    next_number INT;
    year_prefix VARCHAR(4);
BEGIN
    year_prefix := TO_CHAR(NOW(), 'YYYY');
    SELECT COALESCE(MAX(CAST(SUBSTRING(mrn FROM 5) AS INT)), 0) + 1 INTO next_number
    FROM patients
    WHERE mrn LIKE year_prefix || '%';
    NEW.mrn := year_prefix || LPAD(next_number::TEXT, 6, '0');
    RETURN NEW;
END;
$$ LANGUAGE plpgsql;

CREATE TRIGGER trg_generate_mrn
BEFORE INSERT ON patients
FOR EACH ROW
WHEN (NEW.mrn IS NULL)
EXECUTE FUNCTION generate_mrn();

-- Function to check bed availability and assign
CREATE OR REPLACE FUNCTION assign_bed(
    p_patient_id UUID,
    p_bed_class bed_class,
    p_branch_id UUID,
    p_admission_id UUID
)
RETURNS UUID AS $$
DECLARE
    v_bed_id UUID;
BEGIN
    -- Find available bed
    SELECT id INTO v_bed_id
    FROM beds
    WHERE branch_id = p_branch_id
      AND bed_class = p_bed_class
      AND status = 'available'
    LIMIT 1
    FOR UPDATE;
    
    IF v_bed_id IS NULL THEN
        RAISE EXCEPTION 'No available bed of class % in branch %', p_bed_class, p_branch_id;
    END IF;
    
    -- Update bed status
    UPDATE beds SET status = 'occupied', updated_at = NOW() WHERE id = v_bed_id;
    
    -- Create bed assignment record
    INSERT INTO bed_assignments (admission_id, bed_id, patient_id, assigned_at, is_current)
    VALUES (p_admission_id, v_bed_id, p_patient_id, NOW(), TRUE);
    
    RETURN v_bed_id;
END;
$$ LANGUAGE plpgsql;

-- Function to process discharge and free bed
CREATE OR REPLACE FUNCTION process_discharge(
    p_admission_id UUID,
    p_discharge_type VARCHAR(50)
)
RETURNS VOID AS $$
DECLARE
    v_bed_id UUID;
    v_encounter_id UUID;
BEGIN
    -- Get current bed assignment
    SELECT bed_id, encounter_id INTO v_bed_id, v_encounter_id
    FROM bed_assignments ba
    JOIN admissions a ON a.id = ba.admission_id
    WHERE ba.admission_id = p_admission_id AND ba.is_current = TRUE;
    
    -- Update admission record
    UPDATE admissions 
    SET status = 'discharged', 
        discharge_date = CURRENT_DATE,
        discharge_time = NOW(),
        discharge_type = p_discharge_type
    WHERE id = p_admission_id;
    
    -- Update encounter
    UPDATE encounters SET status = 'completed', check_out_time = NOW() 
    WHERE id = v_encounter_id;
    
    -- Release bed and trigger cleaning
    UPDATE beds SET status = 'vacated_cleaning', updated_at = NOW() WHERE id = v_bed_id;
    
    -- Create housekeeping request
    INSERT INTO housekeeping_requests (bed_id, request_type, requested_at, status)
    VALUES (v_bed_id, 'discharge_cleaning', NOW(), 'pending');
    
    -- End current bed assignment
    UPDATE bed_assignments SET is_current = FALSE, released_at = NOW()
    WHERE admission_id = p_admission_id AND is_current = TRUE;
END;
$$ LANGUAGE plpgsql;

-- Function to auto-generate purchase requisition when stock low
CREATE OR REPLACE FUNCTION check_reorder_levels()
RETURNS TRIGGER AS $$
DECLARE
    v_reorder_level INT;
    v_reorder_qty INT;
    v_pr_number VARCHAR(50);
BEGIN
    -- Get reorder settings for this item
    SELECT reorder_level, reorder_quantity INTO v_reorder_level, v_reorder_qty
    FROM inventory_items WHERE id = NEW.item_id;
    
    -- If current stock falls below reorder level, create requisition
    IF NEW.quantity_available <= v_reorder_level THEN
        -- Generate requisition number
        v_pr_number := 'PR-' || TO_CHAR(NOW(), 'YYYYMMDD') || '-' || LPAD(CAST(FLOOR(RANDOM()*1000) AS TEXT), 3, '0');
        
        INSERT INTO purchase_requisitions (pr_number, department_id, requested_by, required_by_date, justification, status)
        VALUES (v_pr_number, (SELECT department_id FROM inventory_items WHERE id = NEW.item_id), 
                (SELECT created_by FROM stock_movements WHERE batch_id = NEW.id ORDER BY created_at DESC LIMIT 1),
                NOW() + INTERVAL '7 days', 'Auto-generated due to low stock', 'submitted');
        
        -- Link requisition item
        INSERT INTO requisition_items (requisition_id, item_id, quantity_requested)
        VALUES ((SELECT id FROM purchase_requisitions WHERE pr_number = v_pr_number), NEW.item_id, v_reorder_qty);
    END IF;
    
    RETURN NEW;
END;
$$ LANGUAGE plpgsql;

CREATE TRIGGER trg_check_reorder
AFTER UPDATE OF quantity_available ON stock_batches
FOR EACH ROW
EXECUTE FUNCTION check_reorder_levels();

-- Prevent duplicate active admission for same patient
CREATE OR REPLACE FUNCTION prevent_duplicate_active_admission()
RETURNS TRIGGER AS $$
BEGIN
    IF EXISTS (
        SELECT 1 FROM admissions 
        WHERE patient_id = NEW.patient_id 
          AND status IN ('admitted', 'under_treatment', 'discharge_planned')
          AND id != COALESCE(NEW.id, '00000000-0000-0000-0000-000000000000')
    ) THEN
        RAISE EXCEPTION 'Patient already has an active admission';
    END IF;
    RETURN NEW;
END;
$$ LANGUAGE plpgsql;

CREATE TRIGGER trg_prevent_duplicate_admission
BEFORE INSERT OR UPDATE OF status ON admissions
FOR EACH ROW
WHEN (NEW.status IN ('admitted', 'under_treatment', 'discharge_planned'))
EXECUTE FUNCTION prevent_duplicate_active_admission();

-- Prevent discharge without financial clearance
CREATE OR REPLACE FUNCTION check_discharge_financial_clearance()
RETURNS TRIGGER AS $$
DECLARE
    v_due_amount DECIMAL(10,2);
BEGIN
    IF NEW.status = 'discharged' AND OLD.status != 'discharged' THEN
        -- Get outstanding balance
        SELECT COALESCE(SUM(due_amount), 0) INTO v_due_amount
        FROM bills
        WHERE encounter_id = (SELECT encounter_id FROM admissions WHERE id = NEW.id)
          AND status NOT IN ('paid', 'written_off');
        
        IF v_due_amount > 0 THEN
            RAISE EXCEPTION 'Cannot discharge patient: outstanding balance of %', v_due_amount;
        END IF;
    END IF;
    RETURN NEW;
END;
$$ LANGUAGE plpgsql;

CREATE TRIGGER trg_check_discharge_clearance
BEFORE UPDATE OF status ON admissions
FOR EACH ROW
EXECUTE FUNCTION check_discharge_financial_clearance();

-- Auto-update stock when prescription is dispensed
CREATE OR REPLACE FUNCTION auto_deduct_stock_on_dispense()
RETURNS TRIGGER AS $$
DECLARE
    v_batch_id UUID;
    v_available_qty INT;
BEGIN
    -- Find oldest batch with sufficient quantity (FEFO)
    SELECT id, quantity_available INTO v_batch_id, v_available_qty
    FROM stock_batches
    WHERE item_id = (SELECT medicine_id FROM prescription_items WHERE id = NEW.prescription_item_id)
      AND quantity_available >= NEW.quantity
      AND expiry_date > CURRENT_DATE
    ORDER BY expiry_date ASC
    LIMIT 1;
    
    IF v_batch_id IS NULL THEN
        RAISE EXCEPTION 'Insufficient stock for medicine';
    END IF;
    
    -- Record stock movement
    INSERT INTO stock_movements (batch_id, item_id, branch_id, movement_type, quantity, reference_document_type, reference_document_id, performed_by)
    VALUES (v_batch_id, 
            (SELECT medicine_id FROM prescription_items WHERE id = NEW.prescription_item_id),
            (SELECT branch_id FROM encounters WHERE id = (SELECT encounter_id FROM prescriptions WHERE id = (SELECT prescription_id FROM prescription_items WHERE id = NEW.prescription_item_id))),
            'issue', NEW.quantity, 'prescription', NEW.prescription_item_id, NEW.nurse_id);
    
    -- Update batch quantity
    UPDATE stock_batches SET quantity_available = quantity_available - NEW.quantity
    WHERE id = v_batch_id;
    
    RETURN NEW;
END;
$$ LANGUAGE plpgsql;

-- Procedure to safely merge duplicate patient records
CREATE OR REPLACE PROCEDURE merge_patients(
    p_master_patient_id UUID,
    p_duplicate_patient_id UUID,
    p_merged_by UUID
)
LANGUAGE plpgsql AS $$
DECLARE
    v_master_mrn VARCHAR(50);

BEGIN

    SELECT mrn INTO v_master_mrn FROM patients WHERE id = p_master_patient_id;

    END IF;

    -- Reassign all encounters to master patient
    UPDATE encounters SET patient_id = p_master_patient_id 
    WHERE patient_id = p_duplicate_patient_id;
    GET DIAGNOSTICS v_record_count = ROW_COUNT;

    UPDATE prescriptions SET patient_id = p_master_patient_id 
    WHERE patient_id = p_duplicate_patient_id;

    WHERE patient_id = p_duplicate_patient_id;

    UPDATE radiology_orders SET patient_id = p_master_patient_id 
    WHERE patient_id = p_duplicate_patient_id;
    
    -- Reassign bills
    UPDATE bills SET patient_id = p_master_patient_id 
    WHERE patient_id = p_duplicate_patient_id;
    
    -- Reassign admissions
    UPDATE admissions SET patient_id = p_master_patient_id 
    WHERE patient_id = p_duplicate_patient_id;

    UPDATE bed_assignments SET patient_id = p_master_patient_id 
    WHERE patient_id = p_duplicate_patient_id;
    
    -- Reassign insurance policies
    UPDATE insurance_policies SET patient_id = p_master_patient_id 
    WHERE patient_id = p_duplicate_patient_id;
    
    -- Reassign helpdesk tickets
    UPDATE helpdesk_tickets SET patient_id = p_master_patient_id 
    WHERE patient_id = p_duplicate_patient_id;
    
    -- Merge alerts
    UPDATE patient_alerts SET patient_id = p_master_patient_id 
    WHERE patient_id = p_duplicate_patient_id;

    UPDATE patient_allergies SET patient_id = p_master_patient_id

        updated_at = NOW()
    WHERE id = p_duplicate_patient_id;

        'MERGE_PATIENT',
        'patients',

        jsonb_build_object('mrn', v_duplicate_mrn, 'merged_into', v_master_mrn)
    );
    
    RAISE NOTICE 'Successfully merged patient % into %', v_duplicate_mrn, v_master_mrn;
END;

-- Procedure to generate bulk invoices for corporate clients
CREATE OR REPLACE FUNCTION generate_corporate_invoice(
    p_corporate_id UUID,
    p_period_start DATE,
    p_period_end DATE
)
RETURNS TABLE(invoice_number TEXT, total_amount DECIMAL) AS $$
DECLARE
    v_invoice_number TEXT;
    v_total DECIMAL;
BEGIN
    -- Generate unique invoice number
    v_invoice_number := 'INV-' || TO_CHAR(NOW(), 'YYYYMMDD') || '-' || LPAD(CAST(FLOOR(RANDOM()*10000) AS TEXT), 4, '0');
    
    -- Calculate total for period
    SELECT COALESCE(SUM(b.total_amount - COALESCE(b.paid_amount, 0)), 0) INTO v_total
    FROM bills b
    JOIN patients p ON p.id = b.patient_id
    WHERE p.corporate_id = p_corporate_id
      AND b.bill_date BETWEEN p_period_start AND p_period_end
      AND b.status NOT IN ('paid', 'written_off');
    
    -- Insert into a corporate_invoices table (create if not exists)
    INSERT INTO corporate_invoices (invoice_number, corporate_id, period_start, period_end, total_amount, status, created_at)
    VALUES (v_invoice_number, p_corporate_id, p_period_start, p_period_end, v_total, 'generated', NOW())
    ON CONFLICT DO NOTHING;
    
    RETURN QUERY SELECT v_invoice_number, v_total;
END;
$$ LANGUAGE plpgsql;

-- Procedure to process refund with approvals
CREATE OR REPLACE FUNCTION process_refund(
    p_payment_id UUID,
    p_refund_amount DECIMAL,
    p_reason TEXT,
    p_requested_by UUID,
    p_approver_id UUID DEFAULT NULL
)
RETURNS UUID AS $$
DECLARE
    v_refund_id UUID;
    v_payment_amount DECIMAL;
    v_bill_id UUID;
BEGIN
    -- Get payment details
    SELECT amount, bill_id INTO v_payment_amount, v_bill_id
    FROM payments WHERE id = p_payment_id;
    
    IF v_payment_amount < p_refund_amount THEN
        RAISE EXCEPTION 'Refund amount exceeds payment amount';
    END IF;
    
    -- Create refund record
    INSERT INTO refunds (payment_id, bill_id, refund_number, refund_amount, refund_reason, status, requested_by, requested_at)
    VALUES (
        p_payment_id, 
        v_bill_id,
        'REF-' || TO_CHAR(NOW(), 'YYYYMMDD') || '-' || LPAD(CAST(FLOOR(RANDOM()*10000) AS TEXT), 4, '0'),
        p_refund_amount,
        p_reason,
        CASE WHEN p_approver_id IS NOT NULL THEN 'approved' ELSE 'pending' END,
        p_requested_by,
        NOW()
    )
    RETURNING id INTO v_refund_id;
    
    -- If approver provided, process immediately
    IF p_approver_id IS NOT NULL THEN
        UPDATE refunds 
        SET approved_by = p_approver_id, 
            approved_at = NOW(),
            status = 'approved'
        WHERE id = v_refund_id;
        
        -- Update payment record (reduce paid amount or mark refunded)
        UPDATE payments SET amount = amount - p_refund_amount, updated_at = NOW()
        WHERE id = p_payment_id;
        
        -- Update bill due amount
        UPDATE bills SET due_amount = due_amount - p_refund_amount, updated_at = NOW()
        WHERE id = v_bill_id;
    END IF;
    
    RETURN v_refund_id;
END;
$$ LANGUAGE plpgsql;
```

        EXIT WHEN NOT exists_flag;
    END LOOP;
    RETURN new_code;
END;

-- Calculate patient age from DOB
CREATE OR REPLACE FUNCTION calculate_age(p_dob DATE)
RETURNS TEXT AS $$
DECLARE
    age_years INT;
    age_months INT;
BEGIN
    IF p_dob IS NULL THEN
        RETURN NULL;
    END IF;
    age_years := EXTRACT(YEAR FROM age(CURRENT_DATE, p_dob));
    age_months := EXTRACT(MONTH FROM age(CURRENT_DATE, p_dob));
    IF age_years > 0 THEN
        RETURN age_years || ' years';
    ELSE
        RETURN age_months || ' months';
    END IF;
END;
$$ LANGUAGE plpgsql;

-- Get current financial year
CREATE OR REPLACE FUNCTION current_financial_year()
RETURNS TEXT AS $$
DECLARE
    current_year INT;
    financial_year TEXT;
BEGIN
    current_year := EXTRACT(YEAR FROM CURRENT_DATE);
    IF EXTRACT(MONTH FROM CURRENT_DATE) >= 7 THEN
        financial_year := current_year || '-' || (current_year + 1);
    ELSE
        financial_year := (current_year - 1) || '-' || current_year;
    END IF;
    RETURN financial_year;
END;
$$ LANGUAGE plpgsql;

CREATE OR REPLACE VIEW vw_monthly_financial_summary AS

-- Doctor-wise collection summary
CREATE OR REPLACE VIEW vw_doctor_collection_summary AS
SELECT 
    d.id as doctor_id,
    d.registration_number,
    DATE_TRUNC('month', e.visit_date) as month,
    COUNT(DISTINCT e.id) as consultations,
    SUM(CASE WHEN b.status = 'paid' THEN b.total_amount ELSE 0 END) as collected_revenue,
    SUM(CASE WHEN b.status = 'finalized' AND b.status != 'paid' THEN b.total_amount ELSE 0 END) as pending_revenue
FROM doctors d
JOIN encounters e ON e.doctor_id = d.id OR e.admitting_doctor_id = d.id
LEFT JOIN bills b ON b.encounter_id = e.id
WHERE e.visit_date >= CURRENT_DATE - INTERVAL '12 months'
GROUP BY d.id, d.registration_number, DATE_TRUNC('month', e.visit_date);

-- Patient revisit rate (loyalty metric)
CREATE OR REPLACE VIEW vw_patient_revisit_rate AS
SELECT 
    DATE_TRUNC('month', first_visit) as cohort_month,
    COUNT(DISTINCT patient_id) as cohort_size,
    COUNT(DISTINCT CASE WHEN revisit_within_30_days THEN patient_id END) as revisited_30d,
    COUNT(DISTINCT CASE WHEN revisit_within_90_days THEN patient_id END) as revisited_90d,
    ROUND(100.0 * COUNT(DISTINCT CASE WHEN revisit_within_30_days THEN patient_id END) / NULLIF(COUNT(DISTINCT patient_id), 0), 2) as retention_rate_30d
FROM (
    SELECT 
        patient_id,
        MIN(visit_date) as first_visit,
        EXISTS(SELECT 1 FROM encounters e2 WHERE e2.patient_id = e1.patient_id AND e2.visit_date BETWEEN e1.visit_date + 1 AND e1.visit_date + 30) as revisit_within_30_days,
        EXISTS(SELECT 1 FROM encounters e2 WHERE e2.patient_id = e1.patient_id AND e2.visit_date BETWEEN e1.visit_date + 1 AND e1.visit_date + 90) as revisit_within_90_days
    FROM encounters e1
    WHERE encounter_type = 'opd'
) cohort
GROUP BY DATE_TRUNC('month', first_visit);
