```sql
    name                VARCHAR(200) NOT NULL,
```

```sql
-- LENGTH OF STAY - Computed separately (not stored in 3NF)
CREATE VIEW clinical.v_admission_los AS
SELECT 
    id,
    admission_datetime,
    discharge_datetime,
    CASE 
        WHEN discharge_datetime IS NOT NULL 
        THEN EXTRACT(DAY FROM (discharge_datetime - admission_datetime))::INTEGER
        ELSE EXTRACT(DAY FROM (NOW() - admission_datetime))::INTEGER
    END AS length_of_stay_days,
    CASE 
        WHEN discharge_datetime IS NOT NULL 
        THEN EXTRACT(EPOCH FROM (discharge_datetime - admission_datetime))/3600
        ELSE EXTRACT(EPOCH FROM (NOW() - admission_datetime))/3600
    END AS length_of_stay_hours
FROM clinical.admissions;
```

```sql
-- Real-Time Bed Status View
CREATE VIEW facility.v_bed_status AS
SELECT
    b.id AS bed_id,
    b.bed_no,
    b.bed_type,
    b.status_code AS status,
    b.daily_rate,
    w.name AS ward_name,
    w.ward_type,
    r.room_no,
    br.name AS branch_name,
    p.full_name AS patient_name,
    p.mrn,
    a.admission_datetime,
    los.length_of_stay_days
FROM facility.beds b
JOIN facility.wards w ON b.ward_id = w.id
LEFT JOIN facility.rooms r ON b.room_id = r.id
JOIN platform.branches br ON w.branch_id = br.id
LEFT JOIN clinical.patients p ON b.current_patient_id = p.id
LEFT JOIN clinical.admissions a ON b.current_admission_id = a.id
LEFT JOIN clinical.v_admission_los los ON a.id = los.id;

CREATE TABLE facility.bed_transfers (
    id                  UUID PRIMARY KEY DEFAULT gen_random_uuid(),
    patient_id          UUID NOT NULL REFERENCES clinical.patients(id),
    admission_id        UUID NOT NULL REFERENCES clinical.admissions(id),
    from_ward_id        UUID REFERENCES facility.wards(id),
    from_bed_id         UUID REFERENCES facility.beds(id),
    to_ward_id          UUID REFERENCES facility.wards(id),
    to_bed_id           UUID REFERENCES facility.beds(id),
    transfer_reason     TEXT,
    transfer_type       VARCHAR(50),
    transferred_at      TIMESTAMPTZ DEFAULT NOW(),
    transferred_by      UUID REFERENCES platform.users(id),
    notes               TEXT
);

-- Bed transfer trigger
CREATE OR REPLACE FUNCTION facility.handle_bed_transfer()
RETURNS TRIGGER AS $$ BEGIN
    -- Free old bed
    UPDATE facility.beds
    SET status_code = 'cleaning',
        current_patient_id = NULL,
        current_admission_id = NULL
    WHERE id = NEW.from_bed_id;
    
    -- Occupy new bed
    UPDATE facility.beds
    SET status_code = 'occupied',
        current_patient_id = NEW.patient_id,
        current_admission_id = NEW.admission_id
    WHERE id = NEW.to_bed_id;
    
    -- Update admission record
    UPDATE clinical.admissions
    SET ward_id = NEW.to_ward_id,
        bed_id = NEW.to_bed_id
    WHERE id = NEW.admission_id;
    
    RETURN NEW;
END;
 $$ LANGUAGE plpgsql;

CREATE TRIGGER trg_bed_transfer
AFTER INSERT ON facility.bed_transfers
FOR EACH ROW EXECUTE FUNCTION facility.handle_bed_transfer();

-- Auto-create cleaning task on discharge
CREATE OR REPLACE FUNCTION facility.auto_cleaning_on_discharge()
RETURNS TRIGGER AS $$ DECLARE v_bed_id UUID;
BEGIN
    SELECT bed_id INTO v_bed_id FROM clinical.admissions WHERE id = NEW.admission_id;
    
    IF v_bed_id IS NOT NULL THEN
        INSERT INTO facility.bed_cleaning_tasks(bed_id, trigger_type)
        VALUES (v_bed_id, 'discharge');
        
        UPDATE facility.beds SET status_code = 'cleaning' WHERE id = v_bed_id;
    END IF;
    RETURN NEW;
END;
 $$ LANGUAGE plpgsql;
```

```sql
-- LEDGER VIEW
CREATE VIEW finance.v_account_ledger AS
SELECT 
    coa.account_code,
    coa.account_name,
    coa.account_type,
    jel.entry_date,
    je.entry_no,
    je.narration AS entry_narration,
    jel.narration AS line_narration,
    jel.debit_amount,
    jel.credit_amount,
    jel.debit_amount - jel.credit_amount AS net_amount,
    SUM(jel.debit_amount - jel.credit_amount) OVER (
        PARTITION BY coa.id 
        ORDER BY jel.created_at, jel.id
    ) AS running_balance
FROM finance.journal_entry_lines jel
JOIN finance.journal_entries je ON jel.entry_id = je.id
JOIN finance.chart_of_accounts coa ON jel.account_id = coa.id
WHERE je.status_code = 'posted'
ORDER BY coa.account_code, jel.created_at;

CREATE VIEW finance.v_patient_outstanding AS
SELECT 
    p.id AS patient_id,
    p.mrn,
    p.full_name,
    b.org_id,
    b.branch_id,
    SUM(b.balance_due) AS total_outstanding,
    COUNT(DISTINCT b.id) AS outstanding_bill_count
FROM clinical.patients p
JOIN finance.patient_bills b ON p.id = b.patient_id
WHERE b.balance_due > 0 AND b.status_code NOT IN ('cancelled', 'written_off')
GROUP BY p.id, p.mrn, p.full_name, b.org_id, b.branch_id;
```

```sql
-- Daily Revenue Summary
CREATE VIEW analytics.v_daily_revenue AS
SELECT 
    b.org_id,
    b.branch_id,
    b.bill_date,
    COUNT(DISTINCT b.id) AS bill_count,
    SUM(b.subtotal) AS gross_revenue,
    SUM(b.discount_amount) AS total_discounts,
    SUM(b.tax_amount) AS total_tax,
    SUM(b.total_amount) AS net_revenue,
    SUM(b.paid_amount) AS collected,
    SUM(b.balance_due) AS outstanding
FROM finance.patient_bills b
WHERE b.status_code NOT IN ('cancelled')
GROUP BY b.org_id, b.branch_id, b.bill_date;

-- Monthly Admission Statistics
CREATE VIEW analytics.v_monthly_admissions AS
SELECT 
    a.org_id,
    a.branch_id,
    TO_CHAR(a.admission_datetime, 'YYYY-MM') AS month,
    COUNT(*) AS total_admissions,
    COUNT(*) FILTER (WHERE a.admission_type = 'elective') AS elective_admissions,
    COUNT(*) FILTER (WHERE a.admission_type = 'emergency') AS emergency_admissions,
    AVG(los.length_of_stay_days) AS avg_los,
    COUNT(DISTINCT a.admitting_doctor_id) AS unique_doctors
FROM clinical.admissions a
LEFT JOIN clinical.v_admission_los los ON a.id = los.id
GROUP BY a.org_id, a.branch_id, TO_CHAR(a.admission_datetime, 'YYYY-MM');

-- Bed Occupancy Trend
CREATE VIEW analytics.v_bed_occupancy_trend AS
SELECT 
    w.branch_id,
    w.id AS ward_id,
    w.name AS ward_name,
    w.ward_type,
    COUNT(b.id) AS total_beds,
    COUNT(b.id) FILTER (WHERE b.status_code = 'occupied') AS occupied_beds,
    ROUND(
        COUNT(b.id) FILTER (WHERE b.status_code = 'occupied') * 100.0 / 
        NULLIF(COUNT(b.id), 0), 2
    ) AS occupancy_pct,
    COUNT(b.id) FILTER (WHERE b.status_code = 'available') AS available_beds
FROM facility.wards w
JOIN facility.beds b ON w.id = b.ward_id
WHERE w.is_active = TRUE AND b.is_active = TRUE
GROUP BY w.branch_id, w.id, w.name, w.ward_type;

-- Department Revenue Ranking
CREATE VIEW analytics.v_department_revenue_ranking AS
SELECT 
    bi.item_type AS department,
    SUM(bi.line_total) AS total_revenue,
    COUNT(DISTINCT bi.bill_id) AS bill_count,
    AVG(bi.line_total) AS avg_line_value
FROM finance.bill_items bi
JOIN finance.patient_bills b ON bi.bill_id = b.id
WHERE b.status_code NOT IN ('cancelled', 'draft')
GROUP BY bi.item_type
ORDER BY total_revenue DESC;

-- Doctor Performance
CREATE VIEW analytics.v_doctor_performance AS
SELECT 
    d.id AS doctor_id,
    e.full_name AS doctor_name,
    d.specialization,
    COUNT(DISTINCT a.id) AS total_admissions,
    COUNT(DISTINCT v.id) AS total_opd_visits,
    COUNT(DISTINCT s.id) AS total_surgeries,
    AVG(los.length_of_stay_days) AS avg_patient_los,
    SUM(cf.fee_amount) AS total_consultation_fees
FROM hr.doctors d
JOIN hr.employees e ON d.employee_id = e.id
LEFT JOIN clinical.admissions a ON d.id = a.admitting_doctor_id
LEFT JOIN clinical.patient_visits v ON v.id = (
    SELECT pv.id FROM clinical.patient_visits pv 
    WHERE pv.chief_complaint LIKE '%' || e.full_name || '%'
    LIMIT 1
)
LEFT JOIN clinical.surgeries s ON s.id = (
    SELECT st.surgery_id FROM clinical.surgery_team st 
    WHERE st.staff_id = d.id AND st.role = 'primary_surgeon'
    LIMIT 1
)
LEFT JOIN clinical.v_admission_los los ON a.id = los.id
LEFT JOIN hr.doctor_consultation_fees cf ON d.id = cf.doctor_id
GROUP BY d.id, e.full_name, d.specialization;

-- Lab Performance
CREATE VIEW analytics.v_lab_performance AS
SELECT 
    lo.branch_id,
    tm.category_id,
    DATE(lo.ordered_at) AS order_date,
    COUNT(DISTINCT lo.id) AS total_orders,
    COUNT(DISTINCT loi.id) AS total_tests,
    COUNT(DISTINCT lr.id) AS total_results,
    COUNT(DISTINCT lr.id) FILTER (WHERE lr.verified_at IS NOT NULL) AS verified_results,
    AVG(EXTRACT(EPOCH FROM (lr.verified_at - lo.ordered_at))/3600) AS avg_turnaround_hours,
    COUNT(DISTINCT lr.id) FILTER (WHERE lr.is_critical = TRUE) AS critical_results
FROM clinical.lab_orders lo
JOIN clinical.lab_order_items loi ON lo.id = loi.order_id
JOIN clinical.test_master tm ON loi.test_id = tm.id
LEFT JOIN clinical.lab_results lr ON loi.id = lr.order_item_id
GROUP BY lo.branch_id, tm.category_id, DATE(lo.ordered_at);
```

```sql
-- AUDIT TRIGGER FUNCTION (Generic)
--// ============================================================

CREATE OR REPLACE FUNCTION audit.log_table_changes()
RETURNS TRIGGER AS $$ DECLARE
    v_old_data JSONB;
    v_new_data JSONB;
    v_changed_cols TEXT[] := '{}';
    v_action VARCHAR(20);
BEGIN
    v_action := TG_OP;
    
    IF TG_OP = 'DELETE' THEN
        v_old_data := to_jsonb(OLD);
    ELSIF TG_OP = 'UPDATE' THEN
        v_old_data := to_jsonb(OLD);
        v_new_data := to_jsonb(NEW);
        
        -- Detect changed columns
        SELECT array_agg(key) INTO v_changed_cols
        FROM (
            SELECT key 
            FROM jsonb_each_text(v_old_data) o
            FULL OUTER JOIN jsonb_each_text(v_new_data) n USING (key)
            WHERE o.value IS DISTINCT FROM n.value
        ) changed;
    ELSIF TG_OP = 'INSERT' THEN
        v_new_data := to_jsonb(NEW);
    END IF;
    
    -- Extract org_id and branch_id from the record if available
    INSERT INTO audit.audit_log (
        org_id, branch_id,
        table_schema, table_name, record_id, action,
        old_data, new_data, changed_columns,
        changed_by, ip_address, user_agent
    ) VALUES (
        COALESCE(
            COALESCE(NEW.org_id, OLD.org_id)::UUID,
            (SELECT org_id FROM platform.users WHERE id = NEW.created_by)::UUID,
            (SELECT org_id FROM platform.users WHERE id = OLD.created_by)::UUID
        ),
        COALESCE(
            COALESCE(NEW.branch_id, OLD.branch_id)::UUID,
            (SELECT branch_id FROM platform.users WHERE id = NEW.created_by)::UUID,
            (SELECT branch_id FROM platform.users WHERE id = OLD.created_by)::UUID
        ),
        TG_TABLE_SCHEMA,
        TG_TABLE_NAME,
        COALESCE(NEW.id, OLD.id),
        v_action,
        v_old_data,
        v_new_data,
        v_changed_cols,
        COALESCE(NEW.created_by, NEW.updated_by, OLD.created_by, OLD.updated_by),
        inet_client_addr(),
        current_setting('request.headers.user-agent', true)
    );
    
    RETURN COALESCE(NEW, OLD);
END;
 $$ LANGUAGE plpgsql;

-- Helper function to enable audit on a table
CREATE OR REPLACE FUNCTION audit.enable_audit(p_schema VARCHAR, p_table VARCHAR)
RETURNS VOID AS $$ BEGIN
    EXECUTE format(
        'CREATE TRIGGER trg_audit_%s AFTER INSERT OR UPDATE OR DELETE ON %I.%I 
         FOR EACH ROW EXECUTE FUNCTION audit.log_table_changes()',
        p_table, p_schema, p_table
    );
END;
 $$ LANGUAGE plpgsql;
```

## 2. Security: Row-Level Security (Multi-Tenancy)

Since this is a multi-tenant `PLATFORM -> ORGANIZATION -> BRANCH` architecture, we must enforce data isolation at the database level using PostgreSQL's Row-Level Security (RLS). This prevents Tenant A from accidentally reading Tenant B's data, even if the application has a bug.

```sql
-- ============================================================
-- ROW-LEVEL SECURITY (RLS) SETUP
-- ============================================================

-- 1. Create a custom configuration to pass context from the app to the DB
-- The backend API will execute: SET app.current_org_id = 'uuid'; SET app.current_branch_id = 'uuid';
ALTER DATABASE current_database() SET app.current_org_id TO '';
ALTER DATABASE current_database() SET app.current_branch_id TO '';

-- 2. Helper function to safely fetch the current session's Org ID
CREATE OR REPLACE FUNCTION app.current_org_id() RETURNS UUID AS $$ BEGIN
    RETURN NULLIF(current_setting('app.current_org_id', true), '')::UUID;
END;
 $$ LANGUAGE plpgsql STABLE;

CREATE OR REPLACE FUNCTION app.current_branch_id() RETURNS UUID AS $$ BEGIN
    RETURN NULLIF(current_setting('app.current_branch_id', true), '')::UUID;
END;
 $$ LANGUAGE plpgsql STABLE;

-- 3. Enable RLS on all tenant-isolated tables
ALTER TABLE clinical.patients ENABLE ROW LEVEL SECURITY;
ALTER TABLE clinical.admissions ENABLE ROW LEVEL SECURITY;
ALTER TABLE clinical.patient_visits ENABLE ROW LEVEL SECURITY;
ALTER TABLE clinical.lab_orders ENABLE ROW LEVEL SECURITY;
ALTER TABLE pharmacy.medicines ENABLE ROW LEVEL SECURITY;
ALTER TABLE pharmacy.medicine_stock ENABLE ROW LEVEL SECURITY;
ALTER TABLE finance.patient_bills ENABLE ROW LEVEL SECURITY;
ALTER TABLE finance.receipts ENABLE ROW LEVEL SECURITY;
ALTER TABLE hr.employees ENABLE ROW LEVEL SECURITY;
ALTER TABLE facility.beds ENABLE ROW LEVEL SECURITY;
-- (Repeat ENABLE ROW LEVEL SECURITY for all tables containing org_id or branch_id)

-- 4. Create Policies for Organization-level isolation
-- Super admins bypass RLS by default, but app users do not.

CREATE POLICY tenant_isolation_patients ON clinical.patients
    USING (org_id = app.current_org_id());

CREATE POLICY tenant_isolation_medicines ON pharmacy.medicines
    USING (org_id = app.current_org_id());

CREATE POLICY tenant_isolation_employees ON hr.employees
    USING (org_id = app.current_org_id());

-- 5. Create Policies for Branch-level isolation (where applicable)
-- Users restricted to a branch can only see that branch's data

CREATE POLICY branch_isolation_admissions ON clinical.admissions
    USING (
        org_id = app.current_org_id() 
        AND (app.current_branch_id() IS NULL OR branch_id = app.current_branch_id())
    );

CREATE POLICY branch_isolation_beds ON facility.beds
    USING (
        org_id = app.current_org_id() -- Note: beds link to wards, which link to branches
        AND (app.current_branch_id() IS NULL OR ward_id IN (SELECT id FROM facility.wards WHERE branch_id = app.current_branch_id()))
    );

CREATE POLICY branch_isolation_stock ON pharmacy.medicine_stock
    USING (
        org_id = app.current_org_id()
        AND (app.current_branch_id() IS NULL OR branch_id = app.current_branch_id())
    );

-- 6. Bypass RLS for background workers (e.g., cron jobs generating analytics)
-- Assign this role to service accounts that need cross-tenant access for aggregations.
CREATE ROLE service_bypass_rls;
ALTER ROLE service_bypass_rls BYPASSRLS;

-- Example: GRANT service_bypass_rls TO analytics_worker_user;
```

## 3. Data Integrity: Missing Triggers & Constraints

We need triggers to handle cascading state changes that foreign keys cannot cover (e.g., changing a bed to 'occupied' when admission is created, not just when transferred).

```sql
-- ============================================================
-- DATA INTEGRITY TRIGGERS
-- ============================================================

-- 1. OCCUPY BED ON ADMISSION
CREATE OR REPLACE FUNCTION clinical.occupy_bed_on_admission()
RETURNS TRIGGER AS $$ BEGIN
    IF NEW.bed_id IS NOT NULL AND NEW.status_code = 'admitted' THEN
        UPDATE facility.beds
        SET status_code = 'occupied',
            current_patient_id = NEW.patient_id,
            current_admission_id = NEW.id
        WHERE id = NEW.bed_id;
    END IF;
    RETURN NEW;
END;
 $$ LANGUAGE plpgsql;

CREATE TRIGGER trg_admission_occupy_bed
AFTER INSERT ON clinical.admissions
FOR EACH ROW EXECUTE FUNCTION clinical.occupy_bed_on_admission();

-- 2. RELEASE BED ON DISCHARGE (Enhancing previous trigger)
CREATE OR REPLACE FUNCTION clinical.release_bed_on_discharge()
RETURNS TRIGGER AS $$ BEGIN
    -- Update admission status first
    UPDATE clinical.admissions
    SET status_code = 'discharged',
        discharge_datetime = NEW.discharge_datetime
    WHERE id = NEW.admission_id;

    -- Free up the bed
    UPDATE facility.beds
    SET status_code = 'cleaning',
        current_patient_id = NULL,
        current_admission_id = NULL
    WHERE id = (SELECT bed_id FROM clinical.admissions WHERE id = NEW.admission_id);

    -- Mark patient deceased if applicable
    IF NEW.discharge_type = 'deceased' THEN
        UPDATE clinical.patients
        SET is_deceased = TRUE,
            death_datetime = NEW.discharge_datetime
        WHERE id = NEW.patient_id;
    END IF;

    RETURN NEW;
END;
 $$ LANGUAGE plpgsql;

DROP TRIGGER IF EXISTS trg_auto_cleaning ON clinical.discharges; -- Remove old basic trigger
CREATE TRIGGER trg_discharge_full_process
AFTER INSERT ON clinical.discharges
FOR EACH ROW EXECUTE FUNCTION clinical.release_bed_on_discharge();

-- 3. PREVENT NEGATIVE PHARMACY STOCK
CREATE OR REPLACE FUNCTION pharmacy.check_negative_stock()
RETURNS TRIGGER AS $$ DECLARE
    v_current_qty INTEGER;
    v_medicine_name VARCHAR;
BEGIN
    SELECT quantity INTO v_current_qty FROM pharmacy.medicine_stock WHERE id = NEW.stock_id;
    
    IF v_current_qty - NEW.quantity < 0 THEN
        SELECT generic_name || ' (' || brand_name || ')' INTO v_medicine_name 
        FROM pharmacy.medicines WHERE id = NEW.medicine_id;
        
        RAISE EXCEPTION 'Insufficient stock for % (Batch: %). Available: %, Requested: %', 
            v_medicine_name, NEW.batch_no, v_current_qty, NEW.quantity;
    END IF;
    
    RETURN NEW;
END;
 $$ LANGUAGE plpgsql;

ALTER TABLE pharmacy.dispensing_items 
DROP CONSTRAINT IF EXISTS chk_positive_qty;
ALTER TABLE pharmacy.dispensing_items 
ADD CONSTRAINT chk_positive_qty CHECK (quantity > 0);

-- Change trigger to BEFORE INSERT to catch errors before they happen
DROP TRIGGER IF EXISTS trg_reduce_stock ON pharmacy.dispensing_items;
CREATE TRIGGER trg_reduce_stock
BEFORE INSERT ON pharmacy.dispensing_items
FOR EACH ROW EXECUTE FUNCTION pharmacy.check_negative_stock();

-- 4. SYNC USER FULL NAME WITH EMPLOYEE (Bi-directional sync)
CREATE OR REPLACE FUNCTION platform.sync_user_employee_name()
RETURNS TRIGGER AS $$ BEGIN
    -- If user is linked to an employee, keep names in sync
    IF TG_OP = 'UPDATE' AND NEW.employee_id IS NOT NULL AND NEW.full_name IS DISTINCT FROM OLD.full_name THEN
        UPDATE hr.employees SET full_name = NEW.full_name WHERE id = NEW.employee_id;
    END IF;
    RETURN NEW;
END;
 $$ LANGUAGE plpgsql;

CREATE TRIGGER trg_sync_user_to_employee
AFTER UPDATE ON platform.users
FOR EACH ROW EXECUTE FUNCTION platform.sync_user_employee_name();

CREATE OR REPLACE FUNCTION hr.sync_employee_user_name()
RETURNS TRIGGER AS $$ BEGIN
    IF TG_OP = 'UPDATE' AND NEW.full_name IS DISTINCT FROM OLD.full_name THEN
        UPDATE platform.users SET full_name = NEW.full_name WHERE employee_id = NEW.id;
    END IF;
    RETURN NEW;
END;
 $$ LANGUAGE plpgsql;

CREATE TRIGGER trg_sync_employee_to_user
AFTER UPDATE ON hr.employees
FOR EACH ROW EXECUTE FUNCTION hr.sync_employee_user_name();
```

## 5. ID Generation & Sequence Management

While UUIDs are great for distributed systems and avoiding ID enumeration, they are terrible for database index fragmentation (B-Tree random inserts). For a high-throughput hospital system, a hybrid approach is best.

```sql
-- ============================================================
-- OPTIMIZED ID GENERATION
-- ============================================================

-- 1. Use UUIDv7 (Time-ordered UUIDs) instead of UUIDv4
-- UUIDv7 embeds a Unix timestamp in the first 48 bits, making them naturally sortable!
-- This eliminates B-Tree index fragmentation on the Primary Key.

CREATE EXTENSION IF NOT EXISTS "pgcrypto";

CREATE OR REPLACE FUNCTION gen_uuid_v7()
RETURNS UUID AS $$ DECLARE
    unix_ts_ms BYTEA;
    uuid_bytes BYTEA;
BEGIN
    unix_ts_ms = substring(int8send(floor(extract(epoch from clock_timestamp()) * 1000)::bigint) from 3);
    uuid_bytes = unix_ts_ms || gen_random_bytes(10);
    uuid_bytes = set_byte(uuid_bytes, 6, (b'0111' || get_byte(uuid_bytes, 6)::bit(4))::bit(8)::int);
    uuid_bytes = set_byte(uuid_bytes, 8, (b'10' || get_byte(uuid_bytes, 8)::bit(6))::bit(8)::int);
    RETURN encode(uuid_bytes, 'hex')::UUID;
END;
 $$ LANGUAGE plpgsql VOLATILE;

-- Example: Change default for high-insert tables
ALTER TABLE clinical.vital_signs ALTER COLUMN id SET DEFAULT gen_uuid_v7();
ALTER TABLE clinical.fluid_records ALTER COLUMN id SET DEFAULT gen_uuid_v7();
ALTER TABLE audit.audit_log ALTER COLUMN id SET DEFAULT gen_uuid_v7();

-- 2. Human-Readable Sequential Codes Generator
-- E.g., "ADM-20231025-0001" (Admission for Oct 25, 2023, 1st admission)
CREATE OR REPLACE FUNCTION clinical.generate_admission_no()
RETURNS TRIGGER AS $$ BEGIN
    IF NEW.admission_no IS NULL THEN
        SELECT 'ADM-' || TO_CHAR(NEW.admission_datetime, 'YYYYMMDD') || '-' || LPAD(COUNT(*)::TEXT, 4, '0')
        INTO NEW.admission_no
        FROM clinical.admissions 
        WHERE DATE(admission_datetime) = DATE(NEW.admission_datetime)
        AND branch_id = NEW.branch_id;
    END IF;
    RETURN NEW;
END;
 $$ LANGUAGE plpgsql;

CREATE TRIGGER trg_gen_admission_no
BEFORE INSERT ON clinical.admissions
FOR EACH ROW EXECUTE FUNCTION clinical.generate_admission_no();

-- Same pattern for Bills
CREATE OR REPLACE FUNCTION finance.generate_bill_no()
RETURNS TRIGGER AS $$ BEGIN
    IF NEW.bill_no IS NULL THEN
        SELECT 'BILL-' || TO_CHAR(NEW.bill_date, 'YYYYMMDD') || '-' || LPAD(COUNT(*)::TEXT, 4, '0')
        INTO NEW.bill_no
        FROM finance.patient_bills 
        WHERE bill_date = NEW.bill_date
        AND branch_id = NEW.branch_id;
    END IF;
    RETURN NEW;
END;
 $$ LANGUAGE plpgsql;

CREATE TRIGGER trg_gen_bill_no
BEFORE INSERT ON finance.patient_bills
FOR EACH ROW EXECUTE FUNCTION finance.generate_bill_no();
```
## 6. Analytics: Automated Data Aggregation (Cron Jobs)

The `analytics.daily_branch_stats` table needs to be populated. In production, use `pg_cron` to run this nightly.

```sql
-- ============================================================
-- PG_CRON: AUTOMATED NIGHTLY AGGREGATION
-- ============================================================

-- Enable pg_cron (requires superuser)
-- CREATE EXTENSION pg_cron;

-- Schedule the aggregation job to run at 01:00 AM every day
SELECT cron.schedule(
    'nightly_stats_aggregation',
    '0 1 * * *', -- Cron format
    $$     INSERT INTO analytics.daily_branch_stats (
        org_id, branch_id, stat_date,
        new_registrations, opd_visits, ipd_admissions, ipd_discharges,
        total_billed, total_collected, total_discount
    )
    SELECT 
        b.org_id,
        b.branch_id,
        CURRENT_DATE - 1 AS stat_date,
        
        -- New Registrations
        (SELECT COUNT(*) FROM clinical.patients p WHERE DATE(p.created_at) = CURRENT_DATE - 1 AND p.branch_id = b.branch_id),
        
        -- OPD Visits
        (SELECT COUNT(*) FROM clinical.patient_visits v WHERE DATE(v.visit_date) = CURRENT_DATE - 1 AND v.branch_id = b.branch_id AND v.visit_type = 'opd'),
        
        -- IPD Admissions
        (SELECT COUNT(*) FROM clinical.admissions a WHERE DATE(a.admission_datetime) = CURRENT_DATE - 1 AND a.branch_id = b.branch_id),
        
        -- IPD Discharges
        (SELECT COUNT(*) FROM clinical.discharges d WHERE DATE(d.discharge_datetime) = CURRENT_DATE - 1 AND d.branch_id = (SELECT branch_id FROM clinical.admissions WHERE id = d.admission_id)),
        
        -- Total Billed
        COALESCE((SELECT SUM(total_amount) FROM finance.patient_bills pb WHERE pb.bill_date = CURRENT_DATE - 1 AND pb.branch_id = b.branch_id AND pb.status_code NOT IN ('cancelled')), 0),
        
        -- Total Collected
        COALESCE((SELECT SUM(r.amount) FROM finance.receipts r JOIN finance.receipt_bill_allocations rba ON r.id = rba.receipt_id JOIN finance.patient_bills pb ON rba.bill_id = pb.id WHERE DATE(r.receipt_date) = CURRENT_DATE - 1 AND r.branch_id = b.branch_id), 0),
        
        -- Total Discount
        COALESCE((SELECT SUM(discount_amount) FROM finance.patient_bills pb WHERE pb.bill_date = CURRENT_DATE - 1 AND pb.branch_id = b.branch_id AND pb.status_code NOT IN ('cancelled')), 0)
        
    FROM platform.branches b
    WHERE b.is_active = TRUE
    ON CONFLICT (org_id, branch_id, stat_date) 
    DO UPDATE SET
        new_registrations = EXCLUDED.new_registrations,
        opd_visits = EXCLUDED.opd_visits,
        ipd_admissions = EXCLUDED.ipd_admissions,
        ipd_discharges = EXCLUDED.ipd_discharges,
        total_billed = EXCLUDED.total_billed,
        total_collected = EXCLUDED.total_collected,
        total_discount = EXCLUDED.total_discount;
    $$ );
```

```sql
    height_cm           DECIMAL(6,2),
    gcs_score           INTEGER,
    pain_scale          INTEGER,
    blood_glucose       DECIMAL(6,2),
    notes               TEXT,
    PRIMARY KEY (id, patient_id) -- Partition key MUST be part of PK
) PARTITION BY HASH (patient_id);

-- Create 4 partitions (typically matches the number of CPU cores or disks)
CREATE TABLE clinical.vital_signs_p0 PARTITION OF clinical.vital_signs_partitioned FOR VALUES WITH (MODULUS 4, REMAINDER 0);
```

## 4. Observability: DBA Health Check Queries
```sql
-- ============================================================
-- OBSERVABILITY: Health & Performance Queries
-- ============================================================

-- 1. TABLE BLOAT MONITORING
-- Over time, UPDATEs and DELETEs leave dead tuples. High bloat means slower scans.
SELECT 
    schemaname || '.' || relname AS table_name,
    pg_size_pretty(pg_total_relation_size(relid)) AS total_size,
    n_dead_tup AS dead_rows,
    n_live_tup AS live_rows,
    ROUND(n_dead_tup * 100.0 / NULLIF(n_live_tup + n_dead_tup, 0), 2) AS bloat_percentage,
    last_vacuum,
    last_autovacuum
FROM pg_stat_user_tables
WHERE n_dead_tup > 10000 
ORDER BY n_dead_tup DESC;

-- 2. MISSING INDEXES (Seq Scans on large tables)
-- If a table has > 10k rows and is doing mostly sequential scans, it probably needs an index.
SELECT 
    schemaname || '.' || relname AS table_name,
    seq_scan,
    idx_scan,
    CASE WHEN seq_scan > 0 THEN ROUND((seq_scan * 100.0 / NULLIF(seq_scan + idx_scan, 0)), 2) ELSE 0 END AS seq_scan_pct,
    n_live_tup AS row_count
FROM pg_stat_user_tables
WHERE n_live_tup > 10000 
  AND COALESCE(seq_scan, 0) > COALESCE(idx_scan, 0)
ORDER BY seq_scan DESC;

-- 3. UNUSED INDEXES (Wastes Insert/Update performance)
-- Indexes that exist but are never used should be dropped.
SELECT 
    schemaname || '.' || relname AS table_name,
    indexrelname AS index_name,
    pg_size_pretty(pg_relation_size(indexrelid)) AS index_size,
    idx_scan AS times_used
FROM pg_stat_user_indexes
WHERE idx_scan = 0 
  AND indexrelname NOT LIKE '%_pkey' -- Don't suggest dropping Primary Keys
ORDER BY pg_relation_size(indexrelid) DESC;

-- 4. LONG RUNNING QUERIES (Blocking/Heavy queries)
-- Alerts you if a query has been running for > 5 minutes.
SELECT 
    pid,
    now() - pg_stat_activity.query_start AS duration,
    query,
    state,
    wait_event_type,
    wait_event
FROM pg_stat_activity 
WHERE (now() - pg_stat_activity.query_start) > interval '5 minutes'
  AND state != 'idle'
  AND query NOT LIKE '%pg_stat_activity%';

-- 5. CONNECTION POOL SATURATION
-- Check if PgBouncer or direct connections are maxed out.
SELECT 
    state,
    COUNT(*) AS connections
FROM pg_stat_activity
WHERE datname = current_database()
GROUP BY state
ORDER BY connections DESC;
```
