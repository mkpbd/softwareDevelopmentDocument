## 2.1 OLTP vs OLAP Separation

```
┌──────────────────────────────────────────────────────────────────┐
│                    TWO DATABASE TYPES                             │
├─────────────────────────┬────────────────────────────────────────┤
│ OLTP (Operational DB)   │ OLAP (Analytical / Data Warehouse)     │
├─────────────────────────┼────────────────────────────────────────┤
│ Highly normalized       │ Denormalized (Star Schema)             │
│ Write-heavy             │ Read-heavy                             │
│ Real-time transactions  │ Batch loaded (ETL)                     │
│ Row-level locking       │ Columnar storage preferred             │
│ Primary PostgreSQL DB   │ Separate PostgreSQL DB or TimescaleDB  │
│ UUID primary keys       │ Surrogate integer keys for joins       │
│ Full audit fields       │ Pre-aggregated KPI tables              │
│ RLS enforced            │ Read-only replica access               │
└─────────────────────────┴────────────────────────────────────────┘
```

## 2.2 Multi-Tenancy Strategy

```sql
-- STRATEGY: Row-Level Multi-Tenancy
-- Every table has tenant_id (UUID)
-- Row-Level Security (RLS) enforced at PostgreSQL level
-- Branch-level isolation via branch_id

-- Pattern for ALL tables:
tenant_id   UUID    NOT NULL  REFERENCES tenants(tenant_id)
branch_id   UUID    NOT NULL  REFERENCES branches(branch_id)
```

## 2.3 Universal Audit Fields (ALL Tables)

```sql
-- These columns appear on EVERY table
created_at   TIMESTAMPTZ   NOT NULL  DEFAULT NOW()
created_by   UUID          NOT NULL  REFERENCES users(user_id)
updated_at   TIMESTAMPTZ   NOT NULL  DEFAULT NOW()
updated_by   UUID          NOT NULL  REFERENCES users(user_id)
is_deleted   BOOLEAN       NOT NULL  DEFAULT FALSE
deleted_at   TIMESTAMPTZ
deleted_by   UUID          REFERENCES users(user_id)
```

## 2.4 UUID Strategy

```sql
-- Enable UUID extension
CREATE EXTENSION IF NOT EXISTS "uuid-ossp";
CREATE EXTENSION IF NOT EXISTS "pgcrypto";

-- UUID Generation
DEFAULT uuid_generate_v4()

-- Why UUID over SERIAL?
-- 1. Safe for distributed systems and multi-branch
-- 2. No sequential guessing (security)
-- 3. Can generate IDs client-side before insert
-- 4. Safe for data merging across branches
```

## 4.1 Patients Table

```sql
CREATE TABLE patients (
    patient_id              UUID        PRIMARY KEY DEFAULT uuid_generate_v4(),
    tenant_id               UUID        NOT NULL REFERENCES tenants(tenant_id),
    branch_id               UUID        NOT NULL REFERENCES branches(branch_id),
    mrn                     VARCHAR(30) NOT NULL,
    -- MRN = Medical Record Number (e.g., MRN-2024-000001)

    -- Personal Information
    first_name              VARCHAR(100) NOT NULL,
    middle_name             VARCHAR(100),
    last_name               VARCHAR(100) NOT NULL,
    full_name               VARCHAR(300)
        GENERATED ALWAYS AS
            (first_name || ' ' || COALESCE(middle_name || ' ', '') || last_name)
        STORED,
    date_of_birth           DATE,
    age_years               SMALLINT
        GENERATED ALWAYS AS
            (EXTRACT(YEAR FROM AGE(date_of_birth))::SMALLINT)
        STORED,
    gender                  SMALLINT    NOT NULL
                            CHECK (gender IN (1,2,3)),
                            -- 1=Male, 2=Female, 3=Other
    blood_group             VARCHAR(5)
                            CHECK (blood_group IN
                                ('A+','A-','B+','B-','AB+','AB-','O+','O-','Unknown')),
    marital_status          SMALLINT
                            CHECK (marital_status IN (1,2,3,4)),
                            -- 1=Single, 2=Married, 3=Widowed, 4=Divorced

    -- Contact Information
    primary_phone           VARCHAR(20) NOT NULL,
    secondary_phone         VARCHAR(20),
    email                   VARCHAR(150),
    whatsapp_number         VARCHAR(20),

    -- Identity
    national_id             BYTEA,      -- Encrypted (pgcrypto)
    national_id_type        VARCHAR(50), -- SSN/Aadhar/Passport
    passport_number         BYTEA,      -- Encrypted
    profile_photo_url       VARCHAR(500),

    -- Classification
    patient_category        SMALLINT    NOT NULL DEFAULT 1
                            CHECK (patient_category IN (1,2,3,4,5,6)),
                            -- 1=General, 2=VIP, 3=Staff, 4=Corporate
                            -- 5=Government, 6=International
    nationality_id          UUID        REFERENCES countries(country_id),
    language_preference     VARCHAR(20) DEFAULT 'en',
    religion                VARCHAR(50),
    occupation              VARCHAR(100),

    -- Referral
    referred_by             VARCHAR(200),
    referral_source         SMALLINT
                            CHECK (referral_source IN (1,2,3,4,5,6)),
                            -- 1=Self, 2=Doctor, 3=Corporate, 4=Online
                            -- 5=Social_Media, 6=Walk_In

    -- Status
    is_active               BOOLEAN     NOT NULL DEFAULT TRUE,
    is_deceased             BOOLEAN     NOT NULL DEFAULT FALSE,
    deceased_date           DATE,
    deceased_cause          TEXT,

    -- Portal Access
    portal_user_id          UUID,       -- FK to users table
    portal_registration_date DATE,

    -- Metadata
    metadata                JSONB       DEFAULT '{}',

    -- Audit
    created_at              TIMESTAMPTZ NOT NULL DEFAULT NOW(),
    updated_at              TIMESTAMPTZ NOT NULL DEFAULT NOW(),
    created_by              UUID        NOT NULL,
    updated_by              UUID        NOT NULL,
    is_deleted              BOOLEAN     NOT NULL DEFAULT FALSE,
    deleted_at              TIMESTAMPTZ,
    deleted_by              UUID,

    CONSTRAINT uq_patients_mrn_per_tenant UNIQUE (tenant_id, mrn)
);

-- Critical Indexes
CREATE INDEX idx_patients_tenant_branch
    ON patients(tenant_id, branch_id);

CREATE INDEX idx_patients_mrn
    ON patients(tenant_id, mrn);

CREATE INDEX idx_patients_phone
    ON patients(tenant_id, primary_phone);

CREATE INDEX idx_patients_email
    ON patients(tenant_id, email)
    WHERE email IS NOT NULL;

-- Full-text search index on patient name
CREATE INDEX idx_patients_fullname_trgm
    ON patients USING GIN (full_name gin_trgm_ops);

-- Partial index for active patients only
CREATE INDEX idx_patients_active
    ON patients(tenant_id, branch_id)
    WHERE is_deleted = FALSE AND is_active = TRUE;

-- Date of birth index for age-based queries
CREATE INDEX idx_patients_dob
    ON patients(date_of_birth)
    WHERE date_of_birth IS NOT NULL;
```

## 16.1 Triggers — Auto Audit Logging

```sql
-- Generic Audit Trigger Function
CREATE OR REPLACE FUNCTION fn_audit_trigger()
RETURNS TRIGGER AS $$
DECLARE
    v_old_data   JSONB;
    v_new_data   JSONB;
    v_changed    TEXT[];
    v_user_id    UUID;
    v_tenant_id  UUID;
BEGIN
    -- Get current user from session variable
    BEGIN
        v_user_id  := current_setting('app.current_user_id')::UUID;
        v_tenant_id := current_setting('app.current_tenant_id')::UUID;
    EXCEPTION WHEN OTHERS THEN
        v_user_id  := NULL;
        v_tenant_id := NULL;
    END;

    IF (TG_OP = 'DELETE') THEN
        v_old_data := row_to_json(OLD)::JSONB;
        v_new_data := NULL;
        v_changed  := ARRAY(SELECT key FROM jsonb_each_text(v_old_data));

        INSERT INTO audit_logs (
            tenant_id, user_id, action_type,
            module, table_name, record_id,
            old_value, new_value, changed_fields,
            ip_address, action_at
        ) VALUES (
            COALESCE(v_tenant_id, (v_old_data->>'tenant_id')::UUID),
            v_user_id,
            'DELETE',
            TG_TABLE_SCHEMA,
            TG_TABLE_NAME,
            (v_old_data->>'id')::UUID,
            v_old_data,
            NULL,
            v_changed,
            inet_client_addr(),
            NOW()
        );
        RETURN OLD;

    ELSIF (TG_OP = 'UPDATE') THEN
        v_old_data := row_to_json(OLD)::JSONB;
        v_new_data := row_to_json(NEW)::JSONB;

        -- Find changed fields only
        SELECT ARRAY(
            SELECT key
            FROM jsonb_each_text(v_new_data)
            WHERE value IS DISTINCT FROM (v_old_data->>key)
              AND key NOT IN ('updated_at', 'updated_by')
        ) INTO v_changed;

        IF array_length(v_changed, 1) > 0 THEN
            INSERT INTO audit_logs (
                tenant_id, user_id, action_type,
                module, table_name, record_id,
                old_value, new_value, changed_fields,
                ip_address, action_at
            ) VALUES (
                COALESCE(v_tenant_id, (v_new_data->>'tenant_id')::UUID),
                v_user_id,
                'UPDATE',
                TG_TABLE_SCHEMA,
                TG_TABLE_NAME,
                (v_new_data->(TG_TABLE_NAME || '_id'))::UUID,
                v_old_data,
                v_new_data,
                v_changed,
                inet_client_addr(),
                NOW()
            );
        END IF;
        RETURN NEW;

    ELSIF (TG_OP = 'INSERT') THEN
        v_new_data := row_to_json(NEW)::JSONB;

        INSERT INTO audit_logs (
            tenant_id, user_id, action_type,
            module, table_name, record_id,
            old_value, new_value, changed_fields,
            ip_address, action_at
        ) VALUES (
            COALESCE(v_tenant_id, (v_new_data->>'tenant_id')::UUID),
            v_user_id,
            'CREATE',
            TG_TABLE_SCHEMA,
            TG_TABLE_NAME,
            (v_new_data->(TG_TABLE_NAME || '_id'))::UUID,
            NULL,
            v_new_data,
            ARRAY(SELECT key FROM jsonb_each_text(v_new_data)),
            inet_client_addr(),
            NOW()
        );
        RETURN NEW;
    END IF;

    RETURN NULL;
END;
$$ LANGUAGE plpgsql SECURITY DEFINER;

-- Apply to critical tables
```

```sql
CREATE OR REPLACE FUNCTION fn_manage_bed_status()
RETURNS TRIGGER AS $$
BEGIN
    -- When admission is created, mark bed as Occupied
    IF TG_OP = 'INSERT' THEN
        UPDATE beds
        SET
            status = 2,  -- Occupied
            current_admission_id = NEW.admission_id,
            last_occupied_at = NOW(),
            updated_at = NOW()
        WHERE bed_id = NEW.bed_id;

    -- When admission is discharged, mark bed as Cleaning
    ELSIF TG_OP = 'UPDATE' THEN
        IF OLD.status <> NEW.status AND NEW.status = 2 THEN
            -- Status changed to Discharged
            UPDATE beds
            SET
                status = 5,  -- Cleaning required
                current_admission_id = NULL,
                updated_at = NOW()
            WHERE bed_id = NEW.bed_id;
        END IF;

        -- If bed changed (transfer)
        IF OLD.bed_id <> NEW.bed_id THEN
            -- Release old bed
            UPDATE beds
            SET status = 5,
                current_admission_id = NULL,
                updated_at = NOW()
            WHERE bed_id = OLD.bed_id;

            -- Reserve new bed
            UPDATE beds
            SET status = 2,
                current_admission_id = NEW.admission_id,
                last_occupied_at = NOW(),
                updated_at = NOW()
            WHERE bed_id = NEW.bed_id;

            -- Log the transfer
            INSERT INTO patient_bed_history (
                admission_id, patient_id,
                from_bed_id, to_bed_id,
                from_ward_id, to_ward_id,
                transferred_at, transferred_by
            ) VALUES (
                NEW.admission_id, NEW.patient_id,
                OLD.bed_id, NEW.bed_id,
                OLD.ward_id, NEW.ward_id,
                NOW(), NEW.updated_by
            );
        END IF;
    END IF;

    RETURN NEW;
END;
$$ LANGUAGE plpgsql;
```

## 16.5 Trigger — Pharmacy Stock Deduction

```sql
CREATE OR REPLACE FUNCTION fn_pharmacy_stock_deduction()
RETURNS TRIGGER AS $$
DECLARE
    v_available NUMERIC;
BEGIN
    IF TG_OP = 'INSERT' AND NEW.is_return = FALSE THEN
        -- Check available stock
        SELECT quantity_available INTO v_available
        FROM pharmacy_inventory
        WHERE inventory_id = NEW.inventory_id
        FOR UPDATE;  -- Lock row

        IF v_available < NEW.quantity_dispensed THEN
            RAISE EXCEPTION
                'Insufficient stock. Available: %, Required: %',
                v_available, NEW.quantity_dispensed;
        END IF;

        -- Deduct stock
        UPDATE pharmacy_inventory
        SET
            quantity_in_stock = quantity_in_stock - NEW.quantity_dispensed,
            updated_at = NOW(),
            updated_by = NEW.dispensed_by
        WHERE inventory_id = NEW.inventory_id;

        -- Update prescription item dispensed quantity
        IF NEW.prescription_item_id IS NOT NULL THEN
            UPDATE prescription_items
            SET dispensed_quantity = dispensed_quantity + NEW.quantity_dispensed
            WHERE prescription_item_id = NEW.prescription_item_id;

            -- Update prescription status
            UPDATE prescriptions p
            SET status = CASE
                WHEN (
                    SELECT COUNT(*) FROM prescription_items pi
                    WHERE pi.prescription_id = p.prescription_id
                    AND pi.quantity > pi.dispensed_quantity
                    AND pi.status = 1
                ) = 0 THEN 3  -- Fully Dispensed
                ELSE 2  -- Partially Dispensed
            END
            WHERE prescription_id = (
                SELECT prescription_id FROM prescription_items
                WHERE prescription_item_id = NEW.prescription_item_id
            );
        END IF;

    ELSIF TG_OP = 'INSERT' AND NEW.is_return = TRUE THEN
        -- Return stock
        UPDATE pharmacy_inventory
        SET
            quantity_in_stock = quantity_in_stock + NEW.return_quantity,
            updated_at = NOW(),
            updated_by = NEW.dispensed_by
        WHERE inventory_id = NEW.inventory_id;
    END IF;

    RETURN NEW;
END;
$$ LANGUAGE plpgsql;

CREATE TRIGGER trg_pharmacy_stock_deduction
    AFTER INSERT ON dispensing_records
```

```sql
-- Procedure: Admit Patient
CREATE OR REPLACE PROCEDURE sp_admit_patient(
    p_patient_id        UUID,
    p_consultation_id   UUID,
    p_admitting_doctor  UUID,
    p_treating_doctor   UUID,
    p_department_id     UUID,
    p_ward_id           UUID,
    p_room_id           UUID,
    p_bed_id            UUID,
    p_admission_type    SMALLINT,
    p_diagnosis         VARCHAR,
    p_icd_code          VARCHAR,
    p_branch_id         UUID,
    p_tenant_id         UUID,
    p_created_by        UUID,
    OUT p_admission_id  UUID,
    OUT p_error_message TEXT
)
LANGUAGE plpgsql AS $$
DECLARE
    v_bed_status    SMALLINT;
    v_adm_number    VARCHAR(30);
    v_seq           BIGINT;
BEGIN
    -- Validate bed availability
    SELECT status INTO v_bed_status
    FROM beds
    WHERE bed_id = p_bed_id AND is_deleted = FALSE
    FOR UPDATE;  -- Lock

    IF v_bed_status IS NULL THEN
        p_error_message := 'Bed not found';
        RETURN;
    END IF;

    IF v_bed_status <> 1 THEN  -- Not Vacant
        p_error_message := 'Bed is not available. Current status: ' ||
            CASE v_bed_status
                WHEN 2 THEN 'Occupied'
                WHEN 3 THEN 'Under Maintenance'
                WHEN 4 THEN 'Reserved'
                WHEN 5 THEN 'Cleaning'
            END;
        RETURN;
    END IF;

    -- Generate admission number
    SELECT COALESCE(MAX(SUBSTRING(admission_number FROM 'ADM-\d+-(\d+)')::BIGINT), 0) + 1
    INTO v_seq
    FROM admissions
    WHERE tenant_id = p_tenant_id
    AND EXTRACT(YEAR FROM admission_date) = EXTRACT(YEAR FROM NOW());

    v_adm_number := 'ADM-' || EXTRACT(YEAR FROM NOW())::TEXT ||
                    '-' || LPAD(v_seq::TEXT, 6, '0');

    -- Create admission record
    p_admission_id := uuid_generate_v4();

    INSERT INTO admissions (
        admission_id, tenant_id, branch_id,
        admission_number, patient_id, consultation_id,
        admitting_doctor_id, treating_doctor_id,
        department_id, ward_id, room_id, bed_id,
        admission_date, admission_type, admission_source,
        primary_diagnosis, primary_icd_code,
        status, created_at, created_by, updated_at, updated_by
    ) VALUES (
        p_admission_id, p_tenant_id, p_branch_id,
        v_adm_number, p_patient_id, p_consultation_id,
        p_admitting_doctor, p_treating_doctor,
        p_department_id, p_ward_id, p_room_id, p_bed_id,
        NOW(), p_admission_type, 1,
        p_diagnosis, p_icd_code,
        1, NOW(), p_created_by, NOW(), p_created_by
    );

    -- Update consultation as admitted
    IF p_consultation_id IS NOT NULL THEN
        UPDATE consultations
        SET is_admitted = TRUE,
            admission_id = p_admission_id,
            status = 4,  -- Admitted
            updated_at = NOW(),
            updated_by = p_created_by
        WHERE consultation_id = p_consultation_id;
    END IF;

    p_error_message := NULL;

EXCEPTION WHEN OTHERS THEN
    p_error_message := SQLERRM;
    p_admission_id  := NULL;
END;
$$;

-- Procedure: Discharge Patient
CREATE OR REPLACE PROCEDURE sp_discharge_patient(
    p_admission_id      UUID,
    p_discharge_type    SMALLINT,
    p_discharge_notes   TEXT,
    p_final_diagnosis   TEXT,
    p_created_by        UUID,
    OUT p_error_message TEXT
)
LANGUAGE plpgsql AS $$
DECLARE
    v_admission     RECORD;
    v_pending_bill  NUMERIC;
BEGIN
    -- Get admission info
    SELECT a.*, b.bill_id, b.outstanding_amount
    INTO v_admission
    FROM admissions a
    LEFT JOIN patient_bills b ON b.admission_id = a.admission_id
        AND b.status IN (2,3)  -- Pending or Partial
        AND b.is_deleted = FALSE
    WHERE a.admission_id = p_admission_id
      AND a.is_deleted = FALSE
    FOR UPDATE OF a;

    IF v_admission IS NULL THEN
        p_error_message := 'Admission not found';
        RETURN;
    END IF;

    IF v_admission.status <> 1 THEN
        p_error_message := 'Patient is not currently admitted';
        RETURN;
    END IF;

    -- Check pending bills
    SELECT COALESCE(SUM(outstanding_amount), 0) INTO v_pending_bill
    FROM patient_bills
    WHERE admission_id = p_admission_id
      AND status IN (2,3)
      AND is_deleted = FALSE;

    -- Discharge patient
    UPDATE admissions
    SET
        status = 2,  -- Discharged
        actual_discharge_date = NOW(),
        discharge_type = p_discharge_type,
        notes = p_discharge_notes,
        updated_at = NOW(),
        updated_by = p_created_by
    WHERE admission_id = p_admission_id;

    -- Bed status handled by trigger (fn_manage_bed_status)

    p_error_message := NULL;

EXCEPTION WHEN OTHERS THEN
    p_error_message := SQLERRM;
    ROLLBACK;
END;
$$;

RETURNS TABLE (
    interacting_drug_id     UUID,

BEGIN

    SELECT
        d2.drug_id,
        d2.generic_name::VARCHAR,
        (interaction->>'severity')::VARCHAR,
        (interaction->>'description')::TEXT
    FROM drugs d1
    CROSS JOIN LATERAL jsonb_array_elements(d1.drug_interactions) AS interaction

    -- Check if patient currently has this drug prescribed
    WHERE d1.drug_id = p_drug_id

          FROM prescriptions p
          JOIN prescription_items pi ON p.prescription_id = pi.prescription_id
          WHERE p.patient_id = p_patient_id
            AND p.status IN (1,2)
            AND pi.status = 1
            AND (pi.end_date IS NULL OR pi.end_date >= CURRENT_DATE)
            AND p.is_deleted = FALSE
      );
```

```sql
-- Function to encrypt sensitive data (using pgcrypto)
CREATE OR REPLACE FUNCTION fn_encrypt_sensitive(
    p_data      TEXT,
    p_key       TEXT
)
RETURNS BYTEA AS $$
BEGIN
    IF p_data IS NULL THEN
        RETURN NULL;
    END IF;
    RETURN pgp_sym_encrypt(p_data, p_key, 'cipher-algo=aes256');
END;
$$ LANGUAGE plpgsql SECURITY DEFINER;

-- Function to decrypt sensitive data
CREATE OR REPLACE FUNCTION fn_decrypt_sensitive(
    p_data      BYTEA,
    p_key       TEXT
)
RETURNS TEXT AS $$
BEGIN
    IF p_data IS NULL THEN
        RETURN NULL;
    END IF;
    RETURN pgp_sym_decrypt(p_data, p_key);
EXCEPTION WHEN OTHERS THEN
    RETURN '[DECRYPTION ERROR]';
END;
$$ LANGUAGE plpgsql SECURITY DEFINER;
```

```sql
-- Function to calculate and insert daily KPIs
CREATE OR REPLACE FUNCTION fn_calculate_daily_kpis(
    p_branch_id UUID,
    p_kpi_date  DATE
)
RETURNS void AS $$
DECLARE
    v_tenant_id UUID;
BEGIN
    SELECT tenant_id INTO v_tenant_id
    FROM branches WHERE branch_id = p_branch_id;

    INSERT INTO kpi_daily_summary (
        tenant_id, branch_id, kpi_date,
        new_patients, returning_patients,
        total_opd_visits, total_ipd_admissions,
        total_discharges,
        beds_occupied, beds_total, bed_occupancy_rate,
        avg_length_of_stay,
        total_revenue, total_collection,
        outstanding_amount,
        lab_orders_received, lab_results_reported,
        appointments_scheduled, appointments_completed,
        appointments_cancelled, appointments_no_show,
        calculated_at
    )
    SELECT
        v_tenant_id,
        p_branch_id,
        p_kpi_date,

        -- New patients (registered on this date)
        (SELECT COUNT(*) FROM patients
         WHERE branch_id = p_branch_id
           AND DATE(created_at) = p_kpi_date
           AND is_deleted = FALSE),

        -- Returning patients
        (SELECT COUNT(DISTINCT patient_id) FROM appointments
         WHERE branch_id = p_branch_id
           AND appointment_date = p_kpi_date
           AND status = 5
           AND patient_id NOT IN (
               SELECT patient_id FROM patients
               WHERE branch_id = p_branch_id
               AND DATE(created_at) = p_kpi_date
           )),

        -- OPD visits
        (SELECT COUNT(*) FROM consultations
         WHERE branch_id = p_branch_id
           AND consultation_date = p_kpi_date
           AND is_deleted = FALSE),

        -- IPD admissions
        (SELECT COUNT(*) FROM admissions
         WHERE branch_id = p_branch_id
           AND DATE(admission_date) = p_kpi_date
           AND is_deleted = FALSE),

        -- Discharges
        (SELECT COUNT(*) FROM admissions
         WHERE branch_id = p_branch_id
           AND DATE(actual_discharge_date) = p_kpi_date
           AND status = 2),

        -- Beds occupied (as of end of day)
        (SELECT COUNT(*) FROM beds
         WHERE branch_id = p_branch_id
           AND status = 2 AND is_deleted = FALSE),

        -- Total beds
        (SELECT COUNT(*) FROM beds
         WHERE branch_id = p_branch_id
           AND is_deleted = FALSE AND is_active = TRUE),

        -- Occupancy rate
        (SELECT ROUND(
            COUNT(CASE WHEN status = 2 THEN 1 END)::NUMERIC /
            NULLIF(COUNT(*), 0) * 100, 2
        ) FROM beds WHERE branch_id = p_branch_id AND is_deleted = FALSE),

        -- Avg LOS
        (SELECT AVG(length_of_stay_days)
         FROM admissions
         WHERE branch_id = p_branch_id
           AND DATE(actual_discharge_date) = p_kpi_date),

        -- Total revenue (bills dated today)
        (SELECT COALESCE(SUM(total_amount), 0)
         FROM patient_bills
         WHERE branch_id = p_branch_id
           AND bill_date = p_kpi_date
           AND is_deleted = FALSE),

        -- Total collection
        (SELECT COALESCE(SUM(amount), 0)
         FROM payment_transactions
         WHERE branch_id = p_branch_id
           AND DATE(transaction_date) = p_kpi_date
           AND status = 1 AND transaction_type = 1),

        -- Outstanding
        (SELECT COALESCE(SUM(outstanding_amount), 0)
         FROM patient_bills
         WHERE branch_id = p_branch_id
           AND status IN (2,3)
           AND is_deleted = FALSE),

        -- Lab orders
        (SELECT COUNT(*) FROM lab_orders
         WHERE branch_id = p_branch_id
           AND DATE(ordered_at) = p_kpi_date
           AND is_deleted = FALSE),

        -- Lab results reported
        (SELECT COUNT(*) FROM lab_results lr
         JOIN lab_order_items loi ON lr.lab_order_item_id = loi.lab_order_item_id
         JOIN lab_orders lo ON loi.lab_order_id = lo.lab_order_id
         WHERE lo.branch_id = p_branch_id
           AND DATE(lr.published_at) = p_kpi_date),

        -- Appointments scheduled
        (SELECT COUNT(*) FROM appointments
         WHERE branch_id = p_branch_id
           AND appointment_date = p_kpi_date
           AND is_deleted = FALSE),

        -- Completed
        (SELECT COUNT(*) FROM appointments
         WHERE branch_id = p_branch_id
           AND appointment_date = p_kpi_date
           AND status = 5),

        -- Cancelled
        (SELECT COUNT(*) FROM appointments
         WHERE branch_id = p_branch_id
           AND appointment_date = p_kpi_date
           AND status = 6),

        -- No show
        (SELECT COUNT(*) FROM appointments
         WHERE branch_id = p_branch_id
           AND appointment_date = p_kpi_date
           AND status = 7),

        NOW()

    ON CONFLICT (tenant_id, branch_id, kpi_date)
    DO UPDATE SET
        -- Update all values on conflict (idempotent)
        beds_occupied = EXCLUDED.beds_occupied,
        total_collection = EXCLUDED.total_collection,
        calculated_at = NOW();
END;
$$ LANGUAGE plpgsql;
```

```sql
-- View: Lab TAT Compliance Report
CREATE OR REPLACE VIEW vw_lab_tat_compliance AS
SELECT
    lo.branch_id,
    tm.category_id,
    tc.category_name,
    tm.test_id,
    tm.test_name,
    tm.tat_hours AS target_tat_hours,
    COUNT(lr.lab_result_id) AS total_results,
    COUNT(CASE
        WHEN EXTRACT(EPOCH FROM (lr.published_at - ls.received_in_lab_at))
             / 3600.0 <= tm.tat_hours
        THEN 1 END
    ) AS within_tat_count,
    ROUND(
        COUNT(CASE
            WHEN EXTRACT(EPOCH FROM (lr.published_at - ls.received_in_lab_at))
                 / 3600.0 <= tm.tat_hours
            THEN 1 END
        )::NUMERIC / NULLIF(COUNT(lr.lab_result_id), 0) * 100,
        2
    ) AS tat_compliance_percentage,
    AVG(
        EXTRACT(EPOCH FROM (lr.published_at - ls.received_in_lab_at)) / 3600.0
    )::NUMERIC(8,2) AS avg_actual_tat_hours
FROM lab_results lr
JOIN lab_order_items loi ON lr.lab_order_item_id = loi.lab_order_item_id
JOIN lab_orders lo ON loi.lab_order_id = lo.lab_order_id
JOIN lab_samples ls ON lr.lab_sample_id = ls.lab_sample_id
JOIN test_master tm ON lr.test_id = tm.test_id
JOIN test_categories tc ON tm.category_id = tc.test_category_id
WHERE lr.published_at IS NOT NULL
  AND ls.received_in_lab_at IS NOT NULL
GROUP BY lo.branch_id, tm.category_id, tc.category_name,
         tm.test_id, tm.test_name, tm.tat_hours;

WHERE pb.is_deleted = FALSE
  AND pb.status NOT IN (5)  -- Exclude cancelled

         DATE_TRUNC('month', pb.bill_date),
```


## 2.9 Auto Journal Generation Trigger

```sql
-- Auto-create journal entry when bill is finalized
CREATE OR REPLACE FUNCTION fn_auto_journal_from_bill()
RETURNS TRIGGER AS $$
DECLARE
    v_period_id     UUID;
    v_fy_id         UUID;
    v_journal_id    UUID;
    v_jnl_number    VARCHAR(30);
    v_ar_account    UUID;
    v_revenue_acct  UUID;
    v_tax_account   UUID;
    v_seq           BIGINT;
BEGIN
    -- Only trigger when status changes to Paid (4)
    IF NEW.status = 4 AND (OLD.status IS NULL OR OLD.status <> 4) THEN

        -- Get current accounting period
        SELECT period_id, financial_year_id
        INTO v_period_id, v_fy_id
        FROM accounting_periods
        WHERE tenant_id = NEW.tenant_id
          AND start_date <= NEW.bill_date
          AND end_date >= NEW.bill_date
          AND status = 1  -- Open
        LIMIT 1;

        IF v_period_id IS NULL THEN
            RAISE EXCEPTION 'No open accounting period found for date: %', NEW.bill_date;
        END IF;

        -- Get journal number
        SELECT COALESCE(MAX(SUBSTRING(journal_number FROM '\d+$')::BIGINT),0)+1
        INTO v_seq
        FROM journal_entries
        WHERE tenant_id = NEW.tenant_id
          AND financial_year_id = v_fy_id;

        v_jnl_number := 'JV-' || TO_CHAR(NEW.bill_date, 'YYYYMM') ||
                        '-' || LPAD(v_seq::TEXT, 6, '0');

        -- Create journal entry
        v_journal_id := uuid_generate_v4();

        INSERT INTO journal_entries (
            journal_entry_id, tenant_id, branch_id,
            journal_number, entry_date, period_id, financial_year_id,
            journal_type, description,
            reference_type, reference_id, reference_number,
            total_debit, total_credit,
            status, is_auto_generated,
            created_at, created_by, updated_at, updated_by
        ) VALUES (
            v_journal_id, NEW.tenant_id, NEW.branch_id,
            v_jnl_number, NEW.bill_date, v_period_id, v_fy_id,
            2,  -- Sales Journal
            'Auto-Journal: Bill ' || NEW.bill_number,
            'patient_bill', NEW.bill_id, NEW.bill_number,
            NEW.total_amount, NEW.total_amount,
            2,  -- Posted directly
            TRUE,
            NOW(), NEW.updated_by, NOW(), NEW.updated_by
        );

        -- Line 1: Debit Accounts Receivable (or Cash if paid)
        INSERT INTO journal_entry_lines (
            journal_line_id, journal_entry_id, line_number,
            account_id, debit_amount, credit_amount,
            description, created_at, created_by
        ) VALUES (
            uuid_generate_v4(), v_journal_id, 1,
            (SELECT account_id FROM chart_of_accounts
             WHERE tenant_id = NEW.tenant_id
               AND account_code = 'AR-PATIENT' LIMIT 1),
            NEW.total_amount, 0,
            'Patient Bill: ' || NEW.bill_number,
            NOW(), NEW.updated_by
        );

        -- Line 2: Credit Revenue
        INSERT INTO journal_entry_lines (
            journal_line_id, journal_entry_id, line_number,
            account_id, debit_amount, credit_amount,
            description, created_at, created_by
        ) VALUES (
            uuid_generate_v4(), v_journal_id, 2,
            (SELECT account_id FROM chart_of_accounts
             WHERE tenant_id = NEW.tenant_id
               AND account_code = 'REV-MEDICAL' LIMIT 1),
            0, (NEW.total_amount - NEW.tax_amount),
            'Revenue: ' || NEW.bill_number,
            NOW(), NEW.updated_by
        );

        -- Line 3: Credit Tax (if applicable)
        IF NEW.tax_amount > 0 THEN
            INSERT INTO journal_entry_lines (
                journal_line_id, journal_entry_id, line_number,
                account_id, debit_amount, credit_amount,
                description, created_at, created_by
            ) VALUES (
                uuid_generate_v4(), v_journal_id, 3,
                (SELECT account_id FROM chart_of_accounts
                 WHERE tenant_id = NEW.tenant_id
                   AND account_code = 'TAX-OUTPUT-GST' LIMIT 1),
                0, NEW.tax_amount,
                'GST: ' || NEW.bill_number,
                NOW(), NEW.updated_by
            );
        END IF;

        -- Update bill with journal entry reference
        UPDATE patient_bills
        SET metadata = metadata || jsonb_build_object(
            'journal_entry_id', v_journal_id,
            'journal_number', v_jnl_number
        )
        WHERE bill_id = NEW.bill_id;

    END IF;

    RETURN NEW;
END;
$$ LANGUAGE plpgsql;

CREATE TRIGGER trg_auto_journal_from_bill
    AFTER UPDATE ON patient_bills
    FOR EACH ROW EXECUTE FUNCTION fn_auto_journal_from_bill();
```

```sql
-- ETL Procedure: Load Daily Fact Data
CREATE OR REPLACE PROCEDURE dw.sp_load_daily_facts(
    p_load_date DATE
)
LANGUAGE plpgsql AS $$
DECLARE
    v_date_key INTEGER;
BEGIN
    v_date_key := TO_CHAR(p_load_date, 'YYYYMMDD')::INTEGER;

    -- Ensure date exists in dim_date
    INSERT INTO dw.dim_date (
        date_key, full_date, day_of_week, day_name,
        day_of_month, week_of_year, month_number, month_name,
        quarter_number, year_number, is_weekend, financial_year
    )
    VALUES (
        v_date_key, p_load_date,
        EXTRACT(DOW FROM p_load_date)::SMALLINT,
        TO_CHAR(p_load_date, 'Day'),
        EXTRACT(DAY FROM p_load_date)::SMALLINT,
        EXTRACT(WEEK FROM p_load_date)::SMALLINT,
        EXTRACT(MONTH FROM p_load_date)::SMALLINT,
        TO_CHAR(p_load_date, 'Month'),
        EXTRACT(QUARTER FROM p_load_date)::SMALLINT,
        EXTRACT(YEAR FROM p_load_date)::SMALLINT,
        EXTRACT(DOW FROM p_load_date) IN (0,6),
        TO_CHAR(p_load_date, 'YYYY') || '-' ||
            TO_CHAR(p_load_date + INTERVAL '1 year', 'YY')
    )
    ON CONFLICT (full_date) DO NOTHING;

    -- Delete existing data for this date (idempotent)
    DELETE FROM dw.fact_billing WHERE date_key = v_date_key;
    DELETE FROM dw.fact_patient_visits WHERE date_key = v_date_key;
    DELETE FROM dw.fact_bed_occupancy WHERE date_key = v_date_key;

    -- Load billing facts
    INSERT INTO dw.fact_billing (
        date_key, patient_key, branch_key, service_key,
        bill_id, bill_line_item_id, tenant_id,
        bill_type, quantity, gross_amount, discount_amount,
        tax_amount, net_amount, is_insurance, loaded_at
    )
    SELECT
        v_date_key,
        dp.patient_key,
        db.branch_key,
        ds.service_key,
        pb.bill_id,
        bli.bill_line_item_id,
        pb.tenant_id,
        pb.bill_type,
        bli.quantity,
        bli.unit_price * bli.quantity,
        bli.discount_amount,
        bli.tax_amount,
        bli.line_total,
        pb.is_insurance,
        NOW()
    FROM bill_line_items bli
    JOIN patient_bills pb ON bli.bill_id = pb.bill_id
    JOIN dw.dim_patient dp ON pb.patient_id = dp.patient_id AND dp.is_current
    JOIN dw.dim_branch db ON pb.branch_id = db.branch_id AND db.is_current
    LEFT JOIN dw.dim_service ds ON bli.service_id = ds.service_id AND ds.is_current
    WHERE pb.bill_date = p_load_date
      AND pb.is_deleted = FALSE
      AND pb.status NOT IN (5);  -- Exclude cancelled

    -- Load bed occupancy snapshot (end of day)
    INSERT INTO dw.fact_bed_occupancy (
        date_key, branch_key, ward_id, ward_name, ward_type,
        total_beds, occupied_beds, vacant_beds, occupancy_rate,
        loaded_at
    )
    SELECT
        v_date_key,
        db.branch_key,
        w.ward_id,
        w.ward_name,
        w.ward_type,
        COUNT(b.bed_id),
        COUNT(CASE WHEN b.status = 2 THEN 1 END),
        COUNT(CASE WHEN b.status = 1 THEN 1 END),
        ROUND(COUNT(CASE WHEN b.status = 2 THEN 1 END)::NUMERIC /
              NULLIF(COUNT(b.bed_id), 0) * 100, 2),
        NOW()
    FROM beds b
    JOIN wards w ON b.ward_id = w.ward_id
    JOIN dw.dim_branch db ON b.branch_id = db.branch_id AND db.is_current
    WHERE b.is_deleted = FALSE AND b.is_active = TRUE
    GROUP BY db.branch_key, w.ward_id, w.ward_name, w.ward_type
    ON CONFLICT (date_key, branch_key, ward_id) DO UPDATE
        SET occupied_beds = EXCLUDED.occupied_beds,
            occupancy_rate = EXCLUDED.occupancy_rate,
            loaded_at = NOW();

    RAISE NOTICE 'Daily facts loaded for date: %', p_load_date;
END;
$$;
```

```sql
-- Function to detect brute force login
CREATE OR REPLACE FUNCTION fn_check_brute_force()
RETURNS TRIGGER AS $$
DECLARE
    v_failed_count  INTEGER;
BEGIN
    IF NEW.is_successful = FALSE THEN
        -- Count failed attempts in last 15 minutes
        SELECT COUNT(*) INTO v_failed_count
        FROM login_history
        WHERE user_id = NEW.user_id
          AND is_successful = FALSE
          AND login_at >= NOW() - INTERVAL '15 minutes';

        IF v_failed_count >= 5 THEN
            -- Create security alert
            INSERT INTO security_alerts (
                tenant_id, alert_type, severity, user_id,
                ip_address, description, evidence,
                auto_actions_taken, triggered_at, status
            )
            SELECT
                u.tenant_id, 'BruteForce', 3, NEW.user_id,
                NEW.ip_address,
                'Multiple failed login attempts detected for user: ' ||
                    u.username,
                jsonb_build_object(
                    'failed_attempts', v_failed_count,
                    'time_window', '15 minutes',
                    'ip_address', NEW.ip_address::TEXT
                ),
                ARRAY['account_locked'],
                NOW(), 1
            FROM users u WHERE u.user_id = NEW.user_id;

            -- Lock the account
            UPDATE users
            SET account_locked_at = NOW(),
                account_locked_until = NOW() + INTERVAL '30 minutes',
                updated_at = NOW()
            WHERE user_id = NEW.user_id;
        END IF;
    END IF;

    RETURN NEW;
END;
$$ LANGUAGE plpgsql;

CREATE TRIGGER trg_check_brute_force
    AFTER INSERT ON login_history
    FOR EACH ROW EXECUTE FUNCTION fn_check_brute_force();
```
