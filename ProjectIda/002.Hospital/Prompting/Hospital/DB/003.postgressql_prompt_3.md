    -- VOLUME: 10-1000 rows | PARTITION: No | ENCRYPT: No
    ════════════════════════════════════════════════════════════════
    CREATE TABLE tenants (
        tenant_id               UUID            PRIMARY KEY
                                DEFAULT uuid_generate_v4(),
        tenant_name             VARCHAR(255)    NOT NULL,
        tenant_code             VARCHAR(20)     NOT NULL,
        legal_entity_name       VARCHAR(300),
        company_registration    VARCHAR(100),

        country_iso             iso_country     NOT NULL DEFAULT 'IN',
        primary_currency        iso_currency    NOT NULL DEFAULT 'INR',
        primary_language        iso_language    NOT NULL DEFAULT 'en',
        contact_email           email_address   NOT NULL,
        contact_phone           phone_e164      NOT NULL,
        website_url             VARCHAR(500),
        logo_url                VARCHAR(1000),
        subscription_tier       SMALLINT        NOT NULL DEFAULT 1
                                CHECK (subscription_tier IN (1,2,3,4)),
                                -- 1=Starter, 2=Professional, 3=Enterprise, 4=Custom
        subscription_status     SMALLINT        NOT NULL DEFAULT 1
                                CHECK (subscription_status IN (1,2,3,4,5)),
                                -- 1=Active, 2=Trial, 3=Suspended, 4=Terminated, 5=Pending
        subscription_start_date DATE,
        subscription_end_date   DATE,
        max_branches            SMALLINT        NOT NULL DEFAULT 3,
        max_users               INTEGER         NOT NULL DEFAULT 500,
        max_patients_per_year   INTEGER         NOT NULL DEFAULT 100000,
        feature_flags           JSONB           NOT NULL DEFAULT '{}',
        -- e.g., {"telemedicine":true,"blood_bank":true,"canteen":true}
        nabh_accredited         BOOLEAN         NOT NULL DEFAULT FALSE,
        jci_accredited          BOOLEAN         NOT NULL DEFAULT FALSE,
        hipaa_required          BOOLEAN         NOT NULL DEFAULT FALSE,
        gdpr_required           BOOLEAN         NOT NULL DEFAULT FALSE,
        timezone                VARCHAR(60)     NOT NULL DEFAULT 'Asia/Kolkata',
        date_format             VARCHAR(20)     NOT NULL DEFAULT 'DD-MM-YYYY',
        mrn_prefix              VARCHAR(10),    -- e.g., 'APC', 'MAX'
        mrn_padding             SMALLINT        NOT NULL DEFAULT 8,
        -- Settings for business logic
        settings                JSONB           NOT NULL DEFAULT '{}',
        -- AUDIT
        created_at              TIMESTAMPTZ     NOT NULL DEFAULT NOW(),
        updated_at              TIMESTAMPTZ     NOT NULL DEFAULT NOW(),
        created_by              UUID            NOT NULL,
        is_deleted              BOOLEAN         NOT NULL DEFAULT FALSE,
        deleted_at              TIMESTAMPTZ,
        deleted_by              UUID,
        -- CONSTRAINTS
        CONSTRAINT uq_tenants_code    UNIQUE (tenant_code),
        CONSTRAINT uq_tenants_name    UNIQUE (tenant_name),

            subscription_end_date IS NULL OR
            subscription_end_date >= subscription_start_date

        'Multi-tenancy root. Each row = one hospital group. All clinical data is isolated by tenant_id.';

    CREATE TABLE patients (
        patient_id              UUID            NOT NULL DEFAULT uuid_generate_v4(),
        tenant_id               UUID            NOT NULL,
        branch_id               UUID            NOT NULL,
        -- MEDICAL RECORD NUMBER
        mrn                     VARCHAR(50)     NOT NULL,
        -- e.g., 'APC00012345' — format = tenant.mrn_prefix + zero-padded sequence
        mrn_global              VARCHAR(50),
        -- Group-wide MRN for multi-branch patients (shared identity)
        -- CORE DEMOGRAPHICS
        first_name              VARCHAR(100)    NOT NULL,
        middle_name             VARCHAR(100),
        last_name               VARCHAR(100)    NOT NULL,
        full_name               VARCHAR(350)
            GENERATED ALWAYS AS (
                TRIM(first_name || ' ' || COALESCE(middle_name || ' ', '') || last_name)
            ) STORED,
        date_of_birth           DATE            NOT NULL
                                CHECK (date_of_birth BETWEEN '1900-01-01' AND CURRENT_DATE),
        -- Age computed columns for query convenience
        age_years               SMALLINT
            GENERATED ALWAYS AS (
                EXTRACT(YEAR FROM AGE(CURRENT_DATE, date_of_birth))::SMALLINT
            ) STORED,
        age_months              SMALLINT
            GENERATED ALWAYS AS (
                (EXTRACT(YEAR FROM AGE(CURRENT_DATE, date_of_birth)) * 12 +
                 EXTRACT(MONTH FROM AGE(CURRENT_DATE, date_of_birth)))::SMALLINT
            ) STORED,
        age_days                INTEGER
            GENERATED ALWAYS AS (
                (CURRENT_DATE - date_of_birth)
            ) STORED,
        is_age_estimated        BOOLEAN         NOT NULL DEFAULT FALSE,
        -- For neonates or unknown DOB patients
        gender                  SMALLINT        NOT NULL
                                CHECK (gender IN (1,2,3,4)),

        rh_factor               SMALLINT

                                CHECK (marital_status IN (1,2,3,4,5)),
        -- IDENTIFICATION (PII — ENCRYPTED)

                                CHECK (national_id_type IN (1,2,3,4,5,6)),
                                -- 1=Passport, 2=Aadhaar, 3=Voter ID,
                                -- 4=Driving License, 5=PAN, 6=Other

        -- Stored as pgp_sym_encrypt(national_id, app_key)

        national_id_verified_at TIMESTAMPTZ,
        national_id_verified_by UUID,
        abha_number             VARCHAR(17),
        -- ABDM Health ID (India): XX-XXXX-XXXX-XXXX
        abha_address            VARCHAR(100),

        primary_email           email_address,
        -- ANTHROPOMETRIC (for dosing, BMI calculation)

        weight_kg               NUMERIC(6,2),

        cause_of_death          TEXT,

        ),
        -- REGISTRATION
        registration_date       DATE            NOT NULL DEFAULT CURRENT_DATE,
        registration_type       SMALLINT        NOT NULL DEFAULT 1
                                CHECK (registration_type IN (1,2,3,4)),
                                -- 1=Walk-in, 2=Online, 3=Referred, 4=Emergency

        referral_source_id      UUID,
        -- PREFERENCES
        preferred_language      iso_language    DEFAULT 'en',
        preferred_communication_channel SMALLINT DEFAULT 1
                                CHECK (preferred_communication_channel IN (1,2,3,4)),
                                -- 1=SMS, 2=Email, 3=WhatsApp, 4=App

    CREATE TABLE consultations (

    -- TABLE: vital_signs
    -- PURPOSE: Patient vitals per encounter
    -- VOLUME: 10,000,000+ per year | PARTITION: BY RANGE(recorded_at)
    ════════════════════════════════════════════════════════════════
    CREATE TABLE vital_signs (
        vital_id                UUID            NOT NULL DEFAULT uuid_generate_v4(),
        tenant_id               UUID            NOT NULL,
        branch_id               UUID            NOT NULL,
        patient_id              UUID            NOT NULL,
        consultation_id         UUID,
        admission_id            UUID,
        -- VITALS
        recorded_at             TIMESTAMPTZ     NOT NULL DEFAULT NOW(),
        recorded_by             UUID            NOT NULL,
        -- CARDIOVASCULAR
        systolic_bp             SMALLINT
                                CHECK (systolic_bp BETWEEN 40 AND 300),
        diastolic_bp            SMALLINT
                                CHECK (diastolic_bp BETWEEN 20 AND 200),
        pulse_rate              SMALLINT
                                CHECK (pulse_rate BETWEEN 20 AND 300),
        -- Neonates can have HR up to 180+
        spo2_percent            NUMERIC(4,1)
                                CHECK (spo2_percent BETWEEN 50 AND 100),
        -- RESPIRATORY
        respiratory_rate        SMALLINT
                                CHECK (respiratory_rate BETWEEN 5 AND 80),
        -- TEMPERATURE
        temperature_celsius     NUMERIC(4,1)
                                CHECK (temperature_celsius BETWEEN 30.0 AND 45.0),
        temperature_site        SMALLINT
                                CHECK (temperature_site IN (1,2,3,4)),
                                -- 1=Oral, 2=Axillary, 3=Rectal, 4=Tympanic
        -- ANTHROPOMETRIC
        height_cm               NUMERIC(5,1)
                                CHECK (height_cm BETWEEN 20 AND 280),
        weight_kg               NUMERIC(6,2)
                                CHECK (weight_kg BETWEEN 0.3 AND 500),
        bmi                     NUMERIC(4,1)
            GENERATED ALWAYS AS (
                CASE WHEN height_cm > 0 AND weight_kg > 0
                THEN ROUND(weight_kg / POWER(height_cm / 100.0, 2), 1)
                ELSE NULL END
            ) STORED,
        head_circumference_cm   NUMERIC(4,1),
        -- Pediatrics
        mid_arm_circumference_cm NUMERIC(4,1),
        -- Malnutrition screening
        -- NEUROLOGICAL
        glasgow_coma_scale      SMALLINT
                                CHECK (glasgow_coma_scale BETWEEN 3 AND 15),
        pain_score              SMALLINT
                                CHECK (pain_score BETWEEN 0 AND 10),
        -- NRS pain scale 0-10
        -- GLUCOSE
        blood_glucose_mmol      NUMERIC(5,2)
                                CHECK (blood_glucose_mmol BETWEEN 1 AND 80),
        glucose_measurement_type SMALLINT
                                CHECK (glucose_measurement_type IN (1,2,3)),
                                -- 1=Fasting, 2=Post-prandial, 3=Random
        -- CRITICAL FLAGS (auto-set by trigger)
        is_critical             BOOLEAN         NOT NULL DEFAULT FALSE,
        critical_parameters     TEXT[],
        -- ['systolic_bp', 'spo2_percent'] — parameters that triggered critical flag
        critical_alert_sent_at  TIMESTAMPTZ,
        critical_acknowledged_by UUID,
        critical_acknowledged_at TIMESTAMPTZ,
        -- NOTES
        vital_notes             TEXT,
        entry_method            SMALLINT        NOT NULL DEFAULT 1
                                CHECK (entry_method IN (1,2,3)),
                                -- 1=Manual entry, 2=Device (auto), 3=Patient-reported
        device_identifier       VARCHAR(100),
        -- If auto-capture, which device/sensor ID
        -- AUDIT
        created_at              TIMESTAMPTZ     NOT NULL DEFAULT NOW(),
        is_deleted              BOOLEAN         NOT NULL DEFAULT FALSE,
        PRIMARY KEY (vital_id, recorded_at),
        CONSTRAINT fk_vitals_tenant  FOREIGN KEY (tenant_id)
            REFERENCES tenants(tenant_id) ON DELETE RESTRICT,
        CONSTRAINT fk_vitals_patient FOREIGN KEY (patient_id)
            REFERENCES patients(patient_id) ON DELETE RESTRICT,
        CONSTRAINT fk_vitals_consult FOREIGN KEY (consultation_id)
            REFERENCES consultations(consultation_id) ON DELETE RESTRICT,
        CONSTRAINT fk_vitals_recorder FOREIGN KEY (recorded_by)
            REFERENCES users(user_id) ON DELETE RESTRICT
    ) PARTITION BY RANGE (recorded_at);

    COMMENT ON TABLE vital_signs IS
        'All patient vital sign observations. Trigger auto-sets is_critical on abnormal values.';

    CREATE TABLE vital_signs_2025_01 PARTITION OF vital_signs
        FOR VALUES FROM ('2025-01-01') TO ('2025-02-01');

    CREATE INDEX idx_vitals_patient_time ON vital_signs(
