    -- ════════════════════════════════════════════════════════════
    -- TABLE: rx_prescriptions
    -- DOMAIN: RX Management
    -- PURPOSE: e-Prescription master record — digital, signed, versioned
    -- VOLUME: 2,000,000+ per year
    -- PARTITION: BY RANGE on prescribed_at (monthly)
    -- ════════════════════════════════════════════════════════════

    CREATE TABLE rx_prescriptions (
        prescription_id         UUID            NOT NULL DEFAULT uuid_generate_v4(),
        tenant_id               UUID            NOT NULL REFERENCES tenants(tenant_id) ON DELETE RESTRICT,
        branch_id               UUID            NOT NULL REFERENCES branches(branch_id) ON DELETE RESTRICT,

        rx_number               VARCHAR(30)     NOT NULL,   -- From seq_rx_number
        prescription_type       SMALLINT        NOT NULL,
                                CHECK (prescription_type IN (1,2,3)),
                                -- 1=OPD, 2=IPD, 3=Emergency

        -- Links
        patient_id              UUID            NOT NULL REFERENCES patients(patient_id) ON DELETE RESTRICT,
        encounter_id            UUID,           -- FK to consultation or admission
        prescriber_id           UUID            NOT NULL REFERENCES users(user_id) ON DELETE RESTRICT,

        -- Validity
        prescribed_at           TIMESTAMPTZ     NOT NULL DEFAULT NOW(),
        valid_until             TIMESTAMPTZ     NOT NULL,
        is_electronic           BOOLEAN         NOT NULL DEFAULT TRUE,

        -- Digital Signature
        digital_signature_hash  BYTEA,          -- Prescriber digital signature
        signed_at               TIMESTAMPTZ,
        signing_certificate_id  VARCHAR(100),   -- Certificate serial number

        -- Status
        prescription_status     SMALLINT        NOT NULL DEFAULT 1,
                                CHECK (prescription_status IN (1,2,3,4,5)),
                                -- 1=Active, 2=Partially Dispensed, 3=Fully Dispensed,
                                -- 4=Cancelled, 5=Expired

        cancellation_reason     TEXT,
        cancelled_by            UUID REFERENCES users(user_id),
        cancelled_at            TIMESTAMPTZ,

        -- Refill
        max_refills_allowed     SMALLINT        NOT NULL DEFAULT 0,
        refills_used            SMALLINT        NOT NULL DEFAULT 0,
                                CHECK (refills_used <= max_refills_allowed),

        -- Clinical Notes
        clinical_notes          TEXT,
        diagnosis_codes         TEXT[],         -- Array of ICD-10 codes justifying Rx

        -- Prior Authorization
        requires_prior_auth     BOOLEAN         NOT NULL DEFAULT FALSE,
        prior_auth_number       VARCHAR(100),
        prior_auth_approved_at  TIMESTAMPTZ,

        -- Reconciliation
        is_reconciled           BOOLEAN         NOT NULL DEFAULT FALSE,
        reconciled_by           UUID REFERENCES users(user_id),
        reconciled_at           TIMESTAMPTZ,

        -- Polypharmacy flag
        polypharmacy_flag       BOOLEAN         NOT NULL DEFAULT FALSE,
        -- Set by trigger if patient has ≥5 concurrent active medications

        -- Audit
        created_at              TIMESTAMPTZ     NOT NULL DEFAULT NOW(),
        updated_at              TIMESTAMPTZ     NOT NULL DEFAULT NOW(),
        created_by              UUID            NOT NULL REFERENCES users(user_id),
        updated_by              UUID            NOT NULL REFERENCES users(user_id),
        is_deleted              BOOLEAN         NOT NULL DEFAULT FALSE,
        deleted_at              TIMESTAMPTZ,
        deleted_by              UUID REFERENCES users(user_id),

        PRIMARY KEY (prescription_id, prescribed_at),

        CONSTRAINT uq_rx_number UNIQUE (tenant_id, rx_number),
        CONSTRAINT chk_rx_valid_period CHECK (valid_until > prescribed_at),
        CONSTRAINT chk_rx_signature CHECK (
            (is_electronic = TRUE AND digital_signature_hash IS NOT NULL AND signed_at IS NOT NULL) OR
            (is_electronic = FALSE)
        )
    ) PARTITION BY RANGE (prescribed_at);

    -- Indexes
    CREATE INDEX idx_rx_patient_status ON rx_prescriptions(tenant_id, patient_id, prescription_status)
        WHERE is_deleted = FALSE;
    CREATE INDEX idx_rx_prescriber ON rx_prescriptions(tenant_id, prescriber_id, prescribed_at DESC)
        WHERE is_deleted = FALSE;
    CREATE INDEX idx_rx_active_valid ON rx_prescriptions(tenant_id, patient_id, valid_until DESC)
        WHERE prescription_status IN (1,2) AND is_deleted = FALSE;
    CREATE INDEX idx_rx_prior_auth ON rx_prescriptions(tenant_id, requires_prior_auth, prescription_status)
        WHERE requires_prior_auth = TRUE AND is_deleted = FALSE;

    -- TABLE: rx_prescription_items

    CREATE TABLE rx_prescription_items (

        tenant_id               UUID            NOT NULL REFERENCES tenants(tenant_id),

        drug_id                 UUID            NOT NULL,   -- FK to drug master
        drug_name               VARCHAR(255)    NOT NULL,   -- Denormalized snapshot
        drug_generic_name       VARCHAR(255),
        drug_code               VARCHAR(50),                -- ATC/NDC/local code

        -- Dosage
        dose_amount             NUMERIC(10,3)   NOT NULL,
        dose_unit               VARCHAR(30)     NOT NULL,   -- mg, ml, IU, etc.
        dose_form               VARCHAR(50)     NOT NULL,   -- tablet, capsule, syrup, injection
        route_of_admin          SMALLINT        NOT NULL,
                                CHECK (route_of_admin IN (1,2,3,4,5,6,7,8)),
                                -- 1=Oral, 2=IV, 3=IM, 4=SC, 5=Topical, 6=Inhaled, 7=SL, 8=Rectal

        -- Frequency & Duration
        frequency_code          VARCHAR(20)     NOT NULL,   -- BD, TDS, OD, QID, etc.

        instructions            TEXT,           -- Take with food, etc.

        -- Quantity
        qty_prescribed          NUMERIC(10,2)   NOT NULL,
        qty_dispensed           NUMERIC(10,2)   NOT NULL DEFAULT 0,
                                CHECK (qty_dispensed <= qty_prescribed),
        qty_remaining           NUMERIC(10,2)
            GENERATED ALWAYS AS (qty_prescribed - qty_dispensed) STORED,

        -- Controlled Substance

        prior_auth_status       SMALLINT        CHECK (prior_auth_status IN (1,2,3)),
                                -- 1=Pending, 2=Approved, 3=Denied

        -- Status
        item_status             SMALLINT        NOT NULL DEFAULT 1,
                                CHECK (item_status IN (1,2,3,4)),
                                -- 1=Active, 2=Dispensed, 3=Cancelled, 4=On Hold

        substitution_allowed    BOOLEAN         NOT NULL DEFAULT FALSE,

        -- Audit
        created_at              TIMESTAMPTZ     NOT NULL DEFAULT NOW(),
        updated_at              TIMESTAMPTZ     NOT NULL DEFAULT NOW(),
        created_by              UUID            NOT NULL REFERENCES users(user_id),
        updated_by              UUID            NOT NULL REFERENCES users(user_id),

            (is_controlled_substance = TRUE AND witness_user_id IS NOT NULL) OR
            (is_controlled_substance = FALSE)

    -- ════════════════════════════════════════════════════════════
    -- TABLE: reagent_lots
    -- PURPOSE: Individual reagent lots/batches with expiry tracking
    -- CRITICAL: FEFO enforcement, QC gate before clinical use
    -- ════════════════════════════════════════════════════════════

    CREATE TABLE reagent_lots (
        lot_id                  UUID            PRIMARY KEY DEFAULT uuid_generate_v4(),
        reagent_id              UUID            NOT NULL REFERENCES reagent_master(reagent_id) ON DELETE RESTRICT,
        tenant_id               UUID            NOT NULL REFERENCES tenants(tenant_id),
        branch_id               UUID            NOT NULL REFERENCES branches(branch_id),

        lot_number              VARCHAR(100)    NOT NULL,
        manufacturer_name       VARCHAR(255)    NOT NULL,
        supplier_id             UUID,           -- FK to suppliers

        -- Dates
        manufacture_date        DATE,
        expiry_date             DATE            NOT NULL,
        received_date           DATE            NOT NULL DEFAULT CURRENT_DATE,
        opened_date             DATE,
        open_stability_days     SMALLINT,       -- Stability after opening (in-use expiry)
        in_use_expiry_date      DATE
            GENERATED ALWAYS AS (
                CASE WHEN opened_date IS NOT NULL AND open_stability_days IS NOT NULL
                     THEN opened_date + open_stability_days
                     ELSE NULL
                END
            ) STORED,

        -- Effective expiry (minimum of lot expiry and in-use expiry)
        effective_expiry_date   DATE
            GENERATED ALWAYS AS (
                CASE
                    WHEN opened_date IS NOT NULL AND open_stability_days IS NOT NULL
                    THEN LEAST(expiry_date, opened_date + open_stability_days)
                    ELSE expiry_date
                END
            ) STORED,

        -- Quantity
        quantity_received       NUMERIC(10,2)   NOT NULL,
        quantity_available      NUMERIC(10,2)   NOT NULL,
                                CHECK (quantity_available >= 0),
        unit_of_measure         VARCHAR(30)     NOT NULL,

        -- QC Status (gate before clinical use)
        qc_status               SMALLINT        NOT NULL DEFAULT 1,
                                CHECK (qc_status IN (1,2,3,4)),
                                -- 1=Pending QC, 2=QC Passed, 3=QC Failed, 4=Released for Use

        qc_verified_by          UUID REFERENCES users(user_id),
        qc_verified_at          TIMESTAMPTZ,
        qc_failure_reason       TEXT,

        -- Lot Status
        lot_status              SMALLINT        NOT NULL DEFAULT 1,
                                CHECK (lot_status IN (1,2,3,4,5)),
                                -- 1=Quarantine, 2=Released, 3=In Use, 4=Exhausted, 5=Recalled/Discarded

        recall_reason           TEXT,
        discarded_at            TIMESTAMPTZ,
        discarded_by            UUID REFERENCES users(user_id),

        -- Storage Location
        storage_location        VARCHAR(100),
        refrigerator_id         UUID,           -- FK to asset_equipment

        -- Cost
        unit_cost               currency_type,
        currency_code           VARCHAR(3)      NOT NULL DEFAULT 'USD',

        -- Audit
        created_at              TIMESTAMPTZ     NOT NULL DEFAULT NOW(),
        updated_at              TIMESTAMPTZ     NOT NULL DEFAULT NOW(),
        created_by              UUID            NOT NULL REFERENCES users(user_id),
        updated_by              UUID            NOT NULL REFERENCES users(user_id),

        CONSTRAINT uq_lot_number UNIQUE (tenant_id, branch_id, reagent_id, lot_number),
        CONSTRAINT chk_lot_expiry CHECK (expiry_date > COALESCE(manufacture_date, '1900-01-01'::DATE)),
        CONSTRAINT chk_lot_qty CHECK (quantity_available <= quantity_received),
        CONSTRAINT chk_qc_gate CHECK (
            -- Cannot release for clinical use without QC pass
            (lot_status = 4 AND qc_status = 4) OR (lot_status != 4)
        ),
        CONSTRAINT chk_expiry_block CHECK (
            -- Expired lot cannot be In Use or Released
            NOT (lot_status IN (2,3,4) AND effective_expiry_date < CURRENT_DATE)
        )
    );

        westgard_1_2s           BOOLEAN         NOT NULL DEFAULT FALSE,  -- Warning: 1 value > 2SD

        westgard_r_4s           BOOLEAN         NOT NULL DEFAULT FALSE,  -- Rejection: range > 4SD
        westgard_4_1s           BOOLEAN         NOT NULL DEFAULT FALSE,  -- Warning: 4 consecutive > 1SD
        westgard_10x            BOOLEAN         NOT NULL DEFAULT FALSE,  -- Rejection: 10 on same side of mean

        -- Overall QC Decision
        qc_passed               BOOLEAN         NOT NULL,
        failure_rules           TEXT[],         -- Which Westgard rules fired
        action_taken            TEXT,

        -- If failed: block clinical use
        clinical_use_blocked    BOOLEAN         NOT NULL DEFAULT FALSE,

        unblocked_reason        TEXT,

        performed_at            TIMESTAMPTZ     NOT NULL DEFAULT NOW(),

4.3 BUSINESS RULE TRIGGERS (NEW — v3.0)

    a) trg_rx_polypharmacy_check
       BEFORE INSERT ON rx_prescriptions
       Count active medications for patient; set polypharmacy_flag = TRUE if ≥ 5

    b) trg_rx_refill_limit_check
       BEFORE INSERT ON rx_refill_requests
       Verify refills_used < max_refills_allowed; raise EXCEPTION otherwise

    c) trg_reagent_expiry_block
       BEFORE INSERT ON reagent_usage_logs
       Check effective_expiry_date on lot; RAISE EXCEPTION if expired

    d) trg_reagent_qc_gate
       BEFORE INSERT ON reagent_usage_logs
       Check qc_status = 4 (Released) on lot; RAISE EXCEPTION otherwise

    e) trg_diet_nbm_block
       BEFORE INSERT ON meal_deliveries
       Check is_nil_by_mouth on active diet_order; RAISE EXCEPTION if TRUE

    f) trg_diet_allergen_check
       BEFORE INSERT ON meal_deliveries
       Cross-check menu_item allergens against patient_allergies; log warning/block

    g) trg_blood_tti_gate
       BEFORE UPDATE ON blood_units (when status changes to 2=Available)
       Verify tti_status = 2 (Negative); RAISE EXCEPTION otherwise

    h) trg_blood_expiry_lock
       BEFORE UPDATE ON blood_units
       Prevent status change to 2,3,4 if expiry_date < CURRENT_DATE

    i) trg_blood_unit_issued_once
       BEFORE INSERT ON blood_issuances
       Verify unit_id has no existing issuance; RAISE EXCEPTION on duplicate

    j) trg_ambulance_maintenance_check
       BEFORE INSERT ON ambulance_trips
       Verify vehicle has no overdue maintenance; RAISE WARNING or EXCEPTION

    k) trg_canteen_shift_balance
       BEFORE UPDATE ON canteen_shift_closings (status → 3=Closed)
       Compute cash_variance; flag discrepancy if ABS(variance) > threshold

    l) trg_laundry_sla_breach_detect
       AFTER UPDATE ON laundry_batches
       If actual_return_at > expected_return_at, set sla_breached = TRUE
       and insert escalation record to system_notifications

    m) trg_stock_fefo_enforce (Pharmacy & Reagents)
       BEFORE INSERT ON dispensing_details / reagent_usage_logs
       Selects lot with earliest effective_expiry_date with quantity_available > 0
