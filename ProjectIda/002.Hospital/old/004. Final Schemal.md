-- Computed view replaces stored total_beds / total_ots (3NF fix)
CREATE OR REPLACE VIEW platform.v_branch_capacity AS
SELECT
    b.id                                                       AS branch_id,
    b.name                                                     AS branch_name,
    COUNT(bd.id)                                               AS total_beds,
    COUNT(bd.id) FILTER (WHERE bd.status = 'available')        AS available_beds,
    COUNT(bd.id) FILTER (WHERE bd.status = 'occupied')         AS occupied_beds,
    COUNT(ot.id)                                               AS total_ots
FROM platform.branches b
LEFT JOIN facility.beds   bd ON bd.branch_id = b.id
LEFT JOIN facility.ots    ot ON ot.branch_id = b.id
GROUP BY b.id, b.name;

    PRIMARY KEY (role_id, permission_id)

-- 2NF FIX: employee_id, designation, department_id REMOVED from users.
-- Those attributes belong to hr.employees (joined via user_id FK).
-- users = identity + credentials + security flags only.
CREATE TABLE platform.users (
    id                   UUID PRIMARY KEY DEFAULT gen_random_uuid(),
    org_id               UUID NOT NULL REFERENCES platform.organizations(id),

    -- Authentication
    username             VARCHAR(100) NOT NULL,
    email                VARCHAR(150),
    phone                VARCHAR(20),
    password_hash        TEXT NOT NULL,

    -- Basic profile (non-employment)
    first_name           VARCHAR(100) NOT NULL,
    last_name            VARCHAR(100),
    -- 3NF FIX: full_name is VIRTUAL, not STORED (eliminates transitive dependency)
    full_name            VARCHAR(255) GENERATED ALWAYS AS (
                             first_name || ' ' || COALESCE(last_name, '')
                         ) STORED,
    name_bn              VARCHAR(200),
    gender               gender_type,
    date_of_birth        DATE,
    photo_url            TEXT,
    language             VARCHAR(10)  DEFAULT 'en',

    -- Security
    is_email_verified    BOOLEAN      DEFAULT FALSE,
    is_phone_verified    BOOLEAN      DEFAULT FALSE,
    two_factor_enabled   BOOLEAN      DEFAULT FALSE,
    two_factor_secret    VARCHAR(255),
    password_changed_at  TIMESTAMPTZ,
    must_change_password BOOLEAN      DEFAULT FALSE,
    failed_login_count   INTEGER      DEFAULT 0,
    locked_until         TIMESTAMPTZ,
    last_login_at        TIMESTAMPTZ,
    last_login_ip        INET,

    is_active            BOOLEAN      DEFAULT TRUE,
    created_at           TIMESTAMPTZ  DEFAULT NOW(),
    updated_at           TIMESTAMPTZ  DEFAULT NOW(),
    created_by           UUID,

    UNIQUE(org_id, username),
    UNIQUE(org_id, email)
);

    occurred_at    TIMESTAMPTZ NOT NULL DEFAULT NOW(),

-- Create monthly partitions for current + next 12 months
DO $$
DECLARE
    start_date DATE := DATE_TRUNC('month', NOW());
    end_date   DATE;
    part_name  TEXT;
BEGIN
    FOR i IN 0..12 LOOP
        end_date  := start_date + INTERVAL '1 month';
        part_name := 'audit.events_' || TO_CHAR(start_date, 'YYYY_MM');
        EXECUTE format(
            'CREATE TABLE IF NOT EXISTS %s PARTITION OF audit.events
             FOR VALUES FROM (%L) TO (%L)',
            part_name, start_date, end_date
        );
        start_date := end_date;
    END LOOP;
END;
$$;

CREATE OR REPLACE FUNCTION audit.fn_log_changes()
RETURNS TRIGGER AS $$
DECLARE
    v_old  JSONB := NULL;
    v_new  JSONB := NULL;
    v_changed TEXT[];
BEGIN
    IF TG_OP = 'DELETE' THEN
        v_old := to_jsonb(OLD);
    ELSIF TG_OP = 'INSERT' THEN
        v_new := to_jsonb(NEW);
    ELSE
        v_old := to_jsonb(OLD);
        v_new := to_jsonb(NEW);
        SELECT ARRAY_AGG(key)
        INTO   v_changed
        FROM   jsonb_each(v_new) n
        WHERE  n.value IS DISTINCT FROM (v_old -> n.key);
    END IF;

    INSERT INTO audit.events (
        org_id, event_type, schema_name, table_name, record_id,
        old_values, new_values, changed_fields, occurred_at
    ) VALUES (
        COALESCE((v_new ->> 'org_id')::UUID, (v_old ->> 'org_id')::UUID),
        TG_OP,
        TG_TABLE_SCHEMA,
        TG_TABLE_NAME,
        COALESCE((v_new ->> 'id')::UUID, (v_old ->> 'id')::UUID),
        v_old, v_new, v_changed,
        NOW()
    );

    RETURN COALESCE(NEW, OLD);
END;
$$ LANGUAGE plpgsql SECURITY DEFINER;

BEGIN
    CASE NEW.movement_type
        WHEN 'receive'   THEN UPDATE pharmacy.stock_batches SET qty_available = qty_available + NEW.qty WHERE id = NEW.batch_id;
        WHEN 'dispense'  THEN UPDATE pharmacy.stock_batches SET qty_available = qty_available - NEW.qty WHERE id = NEW.batch_id;
        WHEN 'return'    THEN UPDATE pharmacy.stock_batches SET qty_available = qty_available + NEW.qty WHERE id = NEW.batch_id;
        WHEN 'damage'    THEN UPDATE pharmacy.stock_batches SET qty_damaged   = qty_damaged   + NEW.qty,
                                                                qty_available = qty_available - NEW.qty WHERE id = NEW.batch_id;
        WHEN 'adjust'    THEN UPDATE pharmacy.stock_batches SET qty_available = qty_available + NEW.qty WHERE id = NEW.batch_id;
        ELSE NULL;
    END CASE;
    RETURN NEW;
END;
$$ LANGUAGE plpgsql;

-- Live patient census by ward
CREATE OR REPLACE VIEW clinical.v_ward_census AS
SELECT
    w.id                                                                  AS ward_id,
    w.name                                                                AS ward_name,
    w.branch_id,
    COUNT(b.id)                                                           AS total_beds,
    COUNT(b.id) FILTER (WHERE b.status = 'occupied')                     AS occupied,
    COUNT(b.id) FILTER (WHERE b.status = 'available')                    AS available,
    ROUND(
        COUNT(b.id) FILTER (WHERE b.status = 'occupied')::NUMERIC /
        NULLIF(COUNT(b.id), 0) * 100, 1
    )                                                                     AS occupancy_pct
FROM facility.wards w

GROUP BY w.id, w.name, w.branch_id;

-- Today's OPD queue
CREATE OR REPLACE VIEW clinical.v_opd_queue_today AS
SELECT
    v.id,
    v.visit_no,
    v.token_no,
    p.mrn,
    p.full_name AS patient_name,
    p.phone,
    d.name      AS dept_name,
    v.status,
    v.scheduled_at,
    v.checked_in_at
FROM clinical.visits v
JOIN clinical.patients p ON p.id = v.patient_id
LEFT JOIN platform.departments d ON d.id = v.dept_id
WHERE v.status IN ('scheduled','checked_in','in_progress')
  AND DATE(v.scheduled_at) = CURRENT_DATE
ORDER BY v.token_no NULLS LAST, v.scheduled_at;

-- Pharmacy low-stock alert
CREATE OR REPLACE VIEW pharmacy.v_low_stock_alert AS
SELECT
    dm.id         AS drug_id,
    dm.generic_name,
    dm.brand_name,
    s.name        AS store_name,
    SUM(sb.qty_available) AS total_available,
    MIN(sb.expiry_date)   AS nearest_expiry
FROM pharmacy.stock_batches sb
JOIN pharmacy.drug_master dm ON dm.id = sb.drug_id
JOIN pharmacy.stores s       ON s.id  = sb.store_id
WHERE sb.is_active = TRUE
GROUP BY dm.id, dm.generic_name, dm.brand_name, s.id, s.name
HAVING SUM(sb.qty_available) < 50
ORDER BY total_available ASC;

-- Daily collection summary
CREATE OR REPLACE VIEW finance.v_daily_collection AS
SELECT
    DATE(p.received_at)              AS collection_date,
    p.method,
    COUNT(*)                         AS txn_count,
    SUM(p.amount)                    AS total_amount
FROM finance.payments p
WHERE p.status = 'paid'
GROUP BY DATE(p.received_at), p.method
ORDER BY collection_date DESC, total_amount DESC;

GROUP BY w.id, w.name, w.branch_id;

-- Today's OPD queue
CREATE OR REPLACE VIEW clinical.v_opd_queue_today AS
SELECT
    v.id,
    v.visit_no,
    v.token_no,
    p.mrn,
    p.full_name AS patient_name,
    p.phone,
    d.name      AS dept_name,
    v.status,
    v.scheduled_at,
    v.checked_in_at
FROM clinical.visits v
JOIN clinical.patients p ON p.id = v.patient_id
LEFT JOIN platform.departments d ON d.id = v.dept_id
WHERE v.status IN ('scheduled','checked_in','in_progress')
  AND DATE(v.scheduled_at) = CURRENT_DATE
ORDER BY v.token_no NULLS LAST, v.scheduled_at;

-- Pharmacy low-stock alert
CREATE OR REPLACE VIEW pharmacy.v_low_stock_alert AS
SELECT
    dm.id         AS drug_id,
    dm.generic_name,
    dm.brand_name,
    s.name        AS store_name,
    SUM(sb.qty_available) AS total_available,
    MIN(sb.expiry_date)   AS nearest_expiry
FROM pharmacy.stock_batches sb
JOIN pharmacy.drug_master dm ON dm.id = sb.drug_id
JOIN pharmacy.stores s       ON s.id  = sb.store_id
WHERE sb.is_active = TRUE
GROUP BY dm.id, dm.generic_name, dm.brand_name, s.id, s.name
HAVING SUM(sb.qty_available) < 50
ORDER BY total_available ASC;

-- Daily collection summary
CREATE OR REPLACE VIEW finance.v_daily_collection AS
SELECT
    DATE(p.received_at)              AS collection_date,
    p.method,
    COUNT(*)                         AS txn_count,
    SUM(p.amount)                    AS total_amount
FROM finance.payments p
WHERE p.status = 'paid'
GROUP BY DATE(p.received_at), p.method
ORDER BY collection_date DESC, total_amount DESC;

-- ============================================================
-- SECTION 7: DATABASE SUMMARY
-- ============================================================

CREATE OR REPLACE VIEW analytics.v_schema_summary AS
SELECT
    t.table_schema                       AS schema_name,
    COUNT(DISTINCT t.table_name)         AS table_count,
    COUNT(DISTINCT i.indexname)          AS index_count
FROM information_schema.tables t
LEFT JOIN pg_indexes i
       ON i.schemaname = t.table_schema
WHERE t.table_type = 'BASE TABLE'
  AND t.table_schema IN (
      'platform','clinical','finance','pharmacy',
      'hr','operations','facility','analytics','audit'
  )
GROUP BY t.table_schema
ORDER BY table_count DESC;
