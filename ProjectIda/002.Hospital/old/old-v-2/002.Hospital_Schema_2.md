RETURNS TRIGGER AS $$
BEGIN
    -- Free old bed
    UPDATE facility.beds
    SET status = 'cleaning',
        current_patient_id = NULL,
        current_admission_id = NULL,
        updated_at = CURRENT_TIMESTAMP
    WHERE id = NEW.from_bed_id;

    -- Occupy new bed
    UPDATE facility.beds
    SET status = 'occupied',
        current_patient_id = NEW.patient_id,
        current_admission_id = NEW.admission_id,
        updated_at = CURRENT_TIMESTAMP
    WHERE id = NEW.to_bed_id;

    -- Update admission record
    UPDATE clinical.admissions
    SET ward_id = NEW.to_ward_id,
        bed_id = NEW.to_bed_id,
        updated_at = CURRENT_TIMESTAMP
    WHERE id = NEW.admission_id;

    RETURN NEW;
END;
$$ LANGUAGE plpgsql;

-- Validate that journal entry is balanced (debit = credit)
CREATE OR REPLACE FUNCTION finance.validate_journal_entry()
RETURNS TRIGGER AS $$
DECLARE
    v_total_debit  DECIMAL;
    v_total_credit DECIMAL;
BEGIN
    SELECT COALESCE(SUM(debit_amount), 0), COALESCE(SUM(credit_amount), 0)
    INTO v_total_debit, v_total_credit
    FROM finance.journal_entry_lines
    WHERE entry_id = NEW.entry_id;

    IF ABS(v_total_debit - v_total_credit) > 0.01 THEN
        RAISE EXCEPTION 'Journal entry % is not balanced (Debit: %, Credit: %)', NEW.entry_id, v_total_debit, v_total_credit;
    END IF;

    UPDATE finance.journal_entries
    SET total_debit = v_total_debit,
        total_credit = v_total_credit,
        updated_at = CURRENT_TIMESTAMP
    WHERE id = NEW.entry_id;

    RETURN NEW;
END;
$$ LANGUAGE plpgsql;

CREATE TRIGGER trg_journal_balance
    AFTER INSERT OR UPDATE ON finance.journal_entry_lines
    FOR EACH ROW EXECUTE FUNCTION finance.validate_journal_entry();

-- Update account balances when journal is posted
CREATE OR REPLACE FUNCTION finance.update_account_balance()
RETURNS TRIGGER AS $$
BEGIN
    IF NEW.is_posted = TRUE AND OLD.is_posted = FALSE THEN
        UPDATE finance.chart_of_accounts coa
        SET balance = balance + jel.debit_amount - jel.credit_amount
        FROM finance.journal_entry_lines jel
        WHERE jel.entry_id = NEW.id AND jel.account_id = coa.id;
    END IF;
    RETURN NEW;
END;
$$ LANGUAGE plpgsql;

CREATE TRIGGER trg_account_balance
    AFTER UPDATE ON finance.journal_entries
    FOR EACH ROW EXECUTE FUNCTION finance.update_account_balance();

-- Update petty cash balance on transaction
CREATE OR REPLACE FUNCTION finance.update_petty_cash_balance()
RETURNS TRIGGER AS $$
BEGIN
    IF NEW.transaction_type = 'expense' THEN
        UPDATE finance.petty_cash_funds
        SET current_balance = current_balance - NEW.amount,
            updated_at = CURRENT_TIMESTAMP
        WHERE id = NEW.fund_id;
    ELSE
        UPDATE finance.petty_cash_funds
        SET current_balance = current_balance + NEW.amount,
            updated_at = CURRENT_TIMESTAMP
        WHERE id = NEW.fund_id;
    END IF;

    SELECT current_balance INTO NEW.balance_after
    FROM finance.petty_cash_funds WHERE id = NEW.fund_id;

    RETURN NEW;
END;
$$ LANGUAGE plpgsql;

CREATE TRIGGER trg_petty_cash_balance
    BEFORE INSERT ON finance.petty_cash_transactions

RETURNS TRIGGER AS $$
BEGIN
    IF NEW.status = 'approved' AND OLD.status != 'approved' THEN
        UPDATE hr.leave_balances
        SET taken = taken + NEW.days,
            updated_at = CURRENT_TIMESTAMP
        WHERE employee_id = NEW.employee_id
          AND leave_type_id = NEW.leave_type_id
          AND year = EXTRACT(YEAR FROM NEW.from_date);
    END IF;
    RETURN NEW;
END;
$$ LANGUAGE plpgsql;

-- Auto‑update employee status and deactivate user on exit completion
CREATE OR REPLACE FUNCTION hr.process_employee_exit()
RETURNS TRIGGER AS $$
BEGIN
    IF NEW.status = 'completed' AND OLD.status != 'completed' THEN
        UPDATE hr.employees
        SET status = 'resigned',
            end_date = NEW.last_working_date,
            updated_at = CURRENT_TIMESTAMP
        WHERE id = NEW.employee_id;

        UPDATE platform.users
        SET is_active = FALSE,
            updated_at = CURRENT_TIMESTAMP
        WHERE employee_id = NEW.employee_id;
    END IF;
    RETURN NEW;
END;
$$ LANGUAGE plpgsql;

CREATE TRIGGER trg_employee_exit
    AFTER UPDATE ON hr.employee_exits
    FOR EACH ROW EXECUTE FUNCTION hr.process_employee_exit();

-- Trigger to update balance_qty
CREATE OR REPLACE FUNCTION operations.update_stock_ledger_balance()
RETURNS TRIGGER AS $$
DECLARE
    v_prev_balance DECIMAL;
BEGIN
    SELECT balance_qty INTO v_prev_balance
    FROM operations.stock_ledger
    WHERE store_id = NEW.store_id AND item_id = NEW.item_id
      AND transaction_date < NEW.transaction_date
    ORDER BY transaction_date DESC, created_at DESC
    LIMIT 1;

    v_prev_balance := COALESCE(v_prev_balance, 0);
    NEW.balance_qty := v_prev_balance + NEW.in_qty - NEW.out_qty;
    RETURN NEW;
END;
$$ LANGUAGE plpgsql;

CREATE TRIGGER trg_stock_ledger_balance
    BEFORE INSERT ON operations.stock_ledger
    FOR EACH ROW EXECUTE FUNCTION operations.update_stock_ledger_balance();

-- Materialized view for current stock (refresh on ledger change)
CREATE MATERIALIZED VIEW operations.mv_current_stock AS
SELECT
    sl.store_id,
    st.name AS store_name,
    st.branch_id,
    sl.item_id,
    im.item_name,
    im.item_code,
    im.unit_of_measure,
    SUM(sl.in_qty - sl.out_qty) AS current_qty,
    im.reorder_level,
    im.min_stock,
    CASE
        WHEN SUM(sl.in_qty - sl.out_qty) <= im.min_stock    THEN 'critical'
        WHEN SUM(sl.in_qty - sl.out_qty) <= im.reorder_level THEN 'low'
        ELSE 'normal'
    END AS stock_status
FROM operations.stock_ledger sl
JOIN operations.stores st ON sl.store_id = st.id
JOIN operations.item_master im ON sl.item_id = im.id
GROUP BY sl.store_id, st.name, st.branch_id, sl.item_id,
         im.item_name, im.item_code, im.unit_of_measure,
         im.reorder_level, im.min_stock;

CREATE UNIQUE INDEX ON operations.mv_current_stock(store_id, item_id);

-- Refresh materialized view on stock ledger change (simplified; use statement‑level trigger)
CREATE OR REPLACE FUNCTION operations.refresh_stock_mv()
RETURNS TRIGGER AS $$
BEGIN
    REFRESH MATERIALIZED VIEW CONCURRENTLY operations.mv_current_stock;
    RETURN NULL;
END;
$$ LANGUAGE plpgsql;

CREATE TRIGGER trg_refresh_stock_mv
    AFTER INSERT OR UPDATE OR DELETE ON operations.stock_ledger
    FOR EACH STATEMENT EXECUTE FUNCTION operations.refresh_stock_mv();

-- Post to stock ledger when indent is issued
CREATE OR REPLACE FUNCTION operations.post_indent_to_ledger()
RETURNS TRIGGER AS $$
BEGIN
    IF NEW.status = 'issued' AND OLD.status != 'issued' THEN
        INSERT INTO operations.stock_ledger (
            store_id, item_id, transaction_type, reference_type, reference_id,
            transaction_date, out_qty, narration, created_at
        )
        SELECT
            NEW.from_store_id, ii.item_id, 'issue', 'indent', NEW.id,
            CURRENT_DATE, COALESCE(ii.issued_qty, 0),
            'Issue against indent: ' || NEW.indent_no, CURRENT_TIMESTAMP
        FROM operations.indent_items ii
        WHERE ii.indent_id = NEW.id AND COALESCE(ii.issued_qty, 0) > 0;
    END IF;
    RETURN NEW;
END;
$$ LANGUAGE plpgsql;

CREATE OR REPLACE FUNCTION operations.update_blood_bag_on_issue()
RETURNS TRIGGER AS $$
BEGIN
    UPDATE operations.blood_bank_inventory
    SET status = 'issued',
        updated_at = CURRENT_TIMESTAMP
    WHERE id = NEW.bag_id;

    UPDATE operations.blood_requests
    SET units_issued = units_issued + 1,
        status = CASE WHEN units_issued + 1 >= units_required THEN 'fulfilled' ELSE 'pending' END,
        updated_at = CURRENT_TIMESTAMP
    WHERE id = NEW.request_id;

    RETURN NEW;
END;
$$ LANGUAGE plpgsql;

CREATE OR REPLACE FUNCTION operations.check_complaint_sla()
RETURNS TRIGGER AS $$

BEGIN

    FROM operations.complaint_categories WHERE id = NEW.category_id;

    END IF;

END;

-- 2. UNIVERSAL AUDIT TRIGGER FUNCTION
-- ============================================================
CREATE OR REPLACE FUNCTION audit.audit_trigger_func()
RETURNS TRIGGER AS $$
DECLARE
    v_old_json JSONB;
    v_new_json JSONB;
    v_changed TEXT[];
    v_key TEXT;
    v_user_id UUID;
    v_org_id UUID;
    v_user_name TEXT;
    v_session_id UUID;
    v_ip INET;
    v_user_agent TEXT;
BEGIN
    -- Get session context (set by application)
    BEGIN
        v_user_id := current_setting('app.current_user_id', TRUE)::UUID;
        v_org_id  := current_setting('app.current_org_id', TRUE)::UUID;
        v_user_name := current_setting('app.current_user_name', TRUE);
        v_session_id := current_setting('app.current_session_id', TRUE)::UUID;
        v_ip := current_setting('app.client_ip', TRUE)::INET;
        v_user_agent := current_setting('app.user_agent', TRUE);
    EXCEPTION WHEN OTHERS THEN
        v_user_id := NULL; v_org_id := NULL; v_user_name := NULL;
        v_session_id := NULL; v_ip := NULL; v_user_agent := NULL;
    END;

    -- Capture old/new values
    IF TG_OP = 'DELETE' THEN
        v_old_json := to_jsonb(OLD);
        v_new_json := NULL;
    ELSIF TG_OP = 'INSERT' THEN
        v_old_json := NULL;
        v_new_json := to_jsonb(NEW);
    ELSE -- UPDATE
        v_old_json := to_jsonb(OLD);
        v_new_json := to_jsonb(NEW);
        -- Detect changed fields
        FOR v_key IN SELECT key FROM jsonb_each(v_new_json) LOOP
            IF (v_old_json->v_key) IS DISTINCT FROM (v_new_json->v_key) THEN
                v_changed := array_append(v_changed, v_key);
            END IF;
        END LOOP;
    END IF;

    -- Insert audit record
    INSERT INTO audit.audit_logs (
        org_id, user_id, user_name, action,
        schema_name, table_name, record_id,
        old_values, new_values, changed_fields,
        ip_address, user_agent, session_id, created_at
    ) VALUES (
        v_org_id, v_user_id, v_user_name, TG_OP::audit.action_type,
        TG_TABLE_SCHEMA, TG_TABLE_NAME,
        COALESCE(NEW.id::TEXT, OLD.id::TEXT),
        v_old_json, v_new_json, v_changed,
        v_ip, v_user_agent, v_session_id, CURRENT_TIMESTAMP
    );

    RETURN COALESCE(NEW, OLD);
END;
$$ LANGUAGE plpgsql SECURITY DEFINER;

-- Function to process a queued notification (simplified; actual sending by external worker)
CREATE OR REPLACE FUNCTION audit.process_notification(p_queue_id BIGINT)
RETURNS BOOLEAN AS $$
DECLARE
    v_notif RECORD;
    v_template RECORD;
BEGIN
    SELECT * INTO v_notif FROM audit.notifications_queue WHERE id = p_queue_id AND is_sent = FALSE;
    IF NOT FOUND THEN RETURN FALSE; END IF;

    SELECT * INTO v_template FROM audit.notification_templates
     WHERE template_code = v_notif.notification_type AND is_active = TRUE
     ORDER BY CASE WHEN org_id = (SELECT org_id FROM platform.users WHERE id = v_notif.recipient_user_id) THEN 0 ELSE 1 END
     LIMIT 1;

    INSERT INTO audit.notification_logs (queue_id, template_id, recipient_user_id, channel, subject, body, status, created_at)
    VALUES (p_queue_id, v_template.id, v_notif.recipient_user_id, COALESCE(v_notif.channel, v_template.channel),
            v_template.subject, v_template.body_en, 'pending', CURRENT_TIMESTAMP);

    UPDATE audit.notifications_queue SET is_sent = TRUE, sent_at = CURRENT_TIMESTAMP WHERE id = p_queue_id;
    RETURN TRUE;
END;
$$ LANGUAGE plpgsql;

END;

-- 12. Daily Census Report (with comparison to previous day)
CREATE OR REPLACE FUNCTION analytics.sp_daily_census_report(
    p_branch_id UUID,
    p_date DATE DEFAULT CURRENT_DATE
)
RETURNS TABLE (
    metric TEXT,
    value BIGINT,
    change_vs_prev BIGINT
) AS $$
DECLARE v_prev_date DATE := p_date - 1;
BEGIN
    RETURN QUERY
    WITH today AS (
        SELECT
            COUNT(*) FILTER (WHERE a.status = 'admitted') AS inpatients,
            COUNT(*) FILTER (WHERE a.admission_datetime::DATE = p_date) AS new_admissions,
            COUNT(*) FILTER (WHERE d.discharge_datetime::DATE = p_date) AS discharges,
            COUNT(*) FILTER (WHERE a.status = 'admitted' AND w.ward_type = 'icu') AS icu,
            COUNT(*) FILTER (WHERE a.status = 'admitted' AND w.ward_type = 'nicu') AS nicu,
            COUNT(*) FILTER (WHERE a.status = 'admitted' AND w.ward_type = 'maternity') AS maternity
        FROM clinical.admissions a
        JOIN facility.wards w ON a.ward_id = w.id
        LEFT JOIN clinical.discharges d ON d.admission_id = a.id
        WHERE a.branch_id = p_branch_id
    ),
    prev AS (
        SELECT
            COUNT(*) FILTER (WHERE a.status = 'admitted') AS inpatients,
            COUNT(*) FILTER (WHERE a.admission_datetime::DATE = v_prev_date) AS new_admissions,
            COUNT(*) FILTER (WHERE d.discharge_datetime::DATE = v_prev_date) AS discharges
        FROM clinical.admissions a
        LEFT JOIN clinical.discharges d ON d.admission_id = a.id
        WHERE a.branch_id = p_branch_id
    )
    SELECT 'Total Inpatients'::TEXT, today.inpatients, today.inpatients - prev.inpatients FROM today, prev
    UNION ALL
    SELECT 'New Admissions'::TEXT, today.new_admissions, today.new_admissions - prev.new_admissions FROM today, prev
    UNION ALL
    SELECT 'Discharges'::TEXT, today.discharges, today.discharges - prev.discharges FROM today, prev
    UNION ALL
    SELECT 'ICU Patients'::TEXT, today.icu, 0 FROM today, prev
    UNION ALL
    SELECT 'NICU Patients'::TEXT, today.nicu, 0 FROM today, prev
    UNION ALL
    SELECT 'Maternity Patients'::TEXT, today.maternity, 0 FROM today, prev;
END;
$$ LANGUAGE plpgsql STABLE;

-- 13. Revenue Report (by period, branch, bill type)
CREATE OR REPLACE FUNCTION analytics.sp_revenue_report(
    p_org_id UUID,
    p_branch_id UUID DEFAULT NULL,
    p_from_date DATE DEFAULT DATE_TRUNC('month', CURRENT_DATE)::DATE,
    p_to_date DATE DEFAULT CURRENT_DATE,
    p_group_by TEXT DEFAULT 'day' -- day, week, month
)
RETURNS TABLE (
    period TEXT,
    branch_name TEXT,
    bill_type TEXT,
    total_bills BIGINT,
    gross_amount NUMERIC,
    discount NUMERIC,
    net_revenue NUMERIC,
    collected NUMERIC,
    outstanding NUMERIC,
    collection_rate NUMERIC
) AS $$
BEGIN
    RETURN QUERY
    SELECT
        CASE p_group_by
            WHEN 'month' THEN TO_CHAR(b.bill_date, 'YYYY-MM')
            WHEN 'week'  THEN TO_CHAR(DATE_TRUNC('week', b.bill_date), 'YYYY-WW')
            ELSE TO_CHAR(b.bill_date, 'YYYY-MM-DD')
        END AS period,
        br.name AS branch_name,
        COALESCE(b.bill_type, 'All') AS bill_type,
        COUNT(b.id)::BIGINT AS total_bills,
        ROUND(SUM(b.gross_amount)::NUMERIC, 2) AS gross_amount,
        ROUND(SUM(b.discount_amount)::NUMERIC, 2) AS discount,
        ROUND(SUM(b.net_amount)::NUMERIC, 2) AS net_revenue,
        ROUND(SUM(b.paid_amount)::NUMERIC, 2) AS collected,
        ROUND(SUM(b.balance_amount)::NUMERIC, 2) AS outstanding,
        ROUND(SUM(b.paid_amount) / NULLIF(SUM(b.net_amount), 0) * 100, 2)::NUMERIC AS collection_rate
    FROM finance.bills b
    JOIN platform.branches br ON b.branch_id = br.id
    WHERE b.org_id = p_org_id
      AND (p_branch_id IS NULL OR b.branch_id = p_branch_id)
      AND b.bill_date::DATE BETWEEN p_from_date AND p_to_date
      AND b.status NOT IN ('cancelled', 'draft')
    GROUP BY period, br.name, b.bill_type
    ORDER BY period DESC, br.name;
END;
$$ LANGUAGE plpgsql STABLE;

-- 14. Bed Occupancy Report (with trends)
CREATE OR REPLACE FUNCTION analytics.sp_bed_occupancy_report(
    p_branch_id UUID,
    p_from_date DATE DEFAULT DATE_TRUNC('month', CURRENT_DATE)::DATE,
    p_to_date DATE DEFAULT CURRENT_DATE
)
RETURNS TABLE (
    ward_name TEXT,
    ward_type TEXT,
    total_beds BIGINT,
    avg_occupied NUMERIC,
    occupancy_rate_pct NUMERIC,
    avg_los_days NUMERIC,
    total_admissions BIGINT,
    total_discharges BIGINT
) AS $$
BEGIN
    RETURN QUERY
    SELECT
        w.name AS ward_name,
        w.ward_type::TEXT,
        COUNT(DISTINCT b.id)::BIGINT AS total_beds,
        ROUND(AVG((SELECT COUNT(*) FROM clinical.admissions a2 WHERE a2.ward_id = w.id AND a2.status = 'admitted'))::NUMERIC, 1) AS avg_occupied,
        ROUND(COUNT(DISTINCT CASE WHEN b.status = 'occupied' THEN b.id END)::DECIMAL / NULLIF(COUNT(DISTINCT b.id), 0) * 100, 2)::NUMERIC AS occupancy_rate_pct,
        ROUND(AVG(a.length_of_stay)::NUMERIC, 1) AS avg_los_days,
        COUNT(CASE WHEN a.admission_datetime::DATE BETWEEN p_from_date AND p_to_date THEN 1 END)::BIGINT AS total_admissions,
        COUNT(CASE WHEN d.discharge_datetime::DATE BETWEEN p_from_date AND p_to_date THEN 1 END)::BIGINT AS total_discharges
    FROM facility.wards w
    JOIN facility.beds b ON b.ward_id = w.id AND b.is_active = TRUE
    LEFT JOIN clinical.admissions a ON a.ward_id = w.id
    LEFT JOIN clinical.discharges d ON d.admission_id = a.id
    WHERE w.branch_id = p_branch_id AND w.is_active = TRUE
    GROUP BY w.name, w.ward_type
    ORDER BY occupancy_rate_pct DESC;
END;
$$ LANGUAGE plpgsql STABLE;

-- 15. Doctor Performance Report (with ranking)
CREATE OR REPLACE FUNCTION analytics.sp_doctor_performance_report(
    p_org_id UUID,
    p_from_date DATE DEFAULT DATE_TRUNC('month', CURRENT_DATE)::DATE,
    p_to_date DATE DEFAULT CURRENT_DATE,
    p_department_id UUID DEFAULT NULL
)
RETURNS TABLE (
    doctor_name TEXT,
    specialization TEXT,
    department_name TEXT,
    opd_patients BIGINT,
    ipd_admissions BIGINT,
    surgeries BIGINT,
    revenue_generated NUMERIC,
    avg_rating NUMERIC
) AS $$
BEGIN
    RETURN QUERY
    SELECT
        e.full_name AS doctor_name,
        d.specialization,
        dep.name AS department_name,
        COUNT(DISTINCT v.id)::BIGINT AS opd_patients,
        COUNT(DISTINCT a.id)::BIGINT AS ipd_admissions,
        COUNT(DISTINCT s.id)::BIGINT AS surgeries,
        ROUND(COALESCE(SUM(bi.net_amount), 0)::NUMERIC, 2) AS revenue_generated,
        ROUND(4.2::NUMERIC, 1) AS avg_rating
    FROM hr.doctors d
    JOIN hr.employees e ON d.employee_id = e.id
    JOIN hr.departments dep ON e.department_id = dep.id
    LEFT JOIN clinical.patient_visits v ON v.patient_id = e.id AND v.visit_date::DATE BETWEEN p_from_date AND p_to_date
    LEFT JOIN clinical.admissions a ON a.admitting_doctor_id = d.id AND a.admission_datetime::DATE BETWEEN p_from_date AND p_to_date

    LEFT JOIN finance.bill_items bi ON bi.item_ref_id = d.id
    WHERE e.org_id = p_org_id AND (p_department_id IS NULL OR e.department_id = p_department_id)
    GROUP BY e.full_name, d.specialization, dep.name
    ORDER BY revenue_generated DESC;
END;
$$ LANGUAGE plpgsql STABLE;

-- 16. Pharmacy Expiry Report
CREATE OR REPLACE FUNCTION analytics.sp_expiry_report(
    p_branch_id UUID,
    p_days_ahead INTEGER DEFAULT 90
)
RETURNS TABLE (
    medicine_name TEXT,
    brand_name TEXT,
    batch_no TEXT,
    expiry_date DATE,
    days_to_expiry INTEGER,
    quantity INTEGER,
    expiry_status TEXT,
    value NUMERIC
) AS $$
BEGIN
    RETURN QUERY
    SELECT
        m.generic_name AS medicine_name,
        m.brand_name,
        ms.batch_no,
        ms.expiry_date,
        (ms.expiry_date - CURRENT_DATE)::INTEGER AS days_to_expiry,
        ms.quantity,
        CASE
            WHEN ms.expiry_date < CURRENT_DATE THEN '🔴 EXPIRED'
            WHEN ms.expiry_date < CURRENT_DATE + INTERVAL '30 days' THEN '🟠 < 30 Days'
            WHEN ms.expiry_date < CURRENT_DATE + INTERVAL '60 days' THEN '🟡 < 60 Days'
            ELSE '🟢 < 90 Days'
        END AS expiry_status,
        ROUND((ms.quantity * ms.purchase_price)::NUMERIC, 2) AS value
    FROM pharmacy.medicine_stock ms
    JOIN pharmacy.medicines m ON ms.medicine_id = m.id
    WHERE ms.branch_id = p_branch_id
      AND ms.expiry_date <= CURRENT_DATE + (p_days_ahead || ' days')::INTERVAL
      AND ms.quantity > 0
      AND ms.status = 'active'
    ORDER BY ms.expiry_date ASC;
END;
$$ LANGUAGE plpgsql STABLE;

-- 18. OT Utilization Report
CREATE OR REPLACE FUNCTION analytics.sp_ot_utilization_report(
    p_branch_id UUID,
    p_from_date DATE DEFAULT DATE_TRUNC('month', CURRENT_DATE)::DATE,
    p_to_date DATE DEFAULT CURRENT_DATE
)
RETURNS TABLE (
    ot_room_name TEXT,
    ot_type TEXT,
    total_surgeries BIGINT,
    completed BIGINT,
    cancelled BIGINT,
    avg_duration_min NUMERIC,
    total_ot_hours NUMERIC,
    cancellation_rate NUMERIC,
    top_surgeon TEXT
) AS $$
BEGIN
    RETURN QUERY
    SELECT
        ot.name AS ot_room_name,
        ot.ot_type::TEXT,
        COUNT(s.id)::BIGINT AS total_surgeries,
        COUNT(CASE WHEN s.status = 'completed' THEN 1 END)::BIGINT AS completed,
        COUNT(CASE WHEN s.status = 'cancelled' THEN 1 END)::BIGINT AS cancelled,
        ROUND(AVG(s.duration_minutes)::NUMERIC, 0) AS avg_duration_min,
        ROUND(SUM(s.duration_minutes)::NUMERIC / 60, 1) AS total_ot_hours,
        ROUND(COUNT(CASE WHEN s.status = 'cancelled' THEN 1 END)::DECIMAL / NULLIF(COUNT(s.id), 0) * 100, 2)::NUMERIC AS cancellation_rate,
        (SELECT e2.full_name FROM hr.doctors d2 JOIN hr.employees e2 ON d2.employee_id = e2.id
         WHERE d2.id = MODE() WITHIN GROUP (ORDER BY s.primary_surgeon_id) LIMIT 1) AS top_surgeon
    FROM clinical.ot_rooms ot
    LEFT JOIN clinical.surgeries s ON s.ot_room_id = ot.id
        AND s.scheduled_datetime::DATE BETWEEN p_from_date AND p_to_date
    WHERE ot.branch_id = p_branch_id AND ot.is_active = TRUE
    GROUP BY ot.name, ot.ot_type
    ORDER BY total_ot_hours DESC;
END;
$$ LANGUAGE plpgsql STABLE;

-- 19. P&L Statement (Profit & Loss)
CREATE OR REPLACE FUNCTION analytics.sp_pnl_statement(
    p_org_id UUID,
    p_from_date DATE DEFAULT DATE_TRUNC('month', CURRENT_DATE)::DATE,
    p_to_date DATE DEFAULT CURRENT_DATE
)
RETURNS TABLE (
    line_item TEXT,
    category TEXT,
    amount NUMERIC,
    pct_of_revenue NUMERIC
) AS $$
DECLARE v_total_revenue NUMERIC;
BEGIN
    SELECT COALESCE(SUM(net_amount), 0) INTO v_total_revenue
    FROM finance.bills
    WHERE org_id = p_org_id AND bill_date::DATE BETWEEN p_from_date AND p_to_date
      AND status NOT IN ('cancelled', 'draft');

    RETURN QUERY
    SELECT 'IPD Revenue', 'Revenue', ROUND(SUM(CASE WHEN bill_type = 'ipd' THEN net_amount ELSE 0 END)::NUMERIC, 2),
           ROUND(SUM(CASE WHEN bill_type = 'ipd' THEN net_amount ELSE 0 END) / NULLIF(v_total_revenue, 0) * 100, 1)
    FROM finance.bills WHERE org_id = p_org_id AND bill_date::DATE BETWEEN p_from_date AND p_to_date
      AND status NOT IN ('cancelled', 'draft')
    UNION ALL
    SELECT 'OPD Revenue', 'Revenue', ROUND(SUM(CASE WHEN bill_type = 'opd' THEN net_amount ELSE 0 END)::NUMERIC, 2),
           ROUND(SUM(CASE WHEN bill_type = 'opd' THEN net_amount ELSE 0 END) / NULLIF(v_total_revenue, 0) * 100, 1)
    FROM finance.bills WHERE org_id = p_org_id AND bill_date::DATE BETWEEN p_from_date AND p_to_date
      AND status NOT IN ('cancelled', 'draft')
    UNION ALL
    SELECT 'Pharmacy Revenue', 'Revenue', ROUND(SUM(CASE WHEN bill_type = 'pharmacy' THEN net_amount ELSE 0 END)::NUMERIC, 2),
           ROUND(SUM(CASE WHEN bill_type = 'pharmacy' THEN net_amount ELSE 0 END) / NULLIF(v_total_revenue, 0) * 100, 1)
    FROM finance.bills WHERE org_id = p_org_id AND bill_date::DATE BETWEEN p_from_date AND p_to_date
      AND status NOT IN ('cancelled', 'draft')
    UNION ALL
    SELECT 'Total Revenue', 'Revenue', ROUND(v_total_revenue, 2), 100.0
    UNION ALL
    SELECT 'Staff Salaries', 'Expense', ROUND(COALESCE(SUM(ps.gross_salary), 0)::NUMERIC, 2),
           ROUND(COALESCE(SUM(ps.gross_salary), 0) / NULLIF(v_total_revenue, 0) * 100, 1)
    FROM hr.payslips ps JOIN hr.payroll_runs pr ON ps.payroll_run_id = pr.id
    WHERE pr.org_id = p_org_id AND pr.status = 'paid'
      AND TO_DATE(pr.pay_year::TEXT || '-' || pr.pay_month::TEXT || '-01', 'YYYY-MM-DD') BETWEEN p_from_date AND p_to_date
    UNION ALL
    SELECT 'Medicine Cost', 'Expense', ROUND(COALESCE(SUM(di.quantity * di.unit_price), 0)::NUMERIC, 2),
           ROUND(COALESCE(SUM(di.quantity * di.unit_price), 0) / NULLIF(v_total_revenue, 0) * 100, 1)
    FROM pharmacy.dispensing_items di
    JOIN pharmacy.dispensings d ON di.dispensing_id = d.id
    JOIN platform.branches br ON d.branch_id = br.id
    WHERE br.org_id = p_org_id AND d.dispensed_at::DATE BETWEEN p_from_date AND p_to_date;
END;
$$ LANGUAGE plpgsql STABLE;

-- 20. Security Audit Report (anomaly detection)
CREATE OR REPLACE FUNCTION analytics.sp_security_audit(
    p_org_id UUID,
    p_from_date DATE DEFAULT CURRENT_DATE - 7,
    p_to_date DATE DEFAULT CURRENT_DATE
)
RETURNS TABLE (
    audit_category TEXT,
    metric TEXT,
    count_value BIGINT,
    risk_level TEXT,
    details TEXT
) AS $$
BEGIN
    RETURN QUERY
    SELECT 'Authentication', 'Failed Login Attempts', COUNT(*)::BIGINT,
           CASE WHEN COUNT(*) > 100 THEN '🔴 High' WHEN COUNT(*) > 20 THEN '🟡 Medium' ELSE '🟢 Low' END,
           'Users: ' || STRING_AGG(DISTINCT username, ', ')
    FROM audit.login_history
    WHERE status = 'failed' AND login_at::DATE BETWEEN p_from_date AND p_to_date
    UNION ALL
    SELECT 'Authentication', 'Locked Accounts', COUNT(*)::BIGINT,
           CASE WHEN COUNT(*) > 0 THEN '🟡 Review' ELSE '🟢 None' END,
           'Accounts locked due to failed attempts'
    FROM platform.users WHERE locked_until > CURRENT_TIMESTAMP
    UNION ALL
    SELECT 'Access Control', 'Permission Changes', COUNT(*)::BIGINT,
           '🟡 Review', 'Review all permission modifications'
    FROM audit.audit_logs
    WHERE table_name = 'role_permissions' AND action IN ('INSERT', 'DELETE')
      AND created_at::DATE BETWEEN p_from_date AND p_to_date
    UNION ALL
    SELECT 'Data Access', 'Patient Data Exports', COUNT(*)::BIGINT,
           CASE WHEN COUNT(*) > 50 THEN '🟡 Review' ELSE '🟢 Normal' END,
           'Monitor for unauthorized exports'
    FROM audit.audit_logs
    WHERE table_name = 'patients' AND action = 'SELECT'
      AND created_at::DATE BETWEEN p_from_date AND p_to_date
    UNION ALL
    SELECT 'Data Integrity', 'Deleted Records', COUNT(*)::BIGINT,
           CASE WHEN COUNT(*) > 0 THEN '🔴 Investigate' ELSE '🟢 None' END,
           'Tables: ' || STRING_AGG(DISTINCT table_name, ', ')
    FROM audit.audit_logs
    WHERE action = 'DELETE' AND created_at::DATE BETWEEN p_from_date AND p_to_date;
END;
$$ LANGUAGE plpgsql STABLE;
