## STEP 2: Core Enumerations

> **Design Rationale:** PostgreSQL `ENUM` types provide type safety, storage efficiency (stored as integers internally ~4 bytes), and referential integrity. They prevent invalid data at the database level rather than application level.

### ER Diagram — Enums Relationship

```text
┌─────────────────────────────────────────────────────────┐
│                    ENUM TYPES                            │
├─────────────┬─────────────────┬─────────────────────────┤
│  PERSON     │  CLINICAL       │  OPERATIONAL             │
├─────────────┼─────────────────┼─────────────────────────┤
│gender_type  │visit_status     │order_status              │
│blood_group  │admission_status │payment_status            │
│marital_status│bed_status      │payment_method            │
│             │consent_status   │shift_type                │
│             │specimen_status  │employment_type           │
│             │result_status    │leave_status              │
│             │claim_status     │maintenance_priority      │
│             │priority_type    │ticket_status             │
│             │                 │notification_channel      │
│             │                 │document_type             │
│             │                 │patient_type              │
└─────────────┴─────────────────┴─────────────────────────┘
```

## STEP 3: Multi-Tenancy Foundation

> **Design Pattern:** Shared-database, shared-schema multi-tenancy. Every tenant (hospital organization) shares the same tables but is isolated by `organization_id` on every row. This is the most cost-effective model for SaaS.

### Entity Relationship Diagram

```txt
┌──────────────────┐       1:N       ┌──────────────────┐
│  organizations   │────────────────▶│     branches     │
│──────────────────│                 │──────────────────│
│ id (PK)          │                 │ id (PK)          │
│ code (UQ)        │                 │ organization_id  │──▶ FK
│ name             │       1:N       │ code (UQ)        │
│ settings (JSONB) │────────────────▶│ name             │
│ features (JSONB) │                 │ total_beds       │
│ subscription_*   │                 │ total_ots        │
└──────────────────┘                 └──────────────────┘
         │
         │ 1:N
         ▼
┌──────────────────┐  self-ref  ┌──────────────────┐
│   departments    │◀───┌──────│   departments    │
│──────────────────│    │      │  (parent)        │
│ id (PK)          │    │      └──────────────────┘
│ organization_id  │    │
│ branch_id        │──┐ │
│ parent_dept_id   │──┘ │
│ code (UQ)        │    │
│ is_clinical      │    │
└──────────────────┘    │
                        │
    ┌───────────────────┘
    │ (departments can be
    │  org-level or branch-level)
```

```sql
CREATE TABLE organizations (
    -- Identity
    id              UUID PRIMARY KEY DEFAULT uuid_generate_v4(),
    code            VARCHAR(20) UNIQUE NOT NULL,
    name            VARCHAR(255) NOT NULL,
    name_bangla     VARCHAR(255),
    legal_name      VARCHAR(255),
    organization_type VARCHAR(50) DEFAULT 'hospital',
    -- Values: hospital, clinic, diagnostic_center, polyclinic, group

    -- Contact
    email           VARCHAR(255),
    phone           VARCHAR(20),
    website         VARCHAR(255),

    -- Address
    address_line1   VARCHAR(255),
    address_line2   VARCHAR(255),
    city            VARCHAR(100),
    district        VARCHAR(100),
    division        VARCHAR(100),
    postal_code     VARCHAR(20),
    country         VARCHAR(100) DEFAULT 'Bangladesh',

    -- Legal & Registration
    trade_license   VARCHAR(100),
    tin_number      VARCHAR(50),   -- Tax Identification Number
    bin_number      VARCHAR(50),   -- Business Identification Number
    dghs_license    VARCHAR(100),  -- Directorate General of Health Services

    -- Branding
    logo_url        TEXT,
    favicon_url     TEXT,
    primary_color   VARCHAR(10),
    secondary_color VARCHAR(10),

    -- Flexible Settings (JSONB)
    settings        JSONB DEFAULT '{}',
    -- Example: {"currency": "BDT", "date_format": "DD/MM/YYYY", ...}

    features        JSONB DEFAULT '{}',
    -- Example: {"telemedicine": true, "insurance": false, ...}

    -- Subscription / License
    subscription_plan  VARCHAR(50) DEFAULT 'basic',
    subscription_start DATE,
    subscription_end   DATE,
    max_branches        INTEGER DEFAULT 1,
    max_users           INTEGER DEFAULT 50,

    -- Status
    is_active       BOOLEAN DEFAULT TRUE,
    created_at      TIMESTAMPTZ DEFAULT CURRENT_TIMESTAMP,
    updated_at      TIMESTAMPTZ DEFAULT CURRENT_TIMESTAMP,
    created_by      UUID,
    updated_by      UUID
);

COMMENT ON TABLE organizations IS 'Top-level tenant: hospital, clinic, or healthcare group';
COMMENT ON COLUMN organizations.settings IS 'Flexible key-value store for org-specific settings';
COMMENT ON COLUMN organizations.features IS 'Feature flags controlling which modules are available';

-- ── 3.2 BRANCHES ────────────────────────────

CREATE TABLE branches (
    -- Identity
    id              UUID PRIMARY KEY DEFAULT uuid_generate_v4(),
    organization_id UUID NOT NULL REFERENCES organizations(id),
    code            VARCHAR(20) NOT NULL,
    name            VARCHAR(255) NOT NULL,
    name_bangla     VARCHAR(255),
    branch_type     VARCHAR(50) DEFAULT 'hospital',

    -- Contact
    email           VARCHAR(255),
    phone           VARCHAR(20),
    emergency_phone VARCHAR(20),

    -- Address
    address_line1   VARCHAR(255),
    address_line2   VARCHAR(255),
    city            VARCHAR(100),
    district        VARCHAR(100),
    division        VARCHAR(100),
    postal_code     VARCHAR(20),
    latitude        DECIMAL(10, 8),   -- For maps/integration
    longitude       DECIMAL(11, 8),

    -- Licenses
    dghs_license    VARCHAR(100),

    -- Capacity Planning
    total_beds      INTEGER DEFAULT 0,
    total_ots       INTEGER DEFAULT 0,  -- Operating Theatres

    -- Settings
    settings        JSONB DEFAULT '{}',
    timezone        VARCHAR(50) DEFAULT 'Asia/Dhaka',

    -- Flags
    is_main_branch  BOOLEAN DEFAULT FALSE,
    is_active       BOOLEAN DEFAULT TRUE,
    created_at      TIMESTAMPTZ DEFAULT CURRENT_TIMESTAMP,
    updated_at      TIMESTAMPTZ DEFAULT CURRENT_TIMESTAMP,
    created_by      UUID,
    updated_by      UUID,

    -- Constraints
    UNIQUE(organization_id, code)
);

COMMENT ON TABLE branches IS 'Physical locations/branches of an organization';
COMMENT ON COLUMN branches.total_beds IS 'Used for capacity planning and license limits';

-- ── 3.3 DEPARTMENTS ─────────────────────────

CREATE TABLE departments (
    -- Identity
    id                  UUID PRIMARY KEY DEFAULT uuid_generate_v4(),
    organization_id     UUID NOT NULL REFERENCES organizations(id),
    branch_id           UUID REFERENCES branches(id),       -- NULL = org-level dept
    parent_department_id UUID REFERENCES departments(id),   -- Self-referencing hierarchy

    code                VARCHAR(20) NOT NULL,
    name                VARCHAR(255) NOT NULL,
    name_bangla         VARCHAR(255),
    department_type     VARCHAR(50),
    -- Values: clinical, administrative, support, diagnostic

    -- Leadership
    head_of_department_id UUID,  -- References users.id (set after users created)

    -- Contact
    email               VARCHAR(255),
    phone               VARCHAR(20),
    extension           VARCHAR(10),
    location            VARCHAR(255),  -- Floor, wing, building

    -- Settings
    settings            JSONB DEFAULT '{}',
    color_code          VARCHAR(10),   -- For UI theming
    icon                VARCHAR(50),

    -- Display
    display_order       INTEGER DEFAULT 0,
    is_clinical         BOOLEAN DEFAULT FALSE,

    -- Status
    is_active           BOOLEAN DEFAULT TRUE,
    created_at          TIMESTAMPTZ DEFAULT CURRENT_TIMESTAMP,
    updated_at          TIMESTAMPTZ DEFAULT CURRENT_TIMESTAMP,

    -- Constraints
    UNIQUE(organization_id, code)
);

COMMENT ON TABLE departments IS 'Hierarchical departments; can be org-wide or branch-specific';
COMMENT ON COLUMN departments.branch_id IS 'NULL means the department spans all branches';
```

## STEP 4: User & Access Control (RBAC)

> **Design Pattern:** Role-Based Access Control (RBAC) with scoped permissions. Roles can be restricted to specific branches, departments, or own records only.

### ER Diagram — Access Control

```text
┌─────────┐  M:N  ┌───────┐  M:N  ┌─────────────┐
│  users  │◀────▶│ roles │◀────▶│ permissions  │
│─────────│       │───────│       │─────────────│
│ id (PK) │       │ id    │       │ id (PK)     │
│ org_id  │       │ code  │       │ module      │
│username │       │ level │       │ action      │
│password │       └───────┘       │ code (UQ)   │
└────┬────┘           ▲            └─────────────┘
     │               │
     │  user_roles   │ role_permissions
     │ (with scope)  │ (with scope restrictions)
     │               │
     ▼               │
┌────────────────┐   │
│user_branch_access│ │
│────────────────│   │
│ user_id        │   │
│ branch_id      │   │
│ is_default     │   │
└────────────────┘   │
                     │
┌────────────────┐   │
│ user_sessions  │   │
│────────────────│   │
│ session_token  │   │
│ device info    │   │
│ expires_at     │   │
└────────────────┘
```

```sql
-- ── 4.3 ROLE-PERMISSIONS MAPPING ───────────

CREATE TABLE role_permissions (
    id                      UUID PRIMARY KEY DEFAULT uuid_generate_v4(),
    role_id                 UUID NOT NULL REFERENCES roles(id) ON DELETE CASCADE,
    permission_id           UUID NOT NULL REFERENCES permissions(id) ON DELETE CASCADE,

    -- Scope Restrictions (narrow down the permission)
    branch_restricted       BOOLEAN DEFAULT FALSE,     -- Only own branch
    department_restricted   BOOLEAN DEFAULT FALSE,     -- Only own department
    own_records_only        BOOLEAN DEFAULT FALSE,     -- Only records created by this user

    created_at              TIMESTAMPTZ DEFAULT CURRENT_TIMESTAMP,
    created_by              UUID,

    UNIQUE(role_id, permission_id)
);

COMMENT ON TABLE role_permissions IS 'Maps permissions to roles with optional scope restrictions';

-- ── 4.4 USERS ───────────────────────────────

CREATE TABLE users (
    id              UUID PRIMARY KEY DEFAULT uuid_generate_v4(),
    organization_id UUID NOT NULL REFERENCES organizations(id),

    -- Authentication
    username        VARCHAR(100) NOT NULL,
    email           VARCHAR(255),
    phone           VARCHAR(20),
    password_hash   VARCHAR(255) NOT NULL,   -- bcrypt hash

    -- Profile
    first_name      VARCHAR(100) NOT NULL,
    last_name       VARCHAR(100),
    full_name       VARCHAR(255) GENERATED ALWAYS AS (
        first_name || ' ' || COALESCE(last_name, '')
    ) STORED,
    name_bangla     VARCHAR(255),

    gender          gender_type,
    date_of_birth   DATE,
    photo_url       TEXT,

    -- Employment (if staff member)
    employee_id     VARCHAR(50),
    designation     VARCHAR(100),
    department_id   UUID REFERENCES departments(id),

    -- Security
    is_email_verified   BOOLEAN DEFAULT FALSE,
    is_phone_verified   BOOLEAN DEFAULT FALSE,
    two_factor_enabled  BOOLEAN DEFAULT FALSE,
    two_factor_secret   VARCHAR(255),

    password_changed_at TIMESTAMPTZ,
    must_change_password BOOLEAN DEFAULT FALSE,
    failed_login_attempts INTEGER DEFAULT 0,
    locked_until        TIMESTAMPTZ,

    -- Status
    is_active       BOOLEAN DEFAULT TRUE,
    last_login_at   TIMESTAMPTZ,
    last_login_ip   INET,

    created_at      TIMESTAMPTZ DEFAULT CURRENT_TIMESTAMP,
    updated_at      TIMESTAMPTZ DEFAULT CURRENT_TIMESTAMP,
    created_by      UUID,

    -- Constraints
    UNIQUE(organization_id, username),
    UNIQUE(organization_id, email)
```

```sql
-- ── TRIGGER FUNCTION: Auto-audit on table changes ──

CREATE OR REPLACE FUNCTION audit_trigger_func()
RETURNS TRIGGER AS $$ DECLARE
    old_data JSONB;
    new_data JSONB;
    changed TEXT[];
BEGIN
    -- Build JSONB snapshots
    IF TG_OP = 'DELETE' THEN
        old_data := to_jsonb(OLD);
        new_data := NULL;
    ELSIF TG_OP = 'UPDATE' THEN
        old_data := to_jsonb(OLD);
        new_data := to_jsonb(NEW);
        -- Detect which fields actually changed
        SELECT array_agg(key) INTO changed
        FROM jsonb_each(old_data) e
        WHERE e.value IS DISTINCT FROM new_data->e.key;
    ELSIF TG_OP = 'INSERT' THEN
        old_data := NULL;
        new_data := to_jsonb(NEW);
    END IF;

    -- Insert audit record
    INSERT INTO audit_logs (
        organization_id,
        branch_id,
        user_id,
        username,
        ip_address,
        user_agent,
        action,
        module,
        table_name,
        record_id,
        old_values,
        new_values,
        changed_fields,
        description
    ) VALUES (
        COALESCE(NEW.organization_id, OLD.organization_id),
        COALESCE(NEW.branch_id, OLD.branch_id),
        current_setting('app.current_user_id', TRUE)::UUID,
        current_setting('app.current_username', TRUE),
        current_setting('app.client_ip', TRUE)::INET,
        current_setting('app.user_agent', TRUE),
        TG_OP,
        current_setting('app.module', TRUE),
        TG_TABLE_NAME,
        COALESCE(NEW.id, OLD.id),
        old_data,
        new_data,
        changed,
        TG_OP || ' on ' || TG_TABLE_NAME ||
            CASE WHEN TG_OP = 'UPDATE' THEN ' (fields: ' || COALESCE(array_to_string(changed, ', '), 'none') || ')'
                 ELSE '' END
    );

    RETURN COALESCE(NEW, OLD);
END;
 $$ LANGUAGE plpgsql SECURITY DEFINER;

-- Helper: Attach audit trigger to any table
CREATE OR REPLACE FUNCTION enable_audit(table_name TEXT)
RETURNS VOID AS $$ BEGIN
    EXECUTE format('
        CREATE TRIGGER audit_trigger
        AFTER INSERT OR UPDATE OR DELETE ON %I
        FOR EACH ROW EXECUTE FUNCTION audit_trigger_func();
    ', table_name);
END;
 $$ LANGUAGE plpgsql;
```

```sql
-- MRN auto-generation function
CREATE OR REPLACE FUNCTION generate_mrn(p_org_id UUID)
RETURNS VARCHAR(50) AS $$ DECLARE
    v_org_code VARCHAR(20);
    v_year VARCHAR(4);
    v_seq INTEGER;
    v_mrn VARCHAR(50);
BEGIN
    SELECT code INTO v_org_code FROM organizations WHERE id = p_org_id;
    v_year := to_char(CURRENT_DATE, 'YYYY');

    -- Get next sequence number for this org/year
    SELECT COALESCE(MAX(
        CAST(SUBSTRING(mrn FROM '[0-9]+$') AS INTEGER)
    ), 0) + 1 INTO v_seq
    FROM patients
    WHERE organization_id = p_org_id
      AND mrn LIKE v_org_code || '-' || v_year || '-%';

    v_mrn := v_org_code || '-' || v_year || '-' || LPAD(v_seq::TEXT, 6, '0');
    RETURN v_mrn;
END;
 $$ LANGUAGE plpgsql;
```

```sql
-- ── PRESCRIPTION DRUG SAFETY FUNCTION ───────

CREATE OR REPLACE FUNCTION check_prescription_safety(
    p_patient_id UUID,
    p_drug_ids UUID[]
) RETURNS TABLE (
    warning_type VARCHAR(50),
    severity VARCHAR(20),
    message TEXT,
    drug_ids UUID[]
) AS $$ BEGIN
    RETURN QUERY

    -- Check 1: Patient allergies
    SELECT
        'ALLERGY'::VARCHAR(50),
        pa.severity::VARCHAR(20),
        'Patient has known allergy to ' || pa.allergen_name ||
            CASE WHEN pa.reaction IS NOT NULL THEN ' (Reaction: ' || pa.reaction || ')' ELSE '' END,
        ARRAY[pa.allergen_id]::UUID[]
    FROM patient_allergies pa
    WHERE pa.patient_id = p_patient_id
      AND pa.is_active = TRUE
      AND pa.allergen_id = ANY(p_drug_ids)

    UNION ALL

    -- Check 2: Drug-drug interactions
    SELECT
        'INTERACTION'::VARCHAR(50),
        di.severity::VARCHAR(20),
        'Interaction: ' || dm1.generic_name || ' + ' || dm2.generic_name ||
            ' — ' || COALESCE(di.clinical_effect, di.description),
        ARRAY[di.drug_id_1, di.drug_id_2]::UUID[]
    FROM drug_interactions di
    JOIN drug_master dm1 ON dm1.id = di.drug_id_1
    JOIN drug_master dm2 ON dm2.id = di.drug_id_2
    WHERE di.is_active = TRUE
      AND (
          (di.drug_id_1 = ANY(p_drug_ids) AND di.drug_id_2 = ANY(p_drug_ids))
      )

    UNION ALL

    -- Check 3: Duplicate therapy (same generic, different brands)
    SELECT
        'DUPLICATE'::VARCHAR(50),
        'moderate'::VARCHAR(20),
        'Possible duplicate therapy: ' || d1.generic_name ||
            ' already prescribed as ' || d1.brand_name ||
            ' (similar to ' || d2.brand_name || ')',
        ARRAY[d1.id, d2.id]::UUID[]
    FROM drug_master d1
    JOIN drug_master d2 ON d1.generic_name = d2.generic_name AND d1.id <> d2.id
    WHERE d1.id = ANY(p_drug_ids)
      AND d2.id = ANY(p_drug_ids)

    ORDER BY
        CASE severity
            WHEN 'life_threatening' THEN 1
            WHEN 'severe' THEN 2
            WHEN 'contraindicated' THEN 2
            WHEN 'moderate' THEN 3
            WHEN 'mild' THEN 4
            ELSE 5
        END;
END;
 $$ LANGUAGE plpgsql STABLE;

-- Prescription number generator
CREATE OR REPLACE FUNCTION generate_prescription_number(p_org_id UUID, p_date DATE)
RETURNS VARCHAR(50) AS $$ DECLARE
    v_seq INTEGER;
BEGIN
    SELECT COALESCE(MAX(CAST(SUBSTRING(prescription_number FROM '[0-9]+$') AS INTEGER)), 0) + 1
    INTO v_seq
    FROM prescriptions
    WHERE organization_id = p_org_id
      AND prescription_date = p_date;

    RETURN 'Rx-' || to_char(p_date, 'YYYYMMDD') || '-' || LPAD(v_seq::TEXT, 6, '0');
END;
```

```sql
SELECT

    p.full_name,

                'controlled', pc.is_controlled
            ) ORDER BY pc.created_at
        ) FILTER (WHERE pc.id IS NOT NULL AND pc.is_active = TRUE),
        '[]'::jsonb
    ) AS chronic_conditions,

    -- Last 5 diagnoses
    COALESCE(
        jsonb_agg(
            jsonb_build_object(
                'description', cd.diagnosis_description,
                'icd', ic.code,
                'date', cd.diagnosed_at,
                'is_primary', cd.is_primary
            ) ORDER BY cd.diagnosed_at DESC
        ) FILTER (WHERE cd.id IS NOT NULL),
        '[]'::jsonb
    ) FILTER (WHERE ROW_NUMBER() OVER (ORDER BY cd.diagnosed_at DESC) <= 5),
        '[]'::jsonb
    ) AS recent_diagnoses

FROM patients p
LEFT JOIN patient_allergies pa ON pa.patient_id = p.id
LEFT JOIN patient_medications pm ON pm.patient_id = p.id
LEFT JOIN patient_chronic_conditions pc ON pc.patient_id = p.id

    IF p_dob IS NULL THEN

    RETURN

END;
 $$ LANGUAGE plpgsql IMMUTABLE;

-- Get blood pressure category
CREATE OR REPLACE FUNCTION bp_category(p_systolic INT, p_diastolic INT)
RETURNS VARCHAR(50) AS $$ BEGIN
    IF p_systolic IS NULL OR p_diastolic IS NULL THEN
        RETURN NULL;
    ELSIF p_systolic >= 180 OR p_diastolic >= 120 THEN
        RETURN 'Hypertensive Crisis';
    ELSIF p_systolic >= 140 OR p_diastolic >= 90 THEN
        RETURN 'Hypertension Stage 2';
    ELSIF (p_systolic BETWEEN 130 AND 139) OR (p_diastolic BETWEEN 80 AND 89) THEN
        RETURN 'Hypertension Stage 1';
    ELSIF p_systolic < 120 AND p_diastolic < 80 THEN
        RETURN 'Normal';
    ELSE
        RETURN 'Elevated';
    END IF;
END;
 $$ LANGUAGE plpgsql IMMUTABLE;

-- BMI category

    IF p_bmi IS NULL THEN

    ELSIF p_bmi < 18.5 THEN
        RETURN 'Underweight';
    ELSIF p_bmi < 25 THEN
        RETURN 'Normal';
    ELSIF p_bmi < 30 THEN

    ELSE
        RETURN 'Obese';

-- GCS interpretation
CREATE OR REPLACE FUNCTION gcs_interpretation(p_total INT)
RETURNS VARCHAR(100) AS $$ BEGIN
    IF p_total IS NULL THEN
        RETURN NULL;
    ELSIF p_total = 15 THEN
        RETURN 'Normal (Fully Conscious)';
    ELSIF p_total BETWEEN 13 AND 14 THEN
        RETURN 'Mild Impairment';
    ELSIF p_total BETWEEN 9 AND 12 THEN
        RETURN 'Moderate Impairment';
    ELSIF p_total BETWEEN 3 AND 8 THEN
        RETURN 'Severe Impairment (Coma)';
    ELSE
        RETURN 'Invalid GCS Score';
    END IF;
END;
 $$ LANGUAGE plpgsql IMMUTABLE;

-- Auto-calculate BMI trigger
CREATE OR REPLACE FUNCTION update_bmi()
RETURNS TRIGGER AS $$ BEGIN
    IF NEW.height IS NOT NULL AND NEW.weight IS NOT NULL
       AND NEW.height > 0 AND NEW.weight > 0 THEN
        -- BMI = weight(kg) / (height(m))^2
        NEW.bmi := ROUND(
            NEW.weight / POWER(NEW.height / 100.0, 2), 1
        );
    END IF;
    RETURN NEW;
END;
 $$ LANGUAGE plpgsql;

CREATE TRIGGER trg_patient_vitals_bmi
BEFORE INSERT OR UPDATE OF height, weight ON patient_vitals
FOR EACH ROW EXECUTE FUNCTION update_bmi();

-- Prevent deletion of patients with active admissions
CREATE OR REPLACE FUNCTION prevent_patient_deletion()
RETURNS TRIGGER AS $$ BEGIN
    IF EXISTS (
        SELECT 1 FROM admissions a
        WHERE a.patient_id = OLD.id AND a.status = 'admitted'
    ) THEN
        RAISE EXCEPTION 'Cannot delete patient %: patient has an active admission',
            OLD.mrn;
    END IF;

    -- Soft delete instead
    UPDATE patients SET is_active = FALSE WHERE id = OLD.id;
    RETURN NULL;
END;
 $$ LANGUAGE plpgsql;
```

```sql
-- ── STOCK AUTO-UPDATE TRIGGER ──────────────

CREATE OR REPLACE FUNCTION update_stock_on_transaction()
RETURNS TRIGGER AS $$ DECLARE
    v_stock_id UUID;
    v_new_available DECIMAL(12, 2);
BEGIN
    -- Find or create stock record
    SELECT id INTO v_stock_id
    FROM inventory_stock
    WHERE item_id = NEW.item_id
      AND batch_id = COALESCE(NEW.batch_id, '00000000-0000-0000-0000-000000000000')
      AND location_id = NEW.location_id;

    IF v_stock_id IS NULL THEN
        INSERT INTO inventory_stock (
            organization_id, branch_id, item_id, batch_id, location_id,
            qty_available, updated_by
        ) VALUES (
            NEW.organization_id, NEW.branch_id, NEW.item_id,
            COALESCE(NEW.batch_id, '00000000-0000-0000-0000-000000000000'),
            NEW.location_id,
            0, NEW.created_by
        ) RETURNING id INTO v_stock_id;
    END IF;

    -- Update available quantity
    UPDATE inventory_stock
    SET qty_available = qty_available + NEW.quantity,
        is_near_expiry = CASE
            WHEN expiry_date IS NOT NULL
             AND expiry_date <= CURRENT_DATE + INTERVAL '30 days'
            THEN TRUE
            ELSE is_near_expiry
        END,
        is_expired = CASE
            WHEN expiry_date IS NOT NULL
             AND expiry_date < CURRENT_DATE
            THEN TRUE
            ELSE is_expired
        END,
        updated_at = CURRENT_TIMESTAMP,
        updated_by = NEW.created_by
    WHERE id = v_stock_id;

    RETURN NEW;
END;
 $$ LANGUAGE plpgsql;

CREATE TRIGGER trg_stock_transaction_update
AFTER INSERT ON stock_transactions
FOR EACH ROW EXECUTE FUNCTION update_stock_on_transaction();

-- ── EXPIRY ALERT FUNCTION ──────────────────

CREATE OR REPLACE FUNCTION get_expiring_batches(
    p_org_id UUID,
    p_days_ahead INTEGER DEFAULT 90
) RETURNS TABLE (
    item_name VARCHAR,
    batch_number VARCHAR,
    expiry_date DATE,
    days_until_expiry INTEGER,
    total_qty DECIMAL,
    location_name VARCHAR
) AS $$ BEGIN
    RETURN QUERY
    SELECT
        ii.item_name,
        ib.batch_number,
        ib.expiry_date,
        ib.expiry_date - CURRENT_DATE AS days_until_expiry,
        SUM(is2.qty_available) AS total_qty,
        il.name AS location_name
    FROM inventory_batches ib
    JOIN inventory_items ii ON ii.id = ib.item_id
    JOIN inventory_stock is2 ON is2.batch_id = ib.id
    JOIN inventory_locations il ON il.id = is2.location_id
    WHERE ii.organization_id = p_org_id
      AND ib.expiry_date BETWEEN CURRENT_DATE AND CURRENT_DATE + p_days_ahead
      AND is2.qty_available > 0
    GROUP BY ii.item_name, ib.batch_number, ib.expiry_date, il.name
    ORDER BY ib.expiry_date ASC;
END;
 $$ LANGUAGE plpgsql STABLE;

-- ── REORDER ALERT FUNCTION ─────────────────

CREATE OR REPLACE FUNCTION get_reorder_alerts(p_org_id UUID, p_branch_id UUID)
RETURNS TABLE (
    item_code VARCHAR,
    item_name VARCHAR,
    current_stock DECIMAL,
    reorder_level DECIMAL,
    reorder_qty DECIMAL,
    shortage DECIMAL
) AS $$ BEGIN
    RETURN QUERY
    SELECT
        ii.item_code,
        ii.item_name,
        COALESCE(SUM(is2.qty_available), 0) AS current_stock,
        ii.reorder_level,
        ii.reorder_qty,
        ii.reorder_level - COALESCE(SUM(is2.qty_available), 0) AS shortage
    FROM inventory_items ii
    LEFT JOIN inventory_stock is2 ON is2.item_id = ii.id
    JOIN inventory_locations il ON il.id = is2.location_id
    WHERE ii.organization_id = p_org_id
      AND ii.is_active = TRUE
      AND il.branch_id = p_branch_id
      AND ii.reorder_level > 0
    GROUP BY ii.item_code, ii.item_name, ii.reorder_level, ii.reorder_qty
    HAVING COALESCE(SUM(is2.qty_available), 0) <= ii.reorder_level
    ORDER BY shortage DESC;
END;
 $$ LANGUAGE plpgsql STABLE;
```

```sql
-- ── AUTO-CRITICAL FLAG FUNCTION ────────────

CREATE OR REPLACE FUNCTION flag_critical_lab_results()
RETURNS TRIGGER AS $$ DECLARE
    v_test_rec RECORD;
    v_patient_age_years INTEGER;
    v_patient_gender VARCHAR;
    v_range JSONB;
    v_val DECIMAL;
BEGIN
    -- Only for numeric results
    IF NEW.result_numeric IS NULL THEN
        RETURN NEW;
    END IF;

    -- Get test definition
    SELECT * INTO v_test_rec FROM lab_tests WHERE id = NEW.test_id;
    IF v_test_rec.id IS NULL THEN
        RETURN NEW;
    END IF;

    -- Get patient info
    SELECT
        EXTRACT(YEAR FROM age(CURRENT_DATE, p.date_of_birth))::INTEGER,
        p.gender::VARCHAR
    INTO v_patient_age_years, v_patient_gender
    FROM patients p WHERE p.id = NEW.patient_id;

    -- Check against critical values first
    IF v_test_rec.critical_low IS NOT NULL AND NEW.result_numeric < v_test_rec.critical_low THEN
        NEW.is_critical := TRUE;
        NEW.critical_type := 'low';
        NEW.is_abnormal := TRUE;
    ELSIF v_test_rec.critical_high IS NOT NULL AND NEW.result_numeric > v_test_rec.critical_high THEN
        NEW.is_critical := TRUE;
        NEW.critical_type := 'high';
        NEW.is_abnormal := TRUE;
    END IF;

    -- If not critical, check reference ranges
    IF NOT NEW.is_critical AND v_test_rec.reference_ranges IS NOT NULL THEN
        FOR v_range IN SELECT * FROM jsonb_array_elements(v_test_rec.reference_ranges)
        LOOP
            IF (v_range->>'gender' = 'any' OR v_range->>'gender' = v_patient_gender)
               AND (v_patient_age_years BETWEEN (v_range->>'age_min')::INTEGER
                                               AND (v_range->>'age_max')::INTEGER)
            THEN
                NEW.reference_range := (v_range->>'low') || ' - ' || (v_range->>'high') || ' ' || (v_range->>'unit');

                IF NEW.result_numeric < (v_range->>'low')::DECIMAL
                   OR NEW.result_numeric > (v_range->>'high')::DECIMAL
                THEN
                    NEW.is_abnormal := TRUE;
                END IF;

                EXIT;  -- Found matching range
            END IF;
        END LOOP;
    END IF;

    -- Auto-generate critical alert
    IF NEW.is_critical THEN
        INSERT INTO lab_critical_alerts (
            organization_id, lab_order_id, lab_order_item_id,
            patient_id, ordering_doctor_id,
            test_name, result_value, critical_type, reference_range
        ) VALUES (
            NEW.organization_id,
            (SELECT lab_order_id FROM lab_order_items WHERE id = NEW.id),
            NEW.id,
            NEW.patient_id,
            (SELECT ordering_doctor_id FROM lab_orders
             WHERE id = (SELECT lab_order_id FROM lab_order_items WHERE id = NEW.id)),
            v_test_rec.test_name,
            NEW.result_value,
            NEW.critical_type,
            NEW.reference_range
        );
    END IF;

    RETURN NEW;
END;
 $$ LANGUAGE plpgsql;
```

```sql
        quantity * unit_rate
    ) STORED,

-- ── AUTO-INVOICE BALANCE FUNCTION ──────────

CREATE OR REPLACE FUNCTION update_invoice_balance()
RETURNS TRIGGER AS $$ BEGIN
    IF TG_OP = 'INSERT' THEN
        UPDATE invoices
        SET total_paid = total_paid + NEW.amount,
            balance_due = net_payable - (total_paid + NEW.amount),
            status = CASE
                WHEN (net_payable - (total_paid + NEW.amount)) <= 0 THEN 'paid'
                WHEN (total_paid + NEW.amount) > 0 THEN 'partial'
                ELSE status
            END,
            updated_at = CURRENT_TIMESTAMP
        WHERE id = NEW.invoice_id;
    ELSIF TG_OP = 'DELETE' THEN
        UPDATE invoices
        SET total_paid = GREATEST(total_paid - OLD.amount, 0),
            balance_due = LEAST(net_payable - GREATEST(total_paid - OLD.amount, 0), net_payable),
            status = CASE
                WHEN GREATEST(total_paid - OLD.amount, 0) >= net_payable THEN 'paid'
                WHEN GREATEST(total_paid - OLD.amount, 0) > 0 THEN 'partial'
                ELSE 'pending'
            END,
            updated_at = CURRENT_TIMESTAMP
        WHERE id = OLD.invoice_id;
    END IF;
    RETURN COALESCE(NEW, OLD);
END;
 $$ LANGUAGE plpgsql;
```

```sql
-- ── NOTIFICATION DISPATCHER FUNCTION ────────

CREATE OR REPLACE FUNCTION dispatch_notification(
    p_org_id UUID,
    p_recipient_id UUID,
    p_recipient_type VARCHAR(20),
    p_template_code VARCHAR(100),
    p_channel notification_channel_type,
    p_variables JSONB DEFAULT '{}',
    p_reference_type VARCHAR(50) DEFAULT NULL,
    p_reference_id UUID DEFAULT NULL,
    p_scheduled_at TIMESTAMPTZ DEFAULT NULL
) RETURNS UUID AS $$ DECLARE
    v_template RECORD;
    v_phone VARCHAR(20);
    v_email VARCHAR(255);
    v_body TEXT;
    v_subject VARCHAR(255);
    v_notification_id UUID;
BEGIN
    -- Get template
    SELECT * INTO v_template
    FROM notification_templates
    WHERE template_code = p_template_code AND is_active = TRUE;

    IF v_template.id IS NULL THEN
        RAISE EXCEPTION 'Notification template % not found', p_template_code;
    END IF;

    -- Get recipient contact info
    IF p_recipient_type = 'patient' THEN
        SELECT primary_phone, email INTO v_phone, v_email
        FROM patients WHERE id = p_recipient_id;
    ELSIF p_recipient_type = 'user' THEN
        SELECT phone, email INTO v_phone, v_email
        FROM users WHERE id = p_recipient_id;
    END IF;

    -- Render template by replacing variables
    v_body := COALESCE(
        CASE p_channel
            WHEN 'sms' THEN v_template.sms_template
            WHEN 'email' THEN v_template.email_body
            WHEN 'whatsapp' THEN v_template.whatsapp_template
            WHEN 'push' THEN v_template.push_body
            WHEN 'in_app' THEN v_template.push_body
        END,
        'Notification'
    );

    -- Simple variable replacement ({{var}} → value)
    SELECT jsonb_object_keys(p_variables) AS key INTO v_body;
    FOR v_body IN SELECT key FROM jsonb_object_keys(p_variables) AS key
    LOOP
        v_body := REPLACE(v_body, '{{' || v_body || '}}',
            COALESCE(p_variables->>v_body, ''));
    END LOOP;

    v_subject := COALESCE(v_template.email_subject, '');
    IF p_channel = 'email' THEN
        FOR v_subject IN SELECT key FROM jsonb_object_keys(p_variables) AS key
        LOOP
            v_subject := REPLACE(v_subject, '{{' || v_subject || '}}',
                COALESCE(p_variables->>v_subject, ''));
        END LOOP;
    END IF;

    -- Insert notification
    INSERT INTO notifications (
        organization_id, recipient_id, recipient_type,
        recipient_phone, recipient_email,
        template_code, channel, subject, body,
        reference_type, reference_id,
        scheduled_at, status
    ) VALUES (
        p_org_id, p_recipient_id, p_recipient_type,
        v_phone, v_email,
        p_template_code, p_channel, v_subject, v_body,
        p_reference_type, p_reference_id,
        p_scheduled_at,
        CASE WHEN p_scheduled_at IS NOT NULL THEN 'scheduled' ELSE 'pending' END
    ) RETURNING id INTO v_notification_id;

    RETURN v_notification_id;
END;
 $$ LANGUAGE plpgsql;
```

```sql
-- ── 22.1 DASHBOARD: TODAY'S SUMMARY ─────────

CREATE OR REPLACE VIEW v_dashboard_today AS
SELECT
    (SELECT id FROM organizations LIMIT 1) AS organization_id,
    (SELECT id FROM branches WHERE is_main_branch = TRUE LIMIT 1) AS branch_id,
    CURRENT_DATE AS report_date,

    -- OPD
    (SELECT COUNT(*) FROM opd_visits WHERE visit_date = CURRENT_DATE) AS total_opd_visits,
    (SELECT COUNT(*) FROM opd_visits WHERE visit_date = CURRENT_DATE AND status = 'completed') AS completed_opd,

    -- IPD
    (SELECT COUNT(*) FROM admissions WHERE status = 'admitted') AS current_admissions,
    (SELECT COUNT(*) FROM admissions WHERE admission_date = CURRENT_DATE AND status = 'admitted') AS new_admissions,
    (SELECT COUNT(*) FROM admissions WHERE actual_discharge_date = CURRENT_DATE) AS discharges_today,

    -- Emergency
    (SELECT COUNT(*) FROM emergency_visits WHERE arrival_datetime::DATE = CURRENT_DATE) AS emergency_visits_today,

    -- Revenue
    (SELECT COALESCE(SUM(amount), 0) FROM payments WHERE payment_date = CURRENT_DATE) AS total_collections,
    (SELECT COALESCE(SUM(amount), 0) FROM payments WHERE payment_date = CURRENT_DATE AND payment_method = 'cash') AS cash_collections,
    (SELECT COALESCE(SUM(amount), 0) FROM payments WHERE payment_date = CURRENT_DATE AND payment_method = 'card') AS card_collections,
    (SELECT COALESCE(SUM(amount), 0) FROM payments WHERE payment_date = CURRENT_DATE AND payment_method = 'upi') AS upi_collections,

    -- Lab
    (SELECT COUNT(*) FROM lab_orders WHERE ordered_at::DATE = CURRENT_DATE) AS lab_orders_today,
    (SELECT COUNT(*) FROM lab_order_items WHERE result_status = 'pending'
     AND lab_order_id IN (SELECT id FROM lab_orders WHERE ordered_at::DATE = CURRENT_DATE)) AS pending_lab_results,

    -- Radiology
    (SELECT COUNT(*) FROM radiology_orders WHERE ordered_at::DATE = CURRENT_DATE) AS radiology_orders_today,

    -- OT
    (SELECT COUNT(*) FROM ot_surgeries WHERE scheduled_date = CURRENT_DATE AND status = 'completed') AS surgeries_completed,
    (SELECT COUNT(*) FROM ot_surgeries WHERE scheduled_date = CURRENT_DATE AND status = 'scheduled') AS surgeries_scheduled,

    -- Bed Occupancy
    (SELECT COUNT(*) FROM inventory_stock WHERE qty_available > 0
     AND location_id IN (
         SELECT id FROM inventory_locations WHERE location_type = 'pharmacy_dispensary'
     )) AS pharmacy_items_in_stock;

-- ── 22.2 MONTHLY REVENUE REPORT ─────────────

CREATE OR REPLACE VIEW v_monthly_revenue AS
SELECT
    to_char(payment_date, 'YYYY-MM') AS month,
    payment_method,
    COUNT(*) AS transaction_count,
    SUM(amount) AS total_amount,
    AVG(amount) AS avg_transaction
FROM payments
WHERE payment_date >= CURRENT_DATE - INTERVAL '12 months'
GROUP BY to_char(payment_date, 'YYYY-MM'), payment_method

-- ── 22.3 DEPARTMENT-WISE REVENUE ───────────

CREATE OR REPLACE VIEW v_department_revenue AS
SELECT
    cc.name AS charge_category,
    COUNT(DISTINCT pc.patient_id) AS unique_patients,
    COUNT(*) AS total_charges,
    SUM(pc.gross_amount) AS gross_revenue,
    SUM(pc.discount_amount) AS total_discounts,
    SUM(pc.vat_amount) AS total_tax,
    SUM(pc.net_amount) AS net_revenue
FROM patient_charges pc
JOIN charge_categories cc ON cc.id = pc.category_id
WHERE pc.status = 'active'
  AND pc.charge_date >= CURRENT_DATE - INTERVAL '30 days'
GROUP BY cc.name
ORDER BY net_revenue DESC;

-- ── 22.4 DOCTOR PERFORMANCE ────────────────

CREATE OR REPLACE VIEW v_doctor_performance AS
SELECT
    d.id AS doctor_id,
    u.full_name AS doctor_name,
    d.specialization,

    -- OPD
    (SELECT COUNT(*) FROM opd_visits ov WHERE ov.doctor_id = d.id
     AND ov.visit_date >= CURRENT_DATE - INTERVAL '30 days') AS opd_visits_30d,

    (SELECT COUNT(*) FROM prescriptions p WHERE p.doctor_id = d.id
     AND p.prescription_date >= CURRENT_DATE - INTERVAL '30 days') AS prescriptions_30d,

    -- Revenue from consultations
    (SELECT COALESCE(SUM(pc.net_amount), 0) FROM patient_charges pc
     WHERE pc.reference_type = 'consultation'
       AND pc.reference_id IN (SELECT id FROM opd_visits WHERE doctor_id = d.id)
       AND pc.charge_date >= CURRENT_DATE - INTERVAL '30 days') AS consultation_revenue_30d,

    -- IPD
    (SELECT COUNT(*) FROM admissions a WHERE a.primary_doctor_id = d.id
     AND a.admission_date >= CURRENT_DATE - INTERVAL '30 days') AS admissions_30d,

    -- Lab orders
    (SELECT COUNT(*) FROM lab_orders lo WHERE lo.ordering_doctor_id = d.id
     AND lo.ordered_at >= CURRENT_DATE - INTERVAL '30 days') AS lab_orders_30d,

    -- Radiology orders
    (SELECT COUNT(*) FROM radiology_orders ro WHERE ro.ordering_doctor_id = d.id
     AND ro.ordered_at >= CURRENT_DATE - INTERVAL '30 days') AS radiology_orders_30d,

    -- Surgeries
    (SELECT COUNT(*) FROM ot_surgeries os WHERE os.primary_surgeon_id = d.id
     AND os.scheduled_date >= CURRENT_DATE - INTERVAL '30 days'
     AND os.status = 'completed') AS surgeries_30d,

    -- Average TAT for lab orders
    (SELECT AVG(EXTRACT(EPOCH FROM (lo.results_completed_at - lo.ordered_at)) / 60)
     FROM lab_orders lo WHERE lo.ordering_doctor_id = d.id
     AND lo.results_completed_at IS NOT NULL
     AND lo.ordered_at >= CURRENT_DATE - INTERVAL '30 days') AS avg_lab Tat_minutes

FROM doctors d
JOIN users u ON u.id = d.user_id
WHERE d.is_active = TRUE;

-- ── 22.5 BED OCCUPANCY REPORT ──────────────

CREATE OR REPLACE VIEW v_bed_occupancy AS
SELECT
    b.id AS branch_id,
    b.name AS branch_name,
    b.total_beds,

    (SELECT COUNT(*) FROM admissions WHERE status = 'admitted' AND branch_id = b.id) AS occupied_beds,

    CASE WHEN b.total_beds > 0 THEN
        ROUND(
            (SELECT COUNT(*) FROM admissions WHERE status = 'admitted' AND branch_id = b.id)::DECIMAL
            / b.total_beds * 100, 1
        )
    ELSE 0 END AS occupancy_percentage,

    b.total_beds - (SELECT COUNT(*) FROM admissions WHERE status = 'admitted' AND branch_id = b.id) AS available_beds

FROM branches b
WHERE b.is_active = TRUE;

-- ── 22.6 PATIENT DEMOGRAPHICS REPORT ───────

CREATE OR REPLACE VIEW v_patient_demographics AS
SELECT
    gender,
    CASE
        WHEN date_of_birth IS NULL THEN 'Unknown'
        WHEN EXTRACT(YEAR FROM age(CURRENT_DATE, date_of_birth)) < 18 THEN '0-17'
        WHEN EXTRACT(YEAR FROM age(CURRENT_DATE, date_of_birth)) BETWEEN 18 AND 30 THEN '18-30'
        WHEN EXTRACT(YEAR FROM age(CURRENT_DATE, date_of_birth)) BETWEEN 31 AND 45 THEN '31-45'
        WHEN EXTRACT(YEAR FROM age(CURRENT_DATE, date_of_birth)) BETWEEN 46 AND 60 THEN '46-60'
        WHEN EXTRACT(YEAR FROM age(CURRENT_DATE, date_of_birth)) BETWEEN 61 AND 75 THEN '61-75'
        ELSE '75+'
    END AS age_group,

    COUNT(*) AS patient_count,

    -- Registration trend
    COUNT(*) FILTER (WHERE registration_date >= CURRENT_DATE - INTERVAL '30 days') AS new_30d,
    COUNT(*) FILTER (WHERE registration_date >= CURRENT_DATE - INTERVAL '90 days') AS new_90d,
    COUNT(*) FILTER (WHERE registration_date >= CURRENT_DATE - INTERVAL '365 days') AS new_1y

FROM patients
WHERE is_active = TRUE
GROUP BY gender, age_group
ORDER BY age_group, gender;

-- ── 22.7 LAB TAT REPORT ────────────────────

CREATE OR REPLACE VIEW v_lab_tat_report AS
SELECT
    lt.test_code,
    lt.test_name,
    lt.department,
    lt.turnaround_time_minutes AS target_tat_minutes,

    COUNT(*) AS total_tests,
    COUNT(*) FILTER (WHERE loi.result_status = 'final') AS completed_tests,

    ROUND(AVG(
        EXTRACT(EPOCH FROM (loi.verified_at - lo.ordered_at)) / 60
    ) FILTER (WHERE loi.verified_at IS NOT NULL), 0) AS avg_tat_minutes,

    ROUND(MAX(
        EXTRACT(EPOCH FROM (loi.verified_at - lo.ordered_at)) / 60
    ) FILTER (WHERE loi.verified_at IS NOT NULL), 0) AS max_tat_minutes,

    -- Percentage within target TAT
    CASE WHEN COUNT(*) FILTER (WHERE loi.verified_at IS NOT NULL) > 0 THEN
        ROUND(
            COUNT(*) FILTER (
                WHERE loi.verified_at IS NOT NULL
                AND EXTRACT(EPOCH FROM (loi.verified_at - lo.ordered_at)) / 60
                    <= lt.turnaround_time_minutes
            )::DECIMAL
            / COUNT(*) FILTER (WHERE loi.verified_at IS NOT NULL) * 100, 1
        )
    ELSE 0 END AS within_tat_percentage

FROM lab_orders lo
JOIN lab_order_items loi ON loi.lab_order_id = lo.id
JOIN lab_tests lt ON lt.id = loi.test_id
WHERE lo.ordered_at >= CURRENT_DATE - INTERVAL '30 days'
GROUP BY lt.test_code, lt.test_name, lt.department, lt.turnaround_time_minutes
ORDER BY within_tat_percentage ASC;

-- ── 22.8 TOP DISEASES REPORT ───────────────

CREATE OR REPLACE VIEW v_top_diseases AS
SELECT
    ic.code AS icd_code,
    ic.short_description AS diagnosis,
    cd.diagnosis_type,
    COUNT(*) AS occurrence_count,
    COUNT(DISTINCT cd.patient_id) AS unique_patients
FROM clinical_diagnoses cd
LEFT JOIN icd_codes ic ON ic.id = cd.icd_code_id
WHERE cd.diagnosed_at >= CURRENT_DATE - INTERVAL '30 days'
GROUP BY ic.code, ic.short_description, cd.diagnosis_type
ORDER BY occurrence_count DESC
LIMIT 50;

-- ── 22.9 INVENTORY CONSUMPTION REPORT ───────

CREATE OR REPLACE VIEW v_inventory_consumption AS
SELECT
    ii.item_code,
    ii.item_name,
    ii.item_category,
    st.transaction_type,
    SUM(ABS(st.quantity)) AS total_quantity,
    SUM(ABS(st.quantity) * st.unit_cost)
    AS total_value
FROM stock_transactions st
JOIN inventory_items ii ON ii.id = st.item_id
WHERE st.quantity < 0
  AND st.transaction_datetime >= CURRENT_DATE - INTERVAL '30 days'
GROUP BY ii.item_code, ii.item_name, ii.item_category, st.transaction_type
ORDER BY total_value DESC;

-- ── 22.10 INSURANCE CLAIMS AGING REPORT ─────

CREATE OR REPLACE VIEW v_claims_aging AS
SELECT
    icl.id AS claim_id,
    icl.claim_number,
    icl.patient_id,
    p.full_name AS patient_name,
    icl.invoice_amount,
    icl.claimed_amount,
    icl.approved_amount,
    icl.status,
    icl.submitted_at,
    
    -- Aging buckets
    CASE
        WHEN icl.submitted_at IS NULL THEN 'Not Submitted'
        WHEN CURRENT_DATE - icl.submitted_at::DATE <= 15 THEN '0-15 days'
        WHEN CURRENT_DATE - icl.submitted_at::DATE <= 30 THEN '16-30 days'
        WHEN CURRENT_DATE - icl.submitted_at::DATE <= 60 THEN '31-60 days'
        WHEN CURRENT_DATE - icl.submitted_at::DATE <= 90 THEN '61-90 days'
        ELSE '90+ days'
    END AS aging_bucket,

    CURRENT_DATE - icl.submitted_at::DATE AS days_since_submission,

    -- Estimated loss
    CASE
        WHEN icl.status = 'rejected' THEN icl.claimed_amount
        WHEN icl.status IN ('partial_approved') THEN (icl.claimed_amount - COALESCE(icl.approved_amount, 0))
        ELSE 0
    END AS estimated_loss

FROM insurance_claims icl
JOIN patients p ON p.id = icl.patient_id
WHERE icl.status NOT IN ('draft', 'settled')
ORDER BY days_since_submission DESC NULLS LAST;
```

### 23.4 Row-Level Security (Multi-Tenant Isolation)

```sql
-- ============================================
-- STEP 23.4: ROW-LEVEL SECURITY (RLS)
-- ============================================
-- RLS ensures that even if a query forgets the WHERE organization_id = ? clause,
-- data from other tenants is NEVER returned.

-- Enable RLS on all tenant-scoped tables
DO $$ DECLARE
    v_table TEXT;
BEGIN
    FOR v_table IN SELECT table_name FROM information_schema.tables
                   WHERE table_schema = 'public'
                     AND table_type = 'BASE TABLE'
                     AND table_name NOT IN (
                         'countries', 'divisions', 'districts', 'upazilas',
                         'icd_codes', 'cpt_codes', 'drug_master', 'drug_interactions',
                         'allergens', 'lab_tests', 'radiology_modalities', 'radiology_exams',
                         'surgical_types', 'charge_categories', 'charge_items',
                         'notification_templates', 'permissions', 'roles',
                         'holidays'
                     )
    LOOP
        EXECUTE format('ALTER TABLE %I ENABLE ROW LEVEL SECURITY', v_table);
    END LOOP;
END;
 $$;

-- ── RLS POLICIES ────────────────────────────
-- These policies use the current_setting function to read a session variable
-- that the application MUST set after authentication.
-- SET LOCAL app.organization_id = 'uuid-here';
-- SET LOCAL app.branch_id = 'uuid-here';  (optional)

-- Helper function to create standard RLS policies
CREATE OR REPLACE FUNCTION apply_rls_policy(
    p_table_name TEXT,
    p_org_column TEXT DEFAULT 'organization_id',
    p_branch_column TEXT DEFAULT NULL
) RETURNS VOID AS $$ BEGIN
    -- Read policy: user can see rows from their org (and optionally branch)
    EXECUTE format('
        CREATE POLICY rls_%s_read ON %s
            FOR SELECT
            USING (
                %s = current_setting(''app.organization_id'', TRUE)::UUID
                %s
            )
    ', p_table_name, p_table_name, p_org_column,
        CASE WHEN p_branch_column IS NOT NULL THEN
            format(' AND (%s IS NULL OR %s = current_setting(''app.branch_id'', TRUE)::UUID)', p_branch_column, p_branch_column)
        ELSE '' END
    );

    -- Insert policy
    EXECUTE format('
        CREATE POLICY rls_%s_insert ON %s
            FOR INSERT
            WITH CHECK (
                %s = current_setting(''app.organization_id'', TRUE)::UUID
            )
    ', p_table_name, p_table_name, p_org_column);

    -- Update policy
    EXECUTE format('
        CREATE POLICY rls_%s_update ON %s
            FOR UPDATE
            USING (
                %s = current_setting(''app.organization_id'', TRUE)::UUID
            )
            WITH CHECK (
                %s = current_setting(''app.organization_id'', TRUE)::UUID
            )
    ', p_table_name, p_table_name, p_org_column, p_org_column);

    -- Delete policy
    EXECUTE format('
        CREATE POLICY rls_%s_delete ON %s
            FOR DELETE
            USING (
                %s = current_setting(''app.organization_id'', TRUE)::UUID
            )
    ', p_table_name, p_table_name, p_org_column);
END;
 $$ LANGUAGE plpgsql;

-- Apply RLS to core tables

SELECT apply_rls_policy('appointments');
SELECT apply_rls_policy('opd_visits');
SELECT apply_rls_policy('admissions');

SELECT apply_rls_policy('ot_surgeries');
SELECT apply_rls_policy('patient_charges');
SELECT apply_rls_policy('invoices');
SELECT apply_rls_policy('payments');
SELECT apply_rls_policy('inventory_items');
```

```sql
CREATE OR REPLACE FUNCTION update_bed_status_on_allocation()
RETURNS TRIGGER AS $$ BEGIN
    IF TG_OP = 'INSERT' THEN
        -- Mark bed as occupied
        UPDATE beds SET status = 'occupied', updated_at = CURRENT_TIMESTAMP
        WHERE id = NEW.bed_id;

        -- Update admission current bed
        UPDATE admissions
        SET current_ward_id = NEW.ward_id,
            current_bed_id = NEW.bed_id,
            updated_at = CURRENT_TIMESTAMP
        WHERE id = NEW.admission_id;

    ELSIF TG_OP = 'UPDATE' THEN
        -- If bed is being released
        IF OLD.release_datetime IS NULL AND NEW.release_datetime IS NOT NULL THEN

    END IF;

END;
 $$ LANGUAGE plpgsql;
```

## STEP 25: Materialized Views (Dashboard Performance)

> **Design Rationale:** Standard `VIEWS` execute the underlying query every time they are accessed. For a hospital dashboard that hundreds of users check simultaneously, this causes massive CPU load. `MATERIALIZED VIEWS` cache the result and are refreshed on a schedule (e.g., every 5 minutes).

```sql
-- ============================================
-- STEP 25: MATERIALIZED VIEWS FOR ANALYTICS
-- ============================================

-- ── 25.1 DASHBOARD SNAPSHOT ─────────────────

CREATE MATERIALIZED VIEW mv_dashboard_today AS
SELECT
    (SELECT id FROM organizations LIMIT 1) AS organization_id,
    CURRENT_DATE AS report_date,
    CURRENT_TIMESTAMP AS last_refreshed,

    -- OPD Metrics
    (SELECT COUNT(*) FROM opd_visits WHERE visit_date = CURRENT_DATE) AS total_opd_visits,
    (SELECT COUNT(*) FROM opd_visits WHERE visit_date = CURRENT_DATE AND status = 'completed') AS completed_opd,
    (SELECT COUNT(*) FROM appointments WHERE appointment_date = CURRENT_DATE AND status IN ('scheduled', 'checked_in')) AS pending_appointments,

    -- IPD Metrics
    (SELECT COUNT(*) FROM admissions WHERE status = 'admitted') AS total_ipd_admissions,
    (SELECT COUNT(*) FROM admissions WHERE admission_date = CURRENT_DATE AND status IN ('admitted', 'discharged')) AS new_admissions_today,
    (SELECT COUNT(*) FROM admissions WHERE actual_discharge_date = CURRENT_DATE) AS discharges_today,

    -- Bed Metrics
    (SELECT COUNT(*) FROM beds WHERE status = 'available') AS available_beds,
    (SELECT COUNT(*) FROM beds WHERE status = 'occupied') AS occupied_beds,

    -- Emergency
    (SELECT COUNT(*) FROM emergency_visits WHERE arrival_datetime::DATE = CURRENT_DATE) AS emergency_visits_today,
    (SELECT COUNT(*) FROM emergency_visits WHERE arrival_datetime::DATE = CURRENT_DATE AND status = 'waiting') AS emergency_waiting,

    -- Revenue Metrics
    (SELECT COALESCE(SUM(amount), 0) FROM payments WHERE payment_date = CURRENT_DATE) AS total_collection_today,
    (SELECT COALESCE(SUM(amount), 0) FROM payments WHERE payment_date = CURRENT_DATE AND payment_method = 'cash') AS cash_collection,
    (SELECT COALESCE(SUM(amount), 0) FROM payments WHERE payment_date = CURRENT_DATE AND payment_method IN ('card', 'upi', 'mobile_banking')) AS digital_collection,

    -- Lab Metrics
    (SELECT COUNT(*) FROM lab_orders WHERE ordered_at::DATE = CURRENT_DATE) AS lab_orders_today,
    (SELECT COUNT(*) FROM lab_order_items WHERE result_status = 'pending'
     AND lab_order_id IN (SELECT id FROM lab_orders WHERE ordered_at::DATE = CURRENT_DATE)) AS pending_lab_results,

    -- OT Metrics
    (SELECT COUNT(*) FROM ot_surgeries WHERE scheduled_date = CURRENT_DATE AND status = 'completed') AS surgeries_completed_today,
    (SELECT COUNT(*) FROM ot_surgeries WHERE scheduled_date = CURRENT_DATE AND status IN ('scheduled', 'in_progress')) AS ongoing_surgeries
WITH DATA;

-- Create a unique index to allow concurrent refresh without locking
CREATE UNIQUE INDEX idx_mv_dashboard_today ON mv_dashboard_today(organization_id, report_date);

-- Refresh function
CREATE OR REPLACE FUNCTION refresh_dashboard_today()
RETURNS VOID AS $$ BEGIN
    REFRESH MATERIALIZED VIEW CONCURRENTLY mv_dashboard_today;
END;
 $$ LANGUAGE plpgsql;

CREATE MATERIALIZED VIEW mv_monthly_revenue AS
SELECT
    to_char(p.payment_date, 'YYYY-MM') AS month,
    p.payment_method,
    COUNT(*) AS transaction_count,
    SUM(p.amount) AS total_amount,
    AVG(p.amount) AS avg_transaction_value
FROM payments p

    ELSE 0 END AS occupancy_pct
```

## STEP 28: Database Backup, Maintenance & Optimization Scripts

```sql
-- ============================================
-- STEP 28: MAINTENANCE & OPTIMIZATION
-- ============================================

-- ── 28.1 COMPREHENSIVE DATABASE HEALTH CHECK ──

CREATE OR REPLACE FUNCTION db_health_check()
RETURNS TABLE (
    check_name VARCHAR,
    status VARCHAR,
    details TEXT
) AS $$ BEGIN
    -- 1. Missing FK Indexes (Foreign keys without indexes cause table locks on deletes)
    RETURN QUERY
    SELECT
        'Missing FK Indexes'::VARCHAR,
        CASE WHEN COUNT(*) > 0 THEN 'WARNING' ELSE 'OK' END,
        COUNT(*)::TEXT || ' foreign key columns lack indexes (can cause slow DELETEs/UPDATEs)'
    FROM (
        SELECT tc.table_schema, tc.constraint_name, tc.table_name, kcu.column_name
        FROM information_schema.table_constraints tc
        JOIN information_schema.key_column_usage kcu ON tc.constraint_name = kcu.constraint_name
        WHERE tc.constraint_type = 'FOREIGN KEY'
          AND NOT EXISTS (
              SELECT 1 FROM pg_indexes pi
              WHERE pi.tablename = tc.table_name
                AND pi.indexdef LIKE '%' || kcu.column_name || '%'
          )
    ) missing_idx;

    -- 2. Unused Indexes
    RETURN QUERY
    SELECT
        'Unused Indexes'::VARCHAR,
        CASE WHEN COUNT(*) > 0 THEN 'WARNING' ELSE 'OK' END,
        COUNT(*)::TEXT || ' indexes have never been used (waste storage and slow writes)'
    FROM pg_stat_user_indexes
    WHERE idx_scan = 0
      AND indexrelname NOT LIKE '%_pkey';

    -- 3. Bloat (Tables needing VACUUM)
    RETURN QUERY
    SELECT
        'Table Bloat'::VARCHAR,
        CASE WHEN COUNT(*) > 0 THEN 'WARNING' ELSE 'OK' END,
        COUNT(*)::TEXT || ' tables have > 20% bloat and need VACUUM'
    FROM (
        SELECT relname, n_live_tup, last_vacuum, last_autovacuum
        FROM pg_stat_user_tables
        WHERE n_dead_tup * 100.0 / NULLIF(n_live_tup + n_dead_tup, 0) > 20
    ) bloat;

    -- 4. Row-Level Security Status
    RETURN QUERY
    SELECT
        'RLS Enabled'::VARCHAR,
        'OK'::VARCHAR,
        COUNT(*)::TEXT || ' tables have Row Level Security enabled'
    FROM pg_policies
    GROUP BY TRUE;

    -- 5. Sequence Exhaustion Check
    RETURN QUERY
    SELECT
        'Sequence Exhaustion'::VARCHAR,
        CASE WHEN COUNT(*) > 0 THEN 'CRITICAL' ELSE 'OK' END,
        string_agg(seq_name || ' (' || remaining::TEXT || ' left)', ', ')
    FROM (
        SELECT s.relname AS seq_name,
               CASE WHEN s.relkind = 'S'
                    THEN (SELECT MAX(id) FROM (
                        SELECT table_name FROM information_schema.columns
                        WHERE column_default LIKE '%nextval(%' || s.relname || '%'
                    ) t)
                    ELSE NULL
               END AS max_id,
               9223372036854775807 AS max_bigint
        FROM pg_class s
        WHERE s.relkind = 'S'
    ) seq_check;
    -- Note: UUID sequences never exhaust. This is mostly for BIGINT serials if any exist.

    -- 6. Long Running Queries
    RETURN QUERY
    SELECT
        'Active Locks'::VARCHAR,
        CASE WHEN COUNT(*) > 0 THEN 'WARNING' ELSE 'OK' END,
        COUNT(*)::TEXT || ' active locks on tables'
    FROM pg_locks l
    JOIN pg_class c ON c.oid = l.relation
    WHERE l.granted = false; -- Blocked locks

END;
 $$ LANGUAGE plpgsql STABLE;
```

## STEP 29: Complex Domain Workflows (Stored Procedures)

> **Design Rationale:** Hospital operations are not simple CRUD. Admitting a patient requires reserving a bed, updating the admission status, triggering an invoice, and notifying the attendant—all atomically. If any step fails, the whole transaction must roll back.

### 29.1 Complete OPD Visit Workflow

```sql
-- ============================================
-- STEP 29.1: OPD VISIT WORKFLOW
-- ============================================
-- Handles: Check-in → Create Visit → Queue Vitals → Generate Charge

CREATE OR REPLACE FUNCTION sp_opd_checkin_and_create_visit(
    p_organization_id UUID,
    p_branch_id UUID,
    p_appointment_id UUID,
    p_recorded_by_user UUID,
    p_chief_complaint TEXT DEFAULT NULL
) RETURNS TABLE (
    visit_id UUID,
    visit_number VARCHAR,
    status VARCHAR,
    message TEXT
) AS $$ DECLARE
    v_appointment RECORD;
    v_visit_number VARCHAR;
    v_visit_id UUID;
    v_visit_charge_id UUID;
BEGIN
    -- 1. Lock the appointment row to prevent double check-in
    SELECT * INTO v_appointment
    FROM appointments
    WHERE id = p_appointment_id
      AND organization_id = p_organization_id
    FOR UPDATE;

    IF v_appointment IS NULL THEN
        RETURN QUERY SELECT NULL::UUID, NULL::VARCHAR, 'error'::VARCHAR, 'Appointment not found'::TEXT;
        RETURN;
    END IF;

    IF v_appointment.status != 'scheduled' THEN
        RETURN QUERY SELECT NULL::UUID, NULL::VARCHAR, 'error'::VARCHAR, 
            'Appointment already processed (Status: ' || v_appointment.status || ')'::TEXT;
        RETURN;
    END IF;

    -- 2. Generate Visit Number
    v_visit_number := 'OPD-' || to_char(CURRENT_DATE, 'YYYYMMDD') || '-' || 
                      LPAD(nextval('pg_temp.visit_seq')::TEXT, 4, '0');
    -- Note: In production, use a sequence generator like generate_visit_number()

    -- 3. Create OPD Visit
    INSERT INTO opd_visits (
        organization_id, branch_id, visit_number, appointment_id,
        patient_id, doctor_id, visit_date, visit_time,
        chief_complaint, vitals_recorded, status
    ) VALUES (
        v_appointment.organization_id, v_appointment.branch_id, v_visit_number,
        p_appointment_id, v_appointment.patient_id, v_appointment.doctor_id,
        v_appointment.appointment_date, v_appointment.appointment_time,
        p_chief_complaint, FALSE, 'checked_in'
    ) RETURNING id INTO v_visit_id;

    -- 4. Update Appointment Status
    UPDATE appointments
    SET status = 'checked_in',
        checked_in_at = CURRENT_TIMESTAMP,
        updated_at = CURRENT_TIMESTAMP
    WHERE id = p_appointment_id;

    -- 5. Auto-create Consultation Charge
    INSERT INTO patient_charges (
        organization_id, branch_id, charge_number, patient_id,
        category_id, item_name, quantity, unit_rate,
        reference_type, reference_id, charged_by
    ) VALUES (
        p_organization_id, p_branch_id, 
        'CHG-' || to_char(CURRENT_TIMESTAMP, 'YYYYMMDDHH24MISS') || '-' || (random() * 1000)::INT,
        v_appointment.patient_id,
        (SELECT id FROM charge_categories WHERE code = 'CONS' LIMIT 1),
        'Consultation Fee - OPD',
        1, v_appointment.consultation_fee,
        'consultation', v_visit_id, p_recorded_by_user
    ) RETURNING id INTO v_visit_charge_id;

    -- 6. Return Success
    RETURN QUERY SELECT 
        v_visit_id, 
        v_visit_number, 
        'success'::VARCHAR, 
        'Patient checked in successfully. Please record vitals.'::TEXT;

EXCEPTION WHEN OTHERS THEN
    RETURN QUERY SELECT NULL::UUID, NULL::VARCHAR, 'error'::VARCHAR, SQLERRM;
END;
 $$ LANGUAGE plpgsql;
```

### 29.2 Complete IPD Admission Workflow

```sql
-- ============================================
-- STEP 29.2: IPD ADMISSION WORKFLOW
-- ============================================
-- Handles: Validate Bed -> Create Admission -> Allocate Bed -> Create Deposit Charge

CREATE OR REPLACE FUNCTION sp_admit_patient(
    p_organization_id UUID,
    p_branch_id UUID,
    p_patient_id UUID,
    p_doctor_id UUID,
    p_ward_id UUID,
    p_bed_id UUID,
    p_admission_type VARCHAR,
    p_admission_source VARCHAR,
    p_provisional_diagnosis TEXT DEFAULT NULL,
    p_deposit_amount DECIMAL DEFAULT 0,
    p_created_by UUID
) RETURNS TABLE (
    admission_id UUID,
    admission_number VARCHAR,
    bed_status VARCHAR,
    message TEXT
) AS $$ DECLARE
    v_admission_number VARCHAR;
    v_admission_id UUID;
    v_bed_status VARCHAR;
    v_daily_rate DECIMAL;
BEGIN
    -- 1. Check if patient already admitted
    IF EXISTS (SELECT 1 FROM admissions WHERE patient_id = p_patient_id AND status = 'admitted') THEN
        RETURN QUERY SELECT NULL::UUID, NULL::VARCHAR, 'error'::VARCHAR, 
            'Patient is already admitted (Active IPD found)'::TEXT;
        RETURN;
    END IF;

    -- 2. Lock bed row to prevent double allocation (Race condition prevention)
    SELECT status INTO v_bed_status FROM beds WHERE id = p_bed_id FOR UPDATE;
    
    IF v_bed_status IS NULL THEN
        RETURN QUERY SELECT NULL::UUID, NULL::VARCHAR, 'error'::VARCHAR, 'Bed not found'::TEXT;
        RETURN;
    ELSIF v_bed_status != 'available' THEN
        RETURN QUERY SELECT NULL::UUID, NULL::VARCHAR, 'error'::VARCHAR, 
            'Bed is not available (Current status: ' || v_bed_status || ')'::TEXT;
        RETURN;
    END IF;

    -- 3. Get daily rate
    SELECT COALESCE(b.rate, bt.default_rate) INTO v_daily_rate
    FROM beds b
    JOIN bed_types bt ON bt.id = b.bed_type_id
    WHERE b.id = p_bed_id;

    -- 4. Generate Admission Number
    v_admission_number := 'IP-DEMO-' || to_char(CURRENT_DATE, 'YY') || 
                         LPAD((SELECT COALESCE(MAX(CAST(SUBSTRING(admission_number FROM '[0-9]+$') AS INTEGER)),0) + 1 
                               FROM admissions WHERE admission_date = CURRENT_DATE)::TEXT, 5, '0');

    -- 5. Create Admission Record
    INSERT INTO admissions (
        organization_id, branch_id, admission_number, patient_id,
        admission_date, admission_time, admission_type, admission_source,
        status, primary_doctor_id, department_id,
        current_ward_id, current_bed_id,
        provisional_diagnosis, created_by
    ) VALUES (
        p_organization_id, p_branch_id, v_admission_number, p_patient_id,
        CURRENT_DATE, CURRENT_TIME, p_admission_type, p_admission_source,
        'admitted', p_doctor_id, 
        (SELECT department_id FROM doctors WHERE id = p_doctor_id),
        p_ward_id, p_bed_id,
        p_provisional_diagnosis, p_created_by
    ) RETURNING id INTO v_admission_id;

    -- 6. Allocate Bed
    INSERT INTO bed_allocations (admission_id, ward_id, bed_id, allocation_type, daily_rate, allocated_by)
    VALUES (v_admission_id, p_ward_id, p_bed_id, 'initial', v_daily_rate, p_created_by);

    -- 7. Mark Bed as Occupied
    UPDATE beds SET status = 'occupied', updated_at = CURRENT_TIMESTAMP WHERE id = p_bed_id;

    -- 8. Create Admission Deposit Record (if amount > 0)
    IF p_deposit_amount > 0 THEN
        INSERT INTO admission_deposits (admission_id, deposit_number, amount, payment_method, deposited_by, created_by)
        VALUES (
            v_admission_id, 
            'DEP-' || to_char(CURRENT_TIMESTAMP, 'YYYYMMDDHH24MISS'),
            p_deposit_amount, 'cash', 'Self', p_created_by
        );
    END IF;

    -- Success
    RETURN QUERY SELECT 
        v_admission_id, 
        v_admission_number, 
        'occupied'::VARCHAR, 
        'Patient admitted successfully to Bed ' || (SELECT bed_number FROM beds WHERE id = p_bed_id)::TEXT;

EXCEPTION WHEN OTHERS THEN
    -- Rollback is automatic in PL/pgSQL on exception
    RETURN QUERY SELECT NULL::UUID, NULL::VARCHAR, 'error'::VARCHAR, 
        'Admission failed: ' || SQLERRM;
END;
 $$ LANGUAGE plpgsql;
```

### 29.3 Complete Discharge Workflow

```sql
-- ============================================
-- STEP 29.3: IPD DISCHARGE WORKFLOW
-- ============================================
-- Handles: Release Bed -> Mark Discharged -> Set Bed to Cleaning

CREATE OR REPLACE FUNCTION sp_discharge_patient(
    p_admission_id UUID,
    p_discharge_type VARCHAR DEFAULT 'normal',
    p_discharged_by UUID
) RETURNS TABLE (
    admission_id UUID,
    bed_id UUID,
    message TEXT
) AS $$ DECLARE
    v_admission RECORD;
    v_bed_id UUID;
BEGIN
    -- 1. Lock admission row
    SELECT * INTO v_admission FROM admissions WHERE id = p_admission_id FOR UPDATE;
    
    IF v_admission IS NULL OR v_admission.status != 'admitted' THEN
        RETURN QUERY SELECT NULL::UUID, NULL::UUID, 'Invalid or non-active admission'::TEXT;
        RETURN;
    END IF;

    v_bed_id := v_admission.current_bed_id;

    -- 2. Update Admission Status
    UPDATE admissions
    SET status = 'discharged',
        actual_discharge_date = CURRENT_DATE,
        actual_discharge_time = CURRENT_TIME,
        discharge_type = p_discharge_type,
        updated_at = CURRENT_TIMESTAMP
    WHERE id = p_admission_id;

    -- 3. Release Bed Allocation
    UPDATE bed_allocations
    SET release_datetime = CURRENT_TIMESTAMP,
        released_by = p_discharged_by
    WHERE admission_id = p_admission_id
      AND release_datetime IS NULL;

    -- 4. Set Bed to Cleaning (Trigger handles status change)
    IF v_bed_id IS NOT NULL THEN
        UPDATE beds SET status = 'cleaning', updated_at = CURRENT_TIMESTAMP WHERE id = v_bed_id;
        
        -- Update admission to clear current bed
        UPDATE admissions SET current_bed_id = NULL, current_ward_id = NULL WHERE id = p_admission_id;
    END IF;

    -- 5. Mark patient as not currently admitted (if applicable)
    -- (Handled by application logic usually, but good for DB consistency)

    RETURN QUERY SELECT 
        p_admission_id, 
        v_bed_id, 
        'Patient discharged. Bed set to cleaning.'::TEXT;

EXCEPTION WHEN OTHERS THEN
    RETURN QUERY SELECT NULL::UUID, NULL::UUID, 'Discharge failed: ' || SQLERRM;
END;
 $$ LANGUAGE plpgsql;
```
