// MULTI-TIMEFRAME BIAS
// ---------------------------------------------------------------------
f_htfBias() =>
    float e   = ta.ema(close, emaLen3Eff)
    int   emaB = na(e) ? 0 : close > e ? 1 : close < e ? -1 : 0
    phH = ta.pivothigh(high, swingLenEff, swingLenEff)
    plL = ta.pivotlow(low,  swingLenEff, swingLenEff)
    var float lH = na
    var float lL = na
    var int   sB = 0
    if not na(phH)
        lH := phH
    if not na(plL)
        lL := plL
    if not na(lH) and close > lH and close[1] <= lH
        sB := 1
    if not na(lL) and close < lL and close[1] >= lL
        sB := -1
    useHTFstruct ? sB : emaB

htfBias(simple string tf) =>
    request.security(syminfo.tickerid, tf, f_htfBias()[1], lookahead=barmerge.lookahead_off)

    if not na(_a) and _a > _p
        r := na(r) ? _a : math.min(r, _a)
    if not na(_b) and _b > _p
        r := na(r) ? _b : math.min(r, _b)
    r
f_maxBelow(_p, _a, _b) =>
    float r = na
    if not na(_a) and _a < _p
        r := na(r) ? _a : math.max(r, _a)
    if not na(_b) and _b < _p
        r := na(r) ? _b : math.max(r, _b)
    r
f_poolAbove(_arr, _p, _tol) =>
    float best = na
    int n = array.size(_arr)
    if n > 0
        for i = 0 to n - 1
            lvl = array.get(_arr, i)
            if lvl > _p
                int touches = 0
                for j = 0 to n - 1
                    if math.abs(array.get(_arr, j) - lvl) <= _tol
                        touches += 1
                if touches >= 2 and (na(best) or lvl < best)
                    best := lvl
    best
f_poolBelow(_arr, _p, _tol) =>
    float best = na
    int n = array.size(_arr)
    if n > 0
        for i = 0 to n - 1
            lvl = array.get(_arr, i)
            if lvl < _p
                int touches = 0
                for j = 0 to n - 1
                    if math.abs(array.get(_arr, j) - lvl) <= _tol
                        touches += 1
                if touches >= 2 and (na(best) or lvl > best)
                    best := lvl
    best

shortScore += ((liqScore and liqSweepBear) or (smtOn and smtBear) or rsiBearDiv or rsiHidBearDiv or inducementBear or judasBear) ? 2 : 0
shortScore += (not useHTF and shortBiasOk) ? 1 : 0
shortScore += ((useEMAscore and emaBear) or (useTLBreak and tlBreakBear) or (useVWAPscore and vwapBear)) ? 1 : 0
shortScore += mssBear ? 1 : 0

// ---------------------------------------------------------------------
// SL / TP (market-on-close basis)
// ---------------------------------------------------------------------

spreadPrice = spreadPipsEff * pip
minSLdist = (minSLatr > 0 and not na(atr)) ? atr * minSLatr : 0.0
slLongAnchor  = (mtfOn and not na(mtfLegLo)) ? math.min(mtfLegLo, low)  : (na(lastPL) ? low  : math.min(lastPL, low))
slShortAnchor = (mtfOn and not na(mtfLegHi)) ? math.max(mtfLegHi, high) : (na(lastPH) ? high : math.max(lastPH, high))
longSL  = math.min(slLongAnchor  - bufferPips * pip - spreadPrice, close - minSLdist)
shortSL = math.max(slShortAnchor + bufferPips * pip + spreadPrice, close + minSLdist)

longRisk  = close - longSL
shortRisk = shortSL - close

longSLok  = maxSLatr == 0 or (not na(atr) and longRisk  <= atr * maxSLatr)
shortSLok = maxSLatr == 0 or (not na(atr) and shortRisk <= atr * maxSLatr)

// cooldown after a tracked loss + volume gate
var int lastLossBar = na

// ---------------------------------------------------------------------
// LIGHTWEIGHT TRADE TRACKER — STATE + EXIT (runs BEFORE the signal decision)
//   FIX #9: exit/management runs here so a position that closes THIS bar frees
//   the slot and a fresh signal can re-enter the same bar (no dead-bar gap).
//   one position at a time; SL/TP/BE/trail/time-stop exit; win/loss + R stats.
// ---------------------------------------------------------------------
var int   posDir    = 0
var float entryPx   = na
var float curSL     = na
var float curTP     = na
var float initRisk  = na
var int   entryBar  = na
var bool  partialDone = false
var float trailStop = na

var int   rtTrades = 0
var int   rtWins   = 0
var int   rtLosses = 0
var int   rtConsec = 0
var int   rtMaxConsec = 0
var float rtSumR   = 0.0
var float rtGainR  = 0.0
var float rtLossR  = 0.0

exitNow   = false
float exitPx = na
isWin     = false

if posDir == 1 and not na(entryPx) and not na(initRisk) and initRisk > 0
    if useBE and high >= entryPx + beTrigR * initRisk
        curSL := math.max(nz(curSL, entryPx), entryPx + beOffPips * pip)
    if useTrail
        trailStop := na(trailStop) ? close - atr * atrMult : math.max(trailStop, close - atr * atrMult)
        curSL := math.max(nz(curSL, trailStop), trailStop)
    if usePartial and not partialDone and high >= entryPx + partialR * initRisk
        partialDone := true
    // exit checks — SL takes priority on an ambiguous bar (conservative)
    if low <= curSL
        exitNow := true
        exitPx  := curSL
        isWin   := curSL > entryPx
    else if high >= curTP
        exitNow := true
        exitPx  := curTP
        isWin   := true
    else if useMaxHold and not na(entryBar) and (bar_index - entryBar) >= maxHoldBars
        exitNow := true
        exitPx  := close
        isWin   := close > entryPx

if posDir == -1 and not na(entryPx) and not na(initRisk) and initRisk > 0
    if useBE and low <= entryPx - beTrigR * initRisk
        curSL := math.min(nz(curSL, entryPx), entryPx - beOffPips * pip)
    if useTrail
        trailStop := na(trailStop) ? close + atr * atrMult : math.min(trailStop, close + atr * atrMult)
        curSL := math.min(nz(curSL, trailStop), trailStop)
    if usePartial and not partialDone and low <= entryPx - partialR * initRisk
        partialDone := true
    if high >= curSL
        exitNow := true
        exitPx  := curSL
        isWin   := curSL < entryPx
    else if low <= curTP
        exitNow := true
        exitPx  := curTP
        isWin   := true
    else if useMaxHold and not na(entryBar) and (bar_index - entryBar) >= maxHoldBars
        exitNow := true
        exitPx  := close
        isWin   := close < entryPx

// finalize the tracked round-trip
if exitNow
    float rMul = (posDir == 1 ? (exitPx - entryPx) : (entryPx - exitPx)) / initRisk
    rtTrades += 1
    rtSumR   += rMul
    if rMul > 0
        rtWins  += 1
        rtGainR += rMul
        rtConsec := 0
    else if rMul < 0
        rtLosses += 1
        rtLossR  += -rMul
        rtConsec += 1
        rtMaxConsec := math.max(rtMaxConsec, rtConsec)
        lastLossBar := bar_index
    else
        rtConsec := 0
    posDir := 0
    entryPx := na
    curSL := na
    curTP := na
    initRisk := na
    entryBar := na
    partialDone := false
    trailStop := na

coolOk = not useCooldown or na(lastLossBar) or (bar_index - lastLossBar) > coolBars
volOk = not useVolFilter or (not na(volume) and volume > ta.sma(volume, volLen) * volMult)

goShortRaw = tradeE and htfSE and extSE and adxE and smtSE and judasSE and shortArmed and shortZone and shortFib and shortTrig and shortScore >= minScoreEff and shortSLok and coolE and volE and flatNow
goLong  = goLongRaw  and (not goShortRaw or longScore  > shortScore)
goShort = goShortRaw and (not goLongRaw  or shortScore > longScore)

longTP    = close + longRisk  * rr - spreadPrice
shortTP   = close - shortRisk * rr + spreadPrice

shortTP := useLiqTP and not na(liqShortTP) and liqShortTP < close - shortRisk ? liqShortTP : shortTP
longTP  := longRisk  > 0 ? math.min(longTP,  close + longRisk  * maxTPr) : longTP
shortTP := shortRisk > 0 ? math.max(shortTP, close - shortRisk * maxTPr) : shortTP

// suggested position size for a manual trader (units), capped by nothing here

shortUnits = shortRisk > 0 ? (acctSize * riskPerc / 100) / shortRisk : na

    initRisk := longRisk
    entryBar := bar_index
