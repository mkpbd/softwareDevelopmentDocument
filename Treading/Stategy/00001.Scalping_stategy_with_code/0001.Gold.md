inSession() =>
    hour_val = hour
    in_london = hour_val >= london_start and hour_val < london_end
    in_ny = hour_val >= ny_start and hour_val < ny_end
    not_asian = avoid_asian ? (hour_val < 23 or hour_val >= 7) : true
    (in_london or in_ny) and not_asian

swingLow = ta.lowest(low, swing_lookback)[1]
swingHigh = ta.highest(high, swing_lookback)[1]

liquiditySweepBuy() =>
    broke_low = low < swingLow[1]
    sufficient_break = (swingLow[1] - low) >= sweep_min_points * syminfo.mintick
    rejection = close > open or (low < open and close > low + (high - low) * 0.4)
    not_continuation = (close - open) > -(high - low) * 0.6
    broke_low and sufficient_break and rejection and not_continuation

liquiditySweepSell() =>
    broke_high = high > swingHigh[1]
    sufficient_break = (high - swingHigh[1]) >= sweep_min_points * syminfo.mintick
    rejection = close < open or (high > open and close < high - (high - low) * 0.4)
    not_continuation = (open - close) > -(high - low) * 0.6
    broke_high and sufficient_break and rejection and not_continuation

isValidMomentumCandle(isBuy) =>
    body = math.abs(close - open)
    candle_range = high - low
    valid_body_ratio = candle_range > 0 ? (body / candle_range) >= min_body_ratio : false
    avg_range = ta.sma(candle_range, 5)
    valid_size = candle_range >= avg_range * min_size_multiplier
    close_from_bottom = (close - low) / candle_range
    close_from_top = (high - close) / candle_range
    valid_close_pos = isBuy ? close_from_bottom >= (1 - 0.25) : close_from_top >= (1 - 0.25)
    valid_vol = volume >= avg_vol * volume_multiplier
    upper_wick = high - math.max(open, close)
    lower_wick = math.min(open, close) - low
    max_wick = math.max(upper_wick, lower_wick)
    valid_wicks = candle_range > 0 ? (max_wick / candle_range) < max_wick_ratio : false
    not_doji = body > candle_range * 0.1
    not_inside_bar = high > high[1] or low < low[1]
    valid_body_ratio and valid_size and valid_close_pos and valid_vol and valid_wicks and not_doji and not_inside_bar

inPullbackZone(isBuy) =>
    if isBuy
        close <= ema20 and close >= ema50 * 0.9998
    else
        close >= ema20 and close <= ema50 * 1.0002

m5BullishTrend() =>
    m5_close > m5_ema200 and m5_ema20 > m5_ema50 and m5_ema50 > m5_ema200

m5BearishTrend() =>
    m5_close < m5_ema200 and m5_ema20 < m5_ema50 and m5_ema50 < m5_ema200

                count += 1
            if high[i] > ta.highest(high[i+1], swing_lookback)
                count += 1
    count

    risk = math.abs(entry - sl)
    tp1 = isBuy ? entry + risk * tp1_rr : entry - risk * tp1_rr
    tp2 = isBuy ? entry + risk * tp2_rr : entry - risk * tp2_rr
    tp3 = isBuy ? entry + risk * tp3_rr : entry - risk * tp3_rr
    [sl, tp1, tp2, tp3]

if buyCondition and strategy.position_size == 0

if sellCondition and strategy.position_size == 0

            running_equity += pnl * math.abs(strategy.position_size)
            strategy.close("Short", comment="TIME")

        if not na(sl_price_var) and trail_sl > sl_price_var
            sl_price_var := trail_sl
        if not na(sl_price_var) and low <= sl_price_var
            pnl            := sl_price_var - entry_price_var
            total_pnl      += pnl
            running_equity += pnl * strategy.position_size
            strategy.close("Long", comment="TRAIL")
            entry_price_var := na
    if strategy.position_size < 0
        trail_sl := ta.lowest(low, 3) + atr_value * trail_atr_mult
        if na(sl_price_var) or trail_sl < sl_price_var
            sl_price_var := trail_sl
        if not na(sl_price_var) and high >= sl_price_var
            pnl            := entry_price_var - sl_price_var

| **Volume Mult ≥** | `1.5` (1.5x) | Volume avg-এর কত গুণ হবে |
| **Max Wick <** | `0.40` (40%) | Upper/lower wick maximum size |

**📐 Visual Explanation:**
```
        High ────────────┐
                          │ ← Upper Wick (<40%)
        Open/Close ──────┤
                          │ ← BODY (≥60% of total range)
        Close/Open ──────┤
                          │
        Low ─────────────┘ ← Lower Wick (<40%)
        
Total Range = High - Low
Valid if: Body / Total Range ≥ 0.60
```

## 5️⃣ **সিগন্যাল কিভাবে কাজ করে (How Signals Work)**

### **🟢 BUY SIGNAL - কখন আসে?**

**সব conditions মিলতে হবে (AND logic):**

```
✅ Condition 1: Session Active
   └─► Current time within London/NY window
   
✅ Condition 2: M5 Bullish Trend (if enabled)
   └─► Price > M5 EMA200
   └─► M5 EMA20 > EMA50 > EMA200
   
✅ Condition 3: Pullback to Zone
   └─► Price between M1 EMA20 and EMA50
   
✅ Condition 4: Liquidity Sweep Detected
   └─► Broke below recent swing low
   └─► By minimum required points
   └─► Shows rejection (not continuation)
   
✅ Condition 5: Valid Momentum Candle
   └─► Body/range ≥ 60%
   └─► Size ≥ 2x average
   └─► Volume ≥ 1.5x average
   └─► Wicks < 40%
   
✅ Condition 6: Volume Spike Confirmed
   └─► Current vol ≥ 2x avg
   └─► Rising for 2 consecutive candles
   
✅ Condition 7: Filters Passed
   └─► RSI < 75 (not overbought)
   └─► < 2 sweeps in last 10 min
   └─► Spread acceptable

🎯 RESULT: GREEN BUY SIGNAL APPEARS!
```

### **🔴 SELL SIGNAL - কখন আসে?**

Same logic but **REVERSED**:

```
✅ M5 Bearish Trend (Price < EMA200, EMAs aligned down)
✅ Price pulled back UP to EMA20-50 zone
✅ Liquidity sweep ABOVE recent swing high
✅ Bearish momentum candle (close in lower 25%)
✅ Volume spike confirmation
✅ RSI > 25 (not oversold)

🎯 RESULT: RED SELL SIGNAL APPEARS!
```

## 7️⃣ **রিস্ক ম্যানেজমেন্ট (Risk Management)**

### **💰 Position Sizing Rules:**

#### **Rule 1: Never Risk More Than 1-2% Per Trade**
```
Account Balance: $1000
Max Risk/Trade:  $1000 × 1% = $10
Max Risk/Trade:  $1000 × 2% = $20
```

#### **Rule 2: Calculate Position Size Correctly**
```
Formula:
Position Size = (Account × Risk%) / (Entry - SL)

Example:
Account: $1000
Risk: 1% ($10)
Entry: 2650.00
SL: 2612.50 (37.5 points risk)

Position Size = $10 / 37.5 = 0.266 units
≈ 0.27 units (or ~27 cents per point)
```

#### **Rule 3: Daily Loss Limit**
```
MAX Daily Loss: 3% of account
$1000 account → Max loss $30/day

After hitting daily limit:
→ STOP TRADING for the day!
→ No exceptions!
```
