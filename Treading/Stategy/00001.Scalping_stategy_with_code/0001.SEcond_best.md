

```python 
//@version=5
// =============================================================================
//  Advanced Trading Suite v2.1 — ICT + SMC + Engulfing Setup
//  Concepts : BOS, CHoCH, Order Block, FVG, Liquidity (BSL/SSL),
//             Kill Zones, Engulfing Pattern, Long/Short Setup with TP/SL
//  Author   : Mostofa
//  Version  : 2.1  (Syntax-fixed: no += operator, no inline commas)
// =============================================================================
indicator("Advanced ICT + SMC + Engulfing Setup [Mostofa v2.1]",
     overlay              = true,
     max_lines_count      = 500,
     max_boxes_count      = 500,
     max_labels_count     = 500)

// ──────────────────────────────────────────────────────────────────────────────
//  INPUT GROUPS
// ──────────────────────────────────────────────────────────────────────────────
g_setup  = "🎯 LONG / SHORT Setup"
g_engulf = "🕯 Engulfing Pattern"
g_rr     = "📐 Risk / Reward"
g_struct = "🔷 Market Structure"
g_ob     = "🟧 Order Blocks"
g_fvg    = "🟦 Fair Value Gap (FVG)"
g_liq    = "💧 Liquidity"
g_kz     = "⏰ Kill Zones"
g_ui     = "🎨 UI / Dashboard"
g_alert  = "🔔 Alerts"

// ── Setup Logic ──────────────────────────────────────────────────────────────
enableSetup    = input.bool(true,  "Enable LONG / SHORT Signals",     group = g_setup)
minConfluence  = input.int(2,      "Minimum Confluence Score (1-5)",  minval = 1, maxval = 5, group = g_setup)
showSetupBox   = input.bool(true,  "Draw Entry / SL / TP",            group = g_setup)

// ── Engulfing ────────────────────────────────────────────────────────────────
enableEngulf   = input.bool(true,  "Detect Engulfing Pattern",        group = g_engulf)
strongOnly     = input.bool(false, "Strong Engulfing Only (1.5x body)", group = g_engulf)
markEngulfing  = input.bool(true,  "Mark Engulfing on Chart",         group = g_engulf)

// ── Risk / Reward ────────────────────────────────────────────────────────────
slBuffer       = input.float(0.0,  "SL Buffer (price)", step = 0.1, group = g_rr)
tp1Ratio       = input.float(1.5,  "TP1 Risk-Reward",   step = 0.1, group = g_rr)
tp2Ratio       = input.float(2.5,  "TP2 Risk-Reward",   step = 0.1, group = g_rr)
tp3Ratio       = input.float(4.0,  "TP3 Risk-Reward",   step = 0.1, group = g_rr)

// ── Market Structure ─────────────────────────────────────────────────────────
showMS         = input.bool(true,  "Show BOS / CHoCH",          group = g_struct)
swingLen       = input.int(10,     "Swing Pivot Length", minval = 2, group = g_struct)

// ── Order Blocks ─────────────────────────────────────────────────────────────
showOB         = input.bool(true,  "Show Order Blocks",         group = g_ob)
obShowLast     = input.int(3,      "Show Last N Order Blocks",  minval = 1, maxval = 10, group = g_ob)

// ── FVG ──────────────────────────────────────────────────────────────────────
showFVG        = input.bool(true,  "Show Fair Value Gaps",      group = g_fvg)
fvgMitigated   = input.bool(true,  "Hide Mitigated FVGs",       group = g_fvg)

// ── Liquidity ────────────────────────────────────────────────────────────────
showLiq        = input.bool(true,  "Show Liquidity Levels",     group = g_liq)
liqLen         = input.int(8,      "Liquidity Pivot Length",    minval = 2, group = g_liq)

// ── Kill Zones (UTC) ─────────────────────────────────────────────────────────
showKZ         = input.bool(true,  "Show Kill Zones",           group = g_kz)
londonKZ       = input.session("0700-1000", "London KZ (UTC)",  group = g_kz)
nyKZ           = input.session("1200-1500", "New York KZ (UTC)", group = g_kz)
asiaKZ         = input.session("0000-0300", "Asia KZ (UTC)",    group = g_kz)

// ── UI / Theme ───────────────────────────────────────────────────────────────
themeMode      = input.string("Dark", "Theme", options = ["Dark", "Light"], group = g_ui)
showPanel      = input.bool(true,  "Show Info Dashboard",       group = g_ui)
panelPos       = input.string("top_right", "Dashboard Position",
     options = ["top_right","top_left","bottom_right","bottom_left","middle_right"], group = g_ui)

// ── Alerts ───────────────────────────────────────────────────────────────────
alertLong      = input.bool(true,  "Alert on LONG Setup",       group = g_alert)
alertShort     = input.bool(true,  "Alert on SHORT Setup",      group = g_alert)
alertBOS       = input.bool(false, "Alert on BOS / CHoCH",      group = g_alert)


// ──────────────────────────────────────────────────────────────────────────────
//  THEME COLORS
// ──────────────────────────────────────────────────────────────────────────────
isDark      = themeMode == "Dark"
bullCol     = color.new(#00c853, 0)
bearCol     = color.new(#ff1744, 0)
neutralCol  = color.new(#9e9e9e, 0)
panelBg     = isDark ? color.new(#0d1117, 10) : color.new(#ffffff, 10)
panelText   = isDark ? color.white : color.black
gridCol     = isDark ? color.new(#1e2530, 0) : color.new(#e0e0e0, 0)

bullOBCol   = color.new(#00c853, 82)
bearOBCol   = color.new(#ff1744, 82)
bullFVGCol  = color.new(#2962ff, 82)
bearFVGCol  = color.new(#ff9100, 82)
bslCol      = color.new(#00bcd4, 0)
sslCol      = color.new(#e91e63, 0)
londonCol   = color.new(#2962ff, 90)
nyCol       = color.new(#ff9100, 90)
asiaCol     = color.new(#9c27b0, 90)


// =============================================================================
//  1. SWING DETECTION
// =============================================================================
ph = ta.pivothigh(swingLen, swingLen)
pl = ta.pivotlow(swingLen,  swingLen)

var float lastPH    = na
var float lastPL    = na
var int   lastPHIdx = na
var int   lastPLIdx = na
var float prevPH    = na
var float prevPL    = na

if not na(ph)
    prevPH    := lastPH
    lastPH    := ph
    lastPHIdx := bar_index - swingLen
if not na(pl)
    prevPL    := lastPL
    lastPL    := pl
    lastPLIdx := bar_index - swingLen


// =============================================================================
//  2. MARKET STRUCTURE — BOS / CHoCH
// =============================================================================
var int trend = 0

bosBull    = false
bosBear    = false
chochBull  = false
chochBear  = false

if showMS
    if not na(lastPH) and close > lastPH and close[1] <= lastPH
        chochBull := trend == -1
        bosBull   := trend != -1
        line.new(lastPHIdx, lastPH, bar_index, lastPH,
             color = bullCol, style = line.style_dashed, width = 1)
        label.new(bar_index, lastPH, chochBull ? "CHoCH ▲" : "BOS ▲",
             color = bullCol, textcolor = color.white,
             size  = size.small, style = label.style_label_down)
        trend := 1
        if alertBOS
            alert((chochBull ? "Bullish CHoCH" : "Bullish BOS") + " on " + syminfo.ticker, alert.freq_once_per_bar)

    if not na(lastPL) and close < lastPL and close[1] >= lastPL
        chochBear := trend == 1
        bosBear   := trend != 1
        line.new(lastPLIdx, lastPL, bar_index, lastPL,
             color = bearCol, style = line.style_dashed, width = 1)
        label.new(bar_index, lastPL, chochBear ? "CHoCH ▼" : "BOS ▼",
             color = bearCol, textcolor = color.white,
             size  = size.small, style = label.style_label_up)
        trend := -1
        if alertBOS
            alert((chochBear ? "Bearish CHoCH" : "Bearish BOS") + " on " + syminfo.ticker, alert.freq_once_per_bar)


// =============================================================================
//  3. ORDER BLOCKS
// =============================================================================
var box[]   bullOBs    = array.new<box>()
var box[]   bearOBs    = array.new<box>()
var float[] bullOBTops = array.new<float>()
var float[] bullOBBots = array.new<float>()
var float[] bearOBTops = array.new<float>()
var float[] bearOBBots = array.new<float>()

bullImpulse = close > high[1] and close[1] < open[1]
bearImpulse = close < low[1]  and close[1] > open[1]

if showOB and bullImpulse
    obTop = math.max(open[1], close[1])
    obBot = math.min(open[1], close[1])
    b = box.new(bar_index - 1, obTop, bar_index + 15, obBot,
         bgcolor      = bullOBCol,
         border_color = color.new(bullCol, 40),
         text         = "Bull OB",
         text_color   = bullCol,
         text_size    = size.tiny,
         text_halign  = text.align_left)
    array.push(bullOBs, b)
    array.push(bullOBTops, obTop)
    array.push(bullOBBots, obBot)
    if array.size(bullOBs) > obShowLast
        box.delete(array.shift(bullOBs))
        array.shift(bullOBTops)
        array.shift(bullOBBots)

if showOB and bearImpulse
    obTop = math.max(open[1], close[1])
    obBot = math.min(open[1], close[1])
    b = box.new(bar_index - 1, obTop, bar_index + 15, obBot,
         bgcolor      = bearOBCol,
         border_color = color.new(bearCol, 40),
         text         = "Bear OB",
         text_color   = bearCol,
         text_size    = size.tiny,
         text_halign  = text.align_left)
    array.push(bearOBs, b)
    array.push(bearOBTops, obTop)
    array.push(bearOBBots, obBot)
    if array.size(bearOBs) > obShowLast
        box.delete(array.shift(bearOBs))
        array.shift(bearOBTops)
        array.shift(bearOBBots)

nearBullOB = false
if array.size(bullOBTops) > 0
    for i = 0 to array.size(bullOBTops) - 1
        if low <= array.get(bullOBTops, i) and low >= array.get(bullOBBots, i) * 0.998
            nearBullOB := true
            break

nearBearOB = false
if array.size(bearOBTops) > 0
    for i = 0 to array.size(bearOBTops) - 1
        if high >= array.get(bearOBBots, i) and high <= array.get(bearOBTops, i) * 1.002
            nearBearOB := true
            break


// =============================================================================
//  4. FAIR VALUE GAP
// =============================================================================
var box[]   bullFVGs    = array.new<box>()
var box[]   bearFVGs    = array.new<box>()
var float[] bullFVGTops = array.new<float>()
var float[] bullFVGBots = array.new<float>()
var float[] bearFVGTops = array.new<float>()
var float[] bearFVGBots = array.new<float>()

bullFVG = low > high[2]
bearFVG = high < low[2]

if showFVG and bullFVG
    b = box.new(bar_index - 2, low, bar_index + 20, high[2],
         bgcolor      = bullFVGCol,
         border_color = color.new(#2962ff, 50),
         border_style = line.style_dotted)
    array.push(bullFVGs, b)
    array.push(bullFVGTops, low)
    array.push(bullFVGBots, high[2])

if showFVG and bearFVG
    b = box.new(bar_index - 2, low[2], bar_index + 20, high,
         bgcolor      = bearFVGCol,
         border_color = color.new(#ff9100, 50),
         border_style = line.style_dotted)
    array.push(bearFVGs, b)
    array.push(bearFVGTops, low[2])
    array.push(bearFVGBots, high)

if fvgMitigated and array.size(bullFVGs) > 0
    for i = array.size(bullFVGs) - 1 to 0
        if low <= array.get(bullFVGBots, i)
            box.delete(array.get(bullFVGs, i))
            array.remove(bullFVGs, i)
            array.remove(bullFVGTops, i)
            array.remove(bullFVGBots, i)

if fvgMitigated and array.size(bearFVGs) > 0
    for i = array.size(bearFVGs) - 1 to 0
        if high >= array.get(bearFVGTops, i)
            box.delete(array.get(bearFVGs, i))
            array.remove(bearFVGs, i)
            array.remove(bearFVGTops, i)
            array.remove(bearFVGBots, i)

nearBullFVG = false
if array.size(bullFVGs) > 0
    for i = 0 to array.size(bullFVGs) - 1
        if low <= array.get(bullFVGTops, i) and low >= array.get(bullFVGBots, i)
            nearBullFVG := true
            break

nearBearFVG = false
if array.size(bearFVGs) > 0
    for i = 0 to array.size(bearFVGs) - 1
        if high >= array.get(bearFVGBots, i) and high <= array.get(bearFVGTops, i)
            nearBearFVG := true
            break


// =============================================================================
//  5. LIQUIDITY  (BSL / SSL) + Sweep Detection
// =============================================================================
liqPH = ta.pivothigh(liqLen, liqLen)
liqPL = ta.pivotlow(liqLen,  liqLen)

var line[]  bslLines  = array.new<line>()
var line[]  sslLines  = array.new<line>()
var float[] bslLevels = array.new<float>()
var float[] sslLevels = array.new<float>()

if showLiq and not na(liqPH)
    l = line.new(bar_index - liqLen, liqPH, bar_index + 5, liqPH,
         color = bslCol, style = line.style_dotted, width = 1, extend = extend.right)
    array.push(bslLines, l)
    array.push(bslLevels, liqPH)

if showLiq and not na(liqPL)
    l = line.new(bar_index - liqLen, liqPL, bar_index + 5, liqPL,
         color = sslCol, style = line.style_dotted, width = 1, extend = extend.right)
    array.push(sslLines, l)
    array.push(sslLevels, liqPL)

bullSweep = false
bearSweep = false

if array.size(sslLevels) > 0
    for i = array.size(sslLevels) - 1 to 0
        lvl = array.get(sslLevels, i)
        if low < lvl and close > lvl
            bullSweep := true
            line.set_color(array.get(sslLines, i), color.new(sslCol, 60))
            line.set_extend(array.get(sslLines, i), extend.none)
            line.set_x2(array.get(sslLines, i), bar_index)
            array.remove(sslLines, i)
            array.remove(sslLevels, i)

if array.size(bslLevels) > 0
    for i = array.size(bslLevels) - 1 to 0
        lvl = array.get(bslLevels, i)
        if high > lvl and close < lvl
            bearSweep := true
            line.set_color(array.get(bslLines, i), color.new(bslCol, 60))
            line.set_extend(array.get(bslLines, i), extend.none)
            line.set_x2(array.get(bslLines, i), bar_index)
            array.remove(bslLines, i)
            array.remove(bslLevels, i)


// =============================================================================
//  6. KILL ZONES
// =============================================================================
inLondon = not na(time(timeframe.period, londonKZ))
inNY     = not na(time(timeframe.period, nyKZ))
inAsia   = not na(time(timeframe.period, asiaKZ))
inKZ     = inLondon or inNY or inAsia

bgcolor(showKZ and inLondon ? londonCol : na, title = "London KZ")
bgcolor(showKZ and inNY     ? nyCol     : na, title = "New York KZ")
bgcolor(showKZ and inAsia   ? asiaCol   : na, title = "Asia KZ")


// =============================================================================
//  7. ENGULFING PATTERN
// =============================================================================
prevBody  = math.abs(close[1] - open[1])
currBody  = math.abs(close - open)
bodyRatio = currBody / math.max(prevBody, syminfo.mintick)

bullishEngulf = enableEngulf and close[1] < open[1] and close > open and open <= close[1] and close >= open[1] and (not strongOnly or bodyRatio >= 1.5)
bearishEngulf = enableEngulf and close[1] > open[1] and close < open and open >= close[1] and close <= open[1] and (not strongOnly or bodyRatio >= 1.5)

if markEngulfing and bullishEngulf
    label.new(bar_index, low, "🟢",
         yloc      = yloc.belowbar,
         color     = color.new(bullCol, 80),
         textcolor = color.white,
         size      = size.tiny,
         style     = label.style_label_up)

if markEngulfing and bearishEngulf
    label.new(bar_index, high, "🔴",
         yloc      = yloc.abovebar,
         color     = color.new(bearCol, 80),
         textcolor = color.white,
         size      = size.tiny,
         style     = label.style_label_down)


// =============================================================================
//  8. LONG / SHORT SETUP — Confluence Scoring
// =============================================================================
longScore  = 0
shortScore = 0

if bullishEngulf
    longScore := longScore + 1
if bearishEngulf
    shortScore := shortScore + 1

if trend == 1 or bosBull or chochBull
    longScore := longScore + 1
if trend == -1 or bosBear or chochBear
    shortScore := shortScore + 1

if inKZ and bullishEngulf
    longScore := longScore + 1
if inKZ and bearishEngulf
    shortScore := shortScore + 1

if (nearBullOB or nearBullFVG) and bullishEngulf
    longScore := longScore + 1
if (nearBearOB or nearBearFVG) and bearishEngulf
    shortScore := shortScore + 1

if bullSweep and bullishEngulf
    longScore := longScore + 1
if bearSweep and bearishEngulf
    shortScore := shortScore + 1

longSignal  = enableSetup and bullishEngulf and longScore  >= minConfluence
shortSignal = enableSetup and bearishEngulf and shortScore >= minConfluence


// =============================================================================
//  9. ENTRY / SL / TP CALCULATION  &  DRAWING
// =============================================================================
var line  entryLine = na
var line  slLine    = na
var line  tp1Line   = na
var line  tp2Line   = na
var line  tp3Line   = na
var label entryLbl  = na
var label slLbl     = na
var label tp1Lbl    = na
var label tp2Lbl    = na
var label tp3Lbl    = na

var float  lastEntry    = na
var float  lastSL       = na
var float  lastTP1      = na
var float  lastTP2      = na
var float  lastTP3      = na
var string lastSide     = "—"
var int    lastScore    = 0
var int    lastSetupBar = na

if longSignal
    entry = close
    sl    = math.min(low, low[1]) - slBuffer
    risk  = entry - sl
    tp1   = entry + risk * tp1Ratio
    tp2   = entry + risk * tp2Ratio
    tp3   = entry + risk * tp3Ratio
    lastEntry    := entry
    lastSL       := sl
    lastTP1      := tp1
    lastTP2      := tp2
    lastTP3      := tp3
    lastSide     := "LONG"
    lastScore    := longScore
    lastSetupBar := bar_index

    if showSetupBox
        line.delete(entryLine)
        line.delete(slLine)
        line.delete(tp1Line)
        line.delete(tp2Line)
        line.delete(tp3Line)
        label.delete(entryLbl)
        label.delete(slLbl)
        label.delete(tp1Lbl)
        label.delete(tp2Lbl)
        label.delete(tp3Lbl)
        xe = bar_index + 25
        entryLine := line.new(bar_index, entry, xe, entry, color = bullCol, width = 2)
        slLine    := line.new(bar_index, sl,    xe, sl,    color = color.new(bearCol, 20), width = 2, style = line.style_dashed)
        tp1Line   := line.new(bar_index, tp1,   xe, tp1,   color = color.new(bullCol, 40), width = 1, style = line.style_dashed)
        tp2Line   := line.new(bar_index, tp2,   xe, tp2,   color = color.new(bullCol, 20), width = 1, style = line.style_dashed)
        tp3Line   := line.new(bar_index, tp3,   xe, tp3,   color = bullCol, width = 1, style = line.style_dashed)
        entryLbl  := label.new(xe, entry, "LONG @ " + str.tostring(entry, format.mintick),
             color = bullCol, textcolor = color.white, size = size.small, style = label.style_label_left)
        slLbl     := label.new(xe, sl,  "SL "  + str.tostring(sl,  format.mintick),
             color = color.new(bearCol, 30), textcolor = color.white, size = size.small, style = label.style_label_left)
        tp1Lbl    := label.new(xe, tp1, "TP1 " + str.tostring(tp1, format.mintick) + "  (" + str.tostring(tp1Ratio, "#.#") + "R)",
             color = color.new(bullCol, 60), textcolor = color.white, size = size.small, style = label.style_label_left)
        tp2Lbl    := label.new(xe, tp2, "TP2 " + str.tostring(tp2, format.mintick) + "  (" + str.tostring(tp2Ratio, "#.#") + "R)",
             color = color.new(bullCol, 40), textcolor = color.white, size = size.small, style = label.style_label_left)
        tp3Lbl    := label.new(xe, tp3, "TP3 " + str.tostring(tp3, format.mintick) + "  (" + str.tostring(tp3Ratio, "#.#") + "R)",
             color = color.new(bullCol, 20), textcolor = color.white, size = size.small, style = label.style_label_left)

    label.new(bar_index, low, "▲ LONG " + str.tostring(longScore) + "/5",
         yloc      = yloc.belowbar,
         color     = bullCol,
         textcolor = color.white,
         size      = size.normal,
         style     = label.style_label_up)

    if alertLong
        alert("🟢 LONG Setup | Score: " + str.tostring(longScore) + "/5 | Entry: " + str.tostring(entry, format.mintick) + " | SL: " + str.tostring(sl, format.mintick) + " | TP1: " + str.tostring(tp1, format.mintick) + " on " + syminfo.ticker, alert.freq_once_per_bar)

if shortSignal
    entry = close
    sl    = math.max(high, high[1]) + slBuffer
    risk  = sl - entry
    tp1   = entry - risk * tp1Ratio
    tp2   = entry - risk * tp2Ratio
    tp3   = entry - risk * tp3Ratio
    lastEntry    := entry
    lastSL       := sl
    lastTP1      := tp1
    lastTP2      := tp2
    lastTP3      := tp3
    lastSide     := "SHORT"
    lastScore    := shortScore
    lastSetupBar := bar_index

    if showSetupBox
        line.delete(entryLine)
        line.delete(slLine)
        line.delete(tp1Line)
        line.delete(tp2Line)
        line.delete(tp3Line)
        label.delete(entryLbl)
        label.delete(slLbl)
        label.delete(tp1Lbl)
        label.delete(tp2Lbl)
        label.delete(tp3Lbl)
        xe = bar_index + 25
        entryLine := line.new(bar_index, entry, xe, entry, color = bearCol, width = 2)
        slLine    := line.new(bar_index, sl,    xe, sl,    color = color.new(bearCol, 20), width = 2, style = line.style_dashed)
        tp1Line   := line.new(bar_index, tp1,   xe, tp1,   color = color.new(bullCol, 40), width = 1, style = line.style_dashed)
        tp2Line   := line.new(bar_index, tp2,   xe, tp2,   color = color.new(bullCol, 20), width = 1, style = line.style_dashed)
        tp3Line   := line.new(bar_index, tp3,   xe, tp3,   color = bullCol, width = 1, style = line.style_dashed)
        entryLbl  := label.new(xe, entry, "SHORT @ " + str.tostring(entry, format.mintick),
             color = bearCol, textcolor = color.white, size = size.small, style = label.style_label_left)
        slLbl     := label.new(xe, sl,  "SL "  + str.tostring(sl,  format.mintick),
             color = color.new(bearCol, 30), textcolor = color.white, size = size.small, style = label.style_label_left)
        tp1Lbl    := label.new(xe, tp1, "TP1 " + str.tostring(tp1, format.mintick) + "  (" + str.tostring(tp1Ratio, "#.#") + "R)",
             color = color.new(bullCol, 60), textcolor = color.white, size = size.small, style = label.style_label_left)
        tp2Lbl    := label.new(xe, tp2, "TP2 " + str.tostring(tp2, format.mintick) + "  (" + str.tostring(tp2Ratio, "#.#") + "R)",
             color = color.new(bullCol, 40), textcolor = color.white, size = size.small, style = label.style_label_left)
        tp3Lbl    := label.new(xe, tp3, "TP3 " + str.tostring(tp3, format.mintick) + "  (" + str.tostring(tp3Ratio, "#.#") + "R)",
             color = color.new(bullCol, 20), textcolor = color.white, size = size.small, style = label.style_label_left)

    label.new(bar_index, high, "▼ SHORT " + str.tostring(shortScore) + "/5",
         yloc      = yloc.abovebar,
         color     = bearCol,
         textcolor = color.white,
         size      = size.normal,
```

## 5. Module Deep-Dive

  

### 5.1 Swing Detection

  

**Function used:** `ta.pivothigh(len, len)` and `ta.pivotlow(len, len)`

  

A pivot high is confirmed when a bar has `len` lower highs on **both sides**. Example: with `swingLen = 10`, the pivot is confirmed **10 bars after it formed**.

  

```

              ← pivot high

             ╱╲

        ────╱  ╲────

        10 bars 10 bars

```

  

**Trade-off:**

- Smaller `swingLen` (5-7) → faster confirmation, more pivots, more noise

- Larger `swingLen` (15-20) → slower confirmation, fewer but stronger pivots

  

The script tracks **last 2 pivots** (current + previous) for both highs and lows, enabling structure analysis.

### 5.2 Market Structure (BOS / CHoCH)

  

**Logic:**

  

| Event | Condition | Meaning |

|---|---|---|

| **BOS ▲** | Price closes above previous swing high while already in uptrend | Bullish continuation |

| **BOS ▼** | Price closes below previous swing low while already in downtrend | Bearish continuation |

| **CHoCH ▲** | Price closes above previous swing high while trend was bearish | Trend reversal up |

| **CHoCH ▼** | Price closes below previous swing low while trend was bullish | Trend reversal down |

  

**Implementation:**

```pine

if not na(lastPH) and close > lastPH and close[1] <= lastPH

    chochBull := trend == -1    // Was bearish? Then this is CHoCH

    bosBull   := trend != -1    // Already bullish? Then it's BOS

    trend := 1

```

  

**Visual:** Dashed horizontal line from the broken pivot to current bar + colored label (▲/▼).

### 5.3 Order Blocks

  

**Definition (simplified):** The last opposing candle before a strong impulse that breaks structure.

  

**Bullish OB detection:**

```pine

bullImpulse = close > high[1] and close[1] < open[1]

//             ^current breaks      ^prev was red

//              prev high

```

  

When detected, the script draws a box covering the **prev red candle's body** (open to close), extending 15 bars forward.

  

**Why this matters:**

- Institutional orders ekhane execute hoyeche

- Price often returns to OB for **re-test** before continuing the move

- High-probability re-entry zone

  

**Limitation:** Eta simplified OB logic. Production-grade ICT OBs additionally require:

- The impulse must break liquidity / structure

- Volume confirmation

- Mitigation tracking

  

Eta upgrade chaile bolun.

### 5.4 Fair Value Gap (FVG)

  

**Definition:** A 3-candle pattern where price moves so fast that an "imbalance" or gap exists between candle 1's high and candle 3's low (or vice versa).

  

**Bullish FVG:**

```

Candle 1: high = 1.0850

Candle 2: strong bullish (low > 1.0850, high = 1.0920)

Candle 3: low = 1.0880 (gap between 1.0850 and 1.0880)

```

  

**Pine condition:**

```pine

bullFVG = low > high[2]    // current low > 2-bar-ago high

bearFVG = high < low[2]    // current high < 2-bar-ago low

```

  

**Mitigation:** When `Hide Mitigated FVGs = ON`, the box auto-deletes once price returns into the gap. This keeps the chart clean.

  

**Trade idea:** Price often returns to fill FVGs before continuing. Entry near FVG = high-probability pullback entry.

### 5.5 Liquidity (BSL / SSL)

  

| Type | Full Name | Where | What |

|---|---|---|---|

| **BSL** | Buy-Side Liquidity | Above swing highs | Stop losses of short sellers |

| **SSL** | Sell-Side Liquidity | Below swing lows | Stop losses of long buyers |

  

**Sweep Detection:**

```pine

// Bullish sweep (sell-side stops taken, price reversed up)

if low < ssl_level and close > ssl_level

    bullSweep := true

```

  

**Why important:**

- Smart money hunts liquidity pools before reversing

- A **sweep + reversal** is one of the highest-probability setups in ICT

- The script awards **+1 confluence** when a sweep aligns with an engulfing pattern

  

**Visual behavior:**

- Active liquidity = bright dotted line extending right

- Swept liquidity = faded line, stops extending (becomes historical)

### 5.7 Engulfing Pattern

  

**Bullish Engulfing:**

```pine

close[1] < open[1]       // Previous = red

and close > open         // Current = green

and open <= close[1]     // Current open ≤ prev close (no gap up)

and close >= open[1]     // Current close ≥ prev open (full body engulf)

```

  

**Bearish Engulfing:** Mirror of above.

  

**Strong Engulfing Filter:** When ON, additionally requires:

```pine

currBody / prevBody >= 1.5

```

  

This filters out marginal engulfing patterns that often fail. **Recommendation:** ON for scalping (1M-15M), OFF for swing (4H-D).

### 5.8 Confluence Scoring

  

**Scoring Table:**

  

| # | Confluence | Points | When Awarded |

|---|---|---|---|

| 1 | Engulfing pattern | +1 | Required (mandatory trigger) |

| 2 | Trend alignment | +1 | `trend == 1` OR `BOS` OR `CHoCH` matches direction |

| 3 | Kill Zone active | +1 | Price in any of 3 sessions |

| 4 | Near OB or FVG | +1 | Current candle touches an unmitigated zone |

| 5 | Liquidity Sweep | +1 | BSL/SSL swept on same bar |

  

**Max Score = 5/5** (A+ setup)

  

**Recommended thresholds:**

  

| Min Score | Frequency | Quality |

|---|---|---|

| 1 | Very high | Low (engulfing only) |

| 2 | High | Decent (default) |

| 3 | Medium | Good |

| 4 | Low | Excellent |

| 5 | Rare | Perfect |

### 5.9 Entry / SL / TP Calculation

  

**LONG Setup:**

```

Entry = close of engulfing candle

SL    = min(current low, previous low) - SL Buffer

Risk  = Entry - SL

  

TP1 = Entry + (Risk × 1.5)

TP2 = Entry + (Risk × 2.5)

TP3 = Entry + (Risk × 4.0)

```

  

**SHORT Setup:** Mirror, swap min/max.

  

**Visual on chart:**

  

```

─── TP3 ────────────────  (1:4 RR)

─── TP2 ────────────────  (1:2.5 RR)

─── TP1 ────────────────  (1:1.5 RR)

═══ ENTRY ══════════════  (solid line)

- - SL - - - - - - - - -  (dashed)

```

  

**Behavior:**

- Lines persist on chart until next signal fires

- Drawn 25 bars forward from signal bar

- Labels show exact price + RR multiplier

### Position Management

  

| Action | When |

|---|---|

| Take partial profit (30-50%) | At TP1 |

| Move SL to breakeven | After TP1 hit |

| Take more profit (30%) | At TP2 |

| Trail SL below recent swing | After TP2 hit |

| Final close | At TP3 or trailing SL hit |

  

### Risk Sizing

  

- **Per trade risk:** 0.5% to 2% of account

- **Daily risk limit:** Max 3 consecutive losses

- **Position size formula:** `Position Size = (Account × Risk%) / (Entry - SL)`

## 9. Performance & Limitations

  

### Computational Costs

  

| Resource | Limit | Current Usage |

|---|---|---|

| `max_lines_count` | 500 | Used by liquidity, structure, TP/SL |

| `max_boxes_count` | 500 | Used by OBs, FVGs |

| `max_labels_count` | 500 | Used by all annotations |

  

**Recommendation:** Apply on charts with ≤ 5000 bars for smooth performance.

  

### Known Limitations

  

1. **Pivot lag:** Swings confirm `swingLen` bars after they form. Live entry must wait for confirmation.

2. **Repaint risk:** The pivot-based BOS/CHoCH does not repaint, but historical signals **may look better in hindsight** than they were live.

3. **Order Block simplification:** Not full ICT-standard OB. Lacks volume confirmation and mitigation logic.

4. **No backtesting:** This is an `indicator()`, not a `strategy()`. For P&L stats, convert to strategy.

5. **Engulfing patterns confirm on candle close:** Cannot fire intra-bar.

  

### What This Indicator Does NOT Do

  

- ❌ Predict the future

- ❌ Guarantee profitable trades

- ❌ Replace risk management

- ❌ Work on every market/timeframe equally

- ❌ Execute trades automatically (needs webhook + bot)

## 10. Troubleshooting

  

### "Indicator not displaying anything"

  

**Check:**

1. Is the script saved and added to chart?

2. Are individual feature toggles ON in settings?

3. For Kill Zones: is the current time within a session window?

4. Try lowering `Minimum Confluence Score` to 1

  

### "Too many signals / chart cluttered"

  

**Solution:**

- Increase `Minimum Confluence Score` to 4-5

- Enable `Strong Engulfing Only`

- Reduce `Show Last N Order Blocks` to 1-2

- Turn off Kill Zone shading or unused modules

  

### "Compile error appears"

  

Pine Script v5 evolves periodically. If TradingView updates break the script:

1. Note the **exact line number** in the error

2. Check that function names match v5 API (e.g., `ta.pivothigh` not `pivothigh`)

3. Verify no `+=` operators remain (Pine doesn't support them)

  

### "Kill Zone time is wrong"

  

- Verify your chart's timezone (right-click time axis → Properties)

- Recalculate session inputs for your broker's UTC offset

- Use UTC inputs always; let TradingView handle display conversion

  

### "Dashboard overlaps with chart"

  

- Change `Dashboard Position` to a different corner

- Or turn off `Show Info Dashboard`

```pine
// 1. Add input

// 3. Add to scoring

    longScore := longScore + 1

    shortScore := shortScore + 1
```

```pine

bullCol     = color.new(#00c853, 0)  // Change hex code

```

  

### Adding More Take Profit Levels

  

```pine

tp4Ratio = input.float(6.0, "TP4 RR", group = g_rr)
```

Replace the first line:

```pine

indicator("...", overlay = true, ...)
```

```pine
strategy("...", overlay = true, initial_capital = 10000,

     default_qty_type = strategy.percent_of_equity,

     default_qty_value = 1)

```

  

Then replace alert blocks with:

```pine

if longSignal

    strategy.entry("LONG", strategy.long)

    strategy.exit("Exit", "LONG", stop = sl, limit = tp2)

```

## 12. Glossary

  

| Term | Definition |

|---|---|

| **ICT** | Inner Circle Trader — Michael Huddleston's institutional trading methodology |

| **SMC** | Smart Money Concept — framework analyzing institutional order flow |

| **BOS** | Break of Structure — price breaks previous swing in trend direction (continuation) |

| **CHoCH** | Change of Character — price breaks previous swing against trend (reversal) |

| **OB** | Order Block — institutional zone of order accumulation |

| **FVG** | Fair Value Gap — price imbalance between 3 candles |

| **BSL** | Buy-Side Liquidity — stop losses above swing highs |

| **SSL** | Sell-Side Liquidity — stop losses below swing lows |

| **Liquidity Sweep** | Price pierces a level to grab stops, then reverses |

| **Kill Zone** | High-volatility session window (institutional activity) |

| **Engulfing** | Reversal candle pattern; current body fully covers previous body |

| **Confluence** | Multiple independent factors aligning on same trade idea |

| **RR** | Risk-Reward ratio (e.g., 1:2 = risk $1 to make $2) |

| **HTF** | Higher Timeframe (4H, Daily, Weekly) |

| **LTF** | Lower Timeframe (1M, 5M, 15M) |

| **Pivot** | Local extreme (high or low) confirmed by surrounding bars |

| **Mitigation** | Price returning to a zone/gap (often "filling" it) |

| **Judas Swing** | Fake move at session open that traps retail before reversal |
