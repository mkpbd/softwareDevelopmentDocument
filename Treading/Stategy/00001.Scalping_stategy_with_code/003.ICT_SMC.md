// --- Fibonacci

fvgKeepBars = input.int(50, "Keep FVG visible (bars)", minval=5, group=gFVG)

// --- Volume

volMultiplier = input.float(1.2, "Volume Spike Multiplier", step=0.1, group=gVol)

// --- Risk
atrLen      = input.int(14, "ATR Length", group=gRisk)
atrMult     = input.float(1.5, "ATR SL Multiplier", step=0.1, group=gRisk)
rrRatio     = input.float(2.0, "Risk : Reward Ratio", step=0.1, minval=1.0, group=gRisk)
riskPercent = input.float(1.0, "Risk Per Trade (%)", step=0.1, minval=0.1, group=gRisk)

// --- Filters
fUseTrend = input.bool(true, "Require Trend Alignment", group=gFilter)
fUseBOS   = input.bool(true, "Require BOS / MSS", group=gFilter)
fUseLiq   = input.bool(true, "Require Liquidity Sweep", group=gFilter)
fUseOB    = input.bool(true, "Require Order Block", group=gFilter)
fUseFib   = input.bool(true, "Require Fib Golden Zone", group=gFilter)
fUseFVG   = input.bool(true, "Require Active FVG", group=gFilter)
fUseVol   = input.bool(true, "Require High Volume", group=gFilter)
fUseEntry = input.bool(true, "Require Entry Candle", group=gFilter)
minScore  = input.int(6, "Minimum Confluence Score (out of 9)", minval=3, maxval=9, group=gFilter, tooltip="Higher = fewer but higher-quality signals. 6-7 recommended.")

showFVG   = input.bool(true, "Show FVG Boxes", group=gVisual)
showFib   = input.bool(true, "Show Fib Zone", group=gVisual)

showMSS   = input.bool(true, "Show MSS Labels", group=gVisual)
showTable = input.bool(true, "Show Info Table", group=gVisual)

// =============================================================================
emaFast = ta.ema(close, emaFastLen)
emaSlow = ta.ema(close, emaSlowLen)
atr     = ta.atr(atrLen)

highVol = volume > volMA * volMultiplier

htfBullish = useHTF ? htfFastRaw > htfSlowRaw : emaFast > emaSlow
htfBearish = useHTF ? htfFastRaw < htfSlowRaw : emaFast < emaSlow

ltfBullish = emaFast > emaSlow and close > emaFast
ltfBearish = emaFast < emaSlow and close < emaFast

// =============================================================================
//  SWING HIGH / LOW DETECTION
// =============================================================================
swingHigh = ta.pivothigh(high, swingLen, swingLen)
swingLow  = ta.pivotlow(low,  swingLen, swingLen)

var float lastSwingHigh    = na
var float lastSwingLow     = na
var int   lastSwingHighBar = na
var int   lastSwingLowBar  = na

if not na(swingHigh)
    lastSwingHigh    := swingHigh
    lastSwingHighBar := bar_index - swingLen
if not na(swingLow)
    lastSwingLow    := swingLow
    lastSwingLowBar := bar_index - swingLen

// =============================================================================
//  BOS  (Break of Structure)
// =============================================================================
var float lastBrokenHigh = na
var float lastBrokenLow  = na

bullishBOS = not na(lastSwingHigh) and close > lastSwingHigh and lastBrokenHigh != lastSwingHigh
bearishBOS = not na(lastSwingLow)  and close < lastSwingLow  and lastBrokenLow  != lastSwingLow

if bullishBOS
    lastBrokenHigh := lastSwingHigh
if bearishBOS
    lastBrokenLow := lastSwingLow

// =============================================================================
//  MSS  (Market Structure Shift)
// =============================================================================
var int trendBias = 0
bullishMSS = bullishBOS and trendBias <= 0
bearishMSS = bearishBOS and trendBias >= 0

if bullishBOS
    trendBias := 1
else if bearishBOS
    trendBias := -1

// =============================================================================
//  LIQUIDITY SWEEP
// =============================================================================
recentHigh = ta.highest(high, liqLookback)[1]
recentLow  = ta.lowest(low,  liqLookback)[1]

bullishLiqSweep = low  < recentLow  and close > recentLow  and close > open
bearishLiqSweep = high > recentHigh and close < recentHigh and close < open

// =============================================================================
//  ORDER BLOCKS
// =============================================================================
strongBullMove = close > open and (close - open) > atr * obStrength
strongBearMove = close < open and (open - close) > atr * obStrength

bullishOBFormed = close[1] < open[1] and strongBullMove
bearishOBFormed = close[1] > open[1] and strongBearMove

var float bullOBTop = na
var float bullOBBot = na
var int   bullOBBar = na

var float bearOBBot = na
var int   bearOBBar = na

    bullOBBot := low[1]
    bullOBBar := bar_index - 1
if bearishOBFormed
    bearOBTop := high[1]
    bearOBBot := low[1]
    bearOBBar := bar_index - 1

inBullOB = not na(bullOBTop) and low  <= bullOBTop and high >= bullOBBot
inBearOB = not na(bearOBTop) and high >= bearOBBot and low  <= bearOBTop

// =============================================================================
//  FAIR VALUE GAP (FVG)
// =============================================================================
bullishFVG = low > high[2]
bearishFVG = high < low[2]

var float bullFVGTop = na
var float bullFVGBot = na
var float bearFVGTop = na
var float bearFVGBot = na
var bool  bullFVGActive = false
var bool  bearFVGActive = false

if bullishFVG
    bullFVGTop := low
    bullFVGBot := high[2]
    bullFVGActive := true
if bearishFVG
    bearFVGTop := low[2]
    bearFVGBot := high
    bearFVGActive := true

if bullFVGActive and low < bullFVGBot
    bullFVGActive := false
if bearFVGActive and high > bearFVGTop
    bearFVGActive := false

// =============================================================================
//  FIBONACCI GOLDEN ZONE (0.5 - 0.618)
// =============================================================================
fibHigh  = ta.highest(high, fibLen)
fibLow   = ta.lowest(low,   fibLen)
fibRange = fibHigh - fibLow

longFibTop = fibHigh - fibRange * fibGoldenLow
longFibBot = fibHigh - fibRange * fibGoldenHigh

shortFibBot = fibLow + fibRange * fibGoldenLow
shortFibTop = fibLow + fibRange * fibGoldenHigh

inLongFib  = close <= longFibTop  and close >= longFibBot
inShortFib = close >= shortFibBot and close <= shortFibTop

// =============================================================================
//  TRENDLINE  (EMA + slope simplified validation)
// =============================================================================

downtrendOK = emaFast < emaSlow and emaFast < emaFast[5] and close < emaFast

//  ENTRY CANDLES

bearEngulf = close < open and close[1] > open[1] and close < open[1] and open >= close[1]

bullMomentum = close > open and (close - open) > (high - low) * 0.6 and close > high[1]
bearMomentum = close < open and (open - close) > (high - low) * 0.6 and close < low[1]

bullEntry = bullEngulf or bullMomentum
bearEntry = bearEngulf or bearMomentum

shortScore = (htfBearish ? 1 : 0) + (downtrendOK ? 1 : 0) + ((bearishBOS or bearishMSS) ? 1 : 0) + (bearishLiqSweep ? 1 : 0) + ((inBearOB or bearishOBFormed) ? 1 : 0) + (inShortFib ? 1 : 0) + (bearFVGActive ? 1 : 0) + (bearEntry ? 1 : 0) + (highVol ? 1 : 0)

// =============================================================================
//  HARD FILTERS
// =============================================================================
longHardOK  = (not fUseTrend or (htfBullish and uptrendOK)) and (not fUseBOS or bullishBOS or bullishMSS) and (not fUseLiq or bullishLiqSweep) and (not fUseOB or inBullOB or bullishOBFormed) and (not fUseFib or inLongFib) and (not fUseFVG or bullFVGActive) and (not fUseVol or highVol) and (not fUseEntry or bullEntry)
shortHardOK = (not fUseTrend or (htfBearish and downtrendOK)) and (not fUseBOS or bearishBOS or bearishMSS) and (not fUseLiq or bearishLiqSweep) and (not fUseOB or inBearOB or bearishOBFormed) and (not fUseFib or inShortFib) and (not fUseFVG or bearFVGActive) and (not fUseVol or highVol) and (not fUseEntry or bearEntry)

shortSignal = shortHardOK and shortScore >= minScore

//  STOP-LOSS / TAKE-PROFIT + POSITION SIZING
// =============================================================================

longTP  = close + (close - longSL) * rrRatio

shortSL = math.max(high, high[1]) + atr * atrMult
shortTP = close - (shortSL - close) * rrRatio

riskCash      = strategy.equity * (riskPercent / 100.0)
longRiskUnit  = math.max(close   - longSL, syminfo.mintick)
shortRiskUnit = math.max(shortSL - close,  syminfo.mintick)

## 1. What This Strategy Does

This script combines **eight institutional-trading concepts** into one filter pipeline and only fires entries when enough of them align:

| Concept                     | What it tells you                                                       |
| --------------------------- | ----------------------------------------------------------------------- |
| **HTF Trend (EMA 20/50)**   | The dominant direction on a higher timeframe                            |
| **Market Structure (BOS)**  | Whether structure is breaking up or down                                |
| **MSS**                     | Whether the trend has just flipped (trend-reversal signal)              |
| **Liquidity Sweep**         | Stop-hunt above a recent high / below a recent low (smart-money entry)  |
| **Order Block**             | Last opposite candle before an impulsive move (institutional zone)      |
| **Fibonacci Golden Zone**   | Retracement between 0.5 and 0.618 — the "fair value" pullback area      |
| **Fair Value Gap (FVG)**    | A 3-bar imbalance the market often returns to fill                      |
| **Entry Candle**            | Engulfing or strong momentum bar confirming the actual entry            |

Each is independently toggleable, and the strategy uses a **0–9 confluence score** so you can dial signal frequency vs. quality.

**Output:** entries with ATR-based stop loss, configurable risk:reward take profit, and **risk-based position sizing** (you risk a fixed % of equity per trade, never more).

## 3. Strategy Components — How Each Module Works

### 3.1 HTF Trend Filter

**Logic:** EMA 20 vs EMA 50 on a higher timeframe (default 1H).

```
HTF Bullish  =  HTF EMA Fast > HTF EMA Slow
HTF Bearish  =  HTF EMA Fast < HTF EMA Slow
```

The script uses `request.security()` with `lookahead_off` — **no repainting**. The HTF value you see today is the same value you would have seen in real time.

**Why it matters:** Trading WITH the higher-timeframe trend filters out 60–70% of fake setups.

---

### 3.2 Swing High / Swing Low

A swing high is a candle whose high is greater than the `swingLen` candles before AND after it (Pine's `ta.pivothigh`). Swing low is the symmetric concept.

**Trade-off:**
- Smaller `swingLen` (3) → more swings, more sensitive, more noise.
- Larger `swingLen` (8) → fewer, cleaner swings, but slower reaction.

---

### 3.3 BOS — Break of Structure

```
Bullish BOS  =  Close > Last Swing High
Bearish BOS  =  Close < Last Swing Low
```

A BOS confirms trend continuation. The script remembers the last broken level so it doesn't double-count the same break.

**Visual:** dashed green line at the broken high, dashed red at the broken low, labeled **BOS Up** / **BOS Dn**.

---

### 3.4 MSS — Market Structure Shift

A BOS is an "MSS" when it happens against the prior trend bias.

```
Bullish MSS  =  Bullish BOS  while  trend was bearish
Bearish MSS  =  Bearish BOS  while  trend was bullish
```

**Why it matters:** MSS is the earliest reliable reversal signal in SMC. The first BOS after a long bearish run is far more powerful than the 5th bullish BOS in a row.

**Visual:** blue **MSS Up** / purple **MSS Dn** labels.

---

### 3.5 Liquidity Sweep

```
Bullish Sweep  =  Low pierces previous N-bar low BUT close stays above it AND candle closes green
Bearish Sweep  =  High pierces previous N-bar high BUT close stays below it AND candle closes red
```

This is the **stop hunt** — price grabs liquidity sitting at the obvious low/high and immediately reverses. It's one of the highest-probability institutional entry signatures.

**Visual:** small × marker below/above the candle.

---

### 3.6 Order Block (OB)

```
Bullish OB  =  The last bearish candle before a strong bullish impulse
Bearish OB  =  The last bullish candle before a strong bearish impulse
```

"Strong impulse" = candle body > `obStrength × ATR` (default 1.0 × ATR).

The script stores the high/low of that opposite candle as the OB zone. When price later returns to that zone, that's a re-entry opportunity.

**Visual:** translucent green (bullish OB) or red (bearish OB) box extending forward.

---

### 3.7 Fair Value Gap (FVG)

```
Bullish FVG  =  Low of current candle > High of candle 2 bars ago
Bearish FVG  =  High of current candle < Low of candle 2 bars ago
```

Translation: a 3-candle pattern where the middle candle is so impulsive it leaves a gap between candle-1's wick and candle-3's wick. Markets often return to fill these gaps.

The script tracks **whether the FVG is still active** (unfilled). If price retraces and closes the gap, the FVG is marked inactive.

**Visual:** lime (bullish) or fuchsia (bearish) box.

---

### 3.8 Fibonacci Golden Zone

```
Range          =  Highest High (last fibLen bars) − Lowest Low (last fibLen bars)
Long zone top  =  High − Range × 0.5
Long zone bot  =  High − Range × 0.618
```

The 0.5 – 0.618 retracement is statistically where the strongest pullbacks reverse. Entries inside this zone have a **better risk:reward** because you're entering deep into the pullback rather than at the extreme.

**Visual:** semi-transparent yellow band on chart.

---

### 3.9 Trendline (EMA-based proxy)

To keep it lightweight, the script uses an EMA-slope test rather than drawing literal trendlines:

```
Uptrend OK    =  EMA Fast > EMA Slow  AND  EMA Fast > EMA Fast 5 bars ago  AND  Close > EMA Fast
Downtrend OK  =  EMA Fast < EMA Slow  AND  EMA Fast < EMA Fast 5 bars ago  AND  Close < EMA Fast
```

This catches the same "trend hold" condition you'd verify by drawing a manual trendline.

---

### 3.10 Entry Candle

The script accepts **either** of two entry candle types:

```
Bullish Engulfing   =  Green candle that fully engulfs the previous red candle
Bullish Momentum    =  Green candle with body > 60% of range AND close > previous high
```

(Bearish versions are symmetric.) The momentum-candle option catches breakouts where there's no formal engulfing, only one strong bar.

---

### 3.11 Volume Filter

```
High Volume  =  Current volume > volMA × volMultiplier
```

Default `volMultiplier = 1.2` means current volume must be at least 20% above the 20-period SMA. This avoids low-conviction signals.

## 4. The Confluence Scoring System

This is the heart of the strategy. Each bar, the script awards **1 point per condition met**:

### LONG score (max 9)

| # | Condition                             | Points |
| - | ------------------------------------- | ------ |
| 1 | HTF trend bullish                     | 1      |
| 2 | LTF uptrend valid                     | 1      |
| 3 | Bullish BOS or MSS just occurred      | 1      |
| 4 | Bullish liquidity sweep present       | 1      |
| 5 | Price in bullish OB                   | 1      |
| 6 | Price in long Fib golden zone         | 1      |
| 7 | Active bullish FVG                    | 1      |
| 8 | Bullish entry candle (engulf/momo)    | 1      |
| 9 | High volume                           | 1      |

(Short score is the mirror image.)

### Two filters, both must pass

1. **Hard filters** — each toggle (`fUseTrend`, `fUseBOS`, etc.) in the Filters group, if ON, is mandatory. If `fUseFib = true` and price is NOT in the fib zone, no signal regardless of score.
2. **Score gate** — total score ≥ `minScore` (default 6).

A signal only fires when **both** pass.

### How to tune

| Goal                        | Action                                                                  |
| --------------------------- | ----------------------------------------------------------------------- |
| Fewer, higher-quality       | `minScore = 7-8`, leave all hard filters ON                             |
| More signals, looser        | `minScore = 5`, turn off `fUseFVG` and `fUseFib`                        |
| Trend-following only        | Keep `fUseTrend` ON, turn off `fUseLiq` (sweeps are reversal-y)         |
| Reversal hunting            | Turn off `fUseTrend`, keep `fUseLiq` and BOS/MSS                        |

## 9. Risk Management

### Position sizing is automatic

The script computes position size each bar:

```
Risk Cash       = Equity × (Risk Per Trade % ÷ 100)
SL Distance     = |Entry − Stop Loss|
Position Size   = Risk Cash ÷ SL Distance
```

So if your equity is $10,000 and Risk Per Trade is 1%, you risk **exactly $100** per trade — whether the SL is 5 pips or 50 pips, the script sizes the position to hit that $100 max loss.

### Key rules to follow manually

1. **Never raise Risk Per Trade above 2%** unless you've validated a positive edge in backtest over 100+ trades.
2. **Avoid news events** — the script doesn't filter for news. Use an external economic calendar.
3. **Honor the stop** — don't move SL further away after entry. Move it closer (to breakeven) once price moves 1R in your favor if you want, but never further.
4. **Daily loss cap** — manually stop trading for the day after 2–3 consecutive losses.

### Default safeguards in the script

- `pyramiding = 0` — only one position at a time
- `process_orders_on_close = true` — no intrabar repaint
- ATR-based SL prevents a fixed-pip SL being too tight in volatile markets

## 11. Backtesting Workflow

1. **Pick a representative period** — at least 6 months of data on the timeframe you trade.
2. Open **Strategy Tester** at the bottom of the chart (next to Pine Editor).
3. Look at these metrics in order:
   - **Net Profit %** — is the edge real?
   - **Profit Factor** — > 1.3 is acceptable, > 1.7 is good.
   - **Max Drawdown %** — must be tolerable for your psychology.
   - **Total Closed Trades** — need 50+ for statistical significance, 100+ ideal.
   - **Percent Profitable** — even 35–45% is fine if R:R is 2+.
   - **Avg Trade** — must be positive after commission.
4. Check the **List of Trades** tab — look for clustering of losses (regime-specific weakness).
5. Re-run on **2–3 different symbols** to check generality.

### What good backtest results look like

| Metric              | Acceptable | Good   | Excellent |
| ------------------- | ---------- | ------ | --------- |
| Profit Factor       | > 1.3      | > 1.7  | > 2.0     |
| Max Drawdown        | < 25%      | < 15%  | < 10%     |
| Win Rate (@ R:R 2)  | > 35%      | > 45%  | > 55%     |
| Trades / month      | 5–30       | 10–40  | 15–50     |

## 12. Optimization Workflow

**Don't curve-fit.** Optimization should be done in this order:

1. **Lock in market regime first.** Backtest on a recent 6-month period with default settings. Note the result.
2. **Tune `minScore` first.** Try 5, 6, 7. Pick the value that maximizes Profit Factor without dropping trade count below 30.
3. **Then tune `swingLen`.** Try 3, 5, 8.
4. **Then `atrMult` (SL).** Try 1.2, 1.5, 1.8, 2.0.
5. **Then `rrRatio`.** Try 1.5, 2.0, 2.5, 3.0.
6. **Walk-forward test.** Take your best settings → apply to the previous 3 months you did NOT optimize on. If results hold, you have a real edge. If they collapse, you curve-fit.

**Never** tune more than 3 parameters at once — the combinatorial space explodes and overfitting becomes guaranteed.

## 13. Worked Example — Long Setup

Imagine 5m chart on BTCUSDT during a London-session pullback:

| Bar | Event                                                  | Score |
| --- | ------------------------------------------------------ | ----- |
| t-20| HTF EMA 20 crossed above EMA 50 (1H trend bullish)     | 1     |
| t-15| Sharp rally up + retrace                                | —     |
| t-8 | Bullish BOS — close above swing high                    | 1+1   |
| t-5 | Price retraces into 0.5–0.618 zone                      | 1     |
| t-3 | Forms a bullish FVG during the bounce attempt           | 1     |
| t-2 | Price wicks below the recent swing low, closes back up  | 1     |
| t-1 | Last bearish candle (the OB) gets revisited             | 1     |
| t-0 | Bullish engulfing candle on high volume                 | 1+1+1 = **9/9** |

The script:
- Fires a 🔼 LONG triangle
- Computes SL = `min(low, low[1]) − ATR × 1.5`
- Computes TP = entry + (entry − SL) × 2.0
- Sizes position to risk exactly 1% of equity
- Opens the long via `strategy.entry()`
- Places bracket order via `strategy.exit()`

If TP hits → +2R win. If SL hits → −1R loss. Either way, **risk was capped at the configured %**.

## 15. Troubleshooting

| Problem                                              | Likely cause / Fix                                                                                                      |
| ---------------------------------------------------- | ----------------------------------------------------------------------------------------------------------------------- |
| "end of line without line continuation" error        | A multi-line statement has indentation that's a multiple of 4. Use the latest cleaned file or join lines together.       |
| No signals firing                                    | `minScore` too high, or too many hard filters ON in low-confluence environment. Drop `minScore` to 5 and turn off `fUseFVG`/`fUseFib` to test.|
| Too many signals, mostly losing                       | `minScore` too low. Bump to 7. Also verify HTF is actually trending — neutral markets are best avoided.                  |
| Boxes maxing out / older OB/FVG disappear            | Pine has limits (max 200 boxes). This is expected — the script keeps the most recent ones.                              |
| Strategy Tester shows zero trades                      | Date range likely doesn't include enough bars for HTF EMA. Scroll back several days/weeks.                              |
| Position size shows as huge or tiny                    | `Risk Per Trade` value misconfigured, or SL distance approaching zero. Check ATR is non-zero on the timeframe.          |
| Different result on re-load                            | The Strategy Tester recomputes on data-window changes. Use a fixed date range for reproducible backtests.                |
| Dashboard not showing                                   | `Show Info Table` toggle is OFF, or browser is zoomed in cutting off the top-right corner.                              |
| HTF trend flickering between bullish/bearish          | EMAs are crossing repeatedly — market is choppy. Avoid trading or wait for a clean trend.                                |

## 16. Limitations & Disclaimers

### Technical limitations

1. **No tick-level execution simulation.** Fills are modeled at the close of the entry bar; in live trading, slippage will differ slightly.
2. **No news-event filter.** The script has no awareness of FOMC, NFP, CPI, earnings, etc. Manual avoidance required.
3. **OB and FVG are simplified.** Production-grade ICT setups use refinements like mitigation tracking, breaker blocks, and inversion FVGs — not implemented here.
4. **Trendline is EMA-proxy.** It doesn't draw literal pivot-to-pivot trendlines.
5. **Single-symbol.** No correlation or portfolio-level risk management.
6. **Past performance ≠ future results.** Always.

### Financial disclaimer

This is a **technical analysis tool** built for educational and development purposes. It is **not** financial advice, not an investment recommendation, and not a guarantee of profitability. Trading carries substantial risk of loss. You are responsible for:

- Validating the strategy on your data before risking capital
- Understanding the regulatory environment in your jurisdiction
- Sizing positions according to your personal risk tolerance
- Recognizing that any backtest result can fail forward

**Use paper trading or a demo account for at least 2–4 weeks** before deploying real capital, and even then start with the minimum position size.
