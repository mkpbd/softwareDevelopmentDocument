f_minAbove(_p, _a, _b) =>
    float r = na
    if not na(_a) and _a > _p
        r := na(r) ? _a : math.min(r, _a)
    if not na(_b) and _b > _p
        r := na(r) ? _b : math.min(r, _b)
    r
f_maxBelow(_p, _a, _b) =>
    float r = na
    if not na(_a) and _a < _p
        r := na(r) ? _a : math.max(r, _a)
    if not na(_b) and _b < _p
        r := na(r) ? _b : math.max(r, _b)
    r
// H1: nearest MULTI-TOUCH pool above/below price (>=2 swings clustered within tol).
f_poolAbove(_arr, _p, _tol) =>
    float best = na
    int n = array.size(_arr)
    if n > 0
        for i = 0 to n - 1
            lvl = array.get(_arr, i)
            if lvl > _p
                int touches = 0
                for j = 0 to n - 1
                    if math.abs(array.get(_arr, j) - lvl) <= _tol
                        touches += 1
                if touches >= 2 and (na(best) or lvl < best)
                    best := lvl
    best
f_poolBelow(_arr, _p, _tol) =>
    float best = na
    int n = array.size(_arr)
    if n > 0
        for i = 0 to n - 1
            lvl = array.get(_arr, i)
            if lvl < _p
                int touches = 0
                for j = 0 to n - 1
                    if math.abs(array.get(_arr, j) - lvl) <= _tol
                        touches += 1
                if touches >= 2 and (na(best) or lvl > best)
                    best := lvl
    best

longTrig = longCandle and longQualOk and (not useEMAfilter or emaBull) and (not useRSIfilter or rsiBull)

// CONFLUENCE = 7 INDEPENDENT buckets (correlated signals collapsed so the score
// actually discriminates quality). Orderflow weighted x2 — it is the strongest read.
// H5 FIX: bucket 3 was near-constant (fib already hard-gated). Now = discount
// location AND RSI momentum agreement, an orthogonal quality read.
// ZONE QUALITY (H/feature: granularity) — count DISTINCT tapped PD-arrays so an
// A+ stacked location (e.g. FVG+OB, or a breaker/IFVG/BPR confluence) scores 2,
// a single weak gap scores 1, nothing scores 0. Previously any tap = flat +1.
longZoneQual = (inBullFVG ? 1 : 0) + (inBullOB ? 1 : 0) + (inBullBreaker ? 1 : 0) + (inBullIFVG ? 1 : 0) + (bprActive ? 1 : 0)
longScore = 0
longScore += longArmed ? 1 : 0                                                                                          // 1 structure armed
longScore += longZoneQual >= 2 ? 2 : longZoneQual >= 1 ? 1 : 0                                                           // 2 PD-array zone tapped (quality-graded 0-2)
longScore += (inDiscount and (not useRSIscore or rsiBull)) ? 1 : 0                                                       // 3 discount location + RSI momentum
longScore += (bullEngulf or fakeoutBull) ? 1 : 0                                                                         // 4 trigger — M1 FIX: the old passive-limit proxy (passiveLimit and longZone and inDiscount) was NEAR-CONSTANT +1 (longZone is already a hard gate, and an up-impulse OTE always sits in discount), so it inflated the score without discriminating quality. Now scores ONLY on a real engulf/fakeout rejection. NOTE: this lowers typical passive-limit scores by ~1 — if trades starve, drop minScore by 1 (the orderflow/zone-stack buckets now do the real discrimination).
longScore += ((liqScore and liqSweepBull) or (smtOn and smtBull) or rsiBullDiv or rsiHidBullDiv or inducementBull or judasBull) ? 2 : 0  // 5 ORDERFLOW (weighted: sweep / SMT / RSI-div / hidden-div / inducement / Judas)
longScore += (not useHTF and longBiasOk) ? 1 : 0                                                                        // 6 HTF (DECORRELATED: scored only when HTF is NOT already a hard gate — no double-count)
longScore += ((useEMAscore and emaBull) or (useTLBreak and tlBreakBull) or (useVWAPscore and vwapBull)) ? 1 : 0          // 7 trend confluence (EMA / TL / VWAP)
longScore += mssBull ? 1 : 0                                                                                            // 8 MSS — CHoCH breaking a classified protected swing (highest-quality reversal; was computed+plotted but unused)

// SHORT confluence
shortZone = (not useFVG and not useOB) or (useFVG and (inBearFVG or bprActive)) or (useOB and inBearOB) or (useIFVG and inBearIFVG) or (useBreaker and inBearBreaker)
// H1 FIX: see long note — require the tighter OTE band when both Golden+OTE are on.
shortFib  = (not useGolden and not useOTE) or (useGolden and useOTE and priceInGolden and priceInOTE) or (useGolden and not useOTE and priceInGolden) or (useOTE and not useGolden and priceInOTE)
shortCandle = passiveLimit or (useEngulf and bearEngulf and trigDispOk) or (useFakeout and fakeoutBear and trigDispOk) or (not useEngulf and not useFakeout)
shortQualOk = not (passiveLimit and useLimitQual) or bearEngulf or fakeoutBear or (liqScore and liqSweepBear) or (smtOn and smtBear) or rsiBearDiv or rsiHidBearDiv or inducementBear or judasBear
shortTrig = shortCandle and shortQualOk and (not useEMAfilter or emaBear) and (not useRSIfilter or rsiBear)

shortZoneQual = (inBearFVG ? 1 : 0) + (inBearOB ? 1 : 0) + (inBearBreaker ? 1 : 0) + (inBearIFVG ? 1 : 0) + (bprActive ? 1 : 0)
shortScore = 0
shortScore += shortArmed ? 1 : 0                                                                                        // 1 structure armed
shortScore += shortZoneQual >= 2 ? 2 : shortZoneQual >= 1 ? 1 : 0                                                        // 2 PD-array zone tapped (quality-graded 0-2)
shortScore += (inPremium and (not useRSIscore or rsiBear)) ? 1 : 0                                                       // 3 premium location + RSI momentum
shortScore += (bearEngulf or fakeoutBear) ? 1 : 0                                                                        // 4 trigger — M1 FIX: scores only on a real engulf/fakeout rejection (dropped the near-constant passive-limit proxy; see long bucket 4).
shortScore += ((liqScore and liqSweepBear) or (smtOn and smtBear) or rsiBearDiv or rsiHidBearDiv or inducementBear or judasBear) ? 2 : 0  // 5 ORDERFLOW (weighted: sweep / SMT / RSI-div / hidden-div / inducement / Judas)
shortScore += (not useHTF and shortBiasOk) ? 1 : 0                                                                      // 6 HTF (DECORRELATED: scored only when HTF is NOT already a hard gate — no double-count)
shortScore += ((useEMAscore and emaBear) or (useTLBreak and tlBreakBear) or (useVWAPscore and vwapBear)) ? 1 : 0         // 7 trend confluence (EMA / TL / VWAP)
shortScore += mssBear ? 1 : 0                                                                                           // 8 MSS — CHoCH breaking a classified protected swing (highest-quality reversal; was computed+plotted but unused)

// ---------------------------------------------------------------------
// SL / TP  (SMC: SL beyond swing wick + buffer; TP by R:R / fib / liquidity)
// Spread cost modeled: stop pushed OUT and target pulled IN by spreadPrice so
// the backtest actually pays the round-trip spread an FX scalp incurs.
// ---------------------------------------------------------------------
// Asia spreads run wider — model that instead of one flat number
spreadPipsEff = spreadPips * (useSessSpread and kzAsia ? spreadAsiaMult : 1.0)
spreadPrice = spreadPipsEff * pip
// structural stop (swing wick + buffer + spread), then floored to a min ATR distance
minSLdist = (minSLatr > 0 and not na(atr)) ? atr * minSLatr : 0.0
// M1 FIX: when reading a higher structure TF, anchor the structural stop to THAT
// TF's impulse-leg extreme (mtfLegLo/Hi) — not the chart-TF pivot — so 1R matches
// the OTE band the entry is built from. Guard with the current bar wick so a stop
// can't sit inside price. Single-TF mode keeps the chart-pivot anchor.
slLongAnchor  = (mtfOn and not na(mtfLegLo)) ? math.min(mtfLegLo, low)  : (na(lastPL) ? low  : math.min(lastPL, low))
slShortAnchor = (mtfOn and not na(mtfLegHi)) ? math.max(mtfLegHi, high) : (na(lastPH) ? high : math.max(lastPH, high))
longSL  = math.min(slLongAnchor  - bufferPips * pip - spreadPrice, close - minSLdist)
shortSL = math.max(slShortAnchor + bufferPips * pip + spreadPrice, close + minSLdist)

// H2 FIX: in passive-limit mode the order fills at the OTE limit price (BELOW close
// for longs / ABOVE for shorts), but qty was sized on close-based risk (close - SL),
// so actual $-risk drifted UNDER the configured riskPerc and R-multiples were off.
// Size risk off the PLANNED FILL BASIS (= the resting limit px, half-spread pessimized)
// so position size matches the real risk taken. Market/ValMode basis stays = close.
// (longLimitPx/shortLimitPx below reuse these so the order and the sizing agree.)
longEntryBasis  = ((useLimitEntry and not useValMode) and legValid) ? math.min(close, fOTEs) - spreadPrice * 0.5 : close
shortEntryBasis = ((useLimitEntry and not useValMode) and legValid) ? math.max(close, fOTEs) + spreadPrice * 0.5 : close

longRisk  = longEntryBasis  - longSL
shortRisk = shortSL - shortEntryBasis

// max-SL guard: reject setups whose stop is wider than maxSLatr × ATR
longSLok  = maxSLatr == 0 or (not na(atr) and longRisk  <= atr * maxSLatr)
shortSLok = maxSLatr == 0 or (not na(atr) and shortRisk <= atr * maxSLatr)

// post-loss cooldown: stamp the bar a losing trade closed, block entries for N bars
var int lastLossBar = na
if strategy.closedtrades > nz(strategy.closedtrades[1], 0)
    if strategy.closedtrades.profit(strategy.closedtrades - 1) < 0
        lastLossBar := bar_index
coolOk = not useCooldown or na(lastLossBar) or (bar_index - lastLossBar) > coolBars

// optional volume hard gate (FX tick-volume — off by default)
volOk = not useVolFilter or (not na(volume) and volume > ta.sma(volume, volLen) * volMult)

// VALIDATION MODE: bypass contextual filters + drop minScore to 3 to measure how
// many trades the CORE SMC logic (armed/zone/fib/trigger/SL) actually produces.
// All *E vars collapse to "always pass" when useValMode is on; live behavior otherwise.
minScoreEff = useValMode ? 3 : minScore
tradeE = useValMode or tradeOk
htfLE  = useValMode or htfLongOk
htfSE  = useValMode or htfShortOk
extLE  = useValMode or extLongOk
extSE  = useValMode or extShortOk
adxE   = useValMode or adxOk
smtLE  = useValMode or smtLongOk
smtSE  = useValMode or smtShortOk
coolE  = useValMode or coolOk
volE   = useValMode or volOk

judasLE = useValMode or judasLongOk
judasSE = useValMode or judasShortOk
goLongRaw  = tradeE and htfLE and extLE and adxE and smtLE and judasLE and longArmed  and longZone  and longFib  and longTrig  and longScore  >= minScoreEff and longSLok  and coolE and volE and strategy.position_size == 0
goShortRaw = tradeE and htfSE and extSE and adxE and smtSE and judasSE and shortArmed and shortZone and shortFib and shortTrig and shortScore >= minScoreEff and shortSLok and coolE and volE and strategy.position_size == 0
// §10 setup ranking: if BOTH sides qualify on the same bar, take the HIGHER-score
// one (tie -> skip, ambiguous). True best-of-session ranking would need lookahead
// (deferring entry to compare against later bars) and is intentionally NOT done —
// that would repaint. This resolves only same-bar long/short conflicts honestly.
goLong  = goLongRaw  and (not goShortRaw or longScore  > shortScore)
goShort = goShortRaw and (not goLongRaw  or shortScore > longScore)

longTP    = close + longRisk  * rr - spreadPrice
shortTP   = close - shortRisk * rr + spreadPrice
// optional: take profit at fib extension instead of fixed R:R
longTP  := useFibExtTP and not na(fibExtTarget) and fibExtTarget > close ? fibExtTarget : longTP
shortTP := useFibExtTP and not na(fibExtTarget) and fibExtTarget < close ? fibExtTarget : shortTP
// optional: draw-on-liquidity target (nearest BSL/SSL beyond >=1R) overrides
longTP  := useLiqTP and not na(liqLongTP)  and liqLongTP  > close + longRisk  ? liqLongTP  : longTP
shortTP := useLiqTP and not na(liqShortTP) and liqShortTP < close - shortRisk ? liqShortTP : shortTP
// FIX: cap TP R:R so a far fib-ext / liquidity pool can't create a fantasy target
longTP  := longRisk  > 0 ? math.min(longTP,  close + longRisk  * maxTPr) : longTP
shortTP := shortRisk > 0 ? math.max(shortTP, close - shortRisk * maxTPr) : shortTP

// position size by risk % of equity, capped so notional never exceeds equity*maxLev
qtyCap   = close > 0 ? strategy.equity * maxLev / close : na
longQty  = longRisk  > 0 ? math.min((strategy.equity * riskPerc / 100) / longRisk,  qtyCap) : na
shortQty = shortRisk > 0 ? math.min((strategy.equity * riskPerc / 100) / shortRisk, qtyCap) : na

// LIMIT-entry price = OTE proximal (0.62), never worse than the current close.
// Long: rest at/below close so we only fill on a deeper discount retrace.
// Short: rest at/above close so we only fill on a deeper premium retrace.
// §3 FIX (fill optimism): TV fills a limit at your EXACT price on a wick touch,
// no slippage/queue/rejection. Real fills are worse. Pessimize the limit basis by
// half-spread (a buy effectively fills nearer the ask, a sell nearer the bid) so
// the backtested entry isn't ~½-spread better than reality on every single trade.
// H2 FIX: reuse the same basis the qty was sized on, so order price and risk sizing agree.
longLimitPx  = longEntryBasis
shortLimitPx = shortEntryBasis

// remember active SL/TP + entry state for management
var float curSL       = na
var float curTP       = na
var float entryPx     = na
var float initRisk    = na   // initial stop distance (1R in price)
var float trailStop   = na
var bool  partialDone = false
var bool  partial2Done = false
var int   entryBar    = na   // §7: bar the position filled (time-stop reference)
var float liqPartPx   = na   // §9.5: price of the nearest liquidity pool for the partial (na = use +R)
// planned (structural) SL/TP captured at ORDER time; re-anchored on the real fill
var float planLongSL  = na
var float planLongTP  = na
var float planShortSL = na
var float planShortTP = na
var int   longPendBar  = na  // bar a resting long limit was placed (na = none pending)
var int   shortPendBar = na

// H1 FIX — HONEST round-trip metric accumulators. strategy.closedtrades counts every
// PARTIAL close (TP1/TP2) as its own trade, so the built-in win%/PF/Sharpe are inflated
// by partial scratches. These track per-POSITION realized pnl (entry-net → flat-net) so
// each REAL trade counts once. R-multiples (rtSumR*) also fix L-1 (dollar-based ratios).
var float rtStartNet      = na   // strategy.netprofit captured at entry fill
var float rtRiskCash      = na   // 1R in account currency, captured at entry
var int   rtTrades        = 0
var int   rtWins          = 0
var int   rtConsecLoss    = 0
var int   rtMaxConsecLoss = 0
var float rtSumR          = 0.0  // Σ R-multiple
var float rtSumR2         = 0.0  // Σ R²
var float rtSumDn2        = 0.0  // Σ R² for losing trades (downside)
var float rtGain          = 0.0  // gross profit (round-trip)
var float rtLoss          = 0.0  // gross loss (round-trip, positive)
var float rtNet           = 0.0  // Σ round-trip pnl

// ----- ORDER PLACEMENT (limit at OTE when enabled, else market-on-close) -----
// Protective exit is attached immediately so the stop is live the instant the
// entry fills (whether now via market or later via the resting limit).
if goLong and longRisk > 0 and not na(longQty) and na(longPendBar) and strategy.position_size == 0
    strategy.entry("LONG", strategy.long, qty=longQty, limit = (useLimitEntry and not useValMode) ? longLimitPx : na)
    strategy.exit("LONG-X", from_entry="LONG", stop=longSL, limit=longTP)
    planLongSL := longSL
    planLongTP := longTP
    curSL := longSL
    curTP := longTP
    entryPx := close          // provisional; refined to avg fill price below
    initRisk := longRisk
    trailStop := na
    partialDone := false
    partial2Done := false
    longPendBar := bar_index
    // NOTE: arm is consumed on FILL (justLong), not here — so a limit that cancels
    // unfilled at window end leaves the arm live for a later trigger in the window.
if goShort and shortRisk > 0 and not na(shortQty) and na(shortPendBar) and strategy.position_size == 0
    strategy.entry("SHORT", strategy.short, qty=shortQty, limit = (useLimitEntry and not useValMode) ? shortLimitPx : na)
    strategy.exit("SHORT-X", from_entry="SHORT", stop=shortSL, limit=shortTP)
    planShortSL := shortSL
    planShortTP := shortTP
    curSL := shortSL
    curTP := shortTP
    entryPx := close
    initRisk := shortRisk
    trailStop := na
    partialDone := false
    partial2Done := false
    shortPendBar := bar_index
    // arm consumed on FILL (justShort), not here — see long note above.

// ----- CANCEL stale resting limits (filled => clear; unfilled past window / opposite break => cancel) -----
if not na(longPendBar)
    if strategy.position_size > 0
        longPendBar := na
    else if (bar_index - longPendBar) > structWinEff or shortBreak
        strategy.cancel("LONG")
        longPendBar := na
if not na(shortPendBar)
    if strategy.position_size < 0
        shortPendBar := na
    else if (bar_index - shortPendBar) > structWinEff or longBreak
        strategy.cancel("SHORT")
        shortPendBar := na

// ----- FILL DETECTION: re-anchor SL/TP/R to the ACTUAL fill price (limit fills better than close) -----
justLong  = strategy.position_size > 0 and strategy.position_size[1] <= 0
justShort = strategy.position_size < 0 and strategy.position_size[1] >= 0
if justLong
    entryPx  := strategy.position_avg_price
    curSL    := planLongSL
    initRisk := math.max(entryPx - planLongSL, syminfo.mintick)
    curTP    := (useFibExtTP or useLiqTP) ? planLongTP : entryPx + initRisk * rr - spreadPrice
    trailStop := na
    partialDone := false
    partial2Done := false
    dayTrades += 1
    longArmBar := na          // consume the arm only once the entry actually fills
    strategy.exit("LONG-X", from_entry="LONG", stop=curSL, limit=curTP)
    rtStartNet := strategy.netprofit                          // H1: round-trip baseline
    rtRiskCash := initRisk * math.abs(strategy.position_size) // H1: 1R in cash
    entryBar  := bar_index                                    // §7 time-stop reference
    // §9.5: nearest liquidity pool between entry and final TP for the partial
    liqPartPx := (usePartialLiq and not na(liqLongTP) and liqLongTP > entryPx and (na(curTP) or liqLongTP < curTP)) ? liqLongTP : na
if justShort
    entryPx  := strategy.position_avg_price
    curSL    := planShortSL
    initRisk := math.max(planShortSL - entryPx, syminfo.mintick)
    curTP    := (useFibExtTP or useLiqTP) ? planShortTP : entryPx - initRisk * rr + spreadPrice
    trailStop := na
    partialDone := false
    partial2Done := false
    dayTrades += 1
    shortArmBar := na         // consume the arm only once the entry actually fills
    strategy.exit("SHORT-X", from_entry="SHORT", stop=curSL, limit=curTP)
    rtStartNet := strategy.netprofit                          // H1: round-trip baseline
    rtRiskCash := initRisk * math.abs(strategy.position_size) // H1: 1R in cash
    entryBar  := bar_index                                    // §7 time-stop reference
    // §9.5: nearest liquidity pool between entry and final TP for the partial
    liqPartPx := (usePartialLiq and not na(liqShortTP) and liqShortTP < entryPx and (na(curTP) or liqShortTP > curTP)) ? liqShortTP : na

// ----- TRADE MANAGEMENT: break-even, ATR trail, partial TP (scalp) -----
if strategy.position_size > 0 and not na(entryPx) and not na(initRisk) and initRisk > 0
    if useBE and high >= entryPx + beTrigR * initRisk
        curSL := math.max(nz(curSL, entryPx), entryPx + beOffPips * pip)
    if useTrail
        trailStop := na(trailStop) ? close - atr * atrMult : math.max(trailStop, close - atr * atrMult)
        curSL := math.max(nz(curSL, trailStop), trailStop)
    strategy.exit("LONG-X", from_entry="LONG", stop=curSL, limit=curTP)
    longPartHit = (usePartialLiq and not na(liqPartPx)) ? high >= liqPartPx : high >= entryPx + partialR * initRisk   // §9.5
    if usePartial and not partialDone and longPartHit
        strategy.close("LONG", qty_percent=partialPct, comment="TP1")
        partialDone := true
        if useBEafterPartial
            curSL := math.max(nz(curSL, entryPx), entryPx + beOffPips * pip)
            strategy.exit("LONG-X", from_entry="LONG", stop=curSL, limit=curTP)
    if usePartial2 and partialDone and not partial2Done and high >= entryPx + partial2R * initRisk
        strategy.close("LONG", qty_percent=partial2Pct, comment="TP2")
        partial2Done := true

if strategy.position_size < 0 and not na(entryPx) and not na(initRisk) and initRisk > 0
    if useBE and low <= entryPx - beTrigR * initRisk
        curSL := math.min(nz(curSL, entryPx), entryPx - beOffPips * pip)
    if useTrail
        trailStop := na(trailStop) ? close + atr * atrMult : math.min(trailStop, close + atr * atrMult)
        curSL := math.min(nz(curSL, trailStop), trailStop)
    strategy.exit("SHORT-X", from_entry="SHORT", stop=curSL, limit=curTP)
    shortPartHit = (usePartialLiq and not na(liqPartPx)) ? low <= liqPartPx : low <= entryPx - partialR * initRisk   // §9.5
    if usePartial and not partialDone and shortPartHit
        strategy.close("SHORT", qty_percent=partialPct, comment="TP1")
        partialDone := true
        if useBEafterPartial
            curSL := math.min(nz(curSL, entryPx), entryPx - beOffPips * pip)
            strategy.exit("SHORT-X", from_entry="SHORT", stop=curSL, limit=curTP)
    if usePartial2 and partialDone and not partial2Done and low <= entryPx - partial2R * initRisk
        strategy.close("SHORT", qty_percent=partial2Pct, comment="TP2")
        partial2Done := true

// §7 TIME-STOP: flatten a stale position open beyond maxHoldBars from entry. A
// scalp that hasn't resolved is dead money tying up margin — cut it.
if useMaxHold and strategy.position_size != 0 and not na(entryBar) and (bar_index - entryBar) >= maxHoldBars
    strategy.close_all(comment="time stop")

// H1 FIX: finalize the round-trip the bar the position returns to flat. tradePnl =
// realized-net delta since entry = the FULL trade incl. every partial, counted ONCE.
if strategy.position_size == 0 and strategy.position_size[1] != 0 and not na(rtStartNet)
    float tradePnl = strategy.netprofit - rtStartNet
    float rMul     = (not na(rtRiskCash) and rtRiskCash > 0) ? tradePnl / rtRiskCash : na
    rtTrades += 1
    rtNet    += tradePnl
    if tradePnl > 0
        rtWins     += 1
        rtGain     += tradePnl
    // LOW FIX: a scratch/break-even round-trip (pnl == 0, e.g. BE stop or time-stop at
    // entry) used to fall into `else` and neither reset NOR extend the loss streak —
    // it just sat in the middle. A non-losing outcome should BREAK a losing run, so
    // reset on pnl >= 0 and only increment the streak on a genuine loss.
    if tradePnl >= 0
        rtConsecLoss := 0
    else
        rtLoss          += -tradePnl
        rtConsecLoss    += 1
        rtMaxConsecLoss := math.max(rtMaxConsecLoss, rtConsecLoss)
    if not na(rMul)
        rtSumR  += rMul
        rtSumR2 += rMul * rMul
        if rMul < 0
            rtSumDn2 += rMul * rMul
    rtStartNet := na
    rtRiskCash := na

if strategy.position_size == 0
    curSL := na
    curTP := na
    entryPx := na
    initRisk := na
    trailStop := na
    partialDone := false
    partial2Done := false
    entryBar := na
    liqPartPx := na

// flatten open position when leaving the backtest window, on daily-loss, or on
// total-drawdown lock (prop-firm survival — stop bleeding from the high-water mark)
if ((useDateFilter and not inDateRange) or dayLossHit or totalDDhit) and strategy.position_size != 0
    strategy.close_all(comment="guard exit")

bgcolor(longArmed  and strategy.position_size == 0 ? color.new(color.green, 92) : na, title="Long armed")

bgcolor(useOOS and time >= oosStart ? color.new(color.purple, 94) : na, title="Out-of-sample region")
