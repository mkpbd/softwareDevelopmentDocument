                    box.delete(ob.bx)
                    label.delete(ob.lb)
                    array.remove(arr, i)

                    label.set_text(ob.lb, (isBull ? "Bull OB " : "Bear OB ") + "✓")

    0

f_htfTrend(tf) =>

        s += 1
    // Engulfing
    if (isBull and bullEngulf) or (not isBull and bearEngulf)
        s += 1
    // Fresh OB

        s += 1
    s
