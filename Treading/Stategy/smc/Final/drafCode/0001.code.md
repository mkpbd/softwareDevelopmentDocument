

```python
//@version=5
indicator("SMC Master Strategy [Bank Trader Edition]", shorttitle="SMC Master", overlay=true, max_boxes_count=500, max_lines_count=500, max_labels_count=500)

// ==================== INPUTS ====================

// General
grp1 = "General Settings"
swingLength = input.int(10, "Swing Length", minval=3, maxval=50, group=grp1)
htfTimeframe = input.timeframe("240", "HTF for Bias", group=grp1)

// Support & Resistance
grp2 = "Support & Resistance"
showSR = input.bool(true, "Show S/R Zones", group=grp2)
srLookback = input.int(50, "S/R Lookback", minval=10, group=grp2)

// Supply & Demand
grp3 = "Supply & Demand"
showSD = input.bool(true, "Show Supply/Demand", group=grp3)
sdStrength = input.float(1.5, "Zone Strength (ATR x)", minval=0.5, step=0.1, group=grp3)
demandColor = input.color(color.new(color.lime, 80), "Demand", group=grp3)
supplyColor = input.color(color.new(color.fuchsia, 80), "Supply", group=grp3)

// BOS / CHoCH
grp4 = "Market Structure"
showBOS = input.bool(true, "Show BOS / CHoCH", group=grp4)
bullColor = input.color(color.green, "Bullish", group=grp4)
bearColor = input.color(color.red, "Bearish", group=grp4)

// FVG
grp5 = "Fair Value Gap"
showFVG = input.bool(true, "Show FVG", group=grp5)
fvgBullColor = input.color(color.new(color.aqua, 80), "Bull FVG", group=grp5)
fvgBearColor = input.color(color.new(color.orange, 80), "Bear FVG", group=grp5)
deleteFilled = input.bool(true, "Auto-delete Filled FVGs", group=grp5)

// Order Blocks
grp6 = "Order Blocks"
showOB = input.bool(true, "Show Order Blocks", group=grp6)
obBullColor = input.color(color.new(color.green, 70), "Bull OB", group=grp6)
obBearColor = input.color(color.new(color.red, 70), "Bear OB", group=grp6)

// Fibonacci
grp7 = "Fibonacci Golden Zone"
showFib = input.bool(true, "Show Fibonacci", group=grp7)
fibLookback = input.int(50, "Fib Lookback", minval=10, group=grp7)
goldenZoneColor = input.color(color.new(color.yellow, 75), "Golden Zone", group=grp7)

// Liquidity
grp8 = "Liquidity"
showLiquidity = input.bool(true, "Show Liquidity (BSL/SSL)", group=grp8)
liqColor = input.color(color.new(color.blue, 50), "Liquidity Line", group=grp8)

// Signals
grp9 = "Signals & Dashboard"
showSignals = input.bool(true, "Show Entry Signals", group=grp9)
showDashboard = input.bool(true, "Show Dashboard", group=grp9)

// ==================== CALCULATIONS ====================

atr = ta.atr(14)
swingHigh = ta.pivothigh(high, swingLength, swingLength)
swingLow = ta.pivotlow(low, swingLength, swingLength)

var float lastSwingHigh = na
var float lastSwingLow = na
var int lastSwingHighBar = na
var int lastSwingLowBar = na

if not na(swingHigh)
    lastSwingHigh := swingHigh
    lastSwingHighBar := bar_index - swingLength

if not na(swingLow)
    lastSwingLow := swingLow
    lastSwingLowBar := bar_index - swingLength

// ==================== SUPPORT & RESISTANCE ====================

var line supportLine = na
var line resistanceLine = na

if showSR and barstate.islast
    highestHigh = ta.highest(high, srLookback)
    lowestLow = ta.lowest(low, srLookback)
    line.delete(supportLine)
    line.delete(resistanceLine)
    supportLine := line.new(bar_index - srLookback, lowestLow, bar_index + 10, lowestLow, color=color.green, width=2, style=line.style_dashed)
    resistanceLine := line.new(bar_index - srLookback, highestHigh, bar_index + 10, highestHigh, color=color.red, width=2, style=line.style_dashed)

// ==================== SUPPLY & DEMAND ====================

bigBullCandle = (close - open) > (atr * sdStrength) and close > open
bigBearCandle = (open - close) > (atr * sdStrength) and close < open

if showSD and bigBullCandle
    box.new(left=bar_index - 1, top=high, right=bar_index + 20, bottom=low, bgcolor=demandColor, border_color=color.lime, text="DEMAND", text_color=color.white, text_size=size.tiny)

if showSD and bigBearCandle
    box.new(left=bar_index - 1, top=high, right=bar_index + 20, bottom=low, bgcolor=supplyColor, border_color=color.fuchsia, text="SUPPLY", text_color=color.white, text_size=size.tiny)

// ==================== BOS & CHoCH ====================

var int trend = 0

if not na(lastSwingHigh) and close > lastSwingHigh and trend != 1
    if showBOS
        labelText = trend == -1 ? "CHoCH Up" : "BOS Up"
        label.new(bar_index, high, labelText, color=bullColor, textcolor=color.white, style=label.style_label_down, size=size.small)
        line.new(lastSwingHighBar, lastSwingHigh, bar_index, lastSwingHigh, color=bullColor, style=line.style_dashed, width=1)
    trend := 1

if not na(lastSwingLow) and close < lastSwingLow and trend != -1
    if showBOS
        labelText2 = trend == 1 ? "CHoCH Down" : "BOS Down"
        label.new(bar_index, low, labelText2, color=bearColor, textcolor=color.white, style=label.style_label_up, size=size.small)
        line.new(lastSwingLowBar, lastSwingLow, bar_index, lastSwingLow, color=bearColor, style=line.style_dashed, width=1)
    trend := -1

// ==================== FAIR VALUE GAP ====================

bullFVG = low > high[2] and close[1] > open[1]
bearFVG = high < low[2] and close[1] < open[1]

var array<box> fvgBoxes = array.new<box>()

if showFVG and bullFVG
    b = box.new(left=bar_index - 2, top=low, right=bar_index + 15, bottom=high[2], bgcolor=fvgBullColor, border_color=color.aqua, text="FVG", text_color=color.white, text_size=size.tiny)
    array.push(fvgBoxes, b)

if showFVG and bearFVG
    b = box.new(left=bar_index - 2, top=low[2], right=bar_index + 15, bottom=high, bgcolor=fvgBearColor, border_color=color.orange, text="FVG", text_color=color.white, text_size=size.tiny)
    array.push(fvgBoxes, b)

// Auto-delete filled FVG (FIXED: proper reverse loop)
if deleteFilled and array.size(fvgBoxes) > 0
    i = array.size(fvgBoxes) - 1
    while i >= 0
        b = array.get(fvgBoxes, i)
        topVal = box.get_top(b)
        botVal = box.get_bottom(b)
        if low <= botVal and high >= topVal
            box.delete(b)
            array.remove(fvgBoxes, i)
        i := i - 1

// ==================== ORDER BLOCKS ====================

bullishOB = close[1] < open[1] and close > high[1] and (close - open) > atr
bearishOB = close[1] > open[1] and close < low[1] and (open - close) > atr

if showOB and bullishOB
    box.new(left=bar_index - 1, top=high[1], right=bar_index + 25, bottom=low[1], bgcolor=obBullColor, border_color=color.green, border_width=2, text="Bull OB", text_color=color.white, text_size=size.tiny)

if showOB and bearishOB
    box.new(left=bar_index - 1, top=high[1], right=bar_index + 25, bottom=low[1], bgcolor=obBearColor, border_color=color.red, border_width=2, text="Bear OB", text_color=color.white, text_size=size.tiny)

// ==================== FIBONACCI GOLDEN ZONE ====================

if showFib and barstate.islast
    highPoint = ta.highest(high, fibLookback)
    lowPoint = ta.lowest(low, fibLookback)
    rangeVal = highPoint - lowPoint
    fib50 = highPoint - rangeVal * 0.5
    fib618 = highPoint - rangeVal * 0.618
    fib786 = highPoint - rangeVal * 0.786
    box.new(left=bar_index - fibLookback, top=fib618, right=bar_index + 15, bottom=fib786, bgcolor=goldenZoneColor, border_color=color.yellow, border_width=1, text="GOLDEN ZONE 61.8-78.6", text_color=color.yellow, text_size=size.small)
    line.new(bar_index - fibLookback, fib50, bar_index + 15, fib50, color=color.gray, style=line.style_dotted)
    line.new(bar_index - fibLookback, fib618, bar_index + 15, fib618, color=color.yellow, style=line.style_solid)
    line.new(bar_index - fibLookback, fib786, bar_index + 15, fib786, color=color.yellow, style=line.style_solid)

// ==================== LIQUIDITY ====================

if showLiquidity and not na(swingHigh)
    line.new(bar_index - swingLength, swingHigh, bar_index + 5, swingHigh, color=liqColor, style=line.style_dotted, width=1)
    label.new(bar_index + 5, swingHigh, "BSL", color=color.new(color.black, 100), textcolor=color.blue, style=label.style_none, size=size.tiny)

if showLiquidity and not na(swingLow)
    line.new(bar_index - swingLength, swingLow, bar_index + 5, swingLow, color=liqColor, style=line.style_dotted, width=1)
    label.new(bar_index + 5, swingLow, "SSL", color=color.new(color.black, 100), textcolor=color.blue, style=label.style_none, size=size.tiny)

// ==================== HTF BIAS ====================

htfClose = request.security(syminfo.tickerid, htfTimeframe, close, lookahead=barmerge.lookahead_off)
htfOpen = request.security(syminfo.tickerid, htfTimeframe, open, lookahead=barmerge.lookahead_off)
htfBullish = htfClose > htfOpen
htfBearish = htfClose < htfOpen
htfBias = htfBullish ? "BULLISH" : "BEARISH"

// ==================== DATE FRAMEWORK SIGNALS ====================

bullishEngulfing = close > open and close[1] < open[1] and close > open[1] and open < close[1]
bearishEngulfing = close < open and close[1] > open[1] and close < open[1] and open > close[1]

confluenceBuy = (htfBullish ? 1 : 0) + (bullishEngulfing ? 1 : 0) + (bullishOB ? 1 : 0) + (bullFVG ? 1 : 0) + (trend == 1 ? 1 : 0)
confluenceSell = (htfBearish ? 1 : 0) + (bearishEngulfing ? 1 : 0) + (bearishOB ? 1 : 0) + (bearFVG ? 1 : 0) + (trend == -1 ? 1 : 0)

strongBuySignal = confluenceBuy >= 3
strongSellSignal = confluenceSell >= 3

if showSignals and strongBuySignal
    label.new(bar_index, low - atr * 0.5, "BUY " + str.tostring(confluenceBuy) + "/5", color=color.green, textcolor=color.white, style=label.style_label_up, size=size.normal)

if showSignals and strongSellSignal
    label.new(bar_index, high + atr * 0.5, "SELL " + str.tostring(confluenceSell) + "/5", color=color.red, textcolor=color.white, style=label.style_label_down, size=size.normal)

// ==================== DASHBOARD ====================

if showDashboard and barstate.islast
    var table dashboard = table.new(position.top_right, 2, 8, bgcolor=color.new(color.black, 20), border_width=1, border_color=color.gray)
    table.cell(dashboard, 0, 0, "SMC MASTER", bgcolor=color.new(color.blue, 30), text_color=color.white, text_size=size.normal)
    table.cell(dashboard, 1, 0, "STATUS", bgcolor=color.new(color.blue, 30), text_color=color.white, text_size=size.normal)
    table.cell(dashboard, 0, 1, "HTF Bias", text_color=color.white, text_size=size.small)
    table.cell(dashboard, 1, 1, htfBias, text_color=htfBullish ? color.lime : color.red, text_size=size.small)
    structureText = trend == 1 ? "Bullish Up" : trend == -1 ? "Bearish Down" : "Neutral"
    structureColor = trend == 1 ? color.lime : trend == -1 ? color.red : color.gray
    table.cell(dashboard, 0, 2, "Structure", text_color=color.white, text_size=size.small)
    table.cell(dashboard, 1, 2, structureText, text_color=structureColor, text_size=size.small)
    table.cell(dashboard, 0, 3, "Buy Score", text_color=color.white, text_size=size.small)
    table.cell(dashboard, 1, 3, str.tostring(confluenceBuy) + "/5", text_color=confluenceBuy >= 3 ? color.lime : color.gray, text_size=size.small)
    table.cell(dashboard, 0, 4, "Sell Score", text_color=color.white, text_size=size.small)
    table.cell(dashboard, 1, 4, str.tostring(confluenceSell) + "/5", text_color=confluenceSell >= 3 ? color.red : color.gray, text_size=size.small)
    table.cell(dashboard, 0, 5, "ATR", text_color=color.white, text_size=size.small)
    table.cell(dashboard, 1, 5, str.tostring(math.round(atr, 5)), text_color=color.yellow, text_size=size.small)
    signalText = strongBuySignal ? "STRONG BUY" : strongSellSignal ? "STRONG SELL" : "WAIT"
    signalColor = strongBuySignal ? color.lime : strongSellSignal ? color.red : color.gray
    table.cell(dashboard, 0, 6, "Signal", text_color=color.white, text_size=size.small)
    table.cell(dashboard, 1, 6, signalText, text_color=signalColor, text_size=size.small)
    table.cell(dashboard, 0, 7, "Rule", bgcolor=color.new(color.yellow, 80), text_color=color.yellow, text_size=size.tiny)
    table.cell(dashboard, 1, 7, "Protect Capital First", bgcolor=color.new(color.yellow, 80), text_color=color.yellow, text_size=size.tiny)

// ==================== ALERTS ====================

alertcondition(strongBuySignal, title="Strong BUY", message="SMC Master: Strong BUY on {{ticker}} TF: {{interval}}")
alertcondition(strongSellSignal, title="Strong SELL", message="SMC Master: Strong SELL on {{ticker}} TF: {{interval}}")
alertcondition(bullFVG, title="Bullish FVG", message="Bullish FVG on {{ticker}}")
alertcondition(bearFVG, title="Bearish FVG", message="Bearish FVG on {{ticker}}")
alertcondition(bullishOB, title="Bullish OB", message="Bullish Order Block on {{ticker}}")
alertcondition(bearishOB, title="Bearish OB", message="Bearish Order Block on {{ticker}}")
```
