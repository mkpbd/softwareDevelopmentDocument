## Step 3: Complete PostgreSQL Schema with Sequences & Indexes

SQL

```sql
-- ============================================
-- COMPLETE POSTGRESQL SCHEMA
-- Project Management System
-- ============================================

-- Enable extensions
CREATE EXTENSION IF NOT EXISTS "uuid-ossp";

-- ============================================
-- SEQUENCES
-- ============================================
CREATE SEQUENCE seq_project_id START 1;
CREATE SEQUENCE seq_stage_id START 1;
CREATE SEQUENCE seq_issue_id START 1;
CREATE SEQUENCE seq_company_id START 1;
CREATE SEQUENCE seq_person_id START 1;
CREATE SEQUENCE seq_workflow_id START 1;
CREATE SEQUENCE seq_dictionary_id START 1;

-- ============================================
-- SCHEMA CREATION
-- ============================================
CREATE SCHEMA IF NOT EXISTS pmanager;

-- ============================================
-- TABLE: pm_project_statuses_all
-- ============================================
CREATE TABLE pmanager.pm_project_statuses_all (
    org_id              INTEGER,
    project_status_id   INTEGER PRIMARY KEY 
                        DEFAULT nextval('seq_project_id'),
    name                VARCHAR(150),
    description         VARCHAR(240),
    is_open             CHAR(1) DEFAULT 'N' 
                        CHECK (is_open IN ('Y','N')),
    code                VARCHAR(150),
    created_by          INTEGER,
    created_date        TIMESTAMP DEFAULT CURRENT_TIMESTAMP,
    modified_by         INTEGER,
    modified_date       TIMESTAMP,
    is_predefined       CHAR(1) DEFAULT 'N' 
                        CHECK (is_predefined IN ('Y','N')),
    attributes          TEXT,
    background          VARCHAR(240),
    project_class       VARCHAR(240)
);

-- ============================================
-- TABLE: pm_project_types_all
-- ============================================
CREATE TABLE pmanager.pm_project_types_all (
    org_id              INTEGER,
    project_type_id     INTEGER PRIMARY KEY,
    name                VARCHAR(150) NOT NULL,
    description         VARCHAR(240),
    created_by          INTEGER,
    created_date        TIMESTAMP DEFAULT CURRENT_TIMESTAMP,
    modified_by         INTEGER,
    modified_date       TIMESTAMP,
    is_predefined       CHAR(1) DEFAULT 'N',
    attributes          TEXT,
    default_status_id   INTEGER REFERENCES 
                        pmanager.pm_project_statuses_all(project_status_id),
    project_class       VARCHAR(240)
);

-- ============================================
-- TABLE: pm_projects_all
-- ============================================
CREATE TABLE pmanager.pm_projects_all (
    org_id                      INTEGER,
    project_id                  INTEGER PRIMARY KEY 
                                DEFAULT nextval('seq_project_id'),
    name                        VARCHAR(240) NOT NULL,
    description                 TEXT,
    variables                   TEXT,
    source                      VARCHAR(240),
    department_id               INTEGER,
    rule_number                 INTEGER,
    contract_date               DATE,
    purchase_order_number       VARCHAR(240),
    customer_id                 INTEGER,
    customer_manager_id         INTEGER,
    responsible_id              INTEGER,
    created_date                TIMESTAMP DEFAULT CURRENT_TIMESTAMP,
    modified_date               TIMESTAMP,
    created_by                  INTEGER,
    modified_by                 INTEGER,
    is_predefined               CHAR(1) DEFAULT 'N' 
                                CHECK (is_predefined IN ('Y','N')),
    project_type_id             INTEGER REFERENCES 
                                pmanager.pm_project_types_all(project_type_id),
    deliverables                TEXT,
    time_budget_h               NUMERIC(10,2),
    actual_end                  DATE,
    cost_manager_id             INTEGER,
    customer_department_id      INTEGER,
```

## Step 5: Create Views for Common Queries

SQL

``` sql
-- ============================================
-- VIEWS
-- ============================================

-- Project Overview View
CREATE VIEW pmanager.v_project_overview AS
SELECT 
    p.project_id,
    p.name AS project_name,
    p.description,
    pt.name AS project_type,
    ps.name AS project_status,
    p.created_date,
    p.responsible_id,
    p.priority,
    p.opportunity_amount,
    COUNT(DISTINCT s.stage_id) AS stage_count
FROM pmanager.pm_projects_all p
LEFT JOIN pmanager.pm_project_types_all pt 
    ON p.project_type_id = pt.project_type_id
LEFT JOIN pmanager.pm_project_statuses_all ps 
    ON pt.default_status_id = ps.project_status_id
LEFT JOIN pmanager.pm_stages_all s 
    ON p.project_id = s.project_id
GROUP BY 
    p.project_id, p.name, p.description,
    pt.name, ps.name, p.created_date,
    p.responsible_id, p.priority, 
    p.opportunity_amount;

-- Issue Activity View
CREATE VIEW pmanager.v_issue_activity AS
SELECT 
    ih.issue_id,
    ih.event_date,
    ih.issue_workflow_id,
    w.name AS workflow_name,
    s.name AS status_name,
    ih.person_id,
    p.first_name || ' ' || p.last_name AS person_name
FROM pmanager.bi_issue_history_all ih
LEFT JOIN pmanager.bi_issue_workflows_all w 
    ON ih.issue_workflow_id = w.issue_workflow_id
LEFT JOIN pmanager.bi_issue_statuses_all s 
    ON ih.state_id = s.issue_status_id
LEFT JOIN pmanager.bi_issue_people_all p 
    ON ih.person_id = p.person_id;
```
